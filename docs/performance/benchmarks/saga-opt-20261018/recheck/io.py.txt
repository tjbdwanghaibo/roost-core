import sys, re, statistics
# 按 progress.txt 的 start/end 切 iostat.txt（epoch + KB/t tps MB/s），输出每个运行的 disk0 平均/峰值
prog, io = sys.argv[1], sys.argv[2]
rows = []
for line in open(io):
    p = line.split()
    if len(p) == 4 and re.match(r"^\d+$", p[0]):
        try: rows.append((int(p[0]), float(p[1]), float(p[2]), float(p[3])))
        except ValueError: pass
rows = rows[1:]  # 第一行是开机以来的平均
runs, starts = [], {}
for line in open(prog):
    p = line.split()
    if len(p) >= 3 and p[2] in ("start", "end", "idle-start", "idle-end"):
        ts = int(p[1])
        if p[2] in ("start", "idle-start"):
            name = p[3] if p[2] == "start" else f"idle@{p[0]}"
            starts[name] = ts; last = name
        else:
            name = p[3] if p[2] == "end" else last
            runs.append((name, starts[name], ts))
print("| 运行 | 秒数 | tps 均值 / P95 | MB/s 均值 | KB/t 均值 |")
print("| --- | --- | --- | --- | --- |")
for name, a, b in runs:
    w = [r for r in rows if a + 15 <= r[0] <= b - 5] or [r for r in rows if a <= r[0] <= b]
    if not w: continue
    tps = sorted(r[2] for r in w)
    print(f"| {name} | {len(w)} | {statistics.mean(tps):.0f} / {tps[int(0.95*(len(tps)-1))]:.0f} | {statistics.mean(r[3] for r in w):.2f} | {statistics.mean(r[1] for r in w):.1f} |")
