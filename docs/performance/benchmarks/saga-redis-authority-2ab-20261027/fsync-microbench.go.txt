// fsyncbench 量 macOS 上 append + fsync 与 append + F_FULLFSYNC 的单次耗时（Redis appendfsync always 在 macOS 上用后者）。
package main

import (
	"fmt"
	"os"
	"path/filepath"
	"sort"
	"time"

	"golang.org/x/sys/unix"
)

func measure(dir string, full bool, n int) []time.Duration {
	f, err := os.CreateTemp(dir, "fsyncbench-*")
	if err != nil {
		panic(err)
	}
	defer os.Remove(f.Name())
	defer f.Close()
	buf := make([]byte, 256)
	out := make([]time.Duration, 0, n)
	for range n {
		start := time.Now()
		if _, err := f.Write(buf); err != nil {
			panic(err)
		}
		if full {
			if _, err := unix.FcntlInt(f.Fd(), unix.F_FULLFSYNC, 0); err != nil {
				panic(err)
			}
		} else if err := f.Sync(); err != nil {
			panic(err)
		}
		out = append(out, time.Since(start))
	}
	sort.Slice(out, func(i, j int) bool { return out[i] < out[j] })
	return out
}

func main() {
	for _, dir := range os.Args[1:] {
		dir, _ = filepath.Abs(dir)
		for _, full := range []bool{false, true} {
			d := measure(dir, full, 300)
			name := "fsync"
			if full {
				name = "F_FULLFSYNC"
			}
			fmt.Printf("%-48s %-12s P50=%-10v P90=%-10v P99=%-10v max=%v\n", dir, name, d[len(d)/2].Round(time.Microsecond), d[len(d)*9/10].Round(time.Microsecond), d[len(d)*99/100].Round(time.Microsecond), d[len(d)-1].Round(time.Microsecond))
		}
	}
}
