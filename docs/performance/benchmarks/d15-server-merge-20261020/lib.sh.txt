MD=$D/md
RUN=$D/drill/run
MONGO="mongodb://127.0.0.1:53117,127.0.0.1:53118,127.0.0.1:53119/?replicaSet=roost-it"
NATS=nats://127.0.0.1:40222
REDIS=127.0.0.1:42379
PREFIX=roost:mergedrill:account

mkdir -p "$RUN"
cd "$MD"
log() { echo "[drill $(date +%H:%M:%S)] $*"; }
ready() { # name port
  local i=0
  while [ $i -lt 120 ]; do
    curl -fsS -o /dev/null "http://127.0.0.1:$2/readyz" 2>/dev/null && return 0
    if ! kill -0 "$(cat $RUN/$1.pid)" 2>/dev/null; then log "$1 exited before ready"; tail -20 $RUN/$1.log; return 1; fi
    i=$((i+1)); sleep 1
  done
  log "$1 not ready"; tail -20 $RUN/$1.log; return 1
}
start() { # name service config port sid
  log "start $1"
  nohup bin/app "$2" --sid "$5" --config "$3" > $RUN/$1.log 2>&1 &
  echo $! > $RUN/$1.pid
  ready "$1" "$4"
}
stop() { # name
  local pid; pid=$(cat $RUN/$1.pid 2>/dev/null) || return 0
  kill "$pid" 2>/dev/null || true
  local i=0; while kill -0 "$pid" 2>/dev/null && [ $i -lt 150 ]; do i=$((i+1)); sleep 1; done
  if kill -0 "$pid" 2>/dev/null; then log "$1 did not stop in 150s"; return 1; fi
  log "stopped $1 (pid $pid)"
}
accountctl() { go run ./cmd/accountctl -redis $REDIS -prefix $PREFIX "$@"; }
drill() { $RUN/mergedrill "$1" -state $RUN/state.json -account-nats $NATS -prefix mergedrill "${@:2}"; }
mongosh_eval() { mongosh --quiet "$MONGO" --eval "$1"; }
