#!/usr/bin/env bash
# 爬坡第二组：更慢的起跑速率；每轮前等所有 saga 到终态、再多等 60 秒（duel 票 TTL）。
set -u
P=<scratch>; cd $P/lt1022; O=$P/ramp
M="mongodb://127.0.0.1:53117,127.0.0.1:53118,127.0.0.1:53119/?replicaSet=roost-it"
pending() { mongosh --quiet "$M" --eval 'print(db.getSiblingDB("lt1022_saga").getCollection("_sagas").countDocuments({status:{$in:[1,2,3]}}))' 2>/dev/null | tail -1; }
for spec in 100:2 100:4 200:4 200:2; do
  n=${spec%%:*}; s=${spec#*:}; lab=r$n-$s
  until [ "$(pending)" = 0 ]; do sleep 5; done; sleep 60
  curl -s 127.0.0.1:9100/metrics > $O/$lab.metrics-before
  l0=$(wc -l < .dev/game.log)
  echo "$(date +%H:%M:%S) $lab start loadavg=$(sysctl -n vm.loadavg)" | tee -a $O/run.log
  $P/bin/lt-loadtest -endpoint 127.0.0.1:7000 -count $n -ramp-step $s -ramp-interval 1s -metrics-addr 127.0.0.1:9300 -account-nats nats://127.0.0.1:40222 > $O/$lab.log 2>&1
  echo "$(date +%H:%M:%S) $lab rc=$? loadavg_end=$(sysctl -n vm.loadavg)" | tee -a $O/run.log
  sleep 3; curl -s 127.0.0.1:9100/metrics > $O/$lab.metrics-after
  tail -n +$((l0+1)) .dev/game.log > $O/$lab.game.log
done
echo done2 >> $O/run.log
