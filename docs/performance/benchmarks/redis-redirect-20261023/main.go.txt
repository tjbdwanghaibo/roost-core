package main

import (
 "context"
 "encoding/json"
 "fmt"
 "os"
 "os/exec"
 "strings"
 "sync"
 "time"

 fredis "github.com/tjbdwanghaibo/roost-core/infra/storage/redis"
 "github.com/tjbdwanghaibo/roost-core/infra/storage/redis/driver"
)

func main() {
 cfg:=fredis.DefaultConfig("")
 for p:=45400;p<45406;p++ {cfg.ClusterAddrs=append(cfg.ClusterAddrs,fmt.Sprintf("127.0.0.1:%d",p))}
 client:=driver.NewRedisClient(cfg);defer client.Close()
 ctx:=context.Background()
 const key="{roost-redirect-diag}:executions"
 var wg sync.WaitGroup
 failovers:=make([]string,0,3)
 wg.Go(func(){for _,p:=range strings.Split(os.Getenv("FAILOVER_PORTS"),","){time.Sleep(3*time.Second);raw,err:=exec.Command("/opt/homebrew/bin/redis-cli","-p",p,"CLUSTER","FAILOVER").CombinedOutput();failovers=append(failovers,fmt.Sprintf("%s: %s %v",p,strings.TrimSpace(string(raw)),err))}})
 errorsByText:=map[string]int{};success:=map[string]bool{};attempts:=0
 ticker:=time.NewTicker(2*time.Millisecond);defer ticker.Stop()
 end:=time.Now().Add(15*time.Second)
 for time.Now().Before(end){<-ticker.C;attempts++;id:=fmt.Sprint(attempts);callCtx,cancel:=context.WithTimeout(ctx,2*time.Second)
 value,err:=client.Eval(callCtx,`return redis.call('HINCRBY',KEYS[1],ARGV[1],1)`,[]string{key},id);cancel()
 if err!=nil {errorsByText[err.Error()]++;continue};if value!=int64(1){panic(fmt.Sprintf("duplicate execution %s=%v",id,value))};success[id]=true
 }
 wg.Wait()
 ledger,err:=client.Raw().HGetAll(ctx,key).Result();if err!=nil{panic(err)}
 for id,n:=range ledger{if n!="1"{panic("replayed "+id+"="+n)}}
 for id:=range success{if ledger[id]!="1"{panic("successful effect missing "+id)}}
 raw,_:=json.MarshalIndent(map[string]any{"attempted":attempts,"successful":len(success),"effects":len(ledger),"errors":errorsByText,"failovers":failovers,"duplicateEffects":0,"missingSuccessfulEffects":0},"","  ")
 fmt.Println(string(raw))
}
