// 正式生成 DAO → Repository → MigrationRunner → 实际文件 WAL/Projector/MongoStore。
// 后端使用框架 mongotest，不冒认为真实 Mongo；注册只在此独立消费者进程中执行。
const reviewKind entity.EntityKind = 245
const collection = "roost_review_20261004_20"
var registerOnce sync.Once
var stepFailure = errors.New("review20 migration step failed")

type reviewEntity struct { *entity.EntityBase; dao *MigratingDao }
func (e *reviewEntity) Base() *entity.EntityBase { return e.EntityBase }
func (*reviewEntity) OnInitFinish(*entity.EntityCreateParam) error { return nil }
func (*reviewEntity) OnDestroy(entity.EntityDestroyReason) {}
func (*reviewEntity) AutoPersist() bool { return true }
func (*reviewEntity) IsRemoved() bool { return false }
func (*reviewEntity) SetRemoved() {}
func (*reviewEntity) Touch() bool { return true }
func (*reviewEntity) UnTouch() {}
func (*reviewEntity) ClearBase() {}
func (*reviewEntity) IsClear() bool { return false }

func registerConsumer() {
	registerOnce.Do(func() {
		entity.RegisterEntityBuilder(&entity.EntityBuilderParam{
			Category: 1, Kind: reviewKind,
			DaoBuilders: []entity.DaoBuilderFunc{func() entity.DaoInterface { return NewMigratingDao() }},
			Builder: func(p *entity.EntityCreateParam) (entity.IThreadSafeEntity, error) {
				return &reviewEntity{entity.NewEntityBaseWithMutex(p.Id, p.Category, false, p.Mutex, p.Kind), p.Dao[collection].(*MigratingDao)}, nil
			},
		})
		for _, from := range []uint32{1, 2} {
			migration.MustRegisterDAO(migration.DAOStep{Collection: collection, From: from, To: from+1,
				Apply: func(ctx context.Context, raw []byte) ([]byte, error) {
					if err := ctx.Err(); err != nil { return nil, err }
					var doc bson.M
					if err := bson.Unmarshal(raw, &doc); err != nil { return nil, err }
					mode, _ := doc["mode"].(string)
					if mode == "step_failure" { return nil, stepFailure }
					doc["name"], doc["score"], doc["_schema"] = "migrated", int64(42), from+1
					if from == 2 {
						switch mode {
						case "bad_bson": return []byte("invalid BSON"), nil
						case "wrong_id": doc["_id"] = doc["_id"].(int64)+1
						case "bad_typed_payload": doc["score"] = "not an int64"
						}
					}
					return bson.Marshal(doc)
				}})
		}
	})
}

type auditedCommitter struct { *engine.Projector; calls atomic.Int32 }
func (c *auditedCommitter) CommitSystem(ctx context.Context, r coredata.CommitRecord) (coredata.ProjectionTicket, error) {
	c.calls.Add(1)
	return c.Projector.CommitSystem(ctx, r)
}

type gatedProjection struct {
	store *engine.MongoStore
	entered chan struct{}
	release chan struct{}
	once sync.Once
}
func (g *gatedProjection) Project(ctx context.Context, r coredata.CommitRecord) error {
	g.once.Do(func() { close(g.entered) })
	select { case <-g.release: case <-ctx.Done(): return ctx.Err() }
	return g.store.Project(ctx, r)
}

type consumerSetup struct {
	store *engine.MongoStore
	client *mongotest.Client
	manager *entity.EntityManager
	repository *engine.EntityRepository
	committer *auditedCommitter
	id int64
}

func setupConsumer(t *testing.T, mode string, schema uint32, gate *gatedProjection) *consumerSetup {
	t.Helper()
	registerConsumer()
	id, err := entity.BuildEntityID(2020, reviewKind)
	if err != nil { t.Fatal(err) }
	client := mongotest.NewClient()
	if err := client.Collection("review", collection).Seed(bson.M{"_id": id, "_schema": schema, "_version": int64(7), "name": "old", "score": int64(1), "mode": mode}); err != nil { t.Fatal(err) }
	store, err := engine.NewMongoStore(client, engine.MongoStoreConfig{DefaultDatabase: "review"})
	if err != nil { t.Fatal(err) }
	opts := nestwal.DefaultOptions(t.TempDir())
	opts.WriterVersion = nestwal.WriterVersionV2
	wal, err := nestwal.Open(opts)
	if err != nil { t.Fatal(err) }
	var projection engine.ProjectionStore = store
	if gate != nil { gate.store = store; projection = gate }
	projector, err := engine.NewProjector(wal, projection, engine.ProjectorOptions{CloseWAL: true})
	if err != nil { _ = wal.Close(context.Background()); t.Fatal(err) }
	t.Cleanup(func() {
		ctx, cancel := context.WithTimeout(context.Background(), 3*time.Second); defer cancel()
		if err := projector.Close(ctx); err != nil { t.Errorf("owned projector/WAL cleanup: %v", err) }
	})
	committer := &auditedCommitter{Projector: projector}
	runner, err := engine.NewMigrationRunner(committer)
	if err != nil { t.Fatal(err) }
	manager := entity.NewEntityManager()
	repository, err := engine.NewEntityRepository(manager, store, runner, nil)
	if err != nil { t.Fatal(err) }
	return &consumerSetup{store, client, manager, repository, committer, id}
}

func (s *consumerSetup) stored(t *testing.T) coredata.RawDocument {
	t.Helper()
	docs, err := s.store.Load(context.Background(), coredata.LoadSpec{Database: "review", Resource: collection, Filter: map[string]any{"_id": s.id}})
	if err != nil || len(docs) != 1 { t.Fatalf("read stored state: %v docs=%d", err, len(docs)) }
	return docs[0]
}

func TestGeneratedRepositoryMigration(t *testing.T) {
	for _, mode := range []string{"success", "current_schema", "newer_schema", "step_failure", "bad_bson", "wrong_id", "bad_typed_payload"} {
		t.Run(mode, func(t *testing.T) {
			schema := uint32(1)
			if mode == "current_schema" { schema = 3 }; if mode == "newer_schema" { schema = 4 }
			s := setupConsumer(t, mode, schema, nil)
			ctx, cancel := context.WithTimeout(context.Background(), time.Second); defer cancel()
			loaded, err := s.repository.LoadEntity(ctx, s.id, reviewKind)
			doc := s.stored(t)
			stats := s.committer.Stats()
			t.Logf("mode=%s load=%v commits=%d storedSchema=%d version=%d projected=%d unacked=%d failures=%d last=%s", mode, err, s.committer.calls.Load(), doc.Schema, doc.Version, stats.Projected, stats.WALUnacked, stats.ProjectionFailures, stats.LastError)
			if mode == "success" || mode == "current_schema" {
				wantScore, wantVersion, wantCommits := int64(42), uint64(8), int32(1)
				if mode == "current_schema" { wantScore, wantVersion, wantCommits = 1, 7, 0 }
				if err != nil || loaded == nil { t.Fatalf("load: %v", err) }
				dao := loaded.(*reviewEntity).dao
				if dao.GetScore() != wantScore || dao.DirtyTracker().Version() != wantVersion || doc.Schema != 3 || doc.Version != wantVersion || s.committer.calls.Load() != wantCommits { t.Fatalf("state: score=%d version=%d doc=%+v", dao.GetScore(), dao.DirtyTracker().Version(), doc) }
				// 再用新 Manager 冷加载，证明读取的是投影结果而非原对象缓存。
				reloaded, e := engine.NewEntityRepository(entity.NewEntityManager(), s.store, nil, nil)
				if e != nil { t.Fatal(e) }
				value, e := reloaded.LoadEntity(ctx, s.id, reviewKind)
				if e != nil || value.(*reviewEntity).dao.GetScore() != wantScore { t.Fatalf("reload: %v", e) }
				return
			}
			if err == nil || loaded != nil || s.manager.Get(s.id) != nil { t.Fatalf("invalid migration published: %v", err) }
			if mode == "step_failure" && !errors.Is(err, stepFailure) { t.Fatal(err) }
			if mode == "bad_typed_payload" {
				retry, e := engine.NewEntityRepository(entity.NewEntityManager(), s.store, nil, nil)
				if e != nil { t.Fatal(e) }
				_, e = retry.LoadEntity(context.Background(), s.id, reviewKind)
				t.Logf("fresh-manager reload still fails: %v", e)
				if e == nil { t.Fatal("bad persisted type unexpectedly recovered") }
			}
			if s.committer.calls.Load() != 0 || doc.Schema != schema || doc.Version != 7 {
				t.Fatalf("unloadable migration reached CommitSystem: calls=%d stored_schema=%d version=%d load=%v", s.committer.calls.Load(), doc.Schema, doc.Version, err)
			}
		})
	}
}

func TestGeneratedRestoreRejectsBadOutputBeforePublishing(t *testing.T) {
	registerConsumer()
	for _, mode := range []string{"bad_bson", "wrong_id", "bad_typed_payload"} {
		t.Run(mode, func(t *testing.T) {
			raw, _ := bson.Marshal(bson.M{"_id": int64(1), "_schema": uint32(1), "mode": mode})
			dao := NewMigratingDao()
			err := dao.RestorePersisted(raw, 1, 7)
			// ID 语义由 Repository 验，不是生成解码器的错误承诺。
			if mode == "wrong_id" { if err != nil || dao.Id() != 2 { t.Fatal(err) }; return }
			if err == nil || dao.DirtyTracker().Version() != 0 { t.Fatalf("restore accepted bad output: %v", err) }
		})
	}
}

func TestGeneratedMigrationCancellationThenProjectionAndReload(t *testing.T) {
	gate := &gatedProjection{entered: make(chan struct{}), release: make(chan struct{})}
	s := setupConsumer(t, "success", 1, gate)
	var releaseOnce sync.Once
	release := func() { releaseOnce.Do(func() { close(gate.release) }) }
	defer release()
	ctx, cancel := context.WithCancel(context.Background()); defer cancel()
	done := make(chan error, 1)
	go func() { _, err := s.repository.LoadEntity(ctx, s.id, reviewKind); done <- err }()
	select { case <-gate.entered: case <-time.After(3*time.Second): t.Fatal("migration never reached projection gate") }
	if s.manager.Get(s.id) != nil { t.Fatal("published before projection") }
	cancel()
	select { case err := <-done: if !errors.Is(err, context.Canceled) { t.Fatal(err) }; case <-time.After(3*time.Second): t.Fatal("caller cancellation did not return") }
	release()
	retryCtx, retryCancel := context.WithTimeout(context.Background(), 3*time.Second); defer retryCancel()
	if err := s.committer.Flush(retryCtx); err != nil { t.Fatal(err) }
	repository, err := engine.NewEntityRepository(entity.NewEntityManager(), s.store, nil, nil)
	if err != nil { t.Fatal(err) }
	value, err := repository.LoadEntity(retryCtx, s.id, reviewKind)
	if err != nil { t.Fatal(err) }
	if value.(*reviewEntity).dao.GetScore() != 42 || value.(*reviewEntity).dao.DirtyTracker().Version() != 8 || s.committer.calls.Load() != 1 { t.Fatal(fmt.Errorf("late projection/reload wrong")) }
}
