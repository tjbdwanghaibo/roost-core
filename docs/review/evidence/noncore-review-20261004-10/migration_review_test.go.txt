package migration

import (
	"context"
	"errors"
	"testing"
)

type review10Data struct { version int32; writes int }
func (d *review10Data) DataVersion() int32 { return d.version }
func (d *review10Data) SetDataVersion(v int32) { d.version = v }

func TestReview10RegistryCompletedPrefixAndFailure(t *testing.T) {
	registry := NewRegistry()
	cause := errors.New("second step refused")
	registry.MustRegister(Step{From: 0, To: 1, Apply: func(_ context.Context, data any) error { data.(*review10Data).writes++; return nil }})
	registry.MustRegister(Step{From: 1, To: 2, Apply: func(_ context.Context, data any) error { data.(*review10Data).writes++; return cause }})
	data := &review10Data{}
	if err := registry.Run(context.Background(), data, 2); !errors.Is(err, cause) { t.Fatal(err) }
	if data.version != 1 || data.writes != 2 { t.Fatalf("version=%d writes=%d", data.version, data.writes) }
	// Run mutates objects in-place. The failing callback's own mutation is NOT
	// automatically rolled back; application needs a detached value or a tx.
}

func TestReview10RegistryCancellationPreventsNextStep(t *testing.T) {
	registry := NewRegistry()
	ctx, cancel := context.WithCancel(context.Background())
	defer cancel()
	registry.MustRegister(Step{From: 0, To: 1, Apply: func(context.Context, any) error { cancel(); return nil }})
	registry.MustRegister(Step{From: 1, To: 2, Apply: func(context.Context, any) error { t.Fatal("canceled context entered next step"); return nil }})
	data := &review10Data{}
	if err := registry.Run(ctx, data, 2); !errors.Is(err, context.Canceled) || data.version != 1 { t.Fatalf("version=%d err=%v", data.version, err) }
}

func TestReview10DAOMigrationOwnsInputAndOutput(t *testing.T) {
	registry := NewDAORegistry()
	var callbackInput []byte
	registry.MustRegisterDAO(DAOStep{Collection: "player", From: 0, To: 1, Apply: func(_ context.Context, raw []byte) ([]byte, error) {
		callbackInput = raw
		raw[0] = 'b'
		return raw, nil
	}})
	input := []byte("a")
	output, version, err := registry.MigrateDAO(context.Background(), "player", input, 0, 1)
	callbackInput[0] = 'c'
	if err != nil || version != 1 || string(input) != "a" || string(output) != "b" { t.Fatalf("input=%q output=%q version=%d err=%v", input, output, version, err) }
}

func TestReview10DAOMigrationFailureDoesNotPublishPartialBytes(t *testing.T) {
	registry := NewDAORegistry()
	cause := errors.New("step refused")
	registry.MustRegisterDAO(DAOStep{Collection: "player", From: 0, To: 1, Apply: func(_ context.Context, raw []byte) ([]byte, error) { raw[0] = 'b'; return raw, nil }})
	registry.MustRegisterDAO(DAOStep{Collection: "player", From: 1, To: 2, Apply: func(_ context.Context, raw []byte) ([]byte, error) { raw[0] = 'c'; return raw, cause }})
	input := []byte("a")
	output, version, err := registry.MigrateDAO(context.Background(), "player", input, 0, 2)
	if !errors.Is(err, cause) || version != 1 || output != nil || string(input) != "a" { t.Fatalf("input=%q output=%q version=%d err=%v", input, output, version, err) }
}
