func TestReview12MatchHistoricalCost(t *testing.T) {
	for _, backend := range []string{"memory", "redis"} {
		for _, history := range []int{64, 256, 1024, 4096, 8192, 16384} {
			t.Run(fmt.Sprintf("%s_%d", backend, history), func(t *testing.T) {
				ctx := context.Background()
				prefix := fmt.Sprintf("review9-match-%d", time.Now().UnixNano())
				var state versionstore.Store[string, queueState]
				if backend == "memory" {
					state = versionstore.NewMemoryStore[string, queueState]()
				} else {
					addr := os.Getenv("ROOST_REVIEW_REDIS")
					if addr == "" {
						t.Fatal("real standalone Redis required")
					}
					client, err := driver.NewClient(redis.DefaultConfig(addr))
					if err != nil {
						t.Fatal(err)
					}
					defer client.Close()
					defer client.Del(ctx, prefix+":queue:review:2:normal")
					state, err = versionstore.NewRedisStore(client, versionstore.RedisConfig[string, queueState]{Prefix: prefix + ":queue:", KeyOf: func(k string) string { return k }, Codec: versionstore.JSONCodec[queueState]{}})
					if err != nil {
						t.Fatal(err)
					}
				}
				now := time.Unix(1700000000, 0)
				queue := Queue{Mode: "review", GroupSize: 2, Partition: "normal"}
				initial := queueState{}
				initial.init()
				for i := 1; i <= history; i++ {
					id := fmt.Sprintf("old-%d", i)
					request := fmt.Sprintf("request-%d", i)
					initial.Tickets[id] = Ticket{ID: id, Queue: queue, Subject: Subject{Kind: "player", ID: int64(i)}, RequestID: request, State: TicketCancelled, CreatedAtUnix: now.Unix() - 60, ExpiresAtUnix: now.Unix() + 3600, ResolvedAtUnix: now.Unix()}
					initial.Requests[request] = id
				}
				if _, created, err := state.Create(ctx, queue.Key(), initial); err != nil || !created {
					t.Fatalf("seed: %v %v", created, err)
				}
				seq := 0
				store, err := NewStore(state, Config{Now: func() time.Time { return now }, NewID: func() (string, error) { seq++; return fmt.Sprintf("new-%d", seq), nil }})
				if err != nil {
					t.Fatal(err)
				}
				replay, err := store.Enqueue(ctx, queue, Subject{Kind: "player", ID: 1}, "request-1")
				if err != nil || replay.ID != "old-1" || replay.State != TicketCancelled {
					t.Fatalf("old replay: %+v %v", replay, err)
				}
				var before, after runtime.MemStats
				runtime.ReadMemStats(&before)
				started := time.Now()
				for i := 0; i < 16; i++ {
					subject := Subject{Kind: "player", ID: int64(history + i + 1)}
					ticket, err := store.Enqueue(ctx, queue, subject, fmt.Sprintf("new-request-%d", i))
					if err != nil {
						t.Fatal(err)
					}
					if _, err := store.Cancel(ctx, queue, ticket.ID, subject); err != nil {
						t.Fatal(err)
					}
				}
				elapsed := time.Since(started)
				runtime.ReadMemStats(&after)
				got, found, err := state.Get(ctx, queue.Key())
				if err != nil || !found {
					t.Fatal(err)
				}
				raw, err := json.Marshal(got.Value)
				if err != nil {
					t.Fatal(err)
				}
				if len(got.Value.Waiting) != 0 || len(got.Value.Tickets) != history+16 || len(got.Value.Requests) != history+16 {
					t.Fatalf("unexpected state sizes: %+v", got.Value)
				}
				t.Logf("COST backend=%s history=%d live=0 tickets=%d requests=%d json_bytes=%d cycles=16 elapsed_us=%d allocated_bytes=%d", backend, history, len(got.Value.Tickets), len(got.Value.Requests), len(raw), elapsed.Microseconds(), after.TotalAlloc-before.TotalAlloc)
			})
		}
	}
}

// A short, serial hotspot sample at a fixed 8,192-entry history. This is
// process-local latency, not production p99 or a long-running capacity SLO.
func TestReview12MatchHotHistorySoak(t *testing.T) {
	ctx := context.Background()
	addr := os.Getenv("ROOST_REVIEW_REDIS")
	if addr == "" {
		t.Fatal("isolated Redis required")
	}
	client, err := driver.NewClient(redis.DefaultConfig(addr))
	if err != nil {
		t.Fatal(err)
	}
	defer client.Close()
	prefix := fmt.Sprintf("review12-match-hot-%d", time.Now().UnixNano())
	queue := Queue{Mode: "review", GroupSize: 2, Partition: "normal"}
	defer client.Del(ctx, prefix+":queue:"+queue.Key())
	state, err := versionstore.NewRedisStore(client, versionstore.RedisConfig[string, queueState]{Prefix: prefix + ":queue:", KeyOf: func(k string) string { return k }, Codec: versionstore.JSONCodec[queueState]{}})
	if err != nil {
		t.Fatal(err)
	}
	now := time.Unix(1700000000, 0)
	initial := queueState{}
	initial.init()
	const history = 8192
	for i := 1; i <= history; i++ {
		id := fmt.Sprintf("old-%d", i)
		request := fmt.Sprintf("request-%d", i)
		initial.Tickets[id] = Ticket{ID: id, Queue: queue, Subject: Subject{Kind: "player", ID: int64(i)}, RequestID: request, State: TicketCancelled, CreatedAtUnix: now.Unix() - 60, ExpiresAtUnix: now.Unix() + 3600, ResolvedAtUnix: now.Unix()}
		initial.Requests[request] = id
	}
	if _, created, err := state.Create(ctx, queue.Key(), initial); err != nil || !created {
		t.Fatalf("seed created=%v err=%v", created, err)
	}
	seq := 0
	store, err := NewStore(state, Config{Now: func() time.Time { return now }, NewID: func() (string, error) { seq++; return fmt.Sprintf("hot-%d", seq), nil }})
	if err != nil {
		t.Fatal(err)
	}
	var before, after runtime.MemStats
	runtime.ReadMemStats(&before)
	const cycles = 128
	latency := make([]int64, 0, cycles)
	started := time.Now()
	for i := 0; i < cycles; i++ {
		begin := time.Now()
		subject := Subject{Kind: "player", ID: int64(history + i + 1)}
		ticket, err := store.Enqueue(ctx, queue, subject, fmt.Sprintf("hot-request-%d", i))
		if err != nil {
			t.Fatal(err)
		}
		if _, err = store.Cancel(ctx, queue, ticket.ID, subject); err != nil {
			t.Fatal(err)
		}
		latency = append(latency, time.Since(begin).Microseconds())
	}
	elapsed := time.Since(started)
	runtime.ReadMemStats(&after)
	result, found, err := state.Get(ctx, queue.Key())
	if err != nil || !found {
		t.Fatal(err)
	}
	if len(result.Value.Waiting) != 0 || len(result.Value.Tickets) != history+cycles || len(result.Value.Requests) != history+cycles {
		t.Fatalf("final maps: waiting=%d tickets=%d requests=%d", len(result.Value.Waiting), len(result.Value.Tickets), len(result.Value.Requests))
	}
	raw, err := json.Marshal(result.Value)
	if err != nil {
		t.Fatal(err)
	}
	sort.Slice(latency, func(i, j int) bool { return latency[i] < latency[j] })
	t.Logf("SOAK backend=redis history=%d cycles=%d elapsed_us=%d allocated_bytes=%d json_bytes=%d p50_us=%d p95_us=%d p99_us=%d max_us=%d", history, cycles, elapsed.Microseconds(), after.TotalAlloc-before.TotalAlloc, len(raw), latency[cycles/2-1], latency[(95*cycles+99)/100-1], latency[(99*cycles+99)/100-1], latency[cycles-1])
}
