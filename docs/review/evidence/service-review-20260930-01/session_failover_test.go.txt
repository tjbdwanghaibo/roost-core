package session

import (
	"context"
	"errors"
	"os"
	"strings"
	"testing"
	"time"

	redis "github.com/tjbdwanghaibo/roost-core/redis"
	"github.com/tjbdwanghaibo/roost-core/redis/driver"
)

// Run as separate go test processes around an owned Redis master's hard exit.
// The shell runner proves WAIT 1 and verifies all 16,384 slots before verify.
func TestReview12SessionAcrossReplicaFailover(t *testing.T) {
	phase := os.Getenv("ROOST_REVIEW12_HA_PHASE")
	if phase != "seed" && phase != "verify" {
		t.Skip("runner sets seed or verify phase")
	}
	addresses := os.Getenv("ROOST_REVIEW_CLUSTER")
	if addresses == "" {
		t.Fatal("Cluster addresses required")
	}
	client, err := driver.NewClient(review12ClusterConfig(addresses))
	if err != nil {
		t.Fatal(err)
	}
	defer client.Close()
	ctx, cancel := context.WithTimeout(context.Background(), 30*time.Second)
	defer cancel()
	prefix := os.Getenv("ROOST_REVIEW12_HA_PREFIX")
	if prefix == "" {
		prefix = "{review12-ha}"
	}
	runID := os.Getenv("ROOST_REVIEW12_HA_RUN")
	if runID == "" {
		runID = "review12-ha-run"
	}
	stores, err := NewRedisStores(client, RedisConfig{Prefix: prefix, RequestTTL: time.Hour})
	if err != nil {
		t.Fatal(err)
	}
	effects := 0
	service, err := New(Config{Runs: stores.Runs, Claims: stores.Claims, Requests: stores.Requests,
		NewRunID: func() (string, error) { return runID, nil }, TTL: 5 * time.Minute,
		Release: ReleaserFunc(func(context.Context, Run, Resource) error { effects++; return nil })})
	if err != nil {
		t.Fatal(err)
	}
	req := EnterRequest{Kind: "dungeon", RequestID: prefix + ":request"}
	if phase == "seed" {
		run, err := service.Enter(ctx, 99, req)
		if err != nil || run.ID != runID {
			t.Fatalf("seed Enter: run=%q err=%v", run.ID, err)
		}
		run, err = service.Attach(ctx, 99, run.ID, Resource{Kind: "scene", ID: "review12-ha-scene"})
		if err != nil || len(run.Pending()) != 1 {
			t.Fatalf("seed Attach: pending=%d err=%v", len(run.Pending()), err)
		}
		t.Logf("RESULT phase=seed run=%s pending=1 effects=%d", run.ID, effects)
		return
	}
	run, found, err := service.Get(ctx, 99, runID)
	if err != nil || !found || len(run.Pending()) != 1 || (run.State != StateOpen && run.State != StateExpired) {
		t.Fatalf("post-failover state: found=%v state=%q pending=%d err=%v", found, run.State, len(run.Pending()), err)
	}
	if run.State == StateExpired && run.DeadlineUnix > time.Now().Unix() {
		t.Fatalf("run expired ahead of deadline %d", run.DeadlineUnix)
	}
	expected := StateSucceeded
	if run.State == StateExpired {
		expected = StateExpired
	}
	finished, err := service.Finish(ctx, 99, run.ID, StateSucceeded, "done")
	if (expected == StateExpired && !errors.Is(err, ErrRunExpired)) || (expected == StateSucceeded && err != nil) || finished.State != expected || len(finished.Pending()) != 0 || effects != 1 {
		t.Fatalf("post-failover Finish: state=%q pending=%d effects=%d err=%v", finished.State, len(finished.Pending()), effects, err)
	}
	replay, err := service.Enter(ctx, 99, req)
	if err != nil || replay.ID != run.ID {
		t.Fatalf("post-failover request replay: run=%q err=%v", replay.ID, err)
	}
	_, found, err = stores.Claims.Get(ctx, 99)
	if err != nil || found {
		t.Fatalf("claim after Finish: found=%v err=%v", found, err)
	}
	t.Logf("RESULT phase=verify state=%s pending=0 effects=1 claim_present=false replay_same=true", finished.State)
}

func review12ClusterConfig(addresses string) *redis.Config {
	list := strings.Split(addresses, ",")
	cfg := redis.DefaultConfig(list[0])
	cfg.ClusterAddrs = list
	return cfg
}
