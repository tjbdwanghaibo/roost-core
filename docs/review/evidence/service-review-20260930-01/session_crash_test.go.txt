	goredis "github.com/redis/go-redis/v9"

// A child process exits after a durable external effect but before Session
// records release completion. Redis is an isolated test backend, not a real
// allocator. Its SETNX simulates the allocator's required durable idempotency.
func TestReview12SessionCrashAfterExternalEffect(t *testing.T) {
	if os.Getenv("ROOST_REVIEW12_CHILD") == "1" {
		review12CrashChild(t)
		return
	}
	for _, mode := range []string{"standalone", "cluster"} {
		t.Run(mode, func(t *testing.T) {
			ctx, cancel := context.WithTimeout(context.Background(), 20*time.Second)
			defer cancel()
			client, raw := review12Client(t, mode)
			defer client.Close()
			prefix := fmt.Sprintf("{review12-session-%d-%s}", time.Now().UnixNano(), mode)
			stores, err := NewRedisStores(client, RedisConfig{Prefix: prefix, RequestTTL: time.Hour})
			if err != nil {
				t.Fatal(err)
			}
			cfg := Config{Runs: stores.Runs, Claims: stores.Claims, Requests: stores.Requests,
				Release: ReleaserFunc(func(context.Context, Run, Resource) error {
					return errors.New("unexpected parent release before child")
				}), TTL: time.Minute}
			first, err := New(cfg)
			if err != nil {
				t.Fatal(err)
			}
			request := EnterRequest{Kind: "dungeon", RequestID: prefix + ":request"}
			run, err := first.Enter(ctx, 42, request)
			if err != nil {
				t.Fatal(err)
			}
			_, err = first.Attach(ctx, 42, run.ID, Resource{Kind: "scene", ID: "scene-42"})
			if err != nil {
				t.Fatal(err)
			}
			attemptsKey, effectKey := prefix+":release_attempts", prefix+":release_effect"
			child := exec.Command(os.Args[0], "-test.run=^TestReview12SessionCrashAfterExternalEffect$", "-test.v")
			child.Env = append(os.Environ(),
				"ROOST_REVIEW12_CHILD=1", "ROOST_REVIEW12_MODE="+mode,
				"ROOST_REVIEW12_PREFIX="+prefix, "ROOST_REVIEW12_RUN="+run.ID)
			output, err := child.CombinedOutput()
			var exitErr *exec.ExitError
			if !errors.As(err, &exitErr) || exitErr.ExitCode() != 73 {
				t.Fatalf("child did not crash at the target point: exit=%v output=%s", err, output)
			}
			t.Logf("child exit=73 after real Redis effect; child_pid=%d", child.Process.Pid)
			durable, found, err := stores.Runs.Get(ctx, run.ID)
			if err != nil || !found || !durable.Value.State.Terminal() || len(durable.Value.Pending()) != 1 {
				t.Fatalf("post-crash durable run: found=%v state=%q pending=%d err=%v", found, durable.Value.State, len(durable.Value.Pending()), err)
			}
			if got := raw.Get(ctx, effectKey).Val(); got != "released" {
				t.Fatalf("external effect missing: %q", got)
			}
			if attempts := raw.Get(ctx, attemptsKey).Val(); attempts != "1" {
				t.Fatalf("first attempt count=%q", attempts)
			}
			// Construct a genuinely new service instance after the process died.
			cfg.Release = ReleaserFunc(func(ctx context.Context, run Run, resource Resource) error {
				if _, err := raw.Incr(ctx, attemptsKey).Result(); err != nil {
					return err
				}
				_, err := raw.SetNX(ctx, effectKey, "released", time.Hour).Result()
				return err
			})
			recovered, err := New(cfg)
			if err != nil {
				t.Fatal(err)
			}
			finished, err := recovered.Finish(ctx, 42, run.ID, StateSucceeded, "done")
			if err != nil || len(finished.Pending()) != 0 || finished.State != StateSucceeded {
				t.Fatalf("recovery finish: state=%q pending=%d err=%v", finished.State, len(finished.Pending()), err)
			}
			if attempts := raw.Get(ctx, attemptsKey).Val(); attempts != "2" {
				t.Fatalf("expected callback retry; attempts=%q", attempts)
			}
			if got := raw.Get(ctx, effectKey).Val(); got != "released" {
				t.Fatalf("external idempotency failed: %q", got)
			}
			if _, err := recovered.Finish(ctx, 42, run.ID, StateSucceeded, "done"); err != nil {
				t.Fatal(err)
			}
			if attempts := raw.Get(ctx, attemptsKey).Val(); attempts != "2" {
				t.Fatalf("finished run retried effect: attempts=%q", attempts)
			}
			claim, found, err := stores.Claims.Get(ctx, 42)
			if err != nil || found {
				t.Fatalf("claim after recovery: found=%v value=%+v err=%v", found, claim.Value, err)
			}
			replay, err := recovered.Enter(ctx, 42, request)
			if err != nil || replay.ID != run.ID {
				t.Fatalf("request replay changed result: run=%q err=%v", replay.ID, err)
			}
			t.Logf("RESULT mode=%s child_exit=73 pending_after_crash=1 attempts=2 external_effects=1 final_pending=0 claim_present=false replay_same=true", mode)
		})
	}
}

func review12CrashChild(t *testing.T) {
	mode, prefix, runID := os.Getenv("ROOST_REVIEW12_MODE"), os.Getenv("ROOST_REVIEW12_PREFIX"), os.Getenv("ROOST_REVIEW12_RUN")
	if mode == "" || prefix == "" || runID == "" {
		t.Fatal("child identity missing")
	}
	client, raw := review12Client(t, mode)
	stores, err := NewRedisStores(client, RedisConfig{Prefix: prefix, RequestTTL: time.Hour})
	if err != nil {
		t.Fatal(err)
	}
	service, err := New(Config{Runs: stores.Runs, Claims: stores.Claims, Requests: stores.Requests,
		Release: ReleaserFunc(func(ctx context.Context, run Run, resource Resource) error {
			if _, err := raw.Incr(ctx, prefix+":release_attempts").Result(); err != nil {
				return err
			}
			first, err := raw.SetNX(ctx, prefix+":release_effect", "released", time.Hour).Result()
			if err != nil || !first {
				return fmt.Errorf("first external effect: applied=%v err=%w", first, err)
			}
			os.Exit(73)
			return nil
		}), TTL: time.Minute})
	if err != nil {
		t.Fatal(err)
	}
	if _, err := service.Finish(context.Background(), 42, runID, StateSucceeded, "done"); err != nil {
		t.Fatal(err)
	}
	t.Fatal("Finish returned; crash callback was not reached")
}

func review12Client(t *testing.T, mode string) (redis.IRedis, goredis.UniversalClient) {
	t.Helper()
	addr := os.Getenv("ROOST_REVIEW_REDIS")
	if mode == "cluster" {
		addr = os.Getenv("ROOST_REVIEW_CLUSTER")
	}
	if addr == "" {
		t.Fatal("isolated Redis address not set")
	}
	cfg := redis.DefaultConfig(strings.Split(addr, ",")[0])
	if mode == "cluster" {
		cfg.ClusterAddrs = strings.Split(addr, ",")
	}
	client, err := driver.NewClient(cfg)
	if err != nil {
		t.Fatal(err)
	}
	return client, client.(*driver.Client).Raw()
}
