// Overlay only: assert intended behavior without changing production tests.
func TestReviewRR14DependencyConsolidationCommitsRelatedFiles(t *testing.T) {
	root := t.TempDir()
	manifest := DefaultManifest("planet", "example.com/planet", nil, nil, nil)
	raw, err := manifest.Marshal()
	if err != nil { t.Fatal(err) }
	writeProjectFile(t, root, ManifestName, string(raw))
	writeProjectFile(t, root, "go.mod", legacyGoMod)
	path := writeProjectFile(t, root, "business.go", "package planet\nimport \"github.com/tjbdwanghaibo/roost-kit/mods\"\nvar _ = mods.ModBus\n")
	var log bytes.Buffer
	var calls []string
	runner := func(_ context.Context, stage string, _, _ io.Writer, args ...string) error {
		calls = append(calls, strings.Join(args, " "))
		staged, err := os.ReadFile(filepath.Join(stage, "business.go"))
		if err != nil { return err }
		if !bytes.Contains(staged, []byte("roost-core/kit/mods")) { t.Fatal("resolver did not see the migrated stage") }
		if args[0] == "get" {
			return os.WriteFile(filepath.Join(stage, "go.mod"), []byte("module example.com/planet\n\ngo 1.27.0\n\nrequire github.com/tjbdwanghaibo/roost-core v1.18.0\n"), 0o644)
		}
		return nil
	}
	if err := updateFrameworkDependenciesTransactional(root, manifest, &log, &log, runner); err != nil { t.Fatal(err) }
	body, _ := os.ReadFile(path)
	actual, _ := loadManifestForUpgrade(root)
	mod, _ := os.ReadFile(filepath.Join(root, "go.mod"))
	t.Logf("resolver=%v\nlog=%s\ncommitted go.mod=%s\nsource=%s\nlegacy policies=%v", calls, log.String(), mod, body, actual.Versions.legacyModulePolicies())
	if strings.Contains(string(body), "roost-kit/mods") { t.Error("RR14: reported success but root still imports the removed roost-kit module") }
	if actual.Versions.legacyModulePolicies() { t.Error("RR14: consolidation dropped legacy policies only in the discarded stage") }
}

func TestReviewDependencyConsolidationFailurePreservesRoot(t *testing.T) {
	root := t.TempDir()
	manifest := DefaultManifest("planet", "example.com/planet", nil, nil, nil)
	raw, err := manifest.Marshal()
	if err != nil { t.Fatal(err) }
	writeProjectFile(t, root, ManifestName, string(raw))
	writeProjectFile(t, root, "go.mod", legacyGoMod)
	source := "package planet\nimport \"github.com/tjbdwanghaibo/roost-kit/mods\"\nvar _ = mods.ModBus\n"
	writeProjectFile(t, root, "business.go", source)
	err = updateFrameworkDependenciesTransactional(root, manifest, io.Discard, io.Discard, func(context.Context, string, io.Writer, io.Writer, ...string) error { return errors.New("review resolver failure") })
	if err == nil || !strings.Contains(err.Error(), "review resolver failure") { t.Fatalf("unexpected error: %v", err) }
	for rel, want := range map[string]string{ "business.go": source, "go.mod": legacyGoMod, ManifestName: string(raw) } {
		got, err := os.ReadFile(filepath.Join(root, rel))
		if err != nil || string(got) != want { t.Fatalf("failure changed root %s: %v\n%s", rel, err, got) }
	}
}
