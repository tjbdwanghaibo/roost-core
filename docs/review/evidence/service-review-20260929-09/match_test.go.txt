package match

import (
 "context"
 "encoding/json"
 "fmt"
 "os"
 "runtime"
 "testing"
 "time"
 redis "github.com/tjbdwanghaibo/roost-core/redis"
 "github.com/tjbdwanghaibo/roost-core/redis/driver"
 "github.com/tjbdwanghaibo/roost-core/versionstore"
)

func TestReview9MatchHistoricalCost(t *testing.T) {
 for _, backend := range []string{"memory","redis"} {
  for _, history := range []int{64,256,1024,4096} {
   t.Run(fmt.Sprintf("%s_%d",backend,history),func(t *testing.T) {
    ctx := context.Background(); prefix := fmt.Sprintf("review9-match-%d",time.Now().UnixNano())
    var state versionstore.Store[string,queueState]
    if backend=="memory" { state=versionstore.NewMemoryStore[string,queueState]() } else {
     addr:=os.Getenv("ROOST_REVIEW_REDIS"); if addr=="" { t.Fatal("real standalone Redis required") }
     client,err:=driver.NewClient(redis.DefaultConfig(addr)); if err!=nil { t.Fatal(err) }; defer client.Close()
     defer client.Del(ctx,prefix+":queue:review:2:normal")
     state,err=versionstore.NewRedisStore(client,versionstore.RedisConfig[string,queueState]{Prefix:prefix+":queue:",KeyOf:func(k string)string{return k},Codec:versionstore.JSONCodec[queueState]{}}); if err!=nil { t.Fatal(err) }
    }
    now:=time.Unix(1700000000,0); queue:=Queue{Mode:"review",GroupSize:2,Partition:"normal"}
    initial:=queueState{}; initial.init()
    for i:=1;i<=history;i++ { id:=fmt.Sprintf("old-%d",i); request:=fmt.Sprintf("request-%d",i); initial.Tickets[id]=Ticket{ID:id,Queue:queue,Subject:Subject{Kind:"player",ID:int64(i)},RequestID:request,State:TicketCancelled,CreatedAtUnix:now.Unix()-60,ExpiresAtUnix:now.Unix()+3600,ResolvedAtUnix:now.Unix()}; initial.Requests[request]=id }
    if _,created,err:=state.Create(ctx,queue.Key(),initial); err!=nil || !created { t.Fatalf("seed: %v %v",created,err) }
    seq:=0; store,err:=NewStore(state,Config{Now:func()time.Time{return now},NewID:func()(string,error){seq++;return fmt.Sprintf("new-%d",seq),nil}}); if err!=nil { t.Fatal(err) }
    replay,err:=store.Enqueue(ctx,queue,Subject{Kind:"player",ID:1},"request-1"); if err!=nil || replay.ID!="old-1" || replay.State!=TicketCancelled { t.Fatalf("old replay: %+v %v",replay,err) }
    var before,after runtime.MemStats; runtime.ReadMemStats(&before); started:=time.Now()
    for i:=0;i<16;i++ { subject:=Subject{Kind:"player",ID:int64(history+i+1)}; ticket,err:=store.Enqueue(ctx,queue,subject,fmt.Sprintf("new-request-%d",i)); if err!=nil { t.Fatal(err) }; if _,err:=store.Cancel(ctx,queue,ticket.ID,subject);err!=nil { t.Fatal(err) } }
    elapsed:=time.Since(started); runtime.ReadMemStats(&after)
    got,found,err:=state.Get(ctx,queue.Key()); if err!=nil || !found { t.Fatal(err) }; raw,err:=json.Marshal(got.Value); if err!=nil { t.Fatal(err) }
    if len(got.Value.Waiting)!=0 || len(got.Value.Tickets)!=history+16 || len(got.Value.Requests)!=history+16 { t.Fatalf("unexpected state sizes: %+v",got.Value) }
    t.Logf("COST backend=%s history=%d live=0 tickets=%d requests=%d json_bytes=%d cycles=16 elapsed_us=%d allocated_bytes=%d",backend,history,len(got.Value.Tickets),len(got.Value.Requests),len(raw),elapsed.Microseconds(),after.TotalAlloc-before.TotalAlloc)
   })
  }
 }
}
