func TestReview9MailClusterPage(t *testing.T) {
 for _, tagged := range []bool{false, true} {
  t.Run(fmt.Sprintf("tagged_%v", tagged), func(t *testing.T) {
   address := os.Getenv("ROOST_REVIEW_CLUSTER"); if address == "" { t.Fatal("real cluster is required") }
   prefix := fmt.Sprintf("review9-mail-%d", time.Now().UnixNano()); if tagged { prefix = "{"+prefix+"}" }
   cfg := viper.New(); cfg.Set("redis.cluster_addrs", address); cfg.Set("mail.key_prefix", prefix); cfg.Set("mail.send_ttl", time.Hour)
   mod := NewMod(nil,nil); if err := mod.Init(cfg); err != nil { t.Fatal(err) }
   rcfg := redis.DefaultConfig(""); rcfg.ClusterAddrs = strings.Split(address,",")
   client, err := driver.NewClient(rcfg); if err != nil { t.Fatal(err) }; defer client.Close()
   registry := app.NewRegistry(cfg); if err := mods.RegisterAll(registry,mods.Capability{Name:mods.ModRedis, Value:client}); err != nil { t.Fatal(err) }; if err := mod.Provide(registry); err != nil { t.Fatal(err) }
   // Mod admission is verified above. Rebuild the same public stores/service
   // with deterministic ids because the Core Service config is private.
   stores,err:=NewRedisStores(client,RedisConfig{Prefix:prefix,SendTTL:time.Hour}); if err!=nil { t.Fatal(err) }
   sequence:=0; service,err:=New(Config{Envelopes:stores.Envelopes,Mailboxes:stores.Mailboxes,Sends:stores.Sends,NewMailID:func()(string,error){sequence++;return fmt.Sprintf("review9-mail-%d",sequence),nil}}); if err!=nil{t.Fatal(err)}; mod.service=service
   ctx, cancel := context.WithTimeout(context.Background(),10*time.Second); defer cancel()
   mails := []Envelope{}
   for i:=0;i<2;i++ { env,err := mod.service.Send(ctx,SendRequest{Audience:AudienceDirect,Recipients:[]int64{1},Subject:"review",ExpiresInSeconds:3600,RequestID:fmt.Sprintf("request-%d",i)}); if err != nil { t.Fatal(err) }; mails=append(mails,env) }
   defer func() { for _, env := range mails { _,_ = client.Del(context.Background(),prefix+":env:"+env.ID); _,_ = client.Del(context.Background(),prefix+":send:"+env.SendRequestID) }; _,_=client.Del(context.Background(),prefix+":box:1") }()
   page, err := mod.service.List(ctx,1,"",1); if err != nil || len(page.Items)!=1 { t.Fatalf("single: %+v %v",page,err) }
   page, err = mod.service.List(ctx,1,"",2)
   t.Logf("tagged=%v single_count=1 batch_count=%d batch_error=%v mail_ids=%s,%s",tagged,len(page.Items),err,mails[0].ID,mails[1].ID)
   if err != nil || len(page.Items)!=2 { t.Fatalf("valid two-mail page unavailable: %+v %v",page,err) }
  })
 }
}

// A candidate implementation using the framework's existing Pipeline API;
// this is evidence for the handoff, not a replacement of production MGet.
func TestReview9MailClusterPipelineCandidate(t *testing.T) {
 address:=os.Getenv("ROOST_REVIEW_CLUSTER"); if address=="" { t.Fatal("real cluster is required") }
 cfg:=redis.DefaultConfig(""); cfg.ClusterAddrs=strings.Split(address,",")
 client,err:=driver.NewClient(cfg); if err!=nil { t.Fatal(err) }; defer client.Close()
 ctx,cancel:=context.WithTimeout(context.Background(),10*time.Second); defer cancel()
 prefix:=fmt.Sprintf("review9-pipeline-%d",time.Now().UnixNano())
 keys:=[]string{prefix+":1",prefix+":2",prefix+":3",prefix+":4"}
 defer func(){for _,key:=range keys{_,_=client.Del(context.Background(),key)}}()
 if err:=client.Set(ctx,keys[0],[]byte{0,255},time.Minute);err!=nil{t.Fatal(err)}
 if err:=client.Set(ctx,keys[1],"",time.Minute);err!=nil{t.Fatal(err)}
 pipe:=client.Pipeline(); futures:=[]*redis.FutureBytes{}
 for _,key:=range keys{futures=append(futures,pipe.Get(ctx,key))}
 if err:=pipe.Exec(ctx);err!=nil{t.Fatal(err)}
 raw,err:=futures[0].Result();if err!=nil || !bytes.Equal(raw,[]byte{0,255}){t.Fatalf("binary: %v %v",raw,err)}
 raw,err=futures[1].Result();if err!=nil || raw==nil || len(raw)!=0{t.Fatalf("empty is not absence: %v %v",raw,err)}
 for _,future:=range futures[2:]{if _,err:=future.Result();!errors.Is(err,redis.ErrNil){t.Fatalf("absent: %v",err)}}
 if err:=client.HSet(ctx,keys[3],"field","value");err!=nil{t.Fatal(err)}
 pipe=client.Pipeline(); good:=pipe.Get(ctx,keys[0]); bad:=pipe.Get(ctx,keys[3])
 if err:=pipe.Exec(ctx);err==nil || !strings.Contains(err.Error(),"WRONGTYPE"){t.Fatalf("wrong type hidden: %v",err)}
 if _,err:=bad.Result();err==nil{t.Fatal("bad future falsely succeeded")}
 if raw,err:=good.Result();err!=nil || !bytes.Equal(raw,[]byte{0,255}){t.Fatalf("good future lost: %v %v",raw,err)}
 t.Log("candidate: cross-slot pipeline positional reads preserve binary/empty/absent; backend errors remain errors")
}
