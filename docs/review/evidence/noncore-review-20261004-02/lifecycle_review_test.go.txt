package lifecycle

import (
	"context"
	"errors"
	"testing"
)

func TestReview2HookSnapshotAndCleanup(t *testing.T) {
	registry := NewRegistry()
	var journal []string
	if err := registry.Register(Hook{Name: "first", Phase: PhaseServiceStopping, Handler: func(context.Context, Event) error {
		journal = append(journal, "first")
		return registry.Register(Hook{Name: "second", Phase: PhaseServiceStopping, Handler: func(context.Context, Event) error { journal = append(journal, "replacement"); return nil }})
	}}); err != nil {
		t.Fatal(err)
	}
	failure := errors.New("second failed")
	if err := registry.Register(Hook{Name: "second", Phase: PhaseServiceStopping, Handler: func(context.Context, Event) error { journal = append(journal, "old"); return failure }}); err != nil {
		t.Fatal(err)
	}
	if err := registry.EmitAll(context.Background(), Event{Phase: PhaseServiceStopping}); !errors.Is(err, failure) {
		t.Fatal(err)
	}
	if len(journal) != 2 || journal[1] != "old" {
		t.Fatal(journal)
	}
	if err := registry.EmitAll(context.Background(), Event{Phase: PhaseServiceStopping}); err != nil {
		t.Fatal(err)
	}
	if len(journal) != 4 || journal[3] != "replacement" {
		t.Fatal(journal)
	}
}

func TestReview2HookPanicDoesNotSkipCleanup(t *testing.T) {
	registry := NewRegistry()
	called := false
	_ = registry.Register(Hook{Name: "panic", Phase: PhaseServiceStopping, Handler: func(context.Context, Event) error { panic("boom") }})
	_ = registry.Register(Hook{Name: "cleanup", Phase: PhaseServiceStopping, Handler: func(context.Context, Event) error { called = true; return nil }})
	if err := registry.EmitAll(nil, Event{Phase: PhaseServiceStopping}); err == nil || !called {
		t.Fatalf("error=%v cleanup=%v", err, called)
	}
}
