package match

import (
	"context"
	"errors"
	"fmt"
	"testing"
)

func TestReview5GroupingRejectsInvalidQueues(t *testing.T) {
	for policyName, policy := range map[string]Grouping{"fifo": FirstComeGrouping{}, "score": ScoreWindowGrouping{NowUnix: func() int64 { return 1700000010 }}} {
		for _, size := range []int{-1, 0, 1, MaxGroupSize + 1} {
			t.Run(fmt.Sprintf("%s/size_%d", policyName, size), func(t *testing.T) {
				defer func() {
					if r := recover(); r != nil {
						t.Errorf("invalid Queue caused panic instead of ErrQueueInvalid: %v", r)
					}
				}()
				_, formed, err := policy.Group(Queue{Mode: "duel", GroupSize: size}, []Ticket{{Subject: player(1, 100)}, {Subject: player(2, 100)}})
				if !errors.Is(err, ErrQueueInvalid) || formed {
					t.Errorf("invalid size %d: formed=%v err=%v", size, formed, err)
				}
			})
		}
	}
}
func TestReview5GroupingValidControl(t *testing.T) {
	for _, policy := range []Grouping{FirstComeGrouping{}, ScoreWindowGrouping{NowUnix: func() int64 { return 1700000010 }}} {
		group, ok, err := policy.Group(ranked(), []Ticket{{ID: "a", Subject: player(1, 100)}, {ID: "b", Subject: player(2, 100)}})
		if err != nil || !ok || len(group) != 2 {
			t.Fatal(group, ok, err)
		}
	}
}
func TestReview5MatchChurnRetainsHistoryObservation(t *testing.T) {
	s, _ := newStore(t)
	ctx := context.Background()
	for i := 0; i < 100; i++ {
		x := enqueue(t, s, ranked(), player(1, 100), fmt.Sprint(i))
		if _, err := s.Cancel(ctx, ranked(), x.ID, player(1, 100)); err != nil {
			t.Fatal(err)
		}
	}
	stored, _, err := s.(*queueStore).state.Get(ctx, ranked().Key())
	if err != nil {
		t.Fatal(err)
	}
	if len(stored.Value.Waiting) != 0 || len(stored.Value.Tickets) != 100 || len(stored.Value.Requests) != 100 {
		t.Fatal(stored.Value)
	}
	t.Log("output depth=0; retained tickets=100 requests=100: limit does not bound history")
}
