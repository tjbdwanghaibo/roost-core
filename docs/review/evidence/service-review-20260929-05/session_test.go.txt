func TestReview5RotatingOwnerSourceClearsEveryPage(t *testing.T) {
	next := int64(1)
	h := newHarness(t, func(c *Config) {
		c.Owners = OwnerSourceFunc(func(_ context.Context, limit int) ([]int64, error) {
			if limit != 1 {
				t.Fatalf("limit %d", limit)
			}
			id := next
			next++
			return []int64{id}, nil
		})
	})
	ctx := context.Background()
	for i := int64(1); i <= 3; i++ {
		run := mustEnter(t, h, i, fmt.Sprint(i))
		if _, err := h.service.Attach(ctx, i, run.ID, scene(fmt.Sprint(i))); err != nil {
			t.Fatal(err)
		}
	}
	h.clock.advance(11 * time.Minute)
	for i := 0; i < 3; i++ {
		resolved, err := h.service.SweepPending(ctx, 1)
		if err != nil || len(resolved) != 1 {
			t.Fatal(resolved, err)
		}
	}
	for i := int64(1); i <= 3; i++ {
		if _, found, err := h.claims.Get(ctx, i); err != nil || found {
			t.Fatal(i, found, err)
		}
	}
	if h.releaser.total() != 3 {
		t.Fatal(h.releaser.total())
	}
}
func TestReview5ConcurrentFinishNeedsResourceSideIdempotency(t *testing.T) {
	var mu sync.Mutex
	calls, effects := 0, 0
	freed := false
	entered := make(chan struct{}, 2)
	release := make(chan struct{})
	h := newHarness(t, func(c *Config) {
		c.Release = ReleaserFunc(func(_ context.Context, r Run, resource Resource) error {
			entered <- struct{}{}
			<-release
			mu.Lock()
			defer mu.Unlock()
			calls++
			if !freed {
				freed = true
				effects++
			}
			return nil
		})
	})
	ctx := context.Background()
	run := mustEnter(t, h, 1, "finish")
	if _, err := h.service.Attach(ctx, 1, run.ID, scene("allocation-1")); err != nil {
		t.Fatal(err)
	}
	done := make(chan error, 2)
	for i := 0; i < 2; i++ {
		go func() { _, err := h.service.Finish(ctx, 1, run.ID, StateSucceeded, "won"); done <- err }()
	}
	<-entered
	<-entered
	close(release)
	for i := 0; i < 2; i++ {
		if err := <-done; err != nil {
			t.Fatal(err)
		}
	}
	if effects != 1 || calls != 2 {
		t.Fatalf("calls=%d effects=%d", calls, effects)
	}
	stored, _, err := h.service.Get(ctx, 1, run.ID)
	if err != nil || len(stored.Pending()) != 0 {
		t.Fatal(stored, err)
	}
	t.Log("two concurrent release calls, one fixture effect; not a production allocator validation")
}
