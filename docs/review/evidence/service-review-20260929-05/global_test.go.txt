package global

import (
	"context"
	"errors"
	"github.com/tjbdwanghaibo/roost-core/versionstore"
	"sync/atomic"
	"testing"
)

type review5LeaseGate struct {
	versionstore.Store[int32, GameLease]
	once             atomic.Bool
	entered, release chan struct{}
}

func (g *review5LeaseGate) Update(ctx context.Context, k int32, m versionstore.Mutate[GameLease]) (versionstore.Versioned[GameLease], bool, error) {
	if g.once.CompareAndSwap(false, true) {
		close(g.entered)
		<-g.release
	}
	return g.Store.Update(ctx, k, m)
}
func TestReview5AcquireMigrationRaceCannotDisplaceNewLease(t *testing.T) {
	gate := &review5LeaseGate{Store: versionstore.NewMemoryStore[int32, GameLease](), entered: make(chan struct{}), release: make(chan struct{})}
	s, _ := newService(t, func(c *Config) { c.Leases = gate })
	ctx := context.Background()
	route := bind(t, s, 100, 1)
	done := make(chan error, 1)
	go func() { _, err := s.AcquireLease(ctx, 100); done <- err }()
	<-gate.entered
	released := false
	defer func() {
		if !released {
			close(gate.release)
			<-done
		}
	}()
	moving, err := s.BeginMigration(ctx, 100, 2, route.Epoch)
	if err != nil {
		t.Fatal(err)
	}
	if _, err := s.CompleteMigration(ctx, 100, moving.Epoch); err != nil {
		t.Fatal(err)
	}
	newLease, err := s.AcquireLease(ctx, 100)
	if err != nil {
		t.Fatal(err)
	}
	close(gate.release)
	released = true
	if err := <-done; !errors.Is(err, ErrConflict) {
		t.Fatal("old route acquisition was not refused", err)
	}
	current, found, err := s.Lease(ctx, 100)
	if err != nil || !found || current.Incarnation != newLease.Incarnation || current.GlobalSID != 2 {
		t.Fatal(current, found, err)
	}
}
