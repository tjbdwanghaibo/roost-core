package activity

import (
	"context"
	"fmt"
	fredis "github.com/tjbdwanghaibo/roost-core/redis"
	"github.com/tjbdwanghaibo/roost-core/redis/driver"
	"github.com/tjbdwanghaibo/roost-core/versionstore"
	"os"
	"testing"
	"time"
)

type review5LateCreate struct {
	versionstore.Store[Key, Activity]
	entered, release chan struct{}
	blocked          Key
}

func (s *review5LateCreate) Create(ctx context.Context, key Key, value Activity) (versionstore.Versioned[Activity], bool, error) {
	if key == s.blocked {
		close(s.entered)
		<-s.release
	}
	return s.Store.Create(ctx, key, value)
}
func review5Activity(t *testing.T, mutate func(*Config)) (*Service, *activityClock) {
	t.Helper()
	return newActivityService(t, func(c *Config) {
		if os.Getenv("ROOST_REVIEW5_BACKEND") == "redis" {
			addr := os.Getenv("ROOST_REVIEW_REDIS")
			if addr == "" {
				t.Fatal("isolated Redis address required")
			}
			client, err := driver.NewClient(fredis.DefaultConfig(addr))
			if err != nil {
				t.Fatal(err)
			}
			t.Cleanup(func() { _ = client.Close() })
			stores, err := NewRedisStores(client, fmt.Sprintf("review5:activity:%d", time.Now().UnixNano()), time.Hour)
			if err != nil {
				t.Fatal(err)
			}
			c.Activities = stores.Activities
			c.Participants = stores.Participants
			c.Ledger = stores.Ledger
			c.Audits = stores.Audits
			c.Dispatches = stores.Dispatches
			c.Windows = stores.Windows
		}
		if mutate != nil {
			mutate(c)
		}
	})
}
func TestReview5LateOpenCannotExceedWindowBound(t *testing.T) {
	key := activityKey("zz-late")
	var delayed *review5LateCreate
	s, clock := review5Activity(t, func(c *Config) {
		delayed = &review5LateCreate{Store: c.Activities, entered: make(chan struct{}), release: make(chan struct{}), blocked: key}
		c.Activities = delayed
	})
	ctx := context.Background()
	done := make(chan error, 1)
	go func() { _, err := s.OpenActivity(ctx, key, []int32{1, 2}); done <- err }()
	<-delayed.entered
	released := false
	defer func() {
		if !released {
			close(delayed.release)
			<-done
		}
	}()
	clock.advance(s.cfg.OpeningGrace + time.Second)
	if _, err := s.AdvanceExpired(ctx, key.GroupID, 1); err != nil {
		t.Fatal(err)
	}
	for i := 0; i < MaxPendingActivities; i++ {
		openActivity(t, s, activityKey(fmt.Sprintf("new-%d", i)), 1)
	}
	close(delayed.release)
	released = true
	err := <-done
	if err != nil {
		t.Logf("late open refusal=%v", err)
	}
	box, found, err := s.cfg.Windows.Get(ctx, key.GroupID)
	if err != nil || !found {
		t.Fatal(err, found)
	}
	if box.Value.pending() > MaxPendingActivities {
		t.Errorf("expired admission returned and exceeded hard bound: pending=%d limit=%d", box.Value.pending(), MaxPendingActivities)
	}
	if _, err := s.NotifyPhase(ctx, key, 1); err != nil {
		t.Fatal(err)
	}
	clock.advance(s.cfg.GraceWindow + time.Second)
	for i := 0; i < 2; i++ {
		if _, err := s.AdvanceExpired(ctx, key.GroupID, 1); err != nil {
			t.Fatal(err)
		}
	}
	late, found, err := s.LookupActivity(ctx, key)
	if err != nil || !found {
		t.Fatal(err, found)
	}
	if late.Status != StatusComplete {
		t.Errorf("expired late aggregation remains %s: scan truncates its key behind %d never-notified activities", late.Status, MaxPendingActivities)
	}
}
func TestReview5UnreclaimedOpeningControl(t *testing.T) {
	s, clock := review5Activity(t, nil)
	ctx := context.Background()
	key := activityKey("control")
	openActivity(t, s, key, 1, 2)
	if _, err := s.NotifyPhase(ctx, key, 1); err != nil {
		t.Fatal(err)
	}
	clock.advance(s.cfg.GraceWindow + time.Second)
	got, err := s.AdvanceExpired(ctx, key.GroupID, 1)
	if err != nil || len(got) != 1 || got[0].Status != StatusComplete {
		t.Fatal(got, err)
	}
}
