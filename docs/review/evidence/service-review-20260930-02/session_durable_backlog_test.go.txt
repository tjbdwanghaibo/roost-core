type review13FlakyClaimStore struct {
	ClaimStore
	failed bool
}

func (s *review13FlakyClaimStore) Get(ctx context.Context, owner int64) (versionstore.Versioned[Claim], bool, error) {
	if owner == 1 && !s.failed {
		s.failed = true
		return versionstore.Versioned[Claim]{}, false, errors.New("one transient claim read failure")
	}
	return s.ClaimStore.Get(ctx, owner)
}

// The cursor is deliberately owned by a test-side Redis OwnerSource. The
// exercise verifies Core SweepPending can resume from an externally persisted
// roster position, not that Core discovers owners on its own.
func TestReview13SessionBacklogCursorAcrossProcessRestart(t *testing.T) {
	phase := os.Getenv("ROOST_REVIEW13_BACKLOG_PHASE")
	if phase != "seed" && phase != "first" && phase != "resume" {
		t.Skip("runner sets backlog phase")
	}
	prefix := os.Getenv("ROOST_REVIEW13_BACKLOG_PREFIX")
	addr := os.Getenv("ROOST_REVIEW_REDIS")
	if prefix == "" || addr == "" {
		t.Fatal("isolated Redis and unique prefix required")
	}
	client, err := driver.NewClient(redis.DefaultConfig(addr))
	if err != nil {
		t.Fatal(err)
	}
	defer client.Close()
	raw := client.(*driver.Client).Raw()
	stores, err := NewRedisStores(client, RedisConfig{Prefix: prefix, RequestTTL: time.Hour})
	if err != nil {
		t.Fatal(err)
	}
	clock := time.Unix(1800000000, 0)
	if phase != "seed" {
		clock = clock.Add(2 * time.Minute)
	}
	seq := 0
	cursorKey, effectsKey := prefix+":owner_cursor", prefix+":release_effects"
	ownerSource := OwnerSourceFunc(func(ctx context.Context, limit int) ([]int64, error) {
		position, err := raw.Get(ctx, cursorKey).Int()
		if err == goredis.Nil {
			position = 0
		} else if err != nil {
			return nil, err
		}
		if position >= 250 {
			position = 0
		}
		end := position + limit
		if end > 250 {
			end = 250
		}
		owners := make([]int64, 0, end-position)
		for id := position + 1; id <= end; id++ {
			owners = append(owners, int64(id))
		}
		if err := raw.Set(ctx, cursorKey, strconv.Itoa(end), 0).Err(); err != nil {
			return nil, err
		}
		return owners, nil
	})
	claims := ClaimStore(stores.Claims)
	if phase == "first" {
		claims = &review13FlakyClaimStore{ClaimStore: claims}
	}
	service, err := New(Config{Runs: stores.Runs, Claims: claims, Requests: stores.Requests,
		TTL: time.Minute, Now: func() time.Time { return clock },
		NewRunID: func() (string, error) { seq++; return fmt.Sprintf("run-%d", seq), nil },
		Owners:   ownerSource,
		Release: ReleaserFunc(func(ctx context.Context, _ Run, resource Resource) error {
			return raw.HIncrBy(ctx, effectsKey, resource.ID, 1).Err()
		}),
	})
	if err != nil {
		t.Fatal(err)
	}
	ctx, cancel := context.WithTimeout(context.Background(), 90*time.Second)
	defer cancel()
	if phase == "seed" {
		for owner := int64(1); owner <= 250; owner++ {
			run, err := service.Enter(ctx, owner, EnterRequest{Kind: "dungeon", RequestID: fmt.Sprintf("%s:request-%d", prefix, owner)})
			if err != nil || run.ID != fmt.Sprintf("run-%d", owner) {
				t.Fatalf("seed owner %d Enter: %+v %v", owner, run, err)
			}
			if _, err := service.Attach(ctx, owner, run.ID, Resource{Kind: "scene", ID: fmt.Sprintf("scene-%d", owner)}); err != nil {
				t.Fatal(err)
			}
		}
		t.Log("RESULT phase=seed owners=250 pending=250")
		return
	}
	if phase == "first" {
		resolved, err := service.SweepPending(ctx, 100)
		if err == nil || !strings.Contains(err.Error(), "one transient claim read failure") || len(resolved) != 99 {
			t.Fatalf("first page: resolved=%d err=%v", len(resolved), err)
		}
		t.Log("RESULT phase=first cursor=100 resolved=99 failed_owner=1")
		return
	}
	for _, want := range []int{100, 50, 1} {
		resolved, err := service.SweepPending(ctx, 100)
		if err != nil || len(resolved) != want {
			t.Fatalf("resume page want=%d got=%d err=%v", want, len(resolved), err)
		}
	}
	for owner := int64(1); owner <= 250; owner++ {
		run, found, err := stores.Runs.Get(ctx, fmt.Sprintf("run-%d", owner))
		if err != nil || !found || run.Value.State != StateExpired || len(run.Value.Pending()) != 0 {
			t.Fatalf("owner %d final run: found=%v state=%q pending=%d err=%v", owner, found, run.Value.State, len(run.Value.Pending()), err)
		}
		_, found, err = stores.Claims.Get(ctx, owner)
		if err != nil || found {
			t.Fatalf("owner %d claim remains=%v err=%v", owner, found, err)
		}
		count, err := raw.HGet(ctx, effectsKey, fmt.Sprintf("scene-%d", owner)).Int()
		if err != nil || count != 1 {
			t.Fatalf("owner %d release effects=%d err=%v", owner, count, err)
		}
	}
	t.Log("RESULT phase=resume pages=100,50,1 final_pending=0 effects_once=250")
}
