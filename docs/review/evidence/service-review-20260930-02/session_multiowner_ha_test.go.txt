package session

import (
	"context"
	"fmt"
	"os"
	"strings"
	"testing"
	"time"

	redis "github.com/tjbdwanghaibo/roost-core/redis"
	"github.com/tjbdwanghaibo/roost-core/redis/driver"
)

// Seed and verify are different go test processes. The runner kills the master
// for owner 1 without WAIT; each owner has its own Redis Cluster hash slot.
func TestReview13SessionSixOwnersAcrossNoWaitFailover(t *testing.T) {
	phase := os.Getenv("ROOST_REVIEW13_HA_PHASE")
	if phase != "seed" && phase != "verify" {
		t.Skip("runner sets HA phase")
	}
	addresses := os.Getenv("ROOST_REVIEW_CLUSTER")
	if addresses == "" {
		t.Fatal("isolated Cluster required")
	}
	cfg := redis.DefaultConfig(strings.Split(addresses, ",")[0])
	cfg.ClusterAddrs = strings.Split(addresses, ",")
	client, err := driver.NewClient(cfg)
	if err != nil {
		t.Fatal(err)
	}
	defer client.Close()
	ctx, cancel := context.WithTimeout(context.Background(), 30*time.Second)
	defer cancel()
	for owner := int64(1); owner <= 6; owner++ {
		prefix := fmt.Sprintf("{review13-ha-owner-%d}", owner)
		runID := fmt.Sprintf("review13-ha-run-%d", owner)
		stores, err := NewRedisStores(client, RedisConfig{Prefix: prefix, RequestTTL: time.Hour})
		if err != nil {
			t.Fatal(err)
		}
		calls := 0
		service, err := New(Config{Runs: stores.Runs, Claims: stores.Claims, Requests: stores.Requests,
			TTL: 5 * time.Minute, NewRunID: func() (string, error) { return runID, nil },
			Release: ReleaserFunc(func(context.Context, Run, Resource) error { calls++; return nil })})
		if err != nil {
			t.Fatal(err)
		}
		request := EnterRequest{Kind: "dungeon", RequestID: prefix + ":request"}
		if phase == "seed" {
			run, err := service.Enter(ctx, owner, request)
			if err != nil || run.ID != runID {
				t.Fatalf("owner %d seed Enter: %+v %v", owner, run, err)
			}
			run, err = service.Attach(ctx, owner, runID, Resource{Kind: "scene", ID: fmt.Sprintf("scene-%d", owner)})
			if err != nil || len(run.Pending()) != 1 {
				t.Fatalf("owner %d seed Attach: %+v %v", owner, run, err)
			}
			continue
		}
		run, found, err := service.Get(ctx, owner, runID)
		if err != nil || !found || run.State != StateOpen || len(run.Pending()) != 1 {
			t.Fatalf("owner %d after failover: found=%v state=%q pending=%d err=%v", owner, found, run.State, len(run.Pending()), err)
		}
		finished, err := service.Finish(ctx, owner, runID, StateSucceeded, "done")
		if err != nil || finished.State != StateSucceeded || len(finished.Pending()) != 0 || calls != 1 {
			t.Fatalf("owner %d Finish: state=%q pending=%d calls=%d err=%v", owner, finished.State, len(finished.Pending()), calls, err)
		}
		replay, err := service.Enter(ctx, owner, request)
		if err != nil || replay.ID != runID {
			t.Fatalf("owner %d replay: %+v %v", owner, replay, err)
		}
		_, found, err = stores.Claims.Get(ctx, owner)
		if err != nil || found {
			t.Fatalf("owner %d claim remains=%v err=%v", owner, found, err)
		}
	}
	t.Logf("RESULT phase=%s owners=6 pending_per_owner=%d", phase, map[bool]int{true: 1, false: 0}[phase == "seed"])
}
