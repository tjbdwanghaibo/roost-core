package session

import (
	"bytes"
	"context"
	"encoding/json"
	"fmt"
	"net/http"
	"os"
	"testing"
	"time"

	redis "github.com/tjbdwanghaibo/roost-core/redis"
	"github.com/tjbdwanghaibo/roost-core/redis/driver"
	"github.com/tjbdwanghaibo/roost-core/versionstore"
)

type review13ToxicRunner struct {
	versionstore.RedisClient
	beforeEval func() error
}

func (r *review13ToxicRunner) Eval(ctx context.Context, script string, keys []string, args ...any) (any, error) {
	if r.beforeEval != nil {
		fire := r.beforeEval
		r.beforeEval = nil
		if err := fire(); err != nil {
			return nil, err
		}
	}
	return r.RedisClient.Eval(ctx, script, keys, args...)
}

func review13ToxicAPI(path string, payload any) error {
	base := os.Getenv("ROOST_REVIEW13_TOXI_API")
	var body bytes.Buffer
	if payload != nil {
		if err := json.NewEncoder(&body).Encode(payload); err != nil {
			return err
		}
	}
	resp, err := http.Post(base+path, "application/json", &body)
	if err != nil {
		return err
	}
	defer resp.Body.Close()
	if resp.StatusCode >= 300 {
		return fmt.Errorf("Toxiproxy %s: status %d", path, resp.StatusCode)
	}
	return nil
}

// The wrapper lets the first Redis read complete, then installs a real TCP
// downstream timeout exactly before Attach's CAS script. It is a test seam;
// the production store/client and the Redis write are unchanged.
func TestReview13SessionLostAttachReplyReconciles(t *testing.T) {
	api, backend, proxy := os.Getenv("ROOST_REVIEW13_TOXI_API"), os.Getenv("ROOST_REVIEW_REDIS"), os.Getenv("ROOST_REVIEW13_TOXI_REDIS")
	if api == "" || backend == "" || proxy == "" {
		t.Fatal("isolated Redis and Toxiproxy addresses required")
	}
	if err := review13ToxicAPI("/reset", nil); err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() {
		if err := review13ToxicAPI("/reset", nil); err != nil {
			t.Errorf("reset proxy: %v", err)
		}
	})
	direct, err := driver.NewClient(redis.DefaultConfig(backend))
	if err != nil {
		t.Fatal(err)
	}
	defer direct.Close()
	proxiedConfig := redis.DefaultConfig(proxy)
	proxiedConfig.ReadTimeout = 2 * time.Second
	proxied, err := driver.NewClient(proxiedConfig)
	if err != nil {
		t.Fatal(err)
	}
	defer proxied.Close()
	prefix := fmt.Sprintf("{review13-session-toxic-%d}", time.Now().UnixNano())
	const runID = "review13-toxic-run"
	cfg := RedisConfig{Prefix: prefix, RequestTTL: time.Hour}
	directStores, err := NewRedisStores(direct, cfg)
	if err != nil {
		t.Fatal(err)
	}
	callbackCount := 0
	serviceConfig := Config{Runs: directStores.Runs, Claims: directStores.Claims, Requests: directStores.Requests,
		TTL: 5 * time.Minute, NewRunID: func() (string, error) { return runID, nil },
		Release: ReleaserFunc(func(context.Context, Run, Resource) error { callbackCount++; return nil })}
	directService, err := New(serviceConfig)
	if err != nil {
		t.Fatal(err)
	}
	ctx := context.Background()
	run, err := directService.Enter(ctx, 77, EnterRequest{Kind: "dungeon", RequestID: prefix + ":request"})
	if err != nil {
		t.Fatal(err)
	}
	toxic := &review13ToxicRunner{RedisClient: proxied}
	toxic.beforeEval = func() error {
		return review13ToxicAPI("/proxies/redis/toxics", map[string]any{
			"name": "drop-attach-reply", "type": "timeout", "stream": "downstream", "toxicity": 1.0,
			"attributes": map[string]any{"timeout": 0},
		})
	}
	proxiedStores, err := NewRedisStores(toxic, cfg)
	if err != nil {
		t.Fatal(err)
	}
	serviceConfig.Runs, serviceConfig.Claims, serviceConfig.Requests = proxiedStores.Runs, proxiedStores.Claims, proxiedStores.Requests
	proxiedService, err := New(serviceConfig)
	if err != nil {
		t.Fatal(err)
	}
	deadline, cancel := context.WithTimeout(ctx, 700*time.Millisecond)
	_, attachErr := proxiedService.Attach(deadline, 77, run.ID, Resource{Kind: "scene", ID: "scene-77"})
	cancel()
	if attachErr == nil {
		t.Fatal("Attach succeeded despite a swallowed CAS reply")
	}
	if toxic.beforeEval != nil {
		t.Fatal("the CAS script was never attempted")
	}
	if err := review13ToxicAPI("/reset", nil); err != nil {
		t.Fatal(err)
	}
	stored, found, err := directService.Get(ctx, 77, run.ID)
	if err != nil || !found || len(stored.Resources) != 1 || stored.Resources[0].ID != "scene-77" {
		t.Fatalf("lost reply outcome: found=%v resources=%+v err=%v", found, stored.Resources, err)
	}
	second, err := directService.Attach(ctx, 77, run.ID, Resource{Kind: "scene", ID: "scene-77"})
	if err != nil || len(second.Resources) != 1 {
		t.Fatalf("Attach replay: resources=%+v err=%v", second.Resources, err)
	}
	finished, err := directService.Finish(ctx, 77, run.ID, StateSucceeded, "done")
	if err != nil || len(finished.Pending()) != 0 || callbackCount != 1 {
		t.Fatalf("Finish after replay: pending=%d callbacks=%d err=%v", len(finished.Pending()), callbackCount, err)
	}
	t.Logf("RESULT lost_reply_error=%v committed_resources=1 replay_resources=1 release_callbacks=1", attachErr)
}
