package match

import (
	"context"
	"errors"
	"fmt"
	"os"
	"runtime"
	"sort"
	"strconv"
	"sync"
	"sync/atomic"
	"testing"
	"time"

	redis "github.com/tjbdwanghaibo/roost-core/redis"
	"github.com/tjbdwanghaibo/roost-core/redis/driver"
	"github.com/tjbdwanghaibo/roost-core/versionstore"
)

// Four independent clients contend for one 4,096-entry queue for a bounded interval.
// Conflict retries here model a caller retrying a stable request/ticket ID;
// the sample measures this machine, not a production throughput promise.
func TestReview13MatchFourClientHotSoak(t *testing.T) {
	addr := os.Getenv("ROOST_REVIEW_REDIS")
	if addr == "" {
		t.Fatal("isolated Redis required")
	}
	ctx, cancel := context.WithTimeout(context.Background(), 90*time.Second)
	defer cancel()
	prefix := fmt.Sprintf("review13-match-%d", time.Now().UnixNano())
	queue := Queue{Mode: "review", GroupSize: 2, Partition: "hot"}
	root, err := driver.NewClient(redis.DefaultConfig(addr))
	if err != nil {
		t.Fatal(err)
	}
	defer root.Close()
	state, err := versionstore.NewRedisStore(root, versionstore.RedisConfig[string, queueState]{
		Prefix: prefix + ":queue:", KeyOf: func(k string) string { return k }, Codec: versionstore.JSONCodec[queueState]{},
	})
	if err != nil {
		t.Fatal(err)
	}
	defer root.Del(context.Background(), prefix+":queue:"+queue.Key())
	const history = 4096
	now := time.Unix(1700000000, 0)
	initial := queueState{}
	initial.init()
	for i := 1; i <= history; i++ {
		id, request := fmt.Sprintf("old-%d", i), fmt.Sprintf("request-%d", i)
		initial.Tickets[id] = Ticket{ID: id, Queue: queue, Subject: Subject{Kind: "player", ID: int64(i)},
			RequestID: request, State: TicketCancelled, CreatedAtUnix: now.Unix() - 60,
			ExpiresAtUnix: now.Unix() + 3600, ResolvedAtUnix: now.Unix()}
		initial.Requests[request] = id
	}
	if _, created, err := state.Create(ctx, queue.Key(), initial); err != nil || !created {
		t.Fatalf("seed created=%v err=%v", created, err)
	}
	var sequence, conflicts int64
	var mu sync.Mutex
	latency := make([]int64, 0, 256)
	results := make(chan error, 4)
	soakSeconds := 20
	if raw := os.Getenv("ROOST_REVIEW13_SOAK_SECONDS"); raw != "" {
		parsed, err := strconv.Atoi(raw)
		if err != nil || parsed < 1 || parsed > 120 {
			t.Fatalf("soak seconds must be 1..120: %q", raw)
		}
		soakSeconds = parsed
	}
	deadline := time.Now().Add(time.Duration(soakSeconds) * time.Second)
	var before, after runtime.MemStats
	runtime.ReadMemStats(&before)
	started := time.Now()
	for worker := 0; worker < 4; worker++ {
		go func(worker int) {
			client, err := driver.NewClient(redis.DefaultConfig(addr))
			if err != nil {
				results <- err
				return
			}
			defer client.Close()
			workerState, err := versionstore.NewRedisStore(client, versionstore.RedisConfig[string, queueState]{
				Prefix: prefix + ":queue:", KeyOf: func(k string) string { return k }, Codec: versionstore.JSONCodec[queueState]{},
			})
			if err != nil {
				results <- err
				return
			}
			store, err := NewStore(workerState, Config{Now: func() time.Time { return now },
				NewID: func() (string, error) { return fmt.Sprintf("hot-%d", atomic.AddInt64(&sequence, 1)), nil }})
			if err != nil {
				results <- err
				return
			}
			for iteration := 0; time.Now().Before(deadline); iteration++ {
				begin := time.Now()
				subject := Subject{Kind: "player", ID: int64(history + 1 + worker*100000 + iteration)}
				request := fmt.Sprintf("worker-%d-iteration-%d", worker, iteration)
				var ticket Ticket
				var opErr error
				for attempt := 0; attempt < 8; attempt++ {
					ticket, opErr = store.Enqueue(ctx, queue, subject, request)
					if !errors.Is(opErr, versionstore.ErrConflict) {
						break
					}
					atomic.AddInt64(&conflicts, 1)
				}
				if opErr != nil {
					results <- fmt.Errorf("worker %d enqueue iteration %d: %w", worker, iteration, opErr)
					return
				}
				for attempt := 0; attempt < 8; attempt++ {
					_, opErr = store.Cancel(ctx, queue, ticket.ID, subject)
					if !errors.Is(opErr, versionstore.ErrConflict) {
						break
					}
					atomic.AddInt64(&conflicts, 1)
				}
				if opErr != nil {
					results <- fmt.Errorf("worker %d cancel iteration %d: %w", worker, iteration, opErr)
					return
				}
				mu.Lock()
				latency = append(latency, time.Since(begin).Microseconds())
				mu.Unlock()
			}
			results <- nil
		}(worker)
	}
	for i := 0; i < 4; i++ {
		if err := <-results; err != nil {
			t.Fatal(err)
		}
	}
	elapsed := time.Since(started)
	runtime.ReadMemStats(&after)
	got, found, err := state.Get(ctx, queue.Key())
	if err != nil || !found {
		t.Fatalf("final state found=%v err=%v", found, err)
	}
	cycles := len(latency)
	if cycles < 4 || len(got.Value.Waiting) != 0 || len(got.Value.Tickets) != history+cycles || len(got.Value.Requests) != history+cycles {
		t.Fatalf("final counts cycles=%d waiting=%d tickets=%d requests=%d", cycles, len(got.Value.Waiting), len(got.Value.Tickets), len(got.Value.Requests))
	}
	sort.Slice(latency, func(i, j int) bool { return latency[i] < latency[j] })
	p95, p99 := latency[(95*cycles+99)/100-1], latency[(99*cycles+99)/100-1]
	t.Logf("RESULT workers=4 history=%d target_seconds=%d cycles=%d elapsed_ms=%d conflict_retries=%d waiting=0 allocated_bytes=%d pair_p50_us=%d pair_p95_us=%d pair_p99_us=%d max_us=%d",
		history, soakSeconds, cycles, elapsed.Milliseconds(), conflicts, after.TotalAlloc-before.TotalAlloc,
		latency[cycles/2], p95, p99, latency[cycles-1])
}
