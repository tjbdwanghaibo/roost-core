package cache

import (
	"context"
	"crypto/rand"
	"fmt"
	"os"
	"reflect"
	"testing"
	"time"

	fredis "github.com/tjbdwanghaibo/roost-core/redis"
	redisdriver "github.com/tjbdwanghaibo/roost-core/redis/driver"
)

type review12PointerText int
func (v *review12PointerText) MarshalText() ([]byte, error) { return []byte(fmt.Sprintf("code=%d", *v)), nil }
func (v *review12PointerText) UnmarshalText(raw []byte) error {
	var n int
	_, err := fmt.Sscanf(string(raw), "code=%d", &n)
	*v = review12PointerText(n)
	return err
}
type review12ValueText int
func (v review12ValueText) MarshalText() ([]byte, error) { return []byte(fmt.Sprintf("code=%d", v)), nil }
func (v *review12ValueText) UnmarshalText(raw []byte) error {
	var n int
	_, err := fmt.Sscanf(string(raw), "code=%d", &n)
	*v = review12ValueText(n)
	return err
}
type review12PointerTextRecord struct { ID int64; Amount review12PointerText }
type review12ValueTextRecord struct { ID int64; Amount review12ValueText }
type review12RootAlias struct { ID int64 `json:"id"`; Root struct { ID int64 `json:"id"` } `json:"root"` }
type review12ReservedField struct { ID int64; Data string `redisdao:"__keys"` }

func TestReview12RefHMapRealRedis(t *testing.T) {
	addr := os.Getenv("ROOST_REDIS_TEST_ADDR")
	if addr == "" { t.Skip("requires an isolated Redis") }
	client, err := redisdriver.NewClient(fredis.DefaultConfig(addr))
	if err != nil { t.Fatal(err) }
	t.Cleanup(func() { _ = client.Close() })
	ctx, cancel := context.WithTimeout(context.Background(), 15*time.Second)
	defer cancel()
	if err := client.Ping(ctx); err != nil { t.Fatal(err) }
	for _, scenario := range []string{"value_control", "pointer_root", "pointer_text", "value_text_control", "patch_nil_parent", "patch_existing_control", "root_alias", "reserved_field"} {
		t.Run(scenario, func(t *testing.T) {
			prefix := "roost:review12:" + rand.Text()
			defer func() { if caught := recover(); caught != nil { t.Errorf("supported generic store panicked: %v", caught) } }()
			switch scenario {
			case "pointer_root":
				store := NewRedisRefHMapStore[int64, *refHMapSession](client, RefHMapConfig[int64, *refHMapSession]{Prefix: prefix, Name: "record", StoreConfig: StoreConfig[int64, *refHMapSession]{KeyOf: func(v *refHMapSession) int64 { return v.ID }}})
				t.Cleanup(func() { _ = store.Delete(context.Background(), 1) })
				want := &refHMapSession{ID: 1, Version: 2}
				if err := store.Set(ctx, want); err != nil { t.Fatal(err) }
				got, held, err := store.Get(ctx, 1)
				if err != nil || !held || !reflect.DeepEqual(got, want) { t.Fatalf("pointer roundtrip: %+v %v %v", got, held, err) }
			case "pointer_text":
				store := NewRedisRefHMapStore(client, RefHMapConfig[int64, review12PointerTextRecord]{Prefix: prefix, Name: "record", StoreConfig: StoreConfig[int64, review12PointerTextRecord]{KeyOf: func(v review12PointerTextRecord) int64 { return v.ID }}})
				t.Cleanup(func() { _ = store.Delete(context.Background(), 1) })
				want := review12PointerTextRecord{ID: 1, Amount: 7}
				if err := store.Set(ctx, want); err != nil { t.Fatal(err) }
				got, held, err := store.Get(ctx, 1)
				t.Logf("pointer TextMarshaler roundtrip: %+v held=%v err=%v", got, held, err)
				if err != nil || !held || got != want { t.Fatal("recognized text scalar did not roundtrip") }
			case "value_text_control":
				store := NewRedisRefHMapStore(client, RefHMapConfig[int64, review12ValueTextRecord]{Prefix: prefix, Name: "record", StoreConfig: StoreConfig[int64, review12ValueTextRecord]{KeyOf: func(v review12ValueTextRecord) int64 { return v.ID }}})
				t.Cleanup(func() { _ = store.Delete(context.Background(), 1) })
				want := review12ValueTextRecord{ID: 1, Amount: 7}
				if err := store.Set(ctx, want); err != nil { t.Fatal(err) }
				got, held, err := store.Get(ctx, 1)
				if err != nil || !held || got != want { t.Fatalf("value TextMarshaler: %+v %v %v", got, held, err) }
			case "root_alias":
				store := NewRedisRefHMapStore(client, RefHMapConfig[int64, review12RootAlias]{Prefix: prefix, Name: "record", StoreConfig: StoreConfig[int64, review12RootAlias]{KeyOf: func(v review12RootAlias) int64 { return v.ID }}})
				t.Cleanup(func() { _ = store.Delete(context.Background(), 1) })
				want := review12RootAlias{ID: 1}; want.Root.ID = 99
				if err := store.Set(ctx, want); err != nil { t.Fatal(err) }
				got, held, err := store.Get(ctx, 1)
				t.Logf("root alias: want=%+v got=%+v held=%v err=%v", want, got, held, err)
				if err != nil || !held || got != want { t.Fatal("accepted layout silently aliased root and child") }
			case "reserved_field":
				store := NewRedisRefHMapStore(client, RefHMapConfig[int64, review12ReservedField]{Prefix: prefix, Name: "record", StoreConfig: StoreConfig[int64, review12ReservedField]{KeyOf: func(v review12ReservedField) int64 { return v.ID }}})
				// Registered keys are precisely this test's root; do not delete arbitrary keys from the payload.
				rootKey := prefix + ":{record:1}:root"
				t.Cleanup(func() { _, _ = client.Del(context.Background(), rootKey) })
				want := review12ReservedField{ID: 1, Data: "payload"}
				if err := store.Set(ctx, want); err != nil { t.Fatal(err) }
				got, held, err := store.Get(ctx, 1)
				t.Logf("reserved field: want=%+v got=%+v held=%v err=%v", want, got, held, err)
				if err != nil || !held || got != want { t.Fatal("accepted field was overwritten by registry metadata") }
			default:
				store := NewRedisRefHMapStore(client, RefHMapConfig[int64, refHMapSession]{Prefix: prefix, Name: "record", StoreConfig: refHMapSessionConfig()})
				t.Cleanup(func() { _ = store.Delete(context.Background(), 1) })
				want := refHMapSession{ID: 1, Version: 2}
				if scenario != "patch_nil_parent" { want.Meta = &refHMapMeta{Label: "before"} }
				if err := store.Set(ctx, want); err != nil { t.Fatal(err) }
				if scenario != "value_control" {
					if err := store.Patch(ctx, 1, "Meta.Label", "after"); err != nil { t.Fatal(err) }
					want.Meta = &refHMapMeta{Label: "after"}
				}
				got, held, err := store.Get(ctx, 1)
				t.Logf("nested patch: got=%+v Meta=%+v held=%v err=%v", got, got.Meta, held, err)
				if err != nil || !held || !reflect.DeepEqual(got, want) { t.Fatal("successful nested patch was not observable") }
			}
		})
	}
}
