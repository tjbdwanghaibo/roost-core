 "github.com/tjbdwanghaibo/roost-core/versionstore"
)

type review4MatchWrite struct {
 versionstore.Store[string, queueState]
 armed atomic.Bool
 after bool
}
func (s *review4MatchWrite) Update(ctx context.Context, key string, mutate versionstore.Mutate[queueState]) (versionstore.Versioned[queueState], bool, error) {
 fail := s.armed.Swap(false)
 if fail && !s.after { return versionstore.Versioned[queueState]{}, false, context.DeadlineExceeded }
 value, applied, err := s.Store.Update(ctx, key, mutate)
 if fail && err == nil { return versionstore.Versioned[queueState]{}, false, context.DeadlineExceeded }
 return value, applied, err
}

func TestReview4MatchUnknownCommitRecovery(t *testing.T) {
 for _, after := range []bool{false, true} { t.Run(fmt.Sprintf("after_write_%v", after), func(t *testing.T){
  var base versionstore.Store[string, queueState] = versionstore.NewMemoryStore[string, queueState]()
  if os.Getenv("ROOST_REVIEW4_BACKEND") == "redis" {
   addr := os.Getenv("ROOST_REVIEW_REDIS"); if addr == "" { t.Fatal("isolated Redis address is required") }
   client, err := driver.NewClient(fredis.DefaultConfig(addr)); if err != nil { t.Fatal(err) }; t.Cleanup(func(){ _ = client.Close() })
   base, err = versionstore.NewRedisStore(client, versionstore.RedisConfig[string, queueState]{Prefix:fmt.Sprintf("review4:match:%d:", time.Now().UnixNano()), KeyOf:func(k string) string{return k}, Codec:versionstore.JSONCodec[queueState]{}})
   if err != nil { t.Fatal(err) }
  }
  gate := &review4MatchWrite{Store:base, after:after}
  store, err := NewStore(gate, Config{}); if err != nil { t.Fatal(err) }
  ctx := context.Background(); q:=ranked()
  a:=enqueue(t,store,q,player(1,100),"a"); b:=enqueue(t,store,q,player(2,100),"b")
  gate.armed.Store(true)
  if _, err := store.Commit(ctx,q,[]string{a.ID,b.ID}); !errors.Is(err,context.DeadlineExceeded) { t.Fatal(err) }
  gotA, foundA, errA := store.Ticket(ctx,q,a.ID,a.Subject)
  gotB, foundB, errB := store.Ticket(ctx,q,b.ID,b.Subject)
  if !foundA || !foundB || errA != nil || errB != nil { t.Fatal(foundA,foundB,errA,errB) }
  retry, retryErr := store.Commit(ctx,q,[]string{a.ID,b.ID})
  if after {
   if gotA.State != TicketMatched || gotA.MatchID == "" || gotA.MatchID != gotB.MatchID || !errors.Is(retryErr,ErrConflict) { t.Fatal(gotA,gotB,retryErr) }
   recovered, found, err := store.Match(ctx,q,gotA.MatchID)
   if !found || err != nil || len(recovered.TicketIDs)!=2 { t.Fatal(recovered,found,err) }
   t.Log("unknown Commit recovered through Ticket.MatchID -> Match; direct replay is a conflict")
  } else if gotA.State != TicketWaiting || gotB.State != TicketWaiting || retryErr != nil || retry.ID == "" { t.Fatal(gotA,gotB,retry,retryErr) }
 }) }
}
