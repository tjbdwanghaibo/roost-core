func TestReview10PipelineMissingMasksWriteError(t *testing.T) {
 for _, cluster := range []bool{false,true} {
  t.Run(fmt.Sprintf("cluster_%v",cluster),func(t *testing.T){
   cfg:=redis.DefaultConfig(os.Getenv("REDIS_ADDR"))
   if cluster { cfg.ClusterAddrs=strings.Split(os.Getenv("ROOST_REVIEW_CLUSTER"),",") }
   if cfg.Addr=="" && len(cfg.ClusterAddrs)==0 { t.Fatal("real Redis required") }
   client,err:=NewClient(cfg); if err!=nil{t.Fatal(err)}; defer client.Close()
   ctx,cancel:=context.WithTimeout(context.Background(),10*time.Second); defer cancel()
   // Same owner fixes command order, on standalone as well as Cluster.
   prefix:=fmt.Sprintf("{review10-%d}",time.Now().UnixNano())
   absent,key:=prefix+":absent",prefix+":string"
   defer func(){_,_=client.Del(context.Background(),key)}()
   if err:=client.Set(ctx,key,"original",time.Minute);err!=nil{t.Fatal(err)}
   control:=client.Pipeline();control.HSet(ctx,key,"field","value")
   if err:=control.Exec(ctx);err==nil || !strings.Contains(err.Error(),"WRONGTYPE"){t.Fatalf("control: %v",err)}
   pipe:=client.Pipeline(); missing:=pipe.Get(ctx,absent);pipe.HSet(ctx,key,"field","value")
   err=pipe.Exec(ctx)
   value,readErr:=client.Get(ctx,key)
   _,missingErr:=missing.Result()
   t.Logf("Exec=%v missing=%v actual_value=%q read_error=%v",err,missingErr,value,readErr)
   if readErr!=nil || string(value)!="original" { t.Fatalf("fixture: %q %v",value,readErr) }
   if err==nil || !strings.Contains(err.Error(),"WRONGTYPE") { t.Fatalf("failed HSet was reported as successful Exec: %v",err) }
  })
 }
}

func TestReview10PipelineReuseAndDiscard(t *testing.T){
 cfg:=redis.DefaultConfig(os.Getenv("REDIS_ADDR"));client,err:=NewClient(cfg);if err!=nil{t.Fatal(err)};defer client.Close()
 ctx:=context.Background();key:=fmt.Sprintf("review10-reuse-%d",time.Now().UnixNano());defer func(){_,_=client.Del(ctx,key)}()
 pipe:=client.Pipeline();pipe.Set(ctx,key,"discarded",time.Minute);pipe.Discard()
 if err:=pipe.Exec(ctx);err!=nil{t.Fatal(err)}
 if _,err:=client.Get(ctx,key);err==nil{t.Fatal("discarded write executed")}
 pipe.Set(ctx,key,"first",time.Minute);first:=pipe.Get(ctx,key);if err:=pipe.Exec(ctx);err!=nil{t.Fatal(err)}
 pipe.Set(ctx,key,"second",time.Minute);second:=pipe.Get(ctx,key);if err:=pipe.Exec(ctx);err!=nil{t.Fatal(err)}
 a,ae:=first.Result();b,be:=second.Result();if ae!=nil||be!=nil||string(a)!="first"||string(b)!="second"{t.Fatalf("futures changed: %q %v %q %v",a,ae,b,be)}
}
