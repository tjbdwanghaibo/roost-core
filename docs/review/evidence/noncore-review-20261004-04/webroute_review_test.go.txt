package webroute

import (
	"net/http"
	"net/http/httptest"
	"strings"
	"testing"

	"github.com/tjbdwanghaibo/roost-core/httpserver"
)

func review4Register(reg *Registrar, path string) (err error, panicked any) {
	defer func() { panicked = recover() }()
	err = reg.Register(http.MethodGet, path, func(http.ResponseWriter, *http.Request) {})
	return
}

func TestReview4InvalidPatternReturnsStartupError(t *testing.T) {
	for _, path := range []string{"/broken/{id", "/broken/{id:[}", "/broken/*/tail"} {
		t.Run(path, func(t *testing.T) {
			err, panicked := review4Register(NewRegistrar(httpserver.NewEngine()), path)
			t.Logf("registration_error=%v panic=%v", err, panicked)
			if panicked != nil || err == nil {
				t.Fatal("invalid route did not return a normal startup error")
			}
		})
	}
}

func TestReview4JSONDocumentControls(t *testing.T) {
	for _, mode := range []string{"valid", "empty", "trailing_whitespace", "trailing_garbage", "second_object", "oversized", "raw_owned_headers"} {
		t.Run(mode, func(t *testing.T) {
			engine := httpserver.NewEngine(httpserver.WithMaxBodyBytes(32))
			calls := 0
			engine.Post("/json", func(w http.ResponseWriter, r *http.Request) {
				if mode == "raw_owned_headers" {
					raw, ok := ReadRaw(w, r)
					if !ok {
						return
					}
					raw.Header.Set("X-Probe", "local")
					if r.Header.Get("X-Probe") != "original" {
						t.Fatal("raw header alias")
					}
					calls++
					WriteResult(w, 200, nil, nil)
					return
				}
				var target map[string]any
				if !DecodeJSON(w, r, &target) {
					return
				}
				calls++
				WriteResult(w, 200, target, nil)
			})
			body := `{"value":1}`
			want := 200
			switch mode {
			case "empty":
				body = ""
			case "trailing_whitespace":
				body += " \n\t"
			case "trailing_garbage":
				body += "junk"
				want = 400
			case "second_object":
				body += `{}`
				want = 400
			case "oversized":
				body = strings.Repeat("x", 33)
				want = 413
			}
			request := httptest.NewRequest("POST", "/json", strings.NewReader(body))
			request.Header.Set("X-Probe", "original")
			recorder := httptest.NewRecorder()
			engine.ServeHTTP(recorder, request)
			if recorder.Code != want || (calls == 1) != (want == 200) {
				t.Fatalf("HTTP=%d calls=%d body=%s", recorder.Code, calls, recorder.Body.String())
			}
		})
	}
}

func TestReview4ParameterShapeCollisionObservation(t *testing.T) {
	engine := httpserver.NewEngine()
	registrar := NewRegistrar(engine)
	first := registrar.Register("GET", "/items/{id}", func(w http.ResponseWriter, _ *http.Request) { _, _ = w.Write([]byte("first")) })
	second := registrar.Register("GET", "/items/{name}", func(w http.ResponseWriter, _ *http.Request) { _, _ = w.Write([]byte("second")) })
	recorder := httptest.NewRecorder()
	engine.ServeHTTP(recorder, httptest.NewRequest("GET", "/items/42", nil))
	t.Logf("first=%v second=%v response=%s", first, second, recorder.Body.String())
}
