type revn03Env struct {
	url, prefix, reqStream, respStream string
}

func revn03NewEnv(t *testing.T) revn03Env {
	t.Helper()
	url := os.Getenv("ROOST_DATAENGINE_IT_NATS_URL")
	if url == "" {
		t.Skip("no NATS")
	}
	suffix := fmt.Sprintf("%d", time.Now().UnixNano())
	env := revn03Env{url: url, prefix: "revn03p" + suffix, reqStream: "REVN03_REQ_" + suffix, respStream: "REVN03_RESP_" + suffix}
	t.Cleanup(func() { revn03DeleteStreams(t, env) })
	return env
}

func revn03DeleteStreams(t *testing.T, env revn03Env) {
	nc, err := gonats.Connect(env.url)
	if err != nil {
		t.Errorf("cleanup connect: %v", err)
		return
	}
	defer nc.Close()
	js, _ := gojs.New(nc)
	ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
	defer cancel()
	for _, s := range []string{env.reqStream, env.respStream} {
		if err := js.DeleteStream(ctx, s); err != nil && !errors.Is(err, gojs.ErrStreamNotFound) {
			t.Errorf("cleanup delete %s: %v", s, err)
		}
	}
}

func revn03Mod(t *testing.T, env revn03Env, sid int32, extra map[string]any) (*NatsMod, *bus.Bus) {
	t.Helper()
	cfg := viper.New()
	cfg.Set("sid", sid)
	cfg.Set("server_type", "rvn")
	cfg.Set("nats.url", env.url)
	cfg.Set("nats.prefix", env.prefix)
	cfg.Set("nats.rpc.transport", "jetstream")
	cfg.Set("nats.rpc.request_stream", env.reqStream)
	cfg.Set("nats.rpc.response_stream", env.respStream)
	cfg.Set("nats.rpc.max_bytes", int64(8<<20))
	cfg.Set("nats.rpc.setup_timeout", 20*time.Second)
	for k, v := range extra {
		cfg.Set(k, v)
	}
	mod := NewNatsMod(nil)
	if err := mod.Init(cfg); err != nil {
		t.Fatal(err)
	}
	registry := app.NewRegistry(cfg)
	if err := mod.Provide(registry); err != nil {
		t.Fatal(err)
	}
	if err := mod.Start(); err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() { _ = mod.StopWithContext(context.Background()) })
	ibus, _ := app.Lookup[bus.IBus](registry, mods.ModBus)
	return mod, ibus.(*bus.Bus)
}

func revn03ConsumerInfo(t *testing.T, env revn03Env, name string) *gojs.ConsumerInfo {
	nc, err := gonats.Connect(env.url)
	if err != nil {
		t.Fatal(err)
	}
	defer nc.Close()
	js, _ := gojs.New(nc)
	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()
	c, err := js.Consumer(ctx, env.reqStream, name)
	if err != nil {
		t.Logf("consumer %s: %v", name, err)
		return nil
	}
	info, err := c.Info(ctx)
	if err != nil {
		t.Logf("consumer info %s: %v", name, err)
		return nil
	}
	return info
}

// P1: a JetStream RPC handler in flight when the serving Mod stops.
func TestRevn03ProbeJetStreamHandlerInFlightAtStop(t *testing.T) {
	env := revn03NewEnv(t)
	server, serverBus := revn03Mod(t, env, 9301, map[string]any{"nats.rpc.ack_wait": 3 * time.Second})
	_, clientBus := revn03Mod(t, env, 9302, nil)

	entered := make(chan struct{}, 4)
	release := make(chan struct{})
	var runs, running atomic.Int32
	var ctxErrAtEntry atomic.Value
	if err := serverBus.HandleRpc("Slow", func(rc *bus.RpcContext) (any, error) {
		runs.Add(1)
		running.Add(1)
		defer running.Add(-1)
		entered <- struct{}{}
		<-release
		ctxErrAtEntry.Store(fmt.Sprint(rc.Context().Err()))
		return map[string]string{"ok": "yes"}, nil
	}); err != nil {
		t.Fatal(err)
	}
	callDone := make(chan error, 1)
	go func() {
		ctx, cancel := context.WithTimeout(context.Background(), 8*time.Second)
		defer cancel()
		var resp map[string]string
		callDone <- clientBus.CallReliable(ctx, "rvn", "Slow", map[string]string{}, &resp)
	}()
	select {
	case <-entered:
	case <-time.After(5 * time.Second):
		t.Fatal("handler never entered")
	}
	stopCtx, cancel := context.WithTimeout(context.Background(), 2*time.Second)
	defer cancel()
	started := time.Now()
	stopErr := server.StopWithContext(stopCtx)
	elapsed := time.Since(started)
	stillRunning := running.Load()
	t.Logf("stop returned err=%v after %s; handler still running=%d", stopErr, elapsed, stillRunning)
	close(release)
	time.Sleep(200 * time.Millisecond)
	t.Logf("handler ctx err when released: %v", ctxErrAtEntry.Load())
	err := <-callDone
	t.Logf("caller result: %v; handler runs=%d", err, runs.Load())
	if stopErr == nil && stillRunning > 0 {
		t.Errorf("StopWithContext returned nil while the JetStream handler was still running")
	}
}

func revn03Logf(t *testing.T, label string, info *gojs.ConsumerInfo) {
	if info == nil {
		t.Logf("%s: no consumer info", label)
		return
	}
	t.Logf("%s: delivered=%d ackfloor=%d ack_pending=%d redelivered=%d pending=%d", label, info.Delivered.Consumer, info.AckFloor.Consumer, info.NumAckPending, info.NumRedelivered, info.NumPending)
}

// P2..P6: ACK / NAK / TERM / redelivery on the real broker.
func TestRevn03ProbeJetStreamSettleMatrix(t *testing.T) {
	env := revn03NewEnv(t)
	_, serverBus := revn03Mod(t, env, 9311, map[string]any{"nats.rpc.ack_wait": time.Second})
	_, clientBus := revn03Mod(t, env, 9312, nil)
	var bizRuns, slowRuns, orphanRuns atomic.Int32
	must := func(err error) {
		if err != nil {
			t.Fatal(err)
		}
	}
	must(serverBus.HandleRpc("Biz", func(*bus.RpcContext) (any, error) { bizRuns.Add(1); return nil, errors.New("biz refused") }))
	must(serverBus.HandleRpc("Slow", func(*bus.RpcContext) (any, error) {
		slowRuns.Add(1)
		time.Sleep(1500 * time.Millisecond)
		return map[string]string{"ok": "1"}, nil
	}))
	must(serverBus.HandleRpc("Orphan", func(*bus.RpcContext) (any, error) { orphanRuns.Add(1); return map[string]string{}, nil }))
	must(serverBus.HandleRpc("Garbage", func(*bus.RpcContext) (any, error) { return nil, nil }))

	// handler business error: envelope back, acked once
	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	var resp map[string]string
	err := clientBus.CallReliable(ctx, "rvn", "Biz", map[string]string{}, &resp)
	cancel()
	time.Sleep(1500 * time.Millisecond)
	t.Logf("biz: err=%v runs=%d", err, bizRuns.Load())
	revn03Logf(t, "biz consumer", revn03ConsumerInfo(t, env, "rpc_rvn_Biz"))

	// undecodable request: NAK until MaxDeliver then TERM
	revn03RawPublish(t, env, env.prefix+".rpc.rvn.Garbage", []byte("not-json"))
	time.Sleep(3 * time.Second)
	revn03Logf(t, "garbage consumer", revn03ConsumerInfo(t, env, "rpc_rvn_Garbage"))

	// reply subject outside every stream: response publish fails each time
	raw := fmt.Sprintf(`{"from_sid":1,"msg_name":"Orphan","session_id":"orphan-1","msg_id":"orphan-1","reply_subject":"revn03nostream.%d.x","deadline_at":%d}`, time.Now().UnixNano(), time.Now().Add(time.Minute).UnixMilli())
	revn03RawPublish(t, env, env.prefix+".rpc.rvn.Orphan", []byte(raw))
	time.Sleep(4 * time.Second)
	t.Logf("orphan handler runs=%d", orphanRuns.Load())
	revn03Logf(t, "orphan consumer", revn03ConsumerInfo(t, env, "rpc_rvn_Orphan"))

	// handler slower than AckWait, caller deadline longer than both
	ctx, cancel = context.WithTimeout(context.Background(), 8*time.Second)
	err = clientBus.CallReliable(ctx, "rvn", "Slow", map[string]string{}, &resp)
	cancel()
	time.Sleep(3 * time.Second)
	t.Logf("slow: err=%v resp=%v runs=%d", err, resp, slowRuns.Load())
	revn03Logf(t, "slow consumer", revn03ConsumerInfo(t, env, "rpc_rvn_Slow"))
}

// P7: broker restart while both Mods stay up (own nats-server, REVN03_NATS_URL).
func TestRevn03ProbeJetStreamReconnect(t *testing.T) {
	url := os.Getenv("REVN03_NATS_URL")
	ctl := os.Getenv("REVN03_NATS_RESTART")
	if url == "" || ctl == "" {
		t.Skip("own nats-server not configured")
	}
	suffix := fmt.Sprintf("%d", time.Now().UnixNano())
	env := revn03Env{url: url, prefix: "revn03r" + suffix, reqStream: "REVN03_RREQ_" + suffix, respStream: "REVN03_RRESP_" + suffix}
	t.Cleanup(func() { revn03DeleteStreams(t, env) })
	_, serverBus := revn03Mod(t, env, 9321, nil)
	_, clientBus := revn03Mod(t, env, 9322, nil)
	var runs atomic.Int32
	if err := serverBus.HandleRpc("Ping", func(*bus.RpcContext) (any, error) { runs.Add(1); return map[string]string{"pong": "1"}, nil }); err != nil {
		t.Fatal(err)
	}
	call := func(d time.Duration) error {
		ctx, cancel := context.WithTimeout(context.Background(), d)
		defer cancel()
		var resp map[string]string
		return clientBus.CallReliable(ctx, "rvn", "Ping", map[string]string{}, &resp)
	}
	if err := call(3 * time.Second); err != nil {
		t.Fatalf("baseline: %v", err)
	}
	// restart the broker: the control file is polled by the shell harness
	if err := os.WriteFile(ctl, []byte("restart"), 0o644); err != nil {
		t.Fatal(err)
	}
	time.Sleep(500 * time.Millisecond)
	started := time.Now()
	err := call(700 * time.Millisecond)
	t.Logf("call during outage: err=%v elapsed=%s", err, time.Since(started))
	deadline := time.Now().Add(60 * time.Second)
	recovered := time.Now()
	for {
		if err := call(2 * time.Second); err == nil {
			t.Logf("recovered %s after restart request; runs=%d", time.Since(recovered), runs.Load())
			break
		} else if time.Now().After(deadline) {
			t.Fatalf("JetStream RPC never recovered after broker restart: %v", err)
		}
		time.Sleep(300 * time.Millisecond)
	}
}

// P9: lightweight RPC refused by a full dispatcher queue.
func TestRevn03ProbeLightweightRPCFullQueue(t *testing.T) {
	url := os.Getenv("ROOST_DATAENGINE_IT_NATS_URL")
	if url == "" {
		t.Skip("no NATS")
	}
	prefix := fmt.Sprintf("revn03q%d", time.Now().UnixNano())
	mk := func(sid int32, workers int) *NatsMod {
		cfg := viper.New()
		cfg.Set("sid", sid)
		cfg.Set("server_type", "rvq")
		cfg.Set("nats.url", url)
		cfg.Set("nats.prefix", prefix)
		cfg.Set("nats.worker_num", workers)
		mod := NewNatsMod(nil)
		if err := mod.Init(cfg); err != nil {
			t.Fatal(err)
		}
		if err := mod.Provide(app.NewRegistry(cfg)); err != nil {
			t.Fatal(err)
		}
		if err := mod.Start(); err != nil {
			t.Fatal(err)
		}
		t.Cleanup(func() { _ = mod.StopWithContext(context.Background()) })
		return mod
	}
	server := mk(9331, 1)
	client := mk(9332, 1)
	release := make(chan struct{})
	defer close(release)
	if err := server.bus.HandleRpc("Block", func(*bus.RpcContext) (any, error) { <-release; return map[string]string{}, nil }); err != nil {
		t.Fatal(err)
	}
	// worker 1, queue 1024: fill it with async calls first
	for i := 0; i < 1100; i++ {
		client.bus.CallAsync("rvq", "Block", map[string]string{}, func([]byte, error) {})
	}
	time.Sleep(500 * time.Millisecond)
	ctx, cancel := context.WithTimeout(context.Background(), 2*time.Second)
	defer cancel()
	started := time.Now()
	var resp map[string]string
	err := client.bus.Call(ctx, "rvq", "Block", map[string]string{}, &resp)
	t.Logf("call refused by full queue: err=%v elapsed=%s", err, time.Since(started))
}

func revn03StartEtcd(t *testing.T) string {
	bin, err := exec.LookPath("etcd")
	if err != nil {
		t.Skip("no etcd")
	}
	port := func() int {
		for {
			l, _ := net.Listen("tcp", "127.0.0.1:0")
			p := l.Addr().(*net.TCPAddr).Port
			_ = l.Close()
			if p != 23791 && p != 2379 && p != 2380 {
				return p
			}
		}
	}
	c, p := fmt.Sprintf("http://127.0.0.1:%d", port()), fmt.Sprintf("http://127.0.0.1:%d", port())
	cmd := exec.Command(bin, "--name", "revn03", "--data-dir", t.TempDir(), "--listen-client-urls", c, "--advertise-client-urls", c,
		"--listen-peer-urls", p, "--initial-advertise-peer-urls", p, "--initial-cluster", "revn03="+p, "--log-level", "error")
	if err := cmd.Start(); err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() { _ = cmd.Process.Kill(); _, _ = cmd.Process.Wait() })
	ep := c[len("http://"):]
	for i := 0; i < 150; i++ {
		if conn, err := net.DialTimeout("tcp", ep, 200*time.Millisecond); err == nil {
			_ = conn.Close()
			return ep
		}
		time.Sleep(100 * time.Millisecond)
	}
	t.Fatal("etcd did not start")
	return ""
}

// P10: discovered call carries the remaining budget to the JetStream handler.
func TestRevn03ProbeDiscoveredCallDeadline(t *testing.T) {
	env := revn03NewEnv(t)
	ep := revn03StartEtcd(t)
	cfg := fetcd.DefaultConfig([]string{ep})
	cfg.ServicePrefix = "/revn03/svc/"
	asm, err := etcddriver.Assemble(cfg)
	if err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() { _ = asm.Close(context.Background()) })
	if err := asm.Start(context.Background(), &fetcd.ServiceInfo{ServiceType: "rvn", Sid: 9341}); err != nil {
		t.Fatal(err)
	}
	_, serverBus := revn03Mod(t, env, 9341, nil)
	_, clientBus := revn03Mod(t, env, 9342, nil)
	var remaining atomic.Int64
	var hasDeadline atomic.Bool
	var gotSid atomic.Int32
	if err := serverBus.HandleRpc("Echo", func(rc *bus.RpcContext) (any, error) {
		if d, ok := rc.Context().Deadline(); ok {
			hasDeadline.Store(true)
			remaining.Store(int64(time.Until(d)))
		}
		gotSid.Store(9341)
		return revn03Status{}, nil
	}); err != nil {
		t.Fatal(err)
	}
	for _, transport := range []servicerpc.Transport{servicerpc.TransportJetStream, servicerpc.TransportLightweight} {
		hasDeadline.Store(false)
		c := servicerpc.NewDiscoveredBusClient(clientBus, "rvn", 800*time.Millisecond, asm.Discovery, servicerpc.WithTransport(transport))
		var resp revn03Status
		if transport == servicerpc.TransportLightweight {
			// lightweight handler is the same method name; register once is enough (JS handler only)
		}
		err := c.CallDiscoveredChecked(context.Background(), "Echo", map[string]string{}, &resp, "x")
		t.Logf("%s: err=%v handler_has_deadline=%v remaining=%s", transport, err, hasDeadline.Load(), time.Duration(remaining.Load()))
	}
}

type revn03Status struct{}

func (revn03Status) StatusCode() int32    { return 0 }
func (revn03Status) StatusReason() string { return "" }

// P11: lightweight Bus.Call against a JetStream-RPC deployment.
func TestRevn03ProbeLightweightCallIntoJetStreamSubjects(t *testing.T) {
	env := revn03NewEnv(t)
	_, serverBus := revn03Mod(t, env, 9351, nil)
	_, clientBus := revn03Mod(t, env, 9352, nil)
	var runs atomic.Int32
	var sawDeadline atomic.Bool
	if err := serverBus.HandleRpc("Grant", func(rc *bus.RpcContext) (any, error) {
		runs.Add(1)
		_, ok := rc.Context().Deadline()
		sawDeadline.Store(ok)
		return map[string]string{"granted": "1"}, nil
	}); err != nil {
		t.Fatal(err)
	}
	ctx, cancel := context.WithTimeout(context.Background(), time.Second)
	defer cancel()
	var resp map[string]string
	err := clientBus.Call(ctx, "rvn", "Grant", map[string]string{}, &resp)
	time.Sleep(time.Second)
	t.Logf("lightweight Call: err=%v resp=%v; handler runs afterwards=%d handler_had_deadline=%v", err, resp, runs.Load(), sawDeadline.Load())
	ctx2, cancel2 := context.WithTimeout(context.Background(), time.Second)
	defer cancel2()
	err = clientBus.CallTo(ctx2, "rvn", 9351, "Grant", map[string]string{}, &resp)
	time.Sleep(time.Second)
	t.Logf("lightweight CallTo: err=%v; handler runs afterwards=%d", err, runs.Load())
}
