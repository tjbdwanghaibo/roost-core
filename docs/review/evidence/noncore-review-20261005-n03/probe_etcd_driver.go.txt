type revn03Etcd struct {
	cmd      *exec.Cmd
	endpoint string
}

func revn03Port(t *testing.T) int {
	for {
		p := freePort(t)
		if p != 23791 && p != 2379 && p != 2380 {
			return p
		}
	}
}

func revn03StartEtcd(t *testing.T) *revn03Etcd {
	t.Helper()
	binary, err := exec.LookPath("etcd")
	if err != nil {
		t.Skip("no etcd")
	}
	clientURL := fmt.Sprintf("http://127.0.0.1:%d", revn03Port(t))
	peerURL := fmt.Sprintf("http://127.0.0.1:%d", revn03Port(t))
	cmd := exec.Command(binary, "--name", "revn03", "--data-dir", t.TempDir(),
		"--listen-client-urls", clientURL, "--advertise-client-urls", clientURL,
		"--listen-peer-urls", peerURL, "--initial-advertise-peer-urls", peerURL,
		"--initial-cluster", "revn03="+peerURL, "--log-level", "error")
	if err := cmd.Start(); err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() {
		_ = cmd.Process.Signal(syscall.SIGCONT)
		_ = cmd.Process.Kill()
		_, _ = cmd.Process.Wait()
	})
	endpoint := clientURL[len("http://"):]
	deadline := time.Now().Add(15 * time.Second)
	for time.Now().Before(deadline) {
		if conn, err := net.DialTimeout("tcp", endpoint, 200*time.Millisecond); err == nil {
			_ = conn.Close()
			return &revn03Etcd{cmd: cmd, endpoint: endpoint}
		}
		time.Sleep(100 * time.Millisecond)
	}
	t.Fatal("etcd did not start")
	return nil
}

func (e *revn03Etcd) freeze(t *testing.T) {
	if err := e.cmd.Process.Signal(syscall.SIGSTOP); err != nil {
		t.Fatal(err)
	}
}
func (e *revn03Etcd) thaw(t *testing.T) { _ = e.cmd.Process.Signal(syscall.SIGCONT) }

// E1: normal Resign while etcd stops answering.
func TestRevn03ProbeResignBudgetFrozenEtcd(t *testing.T) {
	etcd := revn03StartEtcd(t)
	cli := realElectionClient(t, etcd.endpoint)
	e := NewElectionFactory(cli).NewElection("/revn03/resign/")
	if err := e.Campaign(context.Background(), "a"); err != nil {
		t.Fatal(err)
	}
	etcd.freeze(t)
	ctx, cancel := context.WithTimeout(context.Background(), 500*time.Millisecond)
	defer cancel()
	started := time.Now()
	done := make(chan error, 1)
	go func() { done <- e.Resign(ctx) }()
	var err error
	select {
	case err = <-done:
	case <-time.After(20 * time.Second):
		err = fmt.Errorf("still blocked after 20s")
	}
	elapsed := time.Since(started)
	etcd.thaw(t)
	t.Logf("Resign(500ms) with frozen etcd: err=%v elapsed=%s is_leader=%v", err, elapsed, e.IsLeader())
	if elapsed > 2*time.Second {
		t.Errorf("Resign ignored its 500ms budget: %s", elapsed)
	}
}

// E4: lease expires while the process is paused; registration must come back.
func TestRevn03ProbeLeaseRecoveryAfterPause(t *testing.T) {
	etcd := revn03StartEtcd(t)
	cli := realElectionClient(t, etcd.endpoint)
	other := realElectionClient(t, etcd.endpoint)
	d := NewDiscovery(cli, "/revn03/rec/", 2)
	d.SetRetryIntervals(200*time.Millisecond, time.Second)
	if err := d.Register(context.Background(), &fetcd.ServiceInfo{ServiceType: "rvn", Sid: 3}); err != nil {
		t.Fatal(err)
	}
	first := d.currentLeaseID()
	// Revoke out of band = lease expired server side while we were paused.
	ctx, cancel := context.WithTimeout(context.Background(), 2*time.Second)
	_, err := other.Revoke(ctx, first)
	cancel()
	if err != nil {
		t.Fatal(err)
	}
	deadline := time.Now().Add(10 * time.Second)
	for time.Now().Before(deadline) {
		ctx, cancel := context.WithTimeout(context.Background(), time.Second)
		resp, err := other.Get(ctx, "/revn03/rec/rvn/3")
		cancel()
		if err == nil && len(resp.Kvs) == 1 && clientv3.LeaseID(resp.Kvs[0].Lease) != first {
			t.Logf("re-registered with lease %x (old %x) after %s", resp.Kvs[0].Lease, first, 10*time.Second-time.Until(deadline))
			_ = d.Deregister(context.Background())
			return
		}
		time.Sleep(100 * time.Millisecond)
	}
	t.Errorf("registration did not come back after the lease vanished")
}

// pausableProxy forwards TCP to target; Pause drops every connection and
// refuses new ones until Resume.
type pausableProxy struct {
	ln     net.Listener
	target string
	mu     sync.Mutex
	paused bool
	conns  []net.Conn
}

func newPausableProxy(t *testing.T, target string) *pausableProxy {
	ln, err := net.Listen("tcp", fmt.Sprintf("127.0.0.1:%d", revn03Port(t)))
	if err != nil {
		t.Fatal(err)
	}
	p := &pausableProxy{ln: ln, target: target}
	t.Cleanup(func() { _ = ln.Close(); p.Pause() })
	go func() {
		for {
			c, err := ln.Accept()
			if err != nil {
				return
			}
			p.mu.Lock()
			if p.paused {
				p.mu.Unlock()
				_ = c.Close()
				continue
			}
			up, err := net.Dial("tcp", target)
			if err != nil {
				p.mu.Unlock()
				_ = c.Close()
				continue
			}
			p.conns = append(p.conns, c, up)
			p.mu.Unlock()
			go func() { _, _ = io.Copy(up, c); _ = up.Close() }()
			go func() { _, _ = io.Copy(c, up); _ = c.Close() }()
		}
	}()
	return p
}

func (p *pausableProxy) Pause() {
	p.mu.Lock()
	p.paused = true
	conns := p.conns
	p.conns = nil
	p.mu.Unlock()
	for _, c := range conns {
		_ = c.Close()
	}
}

func (p *pausableProxy) Resume() {
	p.mu.Lock()
	p.paused = false
	p.mu.Unlock()
}

// E5: the mirror's watch resumes below a compaction after a network cut.
func TestRevn03ProbeMirrorRecoversFromCompactionAfterCut(t *testing.T) {
	etcd := revn03StartEtcd(t)
	proxy := newPausableProxy(t, etcd.endpoint)
	direct := realElectionClient(t, etcd.endpoint)
	viaProxy, err := clientv3.New(clientv3.Config{Endpoints: []string{proxy.ln.Addr().String()}, DialTimeout: 2 * time.Second})
	if err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() { _ = viaProxy.Close() })
	prefix := "/revn03/mirror/"
	cfg := JSONLocalMirrorConfig[string](prefix)
	cfg.RetryMinInterval, cfg.RetryMaxInterval = 100*time.Millisecond, 500*time.Millisecond
	ctx, cancel := context.WithCancel(context.Background())
	defer cancel()
	mirror, err := NewLocalMirror(ctx, &Client{cli: viaProxy}, cfg)
	if err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() { _ = mirror.(interface{ Close() error }).Close() })
	wctx, wcancel := context.WithTimeout(context.Background(), 5*time.Second)
	if err := mirror.WaitForSync(wctx); err != nil {
		t.Fatal(err)
	}
	wcancel()
	put := func(k, v string) int64 {
		ctx, cancel := context.WithTimeout(context.Background(), 2*time.Second)
		defer cancel()
		resp, err := direct.Put(ctx, prefix+k, `"`+v+`"`)
		if err != nil {
			t.Fatal(err)
		}
		return resp.Header.Revision
	}
	put("a", "1")
	time.Sleep(300 * time.Millisecond)
	proxy.Pause()
	put("a", "2")
	put("b", "x")
	rev := put("unrelated-tail", "z")
	cctx, ccancel := context.WithTimeout(context.Background(), 2*time.Second)
	if _, err := direct.Compact(cctx, rev); err != nil {
		t.Fatal(err)
	}
	ccancel()
	t.Logf("while cut: status=%+v", mirror.Status())
	proxy.Resume()
	deadline := time.Now().Add(20 * time.Second)
	for time.Now().Before(deadline) {
		snap, _ := mirror.Snapshot()
		if snap[prefix+"a"] == "2" && snap[prefix+"b"] == "x" {
			t.Logf("mirror recovered: %v status=%+v", snap, mirror.Status())
			return
		}
		time.Sleep(200 * time.Millisecond)
	}
	snap, _ := mirror.Snapshot()
	t.Errorf("mirror did not recover after compaction: %v status=%+v", snap, mirror.Status())
}
