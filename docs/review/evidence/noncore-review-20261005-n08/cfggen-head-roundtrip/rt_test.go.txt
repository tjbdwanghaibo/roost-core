func stage(t *testing.T) string {
	t.Helper()
	dir := t.TempDir()
	for _, n := range []string{"monster.json", "item.json", "drop.json"} {
		raw, err := os.ReadFile(filepath.Join("data", n))
		if err != nil {
			t.Fatal(err)
		}
		if err := os.WriteFile(filepath.Join(dir, n), raw, 0o644); err != nil {
			t.Fatal(err)
		}
	}
	return dir
}

func open(t *testing.T, dir string, strict bool) (*configdata.Store, *configdata.Snapshot, error) {
	t.Helper()
	r := configdata.NewRegistry()
	MustRegisterGeneratedConfigData(r)
	s := configdata.NewStore(r, dir)
	s.SetStrictJSON(strict)
	snap, err := s.Load(context.Background())
	return s, snap, err
}

func ids(rows []MonsterCfg) []int32 {
	out := make([]int32, 0, len(rows))
	for _, r := range rows {
		out = append(out, r.ID)
	}
	return out
}

func eq(a []int32, b ...int32) bool {
	if len(a) != len(b) {
		return false
	}
	for i := range a {
		if a[i] != b[i] {
			return false
		}
	}
	return true
}

// 前向 ref（monster 声明在 drop/item 之前）、各类型索引、skipempty、显式索引名的真实 JSON 往返。
func TestIndexAndRefRoundTrip(t *testing.T) {
	_, snap, err := open(t, stage(t), false)
	if err != nil {
		t.Fatalf("load: %v", err)
	}
	if got := ids(MonsterBySceneID(snap, math.MaxUint64)); !eq(got, 1) {
		t.Fatalf("uint64 max index = %v", got)
	}
	if got := ids(MonsterBySceneID(snap, 7)); !eq(got, 3) {
		t.Fatalf("scene 7 = %v", got)
	}
	if got := MonsterBySceneID(snap, 0); len(got) != 0 {
		t.Fatalf("skipempty uint64 zero indexed: %v", ids(got))
	}
	if got := ids(MonsterByBoss(snap, true)); !eq(got, 1) {
		t.Fatalf("boss=true = %v", got)
	}
	if got := MonsterByBoss(snap, false); len(got) != 0 {
		t.Fatalf("skipempty bool false indexed: %v", ids(got))
	}
	if got := ids(MonsterByTag(snap, "")); !eq(got, 1, 3) {
		t.Fatalf("string index without skipempty must keep empty values in file order: %v", got)
	}
	if got := ids(MonsterByTag(snap, "elite")); !eq(got, 2) {
		t.Fatalf("tag elite = %v", got)
	}
	if got := ItemByKind(snap, ""); len(got) != 0 {
		t.Fatalf("skipempty string empty indexed: %+v", got)
	}
	if got := ItemByKind(snap, "weapon"); len(got) != 1 || got[0].Code != "sword" {
		t.Fatalf("kind weapon = %+v", got)
	}
	if got := ItemByRarity(snap, false); len(got) != 1 || got[0].Code != "rock" {
		t.Fatalf("explicit index name rarity=false = %+v", got)
	}
	if got := ItemByLevel(snap, -5); len(got) != 1 || got[0].Code != "sword" {
		t.Fatalf("negative int64 index = %+v", got)
	}
	if got := ItemByLevel(snap, 0); len(got) != 1 || got[0].Code != "rock" {
		t.Fatalf("zero int64 without skipempty = %+v", got)
	}
	if got := DropByWeight(snap, math.MaxUint32); len(got) != 1 || got[0].ID != 20 {
		t.Fatalf("uint32 max index = %+v", got)
	}
	tbl, _ := MonsterTableFrom(snap)
	if m, ok := tbl.Get(3); !ok || m.ItemCode != "" || m.DropID != 10 {
		t.Fatalf("monster 3 = %+v ok=%v", m, ok)
	}
}

func rewrite(t *testing.T, dir, name, body string) {
	t.Helper()
	if err := os.WriteFile(filepath.Join(dir, name), []byte(body), 0o644); err != nil {
		t.Fatal(err)
	}
}

func TestRefRequiredRejectionsKeepLiveSnapshotAndIndexes(t *testing.T) {
	cases := []struct {
		name, file, body, want string
	}{
		{"dangling string ref", "monster.json", `[{"id":1,"drop_id":10,"item_code":"axe"}]`, "ItemCode references missing item key axe"},
		{"required zero", "monster.json", `[{"id":1,"drop_id":0,"item_code":"sword"}]`, "required reference to drop is zero"},
		{"required missing key", "monster.json", `[{"id":1,"item_code":"sword"}]`, "required reference to drop is zero"},
		{"target row removed", "drop.json", `[{"id":20,"weight":1}]`, "DropID references missing drop key 10"},
		{"string key removed", "item.json", `[{"code":"rock"}]`, "ItemCode references missing item key sword"},
	}
	for _, tc := range cases {
		t.Run(tc.name, func(t *testing.T) {
			dir := stage(t)
			store, snap, err := open(t, dir, false)
			if err != nil {
				t.Fatal(err)
			}
			rewrite(t, dir, tc.file, tc.body)
			if _, err := store.Reload(context.Background()); err == nil || !strings.Contains(err.Error(), tc.want) {
				t.Fatalf("reload err = %v, want %q", err, tc.want)
			}
			cur := store.Current()
			if cur.Version != snap.Version || !eq(ids(MonsterBySceneID(cur, math.MaxUint64)), 1) {
				t.Fatalf("live snapshot/index moved after refused reload: v%d→v%d", snap.Version, cur.Version)
			}
			// 用新进程加载同一份坏数据也必须拒绝（不是只有 reload 拦）。
			if _, _, err := open(t, dir, false); err == nil {
				t.Fatal("cold load accepted the bad data")
			}
		})
	}
}

// 分组导出：服务端绑定省掉客户端字段；非 strict 加载忽略数据里的客户端列，strict 拒绝（文档化行为）。
func TestGroupOmittedFieldVsStrictJSON(t *testing.T) {
	dir := stage(t)
	if _, _, err := open(t, dir, false); err != nil {
		t.Fatalf("non-strict: %v", err)
	}
	if _, _, err := open(t, dir, true); err == nil || !strings.Contains(err.Error(), "icon") {
		t.Fatalf("strict with client-only column = %v, want unknown field icon", err)
	}
}

// 合法 reload 后新旧快照的索引各自稳定。
func TestReloadMovesIndexesAndPinnedSnapshotKeepsOld(t *testing.T) {
	dir := stage(t)
	store, old, err := open(t, dir, false)
	if err != nil {
		t.Fatal(err)
	}
	rewrite(t, dir, "monster.json", `[{"id":1,"scene_id":5,"drop_id":20,"boss":true,"tag":"x"},{"id":9,"scene_id":5,"drop_id":10,"tag":"x"}]`)
	next, err := store.Reload(context.Background())
	if err != nil {
		t.Fatal(err)
	}
	if got := ids(MonsterBySceneID(next, 5)); !eq(got, 1, 9) {
		t.Fatalf("new scene 5 = %v", got)
	}
	if got := MonsterBySceneID(next, math.MaxUint64); len(got) != 0 {
		t.Fatalf("stale index entry survived reload: %v", ids(got))
	}
	if got := ids(MonsterBySceneID(old, math.MaxUint64)); !eq(got, 1) {
		t.Fatalf("pinned old snapshot index changed: %v", got)
	}
	if got := ids(MonsterByTag(next, "")); len(got) != 0 {
		t.Fatalf("new empty tag = %v", got)
	}
}
