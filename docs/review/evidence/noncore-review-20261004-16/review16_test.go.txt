package mongotest

import (
 "context"
 "errors"
 "testing"
 fmongo "github.com/tjbdwanghaibo/roost-core/mongo"
 "go.mongodb.org/mongo-driver/v2/bson"
)

func TestReview16MongoBoundaries(t *testing.T){
 ctx:=context.Background()
 for _,op:=range []string{"read","set","unset"}{for _,nested:=range []bool{false,true}{name:=op+"_flat_control";if nested{name=op+"_nested"};t.Run(name,func(t *testing.T){
  c:=NewClient().Collection("r16",name);doc:=bson.M{"_id":int64(1),"n":int64(1),"keep":"yes"};path:="n"
  if nested{doc=bson.M{"_id":int64(1),"meta":bson.M{"n":int64(1),"keep":"yes"}};path="meta.n"}
  if err:=c.Seed(doc);err!=nil{t.Fatal(err)}
  if op=="read"{count,err:=c.CountDocuments(ctx,bson.M{path:int64(1)});if err!=nil||count!=1{t.Fatalf("existing nested field not matched: count=%d err=%v",count,err)};return}
  update:=bson.M{"$set":bson.M{path:int64(9)}};if op=="unset"{update=bson.M{"$unset":bson.M{path:1}}}
  if _,err:=c.UpdateOne(ctx,bson.M{"_id":int64(1)},update);err!=nil{t.Fatal(err)}
  var got struct{N int64 `bson:"n"`;Keep string `bson:"keep"`;Meta struct{N int64 `bson:"n"`;Keep string `bson:"keep"`} `bson:"meta"`}
  if err:=c.FindOne(ctx,bson.M{"_id":int64(1)},&got);err!=nil{t.Fatal(err)}
  n,keep:=got.N,got.Keep;if nested{n,keep=got.Meta.N,got.Meta.Keep};want:=int64(9);if op=="unset"{want=0}
  if n!=want||keep!="yes"{t.Fatalf("dotted %s changed wrong fields: n=%d keep=%q",op,n,keep)}
 })}}
 for _,duplicate:=range []bool{false,true}{name:="unique_distinct_control";if duplicate{name="unique_existing_duplicates"};t.Run(name,func(t *testing.T){
  c:=NewClient().Collection("r16",name);for _,id:=range []int64{1,2}{value:=id;if duplicate{value=1};if err:=c.Seed(bson.M{"_id":id,"name":value});err!=nil{t.Fatal(err)}}
  err:=c.EnsureIndexes(ctx,[]fmongo.IndexModel{{Name:"name_unique",Keys:bson.D{{Key:"name",Value:1}},Unique:true}})
  if duplicate{if !errors.Is(err,fmongo.ErrDuplicateKey){t.Fatalf("unique index accepted existing duplicates: err=%v has=%v",err,c.HasIndex("name"))}}else if err!=nil{t.Fatal(err)}
 })}
 for _,unsupported:=range []bool{false,true}{name:="bulk_valid_control";if unsupported{name="bulk_unsupported_preflight"};t.Run(name,func(t *testing.T){
  c:=NewClient().Collection("r16",name);models:=[]fmongo.WriteModel{{Type:fmongo.WriteModelInsertOne,Document:bson.M{"_id":int64(1)}}}
  if unsupported{models=append(models,fmongo.WriteModel{Type:255})}
  _,err:=c.BulkWrite(ctx,models);count,countErr:=c.CountDocuments(ctx,bson.M{})
  if countErr!=nil{t.Fatal(countErr)}
  if unsupported{if !errors.Is(err,ErrUnsupported)||count!=0{t.Fatalf("invalid model caused prefix side effect: count=%d err=%v",count,err)}}else if err!=nil||count!=1{t.Fatalf("count=%d err=%v",count,err)}
 })}
 for _,mode:=range []string{"serial_control","concurrent_existing","concurrent_new_collection"}{t.Run("rollback_"+mode,func(t *testing.T){
  client:=NewClient();c:=client.Collection("r16",mode);if err:=c.Seed(bson.M{"_id":int64(1),"n":int64(1)});err!=nil{t.Fatal(err)}
  session,err:=client.StartSession(ctx);if err!=nil{t.Fatal(err)};defer session.EndSession(ctx)
  abort:=errors.New("abort A");entered:=make(chan struct{});resume:=make(chan struct{});done:=make(chan error,1)
  go func(){done<-session.WithTransaction(ctx,func(tx context.Context)error{if _,err:=c.UpdateOne(tx,bson.M{"_id":int64(1)},bson.M{"$set":bson.M{"n":int64(9)}});err!=nil{return err};close(entered);<-resume;return abort})}()
  <-entered
  target:=c
  if mode=="concurrent_new_collection"{target=client.Collection("r16","created_by_B")}
  if mode!="serial_control"{
   b,err:=client.StartSession(ctx);if err!=nil{close(resume);<-done;t.Fatal(err)}
   err=b.WithTransaction(ctx,func(tx context.Context)error{_,err:=target.InsertOne(tx,bson.M{"_id":int64(2),"n":int64(2)});return err});b.EndSession(ctx)
   if err!=nil{close(resume);<-done;t.Fatal(err)}
  }
  close(resume);if err:=<-done;!errors.Is(err,abort){t.Fatal(err)}
  if mode=="serial_control"{if got:=review16Number(t,c);got!=1{t.Fatalf("serial rollback n=%d",got)};return}
  count,err:=target.CountDocuments(ctx,bson.M{"_id":int64(2)});if err!=nil||count!=1{t.Fatalf("abort A erased committed B: count=%d err=%v",count,err)}
 })}
}

func review16Number(t *testing.T,c *Collection)int64{t.Helper();var doc struct{N int64 `bson:"n"`};if err:=c.FindOne(context.Background(),bson.M{"_id":int64(1)},&doc);err!=nil{t.Fatal(err)};return doc.N}
