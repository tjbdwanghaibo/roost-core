const nestedKind entity.EntityKind = 247
const nestedCollection = "roost_review_20261005_22_nested"
var nestedOnce sync.Once
type nestedEntity struct { *reviewEntity; nested *NestedMigrationDao }

func registerNested() {
 nestedOnce.Do(func() {
  entity.RegisterEntityBuilder(&entity.EntityBuilderParam{
   Category: 1, Kind: nestedKind,
   DaoBuilders: []entity.DaoBuilderFunc{func() entity.DaoInterface {return NewNestedMigrationDao()}},
   Builder: func(p *entity.EntityCreateParam) (entity.IThreadSafeEntity, error) {
    return &nestedEntity{&reviewEntity{EntityBase: entity.NewEntityBaseWithMutex(p.Id, p.Category, false, p.Mutex, p.Kind)}, p.Dao[nestedCollection].(*NestedMigrationDao)}, nil
   },
  })
  for _, from := range []uint32{1, 2} {
   migration.MustRegisterDAO(migration.DAOStep{Collection: nestedCollection, From: from, To: from+1, Apply: func(ctx context.Context, raw []byte) ([]byte, error) {
    if err := ctx.Err(); err != nil {return nil, err}
    var doc bson.M
    decoder := bson.NewDecoder(bson.NewDocumentReader(bytes.NewReader(raw)))
    decoder.DefaultDocumentM()
    if err := decoder.Decode(&doc); err != nil {return nil, err}
    var visit func(any) error
    visit = func(v any) error {
     switch d := v.(type) {
     case bson.M:
      if score, ok := d["score"]; ok {
       if from == 1 { n, err := strconv.ParseInt(score.(string), 10, 64); if err != nil {return err}; d["score"] = n } else {d["score"] = score.(int64)+1}
      }
      for _, child := range d {if err := visit(child); err != nil {return err}}
     case bson.A:
      for _, child := range d {if err := visit(child); err != nil {return err}}
     }
     return nil
    }
    if err := visit(doc); err != nil {return nil, err}
    if from == 2 && doc["mode"] == "bad_nested_type" {doc["node"].(bson.M)["leaf"].(bson.M)["score"] = "invalid int64"}
    doc["_schema"] = from+1
    return bson.Marshal(doc)
   }})
  }
 })
}

func TestGeneratedNestedSchemaMigrationAndPersistedUpdates(t *testing.T) {
 for _, mode := range []string{"success", "bad_nested_type", "invalid_source_number", "null_children"} {
  t.Run(mode, func(t *testing.T) {
   registerNested()
   s := setupConsumerForKind(t, "success", 1, nil, nestedKind)
   var leaf any = bson.M{"score": "41", "label": "kept"}
   if mode == "invalid_source_number" {leaf = bson.M{"score": "not-a-number"}}
   branch := bson.M{"name": "branch", "leaf": leaf, "items": bson.M{"7": leaf}, "kids": bson.A{leaf}}
   if mode == "null_children" {branch = bson.M{"name": "branch", "leaf": nil, "items": bson.M{"7": nil}, "kids": bson.A{nil}}}
   if err := s.client.Collection("review", nestedCollection).Seed(bson.M{"_id": s.id, "_schema": uint32(1), "_version": int64(7), "node": branch, "value": branch, "mode": mode}); err != nil {t.Fatal(err)}
   ctx, cancel := context.WithTimeout(context.Background(), 3*time.Second)
   defer cancel()
   value, err := s.repository.LoadEntity(ctx, s.id, nestedKind)
   stored := func() coredata.RawDocument {
    docs, err := s.store.Load(ctx, coredata.LoadSpec{Database: "review", Resource: nestedCollection, Filter: map[string]any{"_id": s.id}})
    if err != nil || len(docs) != 1 {t.Fatalf("stored nested: %v docs=%d", err, len(docs))}
    return docs[0]
   }
   if mode == "bad_nested_type" || mode == "invalid_source_number" {
    doc := stored()
    if err == nil || value != nil || s.manager.Get(s.id) != nil || s.committer.calls.Load() != 0 || doc.Schema != 1 || doc.Version != 7 || s.committer.Stats().WALUnacked != 0 {t.Fatalf("bad nested output/source published: err=%v doc=%+v commits=%d", err, doc, s.committer.calls.Load())}
    return
   }
   if err != nil {t.Fatal(err)}
   d := value.(*nestedEntity).nested
   if d.Dirty().Dirty() || d.DirtyTracker().Version() != 8 || s.committer.calls.Load() != 1 {t.Fatal("migration tracker/commit not restored")}
   if mode == "null_children" {
    a, ok := d.GetNode().GetItems(7)
    b, ok2 := d.GetValue().GetKids(0)
    if d.GetNode().GetLeaf() != nil || a != nil || b != nil || !ok || !ok2 {t.Fatal("null descendants not preserved")}
   } else {
    for _, p := range []*MigrationBranch{d.GetNode(), d.GetValue()} {
     a, _ := p.GetItems(7); b, _ := p.GetKids(0)
     if p.GetLeaf().GetScore() != 42 || a.GetScore() != 42 || b.GetScore() != 42 || p.GetLeaf().GetLabel() != "kept" {t.Fatal("nested type conversion lost data")}
    }
    if _, err := nest.RunIsolatedTransaction(ctx, s.committer, "post_migration_child", func() (any, error) {d.GetValue().GetLeaf().SetScore(99); return nil, nil}); err != nil {t.Fatal(err)}
    if err := s.committer.Flush(ctx); err != nil {t.Fatal(err)}
   }
   repo, err := engine.NewEntityRepository(entity.NewEntityManager(), s.store, nil, nil)
   if err != nil {t.Fatal(err)}
   reloaded, err := repo.LoadEntity(ctx, s.id, nestedKind)
   if err != nil {t.Fatal(err)}
   fresh := reloaded.(*nestedEntity).nested
   if mode == "success" && (fresh.GetValue().GetLeaf().GetScore() != 99 || fresh.GetNode().GetLeaf().GetScore() != 42 || fresh.DirtyTracker().Version() != 9) {t.Fatal("deep update not persisted/reloaded independently")}
  })
 }
}
