package match

import (
	"context"
	"encoding/json"
	"fmt"
	"os"
	"testing"
	"time"

	fredis "github.com/tjbdwanghaibo/roost-core/redis"
	driver "github.com/tjbdwanghaibo/roost-core/redis/driver"
)

// This is evidence for the already-recorded retention boundary, not a new
// bug or a claim that 64 requests establish production capacity.
func TestReview7MatchTerminalProofRetention(t *testing.T) {
	for _, backend := range []string{"memory", "redis"} {
		t.Run(backend, func(t *testing.T) {
			now := time.Unix(1700000000, 0)
			cfg := Config{Now: func() time.Time { return now }}
			var store Store
			var err error
			if backend == "memory" {
				store, err = NewMemoryStore(cfg)
			} else {
				addr := os.Getenv("ROOST_REVIEW_REDIS")
				if addr == "" {
					t.Fatal("real Redis required")
				}
				client, e := driver.NewClient(fredis.DefaultConfig(addr))
				if e != nil {
					t.Fatal(e)
				}
				t.Cleanup(func() { _ = client.Close() })
				prefix := fmt.Sprintf("review7:match:%d", time.Now().UnixNano())
				store, err = NewRedisStore(client, prefix, cfg)
				t.Cleanup(func() {
					_, _ = client.Del(context.Background(), prefix+":queue:"+(Queue{Mode: "arena", GroupSize: 2}).Key())
				})
			}
			if err != nil {
				t.Fatal(err)
			}
			ctx := context.Background()
			q := Queue{Mode: "arena", GroupSize: 2}
			var original Ticket
			for i := 0; i < 64; i++ {
				subject := Subject{Kind: "player", ID: int64(i + 1)}
				ticket, e := store.Enqueue(ctx, q, subject, fmt.Sprintf("request-%d", i))
				if e != nil {
					t.Fatal(e)
				}
				ticket, e = store.Cancel(ctx, q, ticket.ID, subject)
				if e != nil {
					t.Fatal(e)
				}
				if i == 0 {
					original = ticket
				}
			}
			now = now.Add(365 * 24 * time.Hour)
			n, err := store.Sweep(ctx, q, MaxPageSize)
			if err != nil || n != 0 {
				t.Fatalf("terminal sweep: %d %v", n, err)
			}
			value, found, err := store.(*queueStore).state.Get(ctx, q.Key())
			if err != nil || !found {
				t.Fatalf("state: %v %v", found, err)
			}
			raw, _ := json.Marshal(value.Value)
			t.Logf("waiting=%d tickets=%d requests=%d matches=%d json_bytes=%d", len(value.Value.Waiting), len(value.Value.Tickets), len(value.Value.Requests), len(value.Value.Matches), len(raw))
			if len(value.Value.Waiting) != 0 || len(value.Value.Tickets) != 64 || len(value.Value.Requests) != 64 {
				t.Fatal("retention observation changed")
			}
			replay, err := store.Enqueue(ctx, q, original.Subject, "request-0")
			if err != nil || replay.ID != original.ID || replay.State != TicketCancelled {
				t.Fatalf("terminal request proof: %+v %v", replay, err)
			}
		})
	}
}
