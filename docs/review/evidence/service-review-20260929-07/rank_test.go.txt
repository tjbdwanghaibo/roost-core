func review7ClusterRank(t *testing.T, prefix string) (*Mod, error) {
	t.Helper()
	addr := os.Getenv("ROOST_REVIEW_CLUSTER")
	if addr == "" {
		t.Fatal("ROOST_REVIEW_CLUSTER is required")
	}
	cfg := viper.New()
	cfg.Set("redis.cluster_addrs", addr)
	cfg.Set("rank.key_prefix", prefix)
	mod := NewMod(nil)
	if err := mod.Init(cfg); err != nil {
		return mod, err
	}
	rcfg := fredis.DefaultConfig("")
	rcfg.ClusterAddrs = strings.Split(addr, ",")
	client, err := driver.NewClient(rcfg)
	if err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() { _ = client.Close() })
	registry := app.NewRegistry(cfg)
	if err := mods.RegisterAll(registry, mods.Capability{Name: mods.ModRedis, Value: client}); err != nil {
		t.Fatal(err)
	}
	if err := mod.Provide(registry); err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() {
		_, _ = client.Del(context.Background(), mod.store.boardKey(arena()))
		_, _ = client.Del(context.Background(), mod.store.ownerKey(arena()))
	})
	return mod, nil
}

func TestReview7RankClusterRejectsInvalidPrefix(t *testing.T) {
	for _, suffix := range []string{"rank", "{}:rank"} {
		t.Run(suffix, func(t *testing.T) {
			prefix := fmt.Sprintf("review7:%d:%s", time.Now().UnixNano(), suffix)
			mod, err := review7ClusterRank(t, prefix)
			if err != nil {
				return
			}
			_, err = mod.store.Submit(context.Background(), arena(), Score{OwnerID: 1, Value: 10}, UpdateAdd, "one")
			if err == nil || !strings.Contains(err.Error(), "CROSSSLOT") {
				t.Fatalf("expected actual cluster slot error, got %v", err)
			}
			t.Fatalf("Init/Provide accepted untagged keys; real Submit fails: %v", err)
		})
	}
}

func TestReview7RankTaggedClusterLifecycle(t *testing.T) {
	mod, err := review7ClusterRank(t, fmt.Sprintf("review7:{rank-%d}", time.Now().UnixNano()))
	if err != nil {
		t.Fatal(err)
	}
	s := mod.store
	ctx := context.Background()
	for i := 0; i < 2; i++ {
		e, err := s.Submit(ctx, arena(), Score{OwnerID: 1, Value: 10, Tie: 1}, UpdateAdd, "same")
		if err != nil || e.Score.Value != 10 {
			t.Fatalf("dedup: %+v %v", e, err)
		}
	}
	p, err := s.Page(ctx, arena(), 0, 10)
	if err != nil || p.Total != 1 || len(p.Entries) != 1 {
		t.Fatalf("page: %+v %v", p, err)
	}
	if err := s.Remove(ctx, arena(), 1); err != nil {
		t.Fatal(err)
	}
	if err := s.Remove(ctx, arena(), 1); err != nil {
		t.Fatal(err)
	}
	for _, id := range []int64{2, 3} {
		if _, err := s.Submit(ctx, arena(), Score{OwnerID: id, Value: 10}, UpdateSet, ""); err != nil {
			t.Fatal(err)
		}
	}
	if err := s.Reset(ctx, arena()); err != nil {
		t.Fatal(err)
	}
	size, err := s.Size(ctx, arena())
	if err != nil || size != 0 {
		t.Fatalf("reset: %d %v", size, err)
	}
}

type review7SwapGate struct {
	RedisClient
	entered, release chan struct{}
	once             sync.Once
}

func (c *review7SwapGate) Eval(ctx context.Context, script string, keys []string, args ...any) (any, error) {
	if script == swapScript && fmt.Sprint(args[3]) == "after-maintenance" {
		c.once.Do(func() {
			close(c.entered)
			select {
			case <-c.release:
			case <-ctx.Done():
			}
		})
	}
	return c.RedisClient.Eval(ctx, script, keys, args...)
}

func TestReview7RankMaintenanceOverlappingSubmit(t *testing.T) {
	for _, reset := range []bool{false, true} {
		t.Run(fmt.Sprintf("reset_%v", reset), func(t *testing.T) {
			if os.Getenv("ROOST_REVIEW_REDIS") == "" {
				t.Fatal("real Redis required")
			}
			s := reviewRankStore(t)
			ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
			defer cancel()
			if _, err := s.Submit(ctx, arena(), Score{OwnerID: 1, Value: 100, Tie: 1}, UpdateSet, "before"); err != nil {
				t.Fatal(err)
			}
			gate := &review7SwapGate{RedisClient: s.client, entered: make(chan struct{}), release: make(chan struct{})}
			s.client = gate
			var unblocked sync.Once
			unblock := func() { unblocked.Do(func() { close(gate.release) }) }
			t.Cleanup(unblock)
			done := make(chan error, 1)
			go func() {
				_, err := s.Submit(ctx, arena(), Score{OwnerID: 1, Value: 5, Tie: 2}, UpdateAdd, "after-maintenance")
				done <- err
			}()
			select {
			case <-gate.entered:
			case <-ctx.Done():
				t.Fatal(ctx.Err())
			}
			if reset {
				if err := s.Reset(ctx, arena()); err != nil {
					t.Fatal(err)
				}
			} else {
				if err := s.Remove(ctx, arena(), 1); err != nil {
					t.Fatal(err)
				}
			}
			unblock()
			select {
			case err := <-done:
				if err != nil {
					t.Fatal(err)
				}
			case <-ctx.Done():
				t.Fatal(ctx.Err())
			}
			e, err := s.Submit(ctx, arena(), Score{OwnerID: 1, Value: 5, Tie: 2}, UpdateAdd, "after-maintenance")
			if err != nil || e.Score.Value != 5 {
				t.Fatalf("stale base or lost new proof: %+v %v", e, err)
			}
		})
	}
}
