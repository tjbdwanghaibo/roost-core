package player

import (
	"context"
	"errors"
	"testing"

	"github.com/tjbdwanghaibo/roost-core/nest"
	"example.com/a1demo/game/equipment"
)

type benchCommitter struct{}

func (benchCommitter) Commit(context.Context, nest.CommitRecord) error { return nil }

var errBenchFail = errors.New("fail")

// One undo transaction that levels up (Base + Final written) and commits.
func BenchmarkA1AttributeLevelUpCommit(b *testing.B) {
	t := &testing.T{}
	subject := newAttributePlayer(t, 7901)
	b.ReportAllocs()
	b.ResetTimer()
	for i := 0; i < b.N; i++ {
		if _, err := nest.RunIsolatedTransaction(context.Background(), benchCommitter{}, "level_up", func() (any, error) {
			return subject.ProfileComp().AddExp(ExpPerLevel)
		}); err != nil {
			b.Fatal(err)
		}
	}
}

// The same transaction failing after the level-up: the rollback path.
func BenchmarkA1AttributeLevelUpRollback(b *testing.B) {
	t := &testing.T{}
	subject := newAttributePlayer(t, 7902)
	b.ReportAllocs()
	b.ResetTimer()
	for i := 0; i < b.N; i++ {
		if _, err := nest.RunIsolatedTransaction(context.Background(), benchCommitter{}, "level_up", func() (any, error) {
			if _, err := subject.ProfileComp().AddExp(ExpPerLevel); err != nil {
				return nil, err
			}
			return nil, errBenchFail
		}); !errors.Is(err, errBenchFail) {
			b.Fatal(err)
		}
	}
}

// Equip / unequip the sword in turn: RefreshGear moves Gear and Final.
func BenchmarkA1AttributeRefreshGearCommit(b *testing.B) {
	t := &testing.T{}
	subject := newAttributePlayer(t, 7903)
	const ironSword = int64(2001)
	if _, err := nest.RunIsolatedTransaction(context.Background(), benchCommitter{}, "grant", func() (any, error) {
		return subject.BagComp().AddItem(ironSword, 1)
	}); err != nil {
		b.Fatal(err)
	}
	slot, _ := equipment.SlotFor(ironSword)
	b.ReportAllocs()
	b.ResetTimer()
	for i := 0; i < b.N; i++ {
		if _, err := nest.RunIsolatedTransaction(context.Background(), benchCommitter{}, "toggle", func() (any, error) {
			if i%2 == 0 {
				if _, err := subject.BagComp().RemoveItem(ironSword, 1); err != nil {
					return nil, err
				}
				if _, _, err := subject.EquipmentComp().Equip(ironSword, 1); err != nil {
					return nil, err
				}
			} else {
				if _, ok := subject.EquipmentComp().Unequip(slot); !ok {
					return nil, errors.New("nothing to unequip")
				}
				if _, err := subject.BagComp().AddItem(ironSword, 1); err != nil {
					return nil, err
				}
			}
			subject.AttributeComp().RefreshGear()
			return nil, nil
		}); err != nil {
			b.Fatal(err)
		}
	}
}
