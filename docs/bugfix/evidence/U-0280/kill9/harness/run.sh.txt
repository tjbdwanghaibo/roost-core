#!/usr/bin/env bash
# U-0280 kill -9 演练：单 sid（1000），一个机器人连续赠礼；每轮在一次 debit 的 Nest 事务写进 WAL 之后
# 立即 kill -9 game 进程并马上重启；最后等全部 saga 终结，统计每个 saga 每个方向的成功回执数与背包数。
#   run.sh <tag> [cycles]
set -uo pipefail
tag=$1; cycles=${2:-3}
here=$(cd "$(dirname "$0")" && pwd)
proj=$here/$tag/proj
out=$here/$tag/out
mkdir -p "$out"
source "$HOME/.roost-it/roost-dataengine-it/env.sh" >/dev/null 2>&1
export GOWORK=off SID=1000 READY_TIMEOUT=150
cd "$proj"
uri=$ROOST_DATAENGINE_IT_MONGO_URI
log() { printf '%s %s\n' "$(date -u +%H:%M:%S)" "$*" | tee -a "$out/run.log"; }

log "start all services ($tag)"
sh deploy/dev/run.sh start >"$out/start.log" 2>&1 || { log "start failed"; tail -40 "$out/start.log"; exit 1; }
for cycle in $(seq 1 "$cycles"); do
  log "cycle $cycle: robot gifting"
  ./bin/loadtest -count 1 -scenario u0280 -scenarios loadtest/scenarios -account-nats "$ROOST_DATAENGINE_IT_NATS_URL" \
    -nats-prefix "$tag" -server-id 1000 -endpoint 127.0.0.1:7000 -duration 180s >"$out/robot-$cycle.log" 2>&1 &
  robot=$!
  sleep 4
  pid=$(cat .dev/game.pid)
  ./bin/u0280watch -mongo "$uri" -db "${tag}_game" -wal data/wal/dataengine -pid "$pid" -timeout 60s | tee -a "$out/kills.log"
  cp .dev/game.log "$out/game-cycle$cycle-killed.log"
  log "cycle $cycle: restarting game immediately"
  sh deploy/dev/run.sh start game >"$out/restart-$cycle.log" 2>&1 || { log "restart failed"; tail -40 "$out/restart-$cycle.log"; }
  kill "$robot" 2>/dev/null; wait "$robot" 2>/dev/null
  log "cycle $cycle: game ready again"
done
log "settle: waiting for every gift saga to end"
for i in $(seq 1 90); do
  open=$(/opt/homebrew/bin/mongosh --quiet "$uri" --eval "db.getSiblingDB('${tag}_saga').getCollection('_sagas').countDocuments({status:{\$in:[1,2,3]}})" 2>/dev/null | tail -1)
  [ "$open" = "0" ] && break
  sleep 5
done
log "settle: open sagas=$open"
sleep 3
cp .dev/game.log "$out/game-final.log"
TAG_NAME=$tag /opt/homebrew/bin/mongosh --quiet "$uri" --file "$here/analyze.js" | tee "$out/analysis.txt"
log "late_after_abandon log lines: $(cat "$out"/game-*.log | grep -c "succeeded after the coordinator abandoned")"
log "expired/superseded acks: $(cat "$out"/game-*.log | grep -c "attempt will not run\|step command expired before it ran")"
log "not-waiting drops: $(cat "$out"/game-*.log | grep -c "step is not waiting for a result")"
log "stop all services"
sh deploy/dev/run.sh stop >"$out/stop.log" 2>&1
