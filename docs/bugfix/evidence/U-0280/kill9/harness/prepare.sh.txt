#!/usr/bin/env bash
# U-0280 kill -9 演练：从给定的 roost-core 树生成 game-demo，并把它隔离到本次前缀。
#   prepare.sh <tag> <roost-core-dir> <redis-db>
# tag 只用小写字母数字（u0280a / u0280b）：库名 <tag>_game / <tag>_saga / <tag>_remote，
# 流 <TAG>_SAGA / <TAG>_EFFECTS / <TAG>_SYNC，NATS 前缀 <tag>，Redis 键前缀 <tag>: 且用独立 db。
set -euo pipefail
tag=$1; core=$2; rdb=$3
TAG=$(printf '%s' "$tag" | tr '[:lower:]' '[:upper:]')
here=$(cd "$(dirname "$0")" && pwd)
proj=$here/$tag/proj
source "$HOME/.roost-it/roost-dataengine-it/env.sh" >/dev/null 2>&1
export GOWORK=off
rm -rf "$here/$tag"; mkdir -p "$here/$tag"
(cd "$core" && go run ./codegen/cmd/roost project new u0280demo -module example.com/u0280demo \
  -mods configdata,mongo,nats,dataengine,nest -template game-demo -out "$proj" >/dev/null)
cd "$proj"
go mod edit -replace "github.com/tjbdwanghaibo/roost-core=$core"
go mod tidy >/dev/null 2>&1

mongo_uri=$ROOST_DATAENGINE_IT_MONGO_URI
nats_url=$ROOST_DATAENGINE_IT_NATS_URL
redis_addr=$ROOST_DATAENGINE_IT_REDIS_ADDR
for f in configs/service/config.*.yaml; do
  case "$f" in *.prod.example.yaml) continue ;; esac
  python3 - "$f" "$tag" "$TAG" "$mongo_uri" "$nats_url" "$redis_addr" "$rdb" <<'PY'
import sys, re
path, tag, TAG, mongo, nats, redis, rdb = sys.argv[1:]
s = open(path).read()
s = s.replace("mongodb://127.0.0.1:27017/?replicaSet=rs0", mongo)
s = s.replace("nats://127.0.0.1:4222", nats)
s = re.sub(r"(?m)^(  addr: )127\.0\.0\.1:6379$", r"\g<1>" + redis, s)
s = re.sub(r"(?m)^(redis:\n(?:  .*\n)*?  db: )0$", r"\g<1>" + rdb, s)
s = s.replace("key_prefix: roost:", "key_prefix: " + tag + ":")
s = re.sub(r"(?m)^(  prefix: )roost$", r"\g<1>" + tag, s)
s = s.replace("roost.saga", tag + ".saga").replace("roost.effect", tag + ".effect").replace("roost.sync", tag + ".sync")
s = s.replace("ROOST_SAGA", TAG + "_SAGA").replace("ROOST_EFFECTS", TAG + "_EFFECTS").replace("ROOST_SYNC", TAG + "_SYNC")
s = re.sub(r"(?m)^(  database: )saga$", r"\g<1>" + tag + "_saga", s)
s = re.sub(r"(?m)^(  database: )game$", r"\g<1>" + tag + "_game", s)
s = re.sub(r"(?m)^(    database: )remote_entity$", r"\g<1>" + tag + "_remote", s)
s = s.replace('snapshot_l2_key_prefix: ""', "snapshot_l2_key_prefix: " + tag)
s = s.replace("8589934592", "268435456")
open(path, "w").write(s)
PY
done
# DAO 库名是编译期常量：标记与生成物一起改。
sed -i '' "s/ db=game/ db=${tag}_game/" db/def/*.go
sed -i '' "s/DaoDBName \(  *\)= \"game\"/DaoDBName \1= \"${tag}_game\"/" db/gen_*_dao.go
grep -h "DaoDBName.*=" db/gen_*_dao.go
# 残留检查：开发配置里不应再有共享的库名、流名、前缀或本机默认地址。
if grep -nE "27017|:4222|:6379|ROOST_|roost\.(saga|effect|sync)|database: (game|saga|remote_entity)$|key_prefix: roost:|^  prefix: roost$" configs/service/config.*.yaml | grep -v prod.example | grep -v '^\S*:[0-9]*:\s*#'; then
  echo "isolation incomplete" >&2; exit 1
fi
go build -o bin/app . && go build -o bin/loadtest ./cmd/loadtest
echo "prepared $proj"
mkdir -p cmd/u0280watch && cp "$here/watcher/main.go" cmd/u0280watch/ && go build -o bin/u0280watch ./cmd/u0280watch
cp "$here/u0280.yaml" loadtest/scenarios/
