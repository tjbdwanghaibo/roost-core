package db

	"context"

	"os"
	"testing"
	"time"

	fredis "github.com/tjbdwanghaibo/roost-core/redis"
	redisdriver "github.com/tjbdwanghaibo/roost-core/redis/driver"
)

func TestGeneratedRefHMapPatchConsumer(t *testing.T) {
	addr:=os.Getenv("ROOST_REDIS_TEST_ADDR")
	if addr==""{t.Fatal("consumer requires an isolated Redis")}
	client,err:=redisdriver.NewClient(fredis.DefaultConfig(addr));if err!=nil{t.Fatal(err)};defer client.Close()
	ctx,cancel:=context.WithTimeout(context.Background(),10*time.Second);defer cancel()
	dao:=NewRedisRecordRedisDAO(client,"roost:refconsumer:"+rand.Text(),time.Minute)
	defer dao.DeleteRecord(context.Background(),1)
	if err:=dao.SetRecord(ctx,Record{ID:1});err!=nil{t.Fatal(err)}
	if err:=dao.PatchRecord(ctx,1,"Meta.Label","visible");err!=nil{t.Fatal(err)}
	got,held,err:=dao.GetRecord(ctx,1)
	if err!=nil||!held||got.Meta==nil||got.Meta.Label!="visible"{t.Fatalf("generated Patch not visible: %+v %v %v",got,held,err)}
	if err:=dao.PatchRecord(ctx,1,"Meta.Label","updated");err!=nil{t.Fatal(err)}
	got,held,err=dao.GetRecord(ctx,1)
	if err!=nil||!held||got.Meta==nil||got.Meta.Label!="updated"{t.Fatalf("existing Patch failed: %+v %v %v",got,held,err)}
}

type unknownRedis struct { fredis.IRedis; after func()error }
func (r *unknownRedis) Eval(ctx context.Context,script string,keys []string,args ...any)(any,error){
 value,err:=r.IRedis.Eval(ctx,script,keys,args...);if err!=nil{return value,err}
 if r.after!=nil{hook:=r.after;r.after=nil;if err:=hook();err!=nil{return nil,err};return nil,context.DeadlineExceeded};return value,nil
}

func TestGeneratedRefHMapUnknownWriteConsumer(t *testing.T){
 addr:=os.Getenv("ROOST_REDIS_TEST_ADDR");if addr==""{t.Fatal("requires isolated Redis")}
 client,err:=redisdriver.NewClient(fredis.DefaultConfig(addr));if err!=nil{t.Fatal(err)};defer client.Close()
 ctx,cancel:=context.WithTimeout(context.Background(),10*time.Second);defer cancel()
 prefix:="roost:unknownconsumer:"+rand.Text()
 authority:=NewRedisRecordRedisDAO(client,prefix,time.Minute);defer authority.DeleteRecord(context.Background(),1)
 if err:=authority.SetRecord(ctx,Record{ID:1,Version:1});err!=nil{t.Fatal(err)}
 wrapped:=&unknownRedis{IRedis:client,after:func()error{return authority.SetRecord(ctx,Record{ID:1,Version:3})}}
 writer:=NewRedisRecordRedisDAO(wrapped,prefix,time.Minute)
 err=writer.SetRecord(ctx,Record{ID:1,Version:2})
 got,held,readErr:=authority.GetRecord(ctx,1)
 if !errors.Is(err,context.DeadlineExceeded)||readErr!=nil||!held||got.Version!=3{t.Fatalf("err=%v read=%v held=%v got=%+v",err,readErr,held,got)}
}
