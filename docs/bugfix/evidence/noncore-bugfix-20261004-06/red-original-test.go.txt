package cache

import (
	"context"
	"errors"
	"strconv"
	"sync/atomic"
	"testing"
	"time"
)

// These probes exercise public stores; dependencies only control errors or
// arrival order. They do not replace the cache admission implementation.
type admissionPolicyRemote struct {
	Store[int, staleValue]
	getErr, deleteErr error
}

func (s *admissionPolicyRemote) Get(ctx context.Context, key int) (staleValue, bool, error) {
	if s.getErr != nil {
		return staleValue{}, false, s.getErr
	}
	return s.Store.Get(ctx, key)
}
func (s *admissionPolicyRemote) Delete(ctx context.Context, key int) error {
	if s.deleteErr != nil {
		return s.deleteErr
	}
	return s.Store.Delete(ctx, key)
}

func TestCacheAdmissionFatalRemotePolicy(t *testing.T) {
	for _, operation := range []string{"get", "delete", "set"} {
		t.Run(operation, func(t *testing.T) {
			ctx := context.Background()
			cfg := staleConfig()
			local := NewAtomicLocalStore(AtomicLocalConfig[int, staleValue]{StoreConfig: cfg, Shards: 1})
			remoteStore := NewAtomicLocalStore(AtomicLocalConfig[int, staleValue]{StoreConfig: cfg, Shards: 1})
			remote := &admissionPolicyRemote{Store: remoteStore}
			loads := 0
			if operation == "get" {
				remote.getErr = ErrConflictingWrite
			}
			if operation == "delete" {
				remote.deleteErr = ErrConflictingWrite
			}
			store := NewReadThroughStore[int, staleValue](local, remote, func(context.Context, int) (staleValue, bool, error) {
				loads++
				return staleValue{Key: 1, Version: 1, Payload: "fallback"}, true, nil
			}, cfg, ReadThroughOptions{IgnoreRemoteError: true, FatalRemoteError: func(err error) bool { return errors.Is(err, ErrConflictingWrite) }})
			var err error
			switch operation {
			case "get":
				_, _, err = store.Get(ctx, 1)
			case "delete":
				_ = local.Set(ctx, staleValue{Key: 1, Version: 2, Payload: "held"})
				err = store.Delete(ctx, 1)
			case "set":
				remoteStore.cfg.Conflict = func(staleValue, staleValue) bool { return true }
				_ = remoteStore.Set(ctx, staleValue{Key: 1, Version: 2, Payload: "held"})
				err = store.Set(ctx, staleValue{Key: 1, Version: 2, Payload: "rejected"})
			}
			value, held, _ := local.Get(ctx, 1)
			t.Logf("operation=%s err=%v loads=%d L1=%+v held=%v", operation, err, loads, value, held)
			if !errors.Is(err, ErrConflictingWrite) {
				t.Fatalf("fatal L2 verdict was degraded: %v", err)
			}
			if operation == "get" && (loads != 0 || held) {
				t.Fatal("fatal read caused fallback or L1 publication")
			}
			if operation == "delete" && !held {
				t.Fatal("failed remote delete erased L1")
			}
		})
	}
}

func TestCacheAdmissionStaleWriteAcrossStores(t *testing.T) {
	builders := map[string]func() Store[int, staleValue]{
		"local": func() Store[int, staleValue] { return NewLocalStore(staleConfig()) },
		"grouped": func() Store[int, staleValue] {
			return NewGroupedLocalStore(func(key int) int { return key / 10 }, staleConfig())
		},
		"raw": func() Store[int, staleValue] {
			return NewRedisRawJSONStore(newRefHMapFakeRedis(), 0, strconv.Itoa, staleConfig())
		},
		"hash": func() Store[int, staleValue] {
			return NewRedisJSONHashStore(newRefHMapFakeRedis(), 0, func(key int) RedisHashKey { return RedisHashKey{Key: "records", Field: strconv.Itoa(key)} }, staleConfig())
		},
		"atomic": func() Store[int, staleValue] {
			return NewAtomicLocalStore(AtomicLocalConfig[int, staleValue]{StoreConfig: staleConfig(), Shards: 1})
		},
		"json": func() Store[int, staleValue] {
			return NewRedisJSONStore(newRefHMapFakeRedis(), 0, strconv.Itoa, staleConfig())
		},
	}
	for name, build := range builders {
		for _, version := range []int64{4, 6} {
			t.Run(name+"/v"+strconv.FormatInt(version, 10), func(t *testing.T) {
				ctx := context.Background()
				store := build()
				if err := store.Set(ctx, staleValue{Key: 1, Version: 5, Payload: "current"}); err != nil {
					t.Fatal(err)
				}
				err := store.Set(ctx, staleValue{Key: 1, Version: version, Payload: "incoming"})
				got, ok, getErr := store.Get(ctx, 1)
				t.Logf("Set(v%d) err=%v Get=%+v", version, err, got)
				if getErr != nil || !ok {
					t.Fatal("stored value disappeared")
				}
				if version < 5 {
					if got.Version != 5 {
						t.Fatal("stale value replaced current")
					}
					if !errors.Is(err, ErrStaleWrite) {
						t.Fatalf("refused write reported success: %v", err)
					}
				} else if err != nil || got.Version != version {
					t.Fatal("newer write did not apply")
				}
			})
		}
	}
}

type admissionCapturedRemote struct {
	Store[int, staleValue]
	first            atomic.Bool
	entered, release chan struct{}
}

func (s *admissionCapturedRemote) Get(ctx context.Context, key int) (staleValue, bool, error) {
	value, ok, err := s.Store.Get(ctx, key)
	if s.first.CompareAndSwap(false, true) {
		close(s.entered)
		<-s.release
	}
	return value, ok, err
}

func TestCacheAdmissionBackfillAdmission(t *testing.T) {
	for _, kind := range []string{"layered", "readthrough"} {
		for _, scenario := range []string{"stale", "conflict", "tombstone"} {
			t.Run(kind+"/"+scenario, func(t *testing.T) {
				ctx := context.Background()
				cfg := staleConfig()
				cfg.Conflict = func(old, next staleValue) bool { return old.Version == next.Version && old.Payload != next.Payload }
				var deleted atomic.Bool
				cfg.Superseded = func(staleValue) bool { return deleted.Load() }
				local := NewAtomicLocalStore(AtomicLocalConfig[int, staleValue]{StoreConfig: cfg, Shards: 1})
				remoteStore := NewAtomicLocalStore(AtomicLocalConfig[int, staleValue]{StoreConfig: staleConfig(), Shards: 1})
				_ = remoteStore.Set(ctx, staleValue{Key: 1, Version: 1, Payload: "captured"})
				remote := &admissionCapturedRemote{Store: remoteStore, entered: make(chan struct{}), release: make(chan struct{})}
				var store Store[int, staleValue]
				if kind == "layered" {
					store = NewLayeredStore[int, staleValue](local, remote, time.Minute, cfg)
				} else {
					store = NewReadThroughStore[int, staleValue](local, remote, nil, cfg, ReadThroughOptions{})
				}
				type result struct {
					value staleValue
					ok    bool
					err   error
				}
				results := make(chan result, 1)
				go func() { value, ok, err := store.Get(ctx, 1); results <- result{value, ok, err} }()
				select {
				case <-remote.entered:
				case <-time.After(time.Second):
					t.Fatal("remote read never entered")
				}
				if scenario == "tombstone" {
					deleted.Store(true)
					if err := store.Delete(ctx, 1); err != nil {
						t.Fatal(err)
					}
				} else {
					version := int64(2)
					if scenario == "conflict" {
						version = 1
					}
					if err := store.Set(ctx, staleValue{Key: 1, Version: version, Payload: "published"}); err != nil {
						t.Fatal(err)
					}
				}
				close(remote.release)
				var got result
				select {
				case got = <-results:
				case <-time.After(time.Second):
					t.Fatal("read did not finish")
				}
				current, held, _ := local.Get(ctx, 1)
				t.Logf("returned=%+v ok=%v err=%v L1=%+v held=%v", got.value, got.ok, got.err, current, held)
				if got.err != nil {
					t.Fatal(got.err)
				}
				if scenario == "tombstone" && got.ok {
					t.Fatal("read returned the value rejected by a newer delete")
				}
				if scenario != "tombstone" && (!got.ok || got.value != current) {
					t.Fatal("read returned the captured value rejected by L1 admission")
				}
			})
		}
	}
}
