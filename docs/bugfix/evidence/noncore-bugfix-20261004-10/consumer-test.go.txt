var migrationOnce sync.Once
var migrationCause = errors.New("review19 migration rejected")

func registerMigration(t *testing.T) {
	t.Helper()
	migrationOnce.Do(func() {
		for _, from := range []uint32{1, 2} {
			migration.MustRegisterDAO(migration.DAOStep{Collection: "roost_review_20261004_19", From: from, To: from + 1, Apply: func(ctx context.Context, raw []byte) ([]byte, error) {
				if err := ctx.Err(); err != nil {
					return nil, err
				}
				var doc bson.M
				if err := bson.Unmarshal(raw, &doc); err != nil {
					return nil, err
				}
				mode, _ := doc["mode"].(string)
				if (mode == "first_failure" && from == 1) || (mode == "second_failure" && from == 2) {
					return nil, migrationCause
				}
				if mode == "bad_output" && from == 2 {
					return []byte("not BSON"), nil
				}
				doc["score"] = int64(10 + from)
				doc["name"] = "migrated"
				doc["_schema"] = from + 1
				return bson.Marshal(doc)
			}})
		}
	})
}

func TestGeneratedMigrationHydration(t *testing.T) {
	registerMigration(t)
	for _, mode := range []string{"success", "current_schema", "newer_schema", "first_failure", "second_failure", "missing_path", "bad_output"} {
		t.Run(mode, func(t *testing.T) {
			dao := NewMigratingDao()
			prior, _ := bson.Marshal(bson.M{"_id": int64(1), "name": "prior", "score": int64(50)})
			if err := dao.RestorePersisted(prior, 3, 8); err != nil {
				t.Fatal(err)
			}
			raw, _ := bson.Marshal(bson.M{"_id": int64(1), "name": "old", "score": int64(1), "mode": mode, "_schema": uint32(1)})
			original := bytes.Clone(raw)
			from := uint32(1)
			if mode == "current_schema" {
				from = 3
			}
			if mode == "newer_schema" {
				from = 4
			}
			if mode == "missing_path" {
				from = 0
			}
			err := dao.RestorePersisted(raw, from, 9)
			if !bytes.Equal(raw, original) {
				t.Fatal("migration changed persisted input bytes")
			}
			switch mode {
			case "success":
				if err != nil || dao.GetName() != "migrated" || dao.GetScore() != 12 || dao.DirtyTracker().Version() != 9 || dao.DirtyTracker().HasSyncDirty() {
					t.Fatalf("hydrate: %s %d version=%d err=%v", dao.GetName(), dao.GetScore(), dao.DirtyTracker().Version(), err)
				}
				var out bson.M
				if err := bson.Unmarshal(dao.Marshal(), &out); err != nil {
					t.Fatal(err)
				}
				if bson.Raw(dao.Marshal()).Lookup("_schema").AsInt64() != 3 {
					t.Fatal("generated current schema did not survive remarshal")
				}
			case "current_schema":
				if err != nil || dao.GetName() != "old" || dao.GetScore() != 1 || dao.DirtyTracker().Version() != 9 {
					t.Fatalf("no migration: %v", err)
				}
			default:
				if err == nil || dao.GetName() != "prior" || dao.GetScore() != 50 || dao.DirtyTracker().Version() != 8 {
					t.Fatalf("failed migration published partial DAO: %s %d %d %v", dao.GetName(), dao.GetScore(), dao.DirtyTracker().Version(), err)
				}
				if mode == "first_failure" || mode == "second_failure" {
					if !errors.Is(err, migrationCause) {
						t.Fatalf("cause lost: %v", err)
					}
				}
				if mode == "missing_path" && !errors.Is(err, migration.ErrPathMissing) {
					t.Fatal(err)
				}
				if mode == "newer_schema" && !strings.Contains(err.Error(), "newer than runtime") {
					t.Fatal(err)
				}
			}
		})
	}
}

type consumerNewRecord struct {
	ID      int64
	Version uint64
	Child   *consumerChild
}
type consumerChild struct{ Score int64 }
type consumerRegistryHook struct {
	fredis.IRedis
	publish func() error
}

func (r *consumerRegistryHook) HGet(ctx context.Context, key, field string) ([]byte, error) {
	raw, err := r.IRedis.HGet(ctx, key, field)
	if r.publish != nil {
		hook := r.publish
		r.publish = nil
		if err := hook(); err != nil {
			return nil, err
		}
	}
	return raw, err
}

func TestGeneratedRegistryConflict(t *testing.T) {
	addr := os.Getenv("ROOST_REDIS_TEST_ADDR")
	if addr == "" {
		t.Fatal("requires isolated Redis")
	}
	client, err := redisdriver.NewClient(fredis.DefaultConfig(addr))
	if err != nil {
		t.Fatal(err)
	}
	defer client.Close()
	ctx, cancel := context.WithTimeout(context.Background(), 15*time.Second)
	defer cancel()
	for _, scenario := range []string{"direct_set", "direct_delete", "cached_set", "cached_delete"} {
		t.Run(scenario, func(t *testing.T) {
			prefix := "roost:generated19:" + rand.Text()
			root := prefix + ":{" + RecordRedisDAOName + ":1}:root"
			child := prefix + ":{" + RecordRedisDAOName + ":1}:child"
			t.Cleanup(func() { _, _ = client.Del(context.Background(), root, child) })
			plain := NewRedisRecordRedisDAO(client, prefix, time.Hour)
			if err := plain.SetRecord(ctx, Record{ID: 1, Version: 1}); err != nil {
				t.Fatal(err)
			}
			newer := fcache.NewRedisRefHMapStore(client, fcache.RefHMapConfig[int64, consumerNewRecord]{Prefix: prefix, Name: RecordRedisDAOName, TTL: time.Hour, StoreConfig: fcache.StoreConfig[int64, consumerNewRecord]{KeyOf: func(v consumerNewRecord) int64 { return v.ID }}})
			wrapped := &consumerRegistryHook{IRedis: client, publish: func() error {
				return newer.Set(ctx, consumerNewRecord{ID: 1, Version: 2, Child: &consumerChild{Score: 9}})
			}}
			writer := NewRedisRecordRedisDAO(wrapped, prefix, time.Hour)
			local := NewLocalRecordRedisDAO()
			if err := local.SetRecord(ctx, Record{ID: 1, Version: 1}); err != nil {
				t.Fatal(err)
			}
			if strings.HasPrefix(scenario, "cached_") {
				writer = NewCachedRecordRedisDAOWithTTL(local, writer, time.Hour)
			}
			var opErr error
			if strings.HasSuffix(scenario, "set") {
				opErr = writer.SetRecord(ctx, Record{ID: 1, Version: 3})
			} else {
				opErr = writer.DeleteRecord(ctx, 1)
			}
			if !errors.Is(opErr, fcache.ErrRefHMapRegistryChanged) {
				t.Fatalf("generated wrapper lost registry conflict: %v", opErr)
			}
			stored, held, err := newer.Get(ctx, 1)
			if err != nil || !held || stored.Version != 2 || stored.Child == nil || stored.Child.Score != 9 {
				t.Fatalf("rejection wrote data: %+v %v %v", stored, held, err)
			}
			cached, held, err := local.GetRecord(ctx, 1)
			if err != nil || !held || cached.Version != 1 {
				t.Fatalf("remote rejection changed L1: %+v %v %v", cached, held, err)
			}
			if err := plain.DeleteRecord(ctx, 1); err != nil {
				t.Fatal(err)
			}
			if n, err := client.Exists(ctx, root, child); err != nil || n != 0 {
				t.Fatalf("explicit retry did not clean schema: %d %v", n, err)
			}
		})
	}
}
