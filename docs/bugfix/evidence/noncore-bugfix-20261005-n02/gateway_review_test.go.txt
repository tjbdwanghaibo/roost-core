type reviewSession struct{ p Principal }

func (s reviewSession) Principal() Principal             { return s.p }
func (reviewSession) Reply(context.Context, any) error   { return nil }
func (reviewSession) Close(error) error                  { return nil }

// N02-C1（容量观察）：一个已认证玩家改变 MessageID 就能占满全局 key 表，之后其他玩家首个请求被限流。
func TestReviewN02OneOwnerExhaustsKeyTableForOthers(t *testing.T) {
	limiter := security.NewRateLimiter(security.RateLimitConfig{Capacity: 5, MaxKeys: 1000, IdleTTL: time.Hour})
	calls := 0
	ep := Chain(EndpointFunc(func(context.Context, Session, Request) (any, error) { calls++; return nil, nil }), Recover(nil), RequireAuthenticated, RateLimit(limiter))
	attacker := reviewSession{Principal{PlayerID: 1, SessionID: "a"}}
	for id := uint32(1); id <= 1000; id++ {
		if _, err := ep.Handle(context.Background(), attacker, Request{MessageID: id}); err != nil {
			t.Fatalf("attacker request %d rejected early: %v", id, err)
		}
	}
	victim := reviewSession{Principal{PlayerID: 2, SessionID: "v"}}
	_, err := ep.Handle(context.Background(), victim, Request{MessageID: 1})
	st := limiter.Stats()
	t.Logf("victim first request err=%v stats=%+v", err, st)
	if !errors.Is(err, ErrRateLimited) {
		t.Fatalf("expected the observed collateral rejection, got %v", err)
	}
}

// N02-C2（容量观察）：满表后每个陌生 key 在全局锁内做一次 O(MaxKeys) 扫描。
func BenchmarkReviewN02FullTableUnseenKey(b *testing.B) {
	for _, n := range []int{1000, 100000} {
		b.Run(time.Duration(n).String(), func(b *testing.B) {
			limiter := security.NewRateLimiter(security.RateLimitConfig{Capacity: 5, MaxKeys: n, IdleTTL: time.Hour})
			for i := 0; i < n; i++ {
				limiter.Allow(security.RateLimitKey{OwnerID: int64(i + 1)})
			}
			b.ResetTimer()
			for i := 0; i < b.N; i++ {
				limiter.Allow(security.RateLimitKey{OwnerID: -int64(i + 1)})
			}
		})
	}
}

// 控制：不配合 context 的 endpoint，Timeout 只能在其返回后结算，返回值为 endpoint 原值。
func TestReviewN02TimeoutWithNonCooperativeEndpoint(t *testing.T) {
	release := make(chan struct{})
	var ctxErr error
	ep := Chain(EndpointFunc(func(ctx context.Context, _ Session, _ Request) (any, error) {
		<-release
		ctxErr = ctx.Err()
		return "late", nil
	}), Timeout(20*time.Millisecond))
	var wg sync.WaitGroup
	wg.Add(1)
	var got any
	var err error
	start := time.Now()
	go func() { defer wg.Done(); got, err = ep.Handle(context.Background(), reviewSession{}, Request{}) }()
	time.Sleep(60 * time.Millisecond)
	close(release)
	wg.Wait()
	t.Logf("elapsed=%v got=%v err=%v ctxErr=%v", time.Since(start), got, err, ctxErr)
	if got != "late" || err != nil || !errors.Is(ctxErr, context.DeadlineExceeded) {
		t.Fatalf("got=%v err=%v ctxErr=%v", got, err, ctxErr)
	}
}
