func reviewN02StartOps(t *testing.T, register func(*admin.Registry)) (*OpsMod, string) {
	t.Helper()
	l, err := net.Listen("tcp", "127.0.0.1:0")
	if err != nil {
		t.Fatal(err)
	}
	addr := l.Addr().String()
	_ = l.Close()
	cfg := viper.New()
	cfg.Set("ops.enabled", true)
	cfg.Set("ops.addr", addr)
	cfg.Set("ops.admin_enabled", true)
	cfg.Set("ops.admin_token", "review-n02-token")
	mod := NewOpsMod()
	if err := mod.Init(cfg); err != nil {
		t.Fatal(err)
	}
	registry := app.NewRegistry(cfg)
	commands, _ := app.Lookup[*admin.Registry](registry, mods.ModAdmin)
	register(commands)
	if err := mod.Provide(registry); err != nil {
		t.Fatal(err)
	}
	if err := mod.Start(); err != nil {
		t.Fatal(err)
	}
	deadline := time.Now().Add(3 * time.Second)
	for {
		c, err := net.Dial("tcp", addr)
		if err == nil {
			_ = c.Close()
			break
		}
		if time.Now().After(deadline) {
			t.Fatal("ops did not listen")
		}
		time.Sleep(10 * time.Millisecond)
	}
	return mod, "http://" + addr
}

func reviewN02Execute(base, name string) (*http.Response, []byte, error) {
	req, _ := http.NewRequest(http.MethodPost, base+"/admin/execute", bytes.NewReader([]byte(`{"name":"`+name+`"}`)))
	req.Header.Set("X-Admin-Token", "review-n02-token")
	resp, err := (&http.Client{Timeout: 5 * time.Second}).Do(req)
	if err != nil {
		return nil, nil, err
	}
	body, _ := io.ReadAll(resp.Body)
	resp.Body.Close()
	return resp, body, nil
}

// N02-R4：正式 Ops 装配：命令已执行（副作用发生），结果不可编码时不能回 200 空体。
func TestReviewN02OpsAdminUnencodableResultIsNot2xx(t *testing.T) {
	var executed atomic.Int32
	mod, base := reviewN02StartOps(t, func(r *admin.Registry) {
		_ = r.Register(admin.CommandDef{Name: "review.ratio", Handler: func(context.Context, admin.Command) (admin.Result, error) {
			executed.Add(1)
			return admin.Result{Data: map[string]any{"ratio": math.NaN()}}, nil
		}})
	})
	defer mod.StopWithContext(context.Background())
	resp, body, err := reviewN02Execute(base, "review.ratio")
	if err != nil {
		t.Fatal(err)
	}
	t.Logf("executed=%d status=%d body=%q", executed.Load(), resp.StatusCode, body)
	if executed.Load() != 1 {
		t.Fatalf("command executed %d times", executed.Load())
	}
	if resp.StatusCode >= 200 && resp.StatusCode < 300 {
		t.Fatalf("executed command with unencodable result answered %d body %q", resp.StatusCode, body)
	}
}

// 控制：不配合 context 的命令占住请求时，正式 Kit Stop 超时返回、保留 server；命令返回后重试排空成功，
// 并且客户端拿到命令结果。
func TestReviewN02OpsNonCooperativeCommandStopRetry(t *testing.T) {
	entered := make(chan struct{})
	release := make(chan struct{})
	mod, base := reviewN02StartOps(t, func(r *admin.Registry) {
		_ = r.Register(admin.CommandDef{Name: "review.block", Handler: func(context.Context, admin.Command) (admin.Result, error) {
			close(entered)
			<-release // 故意忽略 ctx
			return admin.Result{Message: "done"}, nil
		}})
	})
	done := make(chan int, 1)
	go func() {
		resp, _, err := reviewN02Execute(base, "review.block")
		if err != nil {
			done <- -1
			return
		}
		done <- resp.StatusCode
	}()
	<-entered
	ctx, cancel := context.WithTimeout(context.Background(), 100*time.Millisecond)
	defer cancel()
	if err := mod.StopWithContext(ctx); !errors.Is(err, context.DeadlineExceeded) {
		t.Fatalf("first stop = %v", err)
	}
	if mod.server == nil {
		t.Fatal("server ownership dropped while handler still running")
	}
	close(release)
	if code := <-done; code != http.StatusOK {
		t.Fatalf("in-flight command status %d", code)
	}
	if err := mod.StopWithContext(context.Background()); err != nil {
		t.Fatalf("retry stop = %v", err)
	}
	if mod.server != nil {
		t.Fatal("server retained after drained stop")
	}
	if _, _, err := reviewN02Execute(base, "review.block"); err == nil {
		t.Fatal("listener still accepting after stop")
	}
}
