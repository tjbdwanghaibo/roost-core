package httpserver

import (
	"encoding/json"
	"net/http"
	"strings"
	"testing"
)

// 修前（50e9a4e8）：先写状态再 Encoder。
func jsonBaseline(w http.ResponseWriter, status int, value any) {
	w.Header().Set("Content-Type", "application/json; charset=utf-8")
	w.WriteHeader(status)
	_ = json.NewEncoder(w).Encode(value)
}

// N02（149f0d18）：Marshal 后 append 换行再写。
func jsonMarshalFirst(w http.ResponseWriter, status int, value any) {
	body, err := json.Marshal(value)
	if err != nil {
		status = http.StatusInternalServerError
		body = []byte(`{"error":"encode response","ok":false}`)
	}
	w.Header().Set("Content-Type", "application/json; charset=utf-8")
	w.WriteHeader(status)
	_, _ = w.Write(append(body, '\n'))
}

// 候选（已采用）：Encoder 写入“首次 Write 时才写状态”的适配器。
type candStatusOnWrite struct {
	w      http.ResponseWriter
	status int
	wrote  bool
}

func (s *candStatusOnWrite) Write(p []byte) (int, error) {
	if !s.wrote {
		s.wrote = true
		s.w.WriteHeader(s.status)
	}
	return s.w.Write(p)
}

func jsonDeferredHeader(w http.ResponseWriter, status int, value any) {
	w.Header().Set("Content-Type", "application/json; charset=utf-8")
	out := &candStatusOnWrite{w: w, status: status}
	if err := json.NewEncoder(out).Encode(value); err != nil && !out.wrote {
		w.WriteHeader(http.StatusInternalServerError)
		_, _ = w.Write([]byte("{\"error\":\"encode response\",\"ok\":false}\n"))
	}
}

type discardWriter struct{ h http.Header }

func (d *discardWriter) Header() http.Header         { return d.h }
func (d *discardWriter) Write(p []byte) (int, error) { return len(p), nil }
func (d *discardWriter) WriteHeader(int)             {}

func BenchmarkN02FinalJSON(b *testing.B) {
	small := map[string]any{"ok": true, "data": map[string]any{"id": 42, "name": "player"}}
	type row struct {
		ID   int64  `json:"id"`
		Name string `json:"name"`
	}
	rows := make([]row, 4096)
	for i := range rows {
		rows[i] = row{ID: int64(i), Name: strings.Repeat("x", 48)}
	}
	large := map[string]any{"ok": true, "data": rows}
	for _, value := range []struct {
		name string
		v    any
	}{{"small", small}, {"large", large}} {
		for _, impl := range []struct {
			name string
			fn   func(http.ResponseWriter, int, any)
		}{{"baseline_encoder", jsonBaseline}, {"n02_marshal", jsonMarshalFirst}, {"deferred_header", jsonDeferredHeader}} {
			b.Run(value.name+"/"+impl.name, func(b *testing.B) {
				w := &discardWriter{h: http.Header{}}
				b.ReportAllocs()
				for i := 0; i < b.N; i++ {
					impl.fn(w, http.StatusOK, value.v)
				}
			})
		}
	}
}
