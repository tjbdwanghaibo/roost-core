package web

import (
	"context"
	"errors"
	"net/http"
	"net/http/httptest"
	"testing"

	"github.com/tjbdwanghaibo/roost-core/httpclient"
	"github.com/tjbdwanghaibo/roost-core/httpserver"
	"github.com/tjbdwanghaibo/roost-core/webroute"
)

func TestGeneratedRouteUnencodableResult(t *testing.T) {
	svc := &Service{}
	engine := httpserver.NewEngine()
	if err := webroute.RegisterModules(webroute.NewRegistrar(engine), svc); err != nil {
		t.Fatal(err)
	}
	srv := httptest.NewServer(engine)
	defer srv.Close()
	client := httpclient.New(httpclient.WithBaseURL(srv.URL))

	var ok StatsResponse
	if err := client.PostJSON(context.Background(), "/gm/stats", StatsRequest{Wins: 1, Games: 2}, &ok); err != nil || ok.WinRate != 0.5 {
		t.Fatalf("normal: err=%v out=%+v", err, ok)
	}
	var out StatsResponse
	err := client.PostJSON(context.Background(), "/gm/stats", StatsRequest{}, &out)
	t.Logf("calls=%d err=%v out=%+v", svc.Calls.Load(), err, out)
	var statusErr *httpclient.StatusError
	if !errors.As(err, &statusErr) || statusErr.StatusCode != http.StatusInternalServerError {
		t.Fatalf("generated route with unencodable result: want 500 StatusError, got err=%v out=%+v", err, out)
	}
}
