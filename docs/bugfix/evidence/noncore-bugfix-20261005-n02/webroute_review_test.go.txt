package webroute_test

import (
	"context"
	"errors"
	"io"
	"math"
	"net/http"
	"net/http/httptest"
	"strings"
	"testing"

	"github.com/tjbdwanghaibo/roost-core/httpclient"

)

type ratioResp struct {
	Ratio float64 `json:"ratio"`
}

type panicMarshal struct{}

func (panicMarshal) MarshalJSON() ([]byte, error) { panic("marshal boom") }

type errMarshal struct{}

func (errMarshal) MarshalJSON() ([]byte, error) { return nil, errors.New("marshal failed") }

func serve(t *testing.T, register func(reg *webroute.Registrar)) *httptest.Server {
	t.Helper()
	engine := httpserver.NewEngine()
	register(webroute.NewRegistrar(engine))
	srv := httptest.NewServer(engine)
	t.Cleanup(srv.Close)
	return srv
}

// N02-R1：业务成功但响应不可编码（NaN），客户端不能看到 2xx 成功。
func TestReviewN02UnencodableSuccessIsNot2xx(t *testing.T) {
	sideEffects := 0
	cases := map[string]any{
		"nan":           ratioResp{Ratio: math.NaN()},
		"inf":           map[string]any{"v": math.Inf(1)},
		"marshal_error": errMarshal{},
		"marshal_panic": panicMarshal{},
		"channel":       map[string]any{"c": make(chan int)},
	}
	for name, value := range cases {
		t.Run(name, func(t *testing.T) {
			srv := serve(t, func(reg *webroute.Registrar) {
				if err := reg.Register(http.MethodPost, "/r", func(w http.ResponseWriter, r *http.Request) {
					var req struct{}
					if !webroute.DecodeJSON(w, r, &req) {
						return
					}
					sideEffects++
					webroute.WriteResult(w, http.StatusOK, value, nil)
				}); err != nil {
					t.Fatal(err)
				}
			})
			resp, err := http.Post(srv.URL+"/r", "application/json", strings.NewReader(`{}`))
			if err != nil {
				t.Fatal(err)
			}
			body, _ := io.ReadAll(resp.Body)
			resp.Body.Close()
			t.Logf("status=%d body=%q", resp.StatusCode, body)
			// 同一路由经正式 httpclient 消费：必须返回 error，不能 nil + 零值。
			var out ratioResp
			cerr := httpclient.New(httpclient.WithBaseURL(srv.URL)).PostJSON(context.Background(), "/r", struct{}{}, &out)
			t.Logf("httpclient err=%v out=%+v", cerr, out)
			if resp.StatusCode >= 200 && resp.StatusCode < 300 {
				t.Fatalf("unencodable response reported as %d with body %q", resp.StatusCode, body)
			}
			if cerr == nil {
				t.Fatalf("httpclient returned nil error for unencodable response; out=%+v", out)
			}
			var statusErr *httpclient.StatusError
			if !errors.As(cerr, &statusErr) || statusErr.StatusCode != http.StatusInternalServerError {
				t.Fatalf("want StatusError 500, got %v", cerr)
			}
		})
	}
	if sideEffects == 0 {
		t.Fatal("handler never ran")
	}
}

// 控制：可编码成功响应字节形状不变（Encoder 风格结尾换行）。
func TestReviewN02EncodableSuccessUnchanged(t *testing.T) {
	srv := serve(t, func(reg *webroute.Registrar) {
		_ = reg.Register(http.MethodGet, "/ok", func(w http.ResponseWriter, r *http.Request) {
			webroute.WriteResult(w, http.StatusOK, map[string]any{"a": "<b>", "n": 1}, nil)
		})
	})
	resp, err := http.Get(srv.URL + "/ok")
	if err != nil {
		t.Fatal(err)
	}
	body, _ := io.ReadAll(resp.Body)
	resp.Body.Close()
	if resp.StatusCode != 200 || string(body) != "{\"a\":\"\\u003cb\\u003e\",\"n\":1}\n" {
		t.Fatalf("status=%d body=%q", resp.StatusCode, body)
	}
	if ct := resp.Header.Get("Content-Type"); ct != "application/json; charset=utf-8" {
		t.Fatalf("content-type %q", ct)
	}
}

// N02-R2：响应已开始后 panic，客户端必须看到传输失败，而不是“完整”的 200 + 拼接错误体。
func TestReviewN02PanicAfterResponseStartedAborts(t *testing.T) {
	engine := httpserver.NewEngine()
	engine.Get("/stream", func(w http.ResponseWriter, r *http.Request) {
		w.Header().Set("Content-Type", "application/json; charset=utf-8")
		w.WriteHeader(http.StatusOK)
		_, _ = w.Write([]byte(`{"items":[1,2`))
		if f, ok := w.(http.Flusher); ok {
			f.Flush()
		}
		panic("boom mid-response")
	})
	srv := httptest.NewServer(engine)
	defer srv.Close()
	resp, err := http.Get(srv.URL + "/stream")
	if err != nil {
		t.Logf("request error (abort observed early): %v", err)
		return
	}
	body, readErr := io.ReadAll(resp.Body)
	resp.Body.Close()
	t.Logf("status=%d body=%q readErr=%v", resp.StatusCode, body, readErr)
	if readErr == nil {
		t.Fatalf("mid-response panic produced a complete response: status=%d body=%q", resp.StatusCode, body)
	}
}

// N02-R3：http.ErrAbortHandler 是标准库约定的中止信号，不能被改写为 500 JSON。
func TestReviewN02ErrAbortHandlerKeepsAbortSemantics(t *testing.T) {
	engine := httpserver.NewEngine()
	engine.Get("/abort", func(w http.ResponseWriter, r *http.Request) { panic(http.ErrAbortHandler) })
	srv := httptest.NewServer(engine)
	defer srv.Close()
	resp, err := http.Get(srv.URL + "/abort")
	if err == nil {
		body, _ := io.ReadAll(resp.Body)
		resp.Body.Close()
		t.Fatalf("ErrAbortHandler answered status=%d body=%q", resp.StatusCode, body)
	}
}

// 控制：响应前 panic 仍是 500 JSON。
func TestReviewN02PanicBeforeResponseIs500(t *testing.T) {
	engine := httpserver.NewEngine()
	engine.Get("/p", func(w http.ResponseWriter, r *http.Request) { panic("boom") })
	srv := httptest.NewServer(engine)
	defer srv.Close()
	resp, err := http.Get(srv.URL + "/p")
	if err != nil {
		t.Fatal(err)
	}
	body, _ := io.ReadAll(resp.Body)
	resp.Body.Close()
	if resp.StatusCode != 500 || !strings.Contains(string(body), "internal server error") {
		t.Fatalf("status=%d body=%q", resp.StatusCode, body)
	}
}
