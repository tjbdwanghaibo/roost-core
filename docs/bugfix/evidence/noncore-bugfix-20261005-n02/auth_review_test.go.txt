type reviewClock struct {
	mu  sync.Mutex
	now time.Time
}

func (c *reviewClock) Now() time.Time { c.mu.Lock(); defer c.mu.Unlock(); return c.now }
func (c *reviewClock) advance(d time.Duration) {
	c.mu.Lock()
	c.now = c.now.Add(d)
	c.mu.Unlock()
}

type reviewAllocator struct{ next atomic.Int64 }

func (a *reviewAllocator) Allocate(context.Context, int32) (int64, error) {
	return a.next.Add(1) + 5_000_000, nil
}

// 业务鉴权全链：真实 account.Service 发票据（Login→CreateRole→SelectRole）→ 真实 TCP 握手 →
// 生成的 demo 认证器 → ValidateSession → Principal → Dispatch 看到的玩家与 sid。
func TestReviewN02BusinessAuthChain(t *testing.T) {
	ctx := context.Background()
	clock := &reviewClock{now: time.Unix(1_700_000_000, 0)}
	names, err := directory.New(versionstore.NewMemoryStore[string, directory.Entry](), directory.Config{Normalize: directory.NormalizeLower, DefaultTTL: time.Minute, Now: clock.Now})
	if err != nil {
		t.Fatal(err)
	}
	service, err := svcaccount.New(svcaccount.Config{
		Accounts: versionstore.NewMemoryStore[string, svcaccount.Account](),
		Roles:    versionstore.NewMemoryStore[int64, svcaccount.Role](),
		Servers:  versionstore.NewMemoryStore[int32, svcaccount.GameServer](),
		Names:    names, Slots: versionstore.NewMemoryStore[string, svcaccount.Slot](),
		Verifier: svcaccount.VerifierFunc(func(_ context.Context, id svcaccount.Identity) (svcaccount.Verified, error) {
			if id.Credential != "good" {
				return svcaccount.Verified{}, fmt.Errorf("bad credential")
			}
			return svcaccount.Verified{Channel: id.Channel, OpenID: id.OpenID}, nil
		}),
		Allocator: &reviewAllocator{},
		NameRules: svcaccount.NameValidatorFunc(func(string) error { return nil }),
		SessionSecret: "review-n02-secret", SessionTTL: time.Minute, Now: clock.Now,
	})
	if err != nil {
		t.Fatal(err)
	}
	if _, err := service.UpsertServer(ctx, svcaccount.GameServer{ID: 1300, Name: "s", Status: svcaccount.ServerOpen}); err != nil {
		t.Fatal(err)
	}
	issue := func(openID, name string) svcaccount.Session {
		account, err := service.Login(ctx, svcaccount.Identity{Channel: "dev", OpenID: openID, Credential: "good"})
		if err != nil {
			t.Fatal(err)
		}
		role, err := service.CreateRole(ctx, account.ID, 1300, name)
		if err != nil {
			t.Fatal(err)
		}
		session, err := service.SelectRole(ctx, account.ID, role.PlayerID)
		if err != nil {
			t.Fatal(err)
		}
		return session
	}
	alice, bob := issue("alice", "alice"), issue("bob", "bob")

	authenticator := &applicationAuthenticator{}
	authenticator.accounts.Store(&accountsHandle{accounts: service})
	type seen struct {
		player int64
		sid    string
	}
	got := make(chan seen, 8)
	cfg := viper.New()
	cfg.Set("player_access.tcp.addr", "127.0.0.1:0")
	config, err := configFromViper(cfg)
	if err != nil {
		t.Fatal(err)
	}
	protocols := player_agent.NewProtocolRegistry()
	if err := player_agent.RegisterProtocol[string, string](protocols, stuckRequestID, stuckResponseID,
		func(p []byte) (string, error) { return string(p), nil }, func(v string) ([]byte, error) { return []byte(v), nil },
		func(c *player_agent.Context, _ string) (string, error) {
			got <- seen{player: c.PlayerID, sid: c.Session.Principal().Claims[svcaccount.ServerIDClaim]}
			return "ok", nil
		}); err != nil {
		t.Fatal(err)
	}
	_ = protocols.Seal()
	server, err := NewServer(config, &accessplayer.Runtime{Protocols: protocols}, authenticator)
	if err != nil {
		t.Fatal(err)
	}
	if err := server.Start(); err != nil {
		t.Fatal(err)
	}
	defer func() { c, cancel := context.WithTimeout(ctx, 3*time.Second); defer cancel(); _ = server.Stop(c) }()

	ticket := func(claimed int64, token string) string {
		return sessionTokenPrefix + strconv.FormatInt(claimed, 10) + ":" + token
	}
	// 正常：Principal 来自角色。
	conn := dialAuthenticated(t, server, ticket(alice.PlayerID, alice.Token))
	sendStuckRequest(t, conn, 2)
	select {
	case s := <-got:
		if s.player != alice.PlayerID || s.sid != "1300" {
			t.Fatalf("dispatch saw %+v, want alice %d sid 1300", s, alice.PlayerID)
		}
	case <-time.After(2 * time.Second):
		t.Fatal("authenticated request never dispatched")
	}
	// 拒绝：每一种都必须在握手阶段被关闭，绝不进入 Dispatch。
	rejects := map[string]string{
		"claim_other_player": ticket(bob.PlayerID, alice.Token),
		"tampered":           ticket(alice.PlayerID, alice.Token[:len(alice.Token)-2]+"xx"),
		"no_prefix":          strconv.FormatInt(alice.PlayerID, 10) + ":" + alice.Token,
		"empty_token":        ticket(alice.PlayerID, ""),
		"negative_player":    ticket(-alice.PlayerID, alice.Token),
	}
	for name, credential := range rejects {
		c := reviewN02SendAuth(t, reviewN02Addr(server), credential)
		if !closedByServer(c, 2*time.Second) {
			t.Fatalf("%s: handshake was not refused", name)
		}
	}
	// 过期：同一票据在 TTL 之后被拒。
	clock.advance(2 * time.Minute)
	expired := reviewN02SendAuth(t, reviewN02Addr(server), ticket(bob.PlayerID, bob.Token))
	if !closedByServer(expired, 2*time.Second) {
		t.Fatal("expired ticket accepted")
	}
	select {
	case s := <-got:
		t.Fatalf("a refused handshake reached dispatch: %+v", s)
	default:
	}
	_ = net.IPv4len
}
