package testdata

import (
	"context"
	"errors"
	"maps"
	"strings"
	"testing"

	"github.com/tjbdwanghaibo/roost-core/dataengine"
	"github.com/tjbdwanghaibo/roost-core/nest"
	"github.com/tjbdwanghaibo/roost-core/nestwal"
	"go.mongodb.org/mongo-driver/v2/bson"
)

type walCommitter struct{ wal *nestwal.WAL }

func (c walCommitter) Commit(ctx context.Context, rec nest.CommitRecord) error {
	if rec.Empty() {
		return nil
	}
	_, err := c.wal.Append(ctx, rec)
	return err
}

// applyMutation applies one generated mutation to a document the way the Mongo
// projection does: Put replaces, Patch runs $set (dotted paths) then $unset.
func applyMutation(t *testing.T, doc bson.M, m dataengine.Mutation) bson.M {
	t.Helper()
	switch m.Kind {
	case dataengine.MutationPut:
		out := bson.M{}
		if err := bson.Unmarshal(m.Data, &out); err != nil {
			t.Fatal(err)
		}
		return out
	case dataengine.MutationPatch:
		if len(m.Patch.SetBSON) > 0 {
			var set bson.D
			if err := bson.Unmarshal(m.Patch.SetBSON, &set); err != nil {
				t.Fatal(err)
			}
			for _, e := range set {
				setPath(doc, e.Key, e.Value)
			}
		}
		for _, p := range m.Patch.Unset {
			unsetPath(doc, p)
		}
		return doc
	default:
		t.Fatalf("unexpected mutation kind %v", m.Kind)
		return nil
	}
}

func setPath(doc bson.M, path string, v any) {
	parts := strings.Split(path, ".")
	cur := doc
	for _, p := range parts[:len(parts)-1] {
		next, ok := cur[p].(bson.M)
		if !ok {
			if d, isD := cur[p].(bson.D); isD {
				next = bson.M{}
				for _, e := range d {
					next[e.Key] = e.Value
				}
			} else {
				next = bson.M{}
			}
			cur[p] = next
		}
		cur = next
	}
	cur[parts[len(parts)-1]] = v
}

func unsetPath(doc bson.M, path string) {
	parts := strings.Split(path, ".")
	cur := doc
	for _, p := range parts[:len(parts)-1] {
		next, ok := cur[p].(bson.M)
		if !ok {
			if d, isD := cur[p].(bson.D); isD {
				next = bson.M{}
				for _, e := range d {
					next[e.Key] = e.Value
				}
				cur[p] = next
			} else {
				return
			}
		}
		cur = next
	}
	delete(cur, parts[len(parts)-1])
}

// FastMap 遍历语义（NC-182）在完整持久链路上的组合：RangeFastItems 回调里
// Set / Delete / 删空（FastMap.Clear）后提交到文件 WAL，再从 WAL 重放到文档，
// 解码出的 DAO 必须与内存一致；回调里改完再失败的事务回滚后内存回到原状，
// 不留 WAL 记录。
func TestFastMapRangeMutationsCommitReplayAndRollBack(t *testing.T) {
	ctx := context.Background()
	dir := t.TempDir()
	opts := nestwal.DefaultOptions(dir)
	opts.WriterVersion = nestwal.WriterVersionV2
	wal, err := nestwal.Open(opts)
	if err != nil {
		t.Fatal(err)
	}
	c := walCommitter{wal}
	d := NewVarietyDao()
	d.SetId(1)
	model := map[int64]int32{}
	run := func(name string, body func()) {
		t.Helper()
		if _, err := nest.RunIsolatedTransaction(ctx, c, name, func() (any, error) { body(); return nil, nil }); err != nil {
			t.Fatalf("%s: %v", name, err)
		}
	}
	snapshot := func() map[int64]int32 {
		got := map[int64]int32{}
		d.RangeFastItems(func(k int64, v int32) bool { got[k] = v; return true })
		return got
	}
	var committed []map[int64]int32 // model after each committed transaction, in WAL order
	check := func(stage string) {
		t.Helper()
		defer func() { committed = append(committed, maps.Clone(model)) }()
		if got := snapshot(); !maps.Equal(got, model) || d.FastItemsLen() != len(model) {
			t.Fatalf("%s: live fast_items = %v (len %d), model %v", stage, got, d.FastItemsLen(), model)
		}
	}

	run("fill", func() {
		d.SetPersistOnly(7)
		for i := int64(1); i <= 6; i++ { // 6 of 8 slots: the next new slot grows the table
			d.SetFastItems(i, int32(i))
			model[i] = int32(i)
		}
	})
	check("fill")

	run("update, delete and grow inside Range", func() {
		first := true
		seen := map[int64]int{}
		d.RangeFastItems(func(k int64, v int32) bool {
			if k == 0 || k >= 100 {
				return true // keys inserted by this callback may or may not be visited
			}
			seen[k]++
			if first {
				first = false
				for n := int64(100); n < 120; n++ {
					d.SetFastItems(n, int32(n))
					model[n] = int32(n)
				}
			}
			if k%3 == 0 {
				d.DelFastItems(k)
				delete(model, k)
			} else {
				d.SetFastItems(k, v*10)
				model[k] = v * 10
			}
			return true
		})
		for k, n := range seen {
			if n != 1 {
				t.Fatalf("key %d visited %d times", k, n)
			}
		}
		if len(seen) != 6 {
			t.Fatalf("visited %v, want each of 1..6 once", seen)
		}
	})
	check("update/delete/grow")

	run("delete everything inside Range, then re-add one", func() {
		d.RangeFastItems(func(k int64, _ int32) bool {
			d.DelFastItems(k)
			delete(model, k)
			if len(model) == 0 { // the map is now empty: FastMap.Clear ran under Range
				d.SetFastItems(42, 42)
				model[42] = 42
			}
			return true
		})
	})
	check("clear")

	before := maps.Clone(model)
	appended := wal.Stats().Appended
	sentinel := errors.New("roll back")
	_, err = nest.RunIsolatedTransaction(ctx, c, "rollback", func() (any, error) {
		d.RangeFastItems(func(k int64, v int32) bool {
			d.SetFastItems(k, v+1)
			for n := int64(200); n < 230; n++ {
				d.SetFastItems(n, 1)
			}
			d.DelFastItems(k)
			return true
		})
		return nil, sentinel
	})
	if !errors.Is(err, sentinel) {
		t.Fatalf("rollback transaction error = %v", err)
	}
	model = before
	if got := snapshot(); !maps.Equal(got, model) || d.FastItemsLen() != len(model) {
		t.Fatalf("rollback: live fast_items = %v, want %v", got, model)
	}
	if wal.Stats().Appended != appended {
		t.Fatalf("rolled back transaction appended %d WAL records", wal.Stats().Appended-appended)
	}

	if err := wal.Close(ctx); err != nil {
		t.Fatal(err)
	}
	reopened, err := nestwal.Open(opts)
	if err != nil {
		t.Fatal(err)
	}
	defer reopened.Close(ctx)
	var doc bson.M
	records := 0
	if err := reopened.Replay(ctx, func(_ nest.CommitFence, rec nest.CommitRecord) error {
		records++
		for _, m := range rec.Mutations {
			doc = applyMutation(t, doc, m)
		}
		raw, err := bson.Marshal(doc)
		if err != nil {
			return err
		}
		step := NewVarietyDao()
		if err := step.Unmarshal(raw); err != nil {
			return err
		}
		got := map[int64]int32{}
		step.RangeFastItems(func(k int64, v int32) bool { got[k] = v; return true })
		if records > len(committed) || !maps.Equal(got, committed[records-1]) {
			t.Errorf("after replaying record %d (%s): fast_items %v, live state after that commit %v", records, rec.Handler, got, committed[min(records, len(committed))-1])
		}
		return nil
	}); err != nil {
		t.Fatal(err)
	}
	if records != 3 {
		t.Fatalf("replayed %d records, want 3", records)
	}
	raw, err := bson.Marshal(doc)
	if err != nil {
		t.Fatal(err)
	}
	replayed := NewVarietyDao()
	if err := replayed.Unmarshal(raw); err != nil {
		t.Fatal(err)
	}
	got := map[int64]int32{}
	replayed.RangeFastItems(func(k int64, v int32) bool { got[k] = v; return true })
	if !maps.Equal(got, model) || replayed.GetPersistOnly() != 7 {
		t.Fatalf("replayed WAL state %v persist_only=%d, live %v", got, replayed.GetPersistOnly(), model)
	}
	t.Logf("replayed %d records; final fast_items %v", records, got)
}
