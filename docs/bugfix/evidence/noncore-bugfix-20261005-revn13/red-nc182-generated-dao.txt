package testdata

import (
	"context"
	"testing"

	"github.com/tjbdwanghaibo/roost-core/nest"
)

// NC-182 在生成 DAO 上的组合：RangeFastItems 回调里 SetFastItems 改已有键。
func TestNC182GeneratedRangeThenSetExisting(t *testing.T) {
	d := NewVarietyDao()
	d.SetId(1)
	c := &ownershipCommitter{}
	if _, err := nest.RunIsolatedTransaction(context.Background(), c, "fill", func() (any, error) {
		for i := int64(1); i <= 6; i++ {
			d.SetFastItems(i, int32(i))
		}
		return nil, nil
	}); err != nil {
		t.Fatal(err)
	}
	seen := map[int64]int{}
	if _, err := nest.RunIsolatedTransaction(context.Background(), c, "double", func() (any, error) {
		d.RangeFastItems(func(k int64, v int32) bool {
			seen[k]++
			d.SetFastItems(k, v*10)
			return true
		})
		return nil, nil
	}); err != nil {
		t.Fatal(err)
	}
	_, phantom := d.GetFastItems(0)
	t.Logf("seen=%v len=%d phantom_key0=%v", seen, d.FastItemsLen(), phantom)
	for i := int64(1); i <= 6; i++ {
		v, _ := d.GetFastItems(i)
		t.Logf("item %d = %d", i, v)
	}
	t.Logf("last commit record: %+v", c.records[len(c.records)-1])
	if phantom || len(seen) != 6 {
		t.Fatalf("generated RangeFastItems + SetFastItems: seen=%v phantom key 0=%v", seen, phantom)
	}
}
