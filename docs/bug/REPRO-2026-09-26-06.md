func aud3TryLockElsewhere(e entity.IThreadSafeEntity) bool {
	done := make(chan bool)

## 5. RR-69：forget 清除重新登记 subject 的脏通知器

package entitysync

import (
	"sync"
	"testing"
)

// Audit probe: RegisterAfterRetirement racing the Flush that completes the
// retirement (forget). When Register lands in the window where forget has
// deleted the subject from the map but not yet cleared the state's dirty
// notifier, forget's SetDirtyNotifier(nil) clobbers the newly registered
// subject's notifier: later MarkDirty never schedules it.
func TestAudRegisterAfterRetirementRacingForget(t *testing.T) {
	const rounds = 3000
	lost, queuedN, direct := 0, 0, 0
	for round := 0; round < rounds; round++ {
		transport := newRecordingTransport()
		manager := newTestManager(t, transport, ManagerConfig{})
		packs := 0
		id := int64(5000 + round)
		state := retiringSubject(t, manager, transport, id, &packs)
		var wg sync.WaitGroup
		start := make(chan struct{})
		var queued bool
		var regErr error
		wg.Add(2)
		go func() { defer wg.Done(); <-start; _ = manager.Flush(t.Context()) }()
		go func() {
			defer wg.Done()
			<-start
			queued, regErr = manager.RegisterAfterRetirement(state, func(error) {})
		}()
		close(start)
		wg.Wait()
		if regErr != nil {
			t.Fatalf("round %d: %v", round, regErr)
		}
		if queued {
			queuedN++
		} else {
			direct++
		}
		// Make sure the retirement finished and the registration is in place.
		_ = manager.Flush(t.Context())
		if manager.subject(id) == nil {
			t.Fatalf("round %d: subject not registered after all", round)
		}
		manager.takePending()
		state.MarkDirty(1)
		manager.pendingMu.Lock()
		_, pending := manager.pending[id]
		manager.pendingMu.Unlock()
		if !pending {
			lost++
			if lost <= 3 {
				t.Logf("round %d: queued=%v: registered subject is not scheduled by MarkDirty (dirty notifier cleared)", round, queued)
			}
		}
		_ = manager.Close(t.Context())
	}
	t.Logf("rounds=%d queued=%d direct=%d lost-notifier=%d", rounds, queuedN, direct, lost)
	if lost > 0 {
		t.Errorf("%d of %d registrations lost their dirty notifier", lost, rounds)
	}
}
