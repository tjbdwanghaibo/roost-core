# RR-20260928-08：strict 下 Remote 已提交、tracker 被淘汰后重新登记报 Overloaded，回复被误标 `ErrRemotePartRejected`

日期：2026-09-28。基线：`611dab1`。来源：第七轮独立审计（[REVIEW-2026-09-28-audit7](../review/REVIEW-2026-09-28-audit7.md) N1、N2）。等级 **P4**：默认 `TransactionTrackLimit=65536` 下几乎不可达，limit 接近并发数时可达。引入：RR-20260928-03。状态：已复现，未修复。

带 Remote 批次的 strict 消息本地已提交，投影器已把 Remote 写进权威（tracker 转 Committed 并关闭）；`batch.Commit` 进入等待前这条已关闭的 tracker 被容量淘汰、剩余名额被 pending 占满，
`waitRemoteTransaction` → `newRemoteTransactionLocked` 返回 `ErrRemoteOverloaded`（不带未知哨兵，`remoteentity/transaction_tracking.go:50-61`）。`remoteentity/batch.go:503` 对 strict 的任何错误按结果未知交 finalizer（随后按已提交收尾），
但 `nest/msg.go:97-98` 只看哨兵，回复打上 `ErrRemotePartRejected`（第 4 行：Remote 没写入），调用方照做会把 Remote 部分重复写一次。探针输出：

```text
PROBE reply=<nil> err=nest: local transaction committed but its remote part was rejected: nest: finish remote write batch: remote entity: capacity exceeded
PROBE first-row=4 sentinels=[nest.ErrRemotePartRejected] overloaded=true localCommits=1
PROBE authority CommitStatus(tx)=4 err=<nil>        （4 = RemoteCommitCommitted）
```

同一轮审计的附带问题：`remoteentity/reply_sentinel_table_test.go` 的 `replySentinels` 仍按改号前（兜底为第 14 行、`ErrNestCanceled` / `ErrNestTimeout` 记为 0）；判别表第 14 行措辞只点了 `Nest.Request`（`RequestMulti*` 同一出口）；
B19 并存形态外层文案“a nested isolated transaction committed before the message failed”与“本地已提交”不符；RR-20260928-05 兼容性只提 `rollback.sh`，没提 install.sh 失败时的自动回滚到不含 `configs/data` 的旧 release。

修复约束：让 Remote 结论的判定与 `batch.go:503` 一致——strict 下 `Commit` 的错误凡不是明确拒绝（`ErrRemoteRejected` 一类）的，一律带 `entity.ErrRemotePersistenceIndeterminate`（在 remoteentity 源头补，自定义实现按 USER_GUIDE 契约同样要求），
使回复落到第 2 行；Durability 0 的 fenced / conflict / rejected 判定不变；`replySentinels` 跟上 1～15 行编号并补第 14 行断言；第 14 行措辞覆盖 `Request*`；B19 并存文案与 RR-05 兼容性如实补写；RR-24 / RR-58 / RR-03 回归不回退。

//go:build probe

package remoteentity

// 审计探针（aud9，非修复方）：判别表第 4 行判据“本地已提交后 Commit 出错且不带 ErrRemotePersistenceIndeterminate = Remote 明确拒绝”。
// strict 下 remoteWriteBatch.Commit 对任何错误都按结果未知处理（b.indeterminate = true，交 finalizer 回源），但 nest 只看哨兵。
// 构造：tracker 容量 1；投影器先把本事务写进权威（Committed，tracker 关闭）；另一笔准入按容量淘汰这条已关闭的 tracker 并占住唯一名额；
// 随后 strict 等待重新登记 tracker 得到 ErrRemoteOverloaded（不带未知哨兵）。

import (
	"context"
	"errors"
	"testing"
	"time"

	"github.com/tjbdwanghaibo/roost-core/entity"
	"github.com/tjbdwanghaibo/roost-core/nest"

func TestZZAud9Row4CounterexampleOverloadedAfterCommitted(t *testing.T) {
	f, live := newReloadFixture(t, 1999)
	f.mgr.remote.txMu.Lock()
	f.mgr.remote.txCapacity = 1
	f.mgr.remote.txMu.Unlock()
	committer := &countingLocalCommitter{}
	var txID entity.RemoteTransactionID
	beforeWait := func(f reloadFixture, commits []entity.RemoteCommit) {
		txID = commits[0].TransactionID
		// 投影器把 Remote 写交给权威并成功（tracker Committed、关闭）。
		done := make(chan error, 1)
		go func() { _, err := f.mgr.ApplyRemoteCommits(context.Background(), txID, commits); done <- err }()
		if err := <-done; err != nil {
			t.Errorf("projector apply: %v", err)
		}
		// 另一笔 Remote 事务准入：容量满，淘汰最旧的已关闭 tracker（本事务），占住唯一名额（pending，不淘汰）。
		if err := f.mgr.trackRemoteTransaction(remoteTestTxID(251)); err != nil {
			t.Errorf("second admission: %v", err)
		}
	}
	manager := boundedConfirmManager{Manager: f.mgr, fixture: f, confirm: 3 * time.Second, beforeWait: beforeWait}
	engine := nest.NewEngine(nest.NestOptionWithGetter(f.access), nest.NestOptionWithWorkerNumAndMsgCap(1, 1, 16),
		nest.NestOptionWithRemoteEntityManager(manager), nest.NestOptionWithTransactionCommitter(committer))
	name := nest.NewHandlerName("aud9_row4_counterexample")
	engine.MustRegisterHandlerWithMeta(name, func([]entity.IThreadSafeEntity, []any, ...nest.HandlerOption) (any, error) {
		live.set("committed-remote", "")
		return "ok", nil
	}, nest.HandlerMeta{Rollback: nest.RollbackUndo, Durability: nest.DurabilityStrict})
	if err := engine.Start(); err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() { _ = engine.Shutdown(context.Background()) })

	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	reply, err := engine.Request(ctx, name, live.ID(), nil)
	cancel()
	row, hits := firstTableRow(err)
	t.Logf("PROBE reply=%v err=%v", reply, err)
	t.Logf("PROBE first-row=%d sentinels=[%s] overloaded=%v localCommits=%d", row, hits, errors.Is(err, entity.ErrRemoteOverloaded), committer.calls.Load())
	status, statusErr := f.storage.CommitStatus(context.Background(), txID)
	t.Logf("PROBE authority CommitStatus(tx)=%v err=%v", status.State, statusErr)
	if errors.Is(err, nest.ErrRemotePartRejected) && status.State == entity.RemoteCommitCommitted {
		t.Errorf("PROBE COUNTEREXAMPLE: reply claims the Remote part was rejected (row 4) but the authority committed the Remote write")
	}
}
