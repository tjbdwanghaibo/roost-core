package remoteentity
import("context";"errors";"testing";"github.com/tjbdwanghaibo/roost-core/entity")
type reviewModeLoss struct{*lostReplyMarkerStore; before bool}
func(s *reviewModeLoss) EnterSharedExpected(ctx context.Context,id int64,e entity.RemoteEntityMarkerLease)(entity.RemoteEntityMarkerLease,error){
 if s.before{return entity.RemoteEntityMarkerLease{},errors.New("before send")};v,err:=s.mockMarkerStore.EnterSharedExpected(ctx,id,e);if err!=nil{return v,err};return entity.RemoteEntityMarkerLease{},errors.New("reply lost")
}
func(s *reviewModeLoss) LeaveSharedExpected(ctx context.Context,id int64,e entity.RemoteEntityMarkerLease)(entity.RemoteEntityMarkerLease,error){
 if s.before{return entity.RemoteEntityMarkerLease{},errors.New("before send")};v,err:=s.mockMarkerStore.LeaveSharedExpected(ctx,id,e);if err!=nil{return v,err};return entity.RemoteEntityMarkerLease{},errors.New("reply lost")
}
func TestReviewModeReplyLoss(t *testing.T){for _,leave:=range []bool{false,true}{for _,before:=range []bool{false,true}{name:="enter";if leave{name="leave"};if before{name+="_before"};t.Run(name,func(t *testing.T){
 ctx:=context.Background();m,s,live:=indeterminateTransferSetup(t,246,15001);if leave{if _,e:=m.EnterRemoteSharedMode(ctx,live.GUId());e!=nil{t.Fatal(e)}}
 m.SetOwnershipStore(&reviewModeLoss{lostReplyMarkerStore:s,before:before})
 var err error;if leave{_,err=m.LeaveRemoteSharedMode(ctx,live.GUId())}else{_,err=m.EnterRemoteSharedMode(ctx,live.GUId())};if err==nil{t.Fatal("injected error absent")}
 authority:=s.leases[live.GUId()];state:=live.RemoteOwnershipState()
 batch,admit:=m.PrepareRemoteWriteBatch(ctx,[]int64{live.GUId()});if batch!=nil{batch.Abort(ctx,errors.New("cleanup"));batch.Close(ctx)}
 t.Logf("before=%v leave=%v authority_shared=%v state=%s admission=%v",before,leave,authority.Shared,state,admit)
 if before{if admit!=nil{t.Fatalf("before-send recovery failed: %v",admit)};return}
 if !leave&&state==entity.RemoteOwnershipLocalOwned&&admit==nil{t.Fatal("entered shared at authority but old local-owned write admitted")}
 })}}}
