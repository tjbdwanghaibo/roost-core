# RR-20260927-27：非 Nest 领头方持有实体锁时触发冷加载，loader 发布要锁该实体则永久死锁（RR-54 相邻形状）

日期：2026-09-27。基线：`d4234c9`（批次 4 修复方复现），现 main `00f67c7`。来源：OPEN-ITEMS B20 的相邻形状（批次 4 修复方发现）。等级 **P4**：仓内正式 loader 的发布只锁被加载实体，未见该调用形态。引入：RR-20260926-54（发布从领头方 goroutine 挪到加载 goroutine）。状态：已修复（[修复记录](../bugfix/RR-20260927-27.md)，分支 `fix-r7-b12`，基线 `7b56060`），本地提交，未推送。

非 Nest 的领头方（`WithGuardScope` 持有 X）触发冷加载 Y，loader 的发布还需要 X 的锁：加载 goroutine 等 X，领头方等加载；领头方自己的 ctx 到期也离不开——`leaderLeft` 要等持有 `leaderMu`
的在途发布结束（`entity/manager_access.go` `runLocal` / `leaderLeft`）。修前发布在领头方 goroutine 上，可重入锁不阻塞。红探针（needX=true）：`leader (own ctx 300ms) still stuck after 3s`。

修复约束：领头方按自己的 ctx 离开的承诺（RR-54）必须成立——`leaderLeft` 不得无界等待在途发布；可选：发布需要领头方已持有的锁时在领头方 goroutine 上执行（恢复修前行为）或让发布失败返回错误，二选一论证；不在快 worker 上等待；RR-54 与 B20 回归不回退。

红探针（批次 4）：

```go
package entity

import (
	"context"
	"errors"
	"testing"
	"time"
)

type lockingPublishLoader struct {
	manager *EntityManager
	extra   IThreadSafeEntity // publish 还要锁的实体（nil 表示只锁新实体）
}

func (l *lockingPublishLoader) LoadEntity(ctx context.Context, id int64, _ EntityKind) (IThreadSafeEntity, error) {
	value := newMgrTestEntity(id, testEntityCategoryPlayer)
	var addErr error
	if err := RunLocal(ctx, func() {
		addErr = WithGuardScope("probe_publish", func(scope *GuardScope) error {
			if l.extra != nil && !scope.Guard().RequireEntity(l.extra) {
				return errors.New("extra lock")
			}
			if !scope.Guard().RequireEntity(value) {
				return errors.New("lock new")
			}
			return l.manager.TryAdd(value)
		})
	}); err != nil {
		return nil, err
	}
	return value, addErr
}

// 探针 OPEN-ITEMS B20：非 Nest 领头方（WithGuardScope 持有 X）冷加载 Y。needX=false：loader 发布只锁 Y；
// needX=true：loader 发布还要锁领头方持有的 X（RR-54 前发布在领头方 goroutine 上、可重入）。
func TestProbeB20(t *testing.T) {
	for _, needX := range []bool{false, true} {
		manager := NewEntityManager()
		access := NewManagerAccess(manager)
		x := newMgrTestEntity(mustBuildTestEntityID(t, 5501, testEntityCategoryPlayer, EntityKindNone), testEntityCategoryPlayer)
		if err := manager.TryAdd(x); err != nil {
			t.Fatal(err)
		}
		loader := &lockingPublishLoader{manager: manager}
		if needX {
			loader.extra = x
		}
		if _, err := access.ConfigureLoader(loader); err != nil {
			t.Fatal(err)
		}
		y := mustBuildTestEntityID(t, 5502, testEntityCategoryPlayer, EntityKindNone)
		done := make(chan error, 1)
		go func() {
			done <- WithGuardScope("probe_leader", func(scope *GuardScope) error {
				if !scope.Guard().RequireEntity(x) {
					return errors.New("lock x")
				}
				ctx, cancel := context.WithTimeout(context.Background(), 300*time.Millisecond)
				defer cancel()
				_, err := access.Get(ctx, y, EntityCategoryNone)
				return err
			})
		}()
		select {
		case err := <-done:
			t.Logf("needX=%v err=%v published=%v", needX, err, manager.Get(y) != nil)
			if err != nil || manager.Get(y) == nil {
				t.Errorf("needX=%v: leader err=%v published=%v", needX, err, manager.Get(y) != nil)
			}
		case <-time.After(3 * time.Second):
			t.Errorf("needX=%v: leader (own ctx 300ms) still stuck after 3s: the load goroutine's publish waits for X held by the leader, and the leader cannot leave because leaderLeft waits for that publish (deadlock)", needX)
		}
	}
}
```
