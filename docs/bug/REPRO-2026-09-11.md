## 新 Mail 边界

将下面代码保存为 mail-boundaries_test.go，映射到 kit/service/mail/review_boundaries_test.go。在 Kit 执行：

```text
go test -race -overlay <mail-boundaries.json> ./service/mail -run '^TestReview' -v -count=1 -timeout=120s
```

两项均 FAIL。容量用例只推进 400 秒，第一封邮件有效期仍为一周。全流程使用 Service API，不直接修改墓碑来构造容量问题。副本隔离用例则有意修改公开读取的返回值，检查这种修改是否泄漏到存储；不代表允许修改原始存储对象。
```go
package mail

import (
	"context"
	"errors"
	"fmt"
	"testing"
	"time"
)

func reviewSettleAndEvict(t *testing.T, h *harness) (Envelope, Claim) {
	t.Helper()
	ctx := context.Background()
	e := mustSend(t, h, withAttachment(directTo(1), "reward"))
	c, err := h.service.ReserveClaim(ctx, 1, e.ID, "")
	if err != nil {
		t.Fatal(err)
	}
	if _, err = h.service.CommitClaim(ctx, 1, e.ID, c.Token); err != nil {
		t.Fatal(err)
	}
	for i := 0; i < MaxMailboxEntries; i++ {
		if err = h.service.Deliver(ctx, 1, fmt.Sprintf("filler-%d", i), 0); err != nil {
			t.Fatal(err)
		}
	}
	box, _, err := h.service.Mailbox(ctx, 1)
	if err != nil {
		t.Fatal(err)
	}
	if _, ok := box.Entries[e.ID]; ok {
		t.Fatal("not evicted")
	}
	if _, ok := box.SettledClaims[e.ID]; !ok {
		t.Fatal("no tombstone")
	}
	return e, c
}
func TestReviewMailboxSnapshotOwnsSettledClaims(t *testing.T) {
	h := newHarness(t)
	e, _ := reviewSettleAndEvict(t, h)
	ctx := context.Background()
	a, _, err := h.service.Mailbox(ctx, 1)
	if err != nil {
		t.Fatal(err)
	}
	delete(a.SettledClaims, e.ID)
	b, _, err := h.service.Mailbox(ctx, 1)
	if err != nil {
		t.Fatal(err)
	}
	if _, ok := b.SettledClaims[e.ID]; !ok {
		t.Fatal("mutating returned mailbox deleted stored settled claim")
	}
}
func TestReviewTombstoneCapacityOutlivesValidEnvelope(t *testing.T) {
	h := newHarness(t)
	ctx := context.Background()
	e, first := reviewSettleAndEvict(t, h)
	if _, err := h.service.ReserveClaim(ctx, 1, e.ID, ""); !errors.Is(err, ErrAlreadyClaimed) {
		t.Fatalf("control: %v", err)
	}
	for i := 0; i < MaxSettledClaims; i++ {
		h.clock.advance(time.Second)
		if _, err := h.service.Delete(ctx, 1, fmt.Sprintf("filler-%d", i)); err != nil {
			t.Fatal(err)
		}
		req := withAttachment(directTo(1), "new")
		req.RequestID = fmt.Sprintf("new-%d", i)
		n := mustSend(t, h, req)
		c, err := h.service.ReserveClaim(ctx, 1, n.ID, "")
		if err != nil {
			t.Fatal(err)
		}
		if _, err = h.service.CommitClaim(ctx, 1, n.ID, c.Token); err != nil {
			t.Fatal(err)
		}
		h.clock.advance(time.Second)
		if err = h.service.Deliver(ctx, 1, fmt.Sprintf("next-filler-%d", i), 0); err != nil {
			t.Fatal(err)
		}
	}
	box, _, err := h.service.Mailbox(ctx, 1)
	if err != nil {
		t.Fatal(err)
	}
	if _, ok := box.SettledClaims[e.ID]; ok {
		t.Fatal("control: oldest tombstone retained")
	}
	if h.clock.Now().Unix() >= e.ExpiresAtUnix {
		t.Fatal("control: envelope expired")
	}
	if _, err = h.service.Delete(ctx, 1, "next-filler-0"); err != nil {
		t.Fatal(err)
	}
	if err = h.service.Deliver(ctx, 1, e.ID, 0); err != nil {
		t.Fatal(err)
	}
	again, err := h.service.ReserveClaim(ctx, 1, e.ID, "")
	if err == nil {
		t.Fatalf("valid claimed mail got new token after tombstone eviction: old=%s new=%s elapsed=400s", first.Token, again.Token)
	}
	if !errors.Is(err, ErrAlreadyClaimed) {
		t.Fatal(err)
	}
}

```

## Remote 容量及 TTL 验收代码

下方包含上一轮容量测试，以及本轮新增 TTL 变体。映射 core/remoteentity/review_external_test.go；若原容量测试已在另一 overlay 文件中，避免重复定义。TTL 变体把 TrackLimit 设为 8，直接向 pruneRemoteTransactionsLocked 传入未来清理时间（2 秒后），确保删除来自 TTL 而不是容量上限。保持与原用例相同的 Context.Done 屏障，验证等待者仍拿到已完成结果。

本轮旧容量用例随整包 race 通过；新增 TTL 用例以 `-run '^TestReviewTransactionWaitSurvivesTTLPrune$'` 单独通过。没有实际等待 2 秒，也没有直接 delete map。
```go
package remoteentity

import (
	"context"
	"fmt"
	"github.com/tjbdwanghaibo/roost-core/entity"
	"testing"
	"time"
)

type reviewSelectBarrier struct {
	context.Context
	entered chan struct{}
	resume  chan struct{}
}

func (c *reviewSelectBarrier) Done() <-chan struct{} { close(c.entered); <-c.resume; return nil }
func TestReviewTransactionWaitSurvivesEviction(t *testing.T) {
	cfg := DefaultConfig()
	cfg.TransactionTrackLimit = 1
	mgr := NewManager(newMockVersionedLockFactory(), cfg, 1000)
	first, second := remoteTestTxID(201), remoteTestTxID(202)
	barrier := &reviewSelectBarrier{Context: context.Background(), entered: make(chan struct{}), resume: make(chan struct{})}
	result := make(chan string, 1)
	go func() {
		defer func() {
			if p := recover(); p != nil {
				result <- fmt.Sprintf("panic: %v", p)
			}
		}()
		err := mgr.FlushRemoteTransaction(barrier, first)
		if err != nil {
			result <- err.Error()
		} else {
			result <- "ok"
		}
	}()
	select {
	case <-barrier.entered:
	case <-time.After(time.Second):
		t.Fatal("wait did not reach select")
	}
	mgr.completeRemoteTransaction(first, entity.RemoteCommitStatus{TransactionID: first, State: entity.RemoteCommitCommitted})
	if err := mgr.trackRemoteTransaction(second); err != nil {
		t.Fatal(err)
	}
	mgr.remote.txMu.Lock()
	_, retained := mgr.remote.txs[first]
	mgr.remote.txMu.Unlock()
	if retained {
		t.Fatal("control: completed tracker should have been evicted")
	}
	close(barrier.resume)
	select {
	case got := <-result:
		if got != "ok" {
			t.Fatalf("completed transaction waiter lost its result after cache eviction: %s", got)
		}
	case <-time.After(time.Second):
		t.Fatal("waiter stuck")
	}
}
func TestReviewTransactionWaitSurvivesTTLPrune(t *testing.T) {
	cfg := DefaultConfig()
	cfg.TransactionTrackLimit = 8
	cfg.TransactionTrackTTL = time.Second
	mgr := NewManager(newMockVersionedLockFactory(), cfg, 1000)
	first, second := remoteTestTxID(201), remoteTestTxID(202)
	barrier := &reviewSelectBarrier{Context: context.Background(), entered: make(chan struct{}), resume: make(chan struct{})}
	result := make(chan string, 1)
	go func() {
		defer func() {
			if p := recover(); p != nil {

		}()
		err := mgr.FlushRemoteTransaction(barrier, first)
		if err != nil {
			result <- err.Error()
		} else {
			result <- "ok"
		}
	}()
	select {
	case <-barrier.entered:
	case <-time.After(time.Second):
		t.Fatal("wait did not reach select")
	}
	mgr.completeRemoteTransaction(first, entity.RemoteCommitStatus{TransactionID: first, State: entity.RemoteCommitCommitted})
	mgr.remote.txMu.Lock()
	mgr.pruneRemoteTransactionsLocked(time.Now().Add(2 * time.Second).UnixNano())
	mgr.remote.txMu.Unlock()
	if err := mgr.trackRemoteTransaction(second); err != nil {
		t.Fatal(err)
	}
```
