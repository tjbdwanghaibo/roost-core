func TestReviewFailureRetry(t *testing.T){
 for _,op:=range []string{"subscribe","profile","unsubscribe","batch"}{for _,failure:=range []string{"reject","panic","cancel"}{t.Run(op+"/"+failure,func(t *testing.T){
  sink:=&recordingEnvelopeSink{};c:=NewSubscriptionCoordinator(sink);n:=0;s:=newSubscriptionTestState(t,101,&n);p:=entity.SyncProfile{Key:"a"};who:=SubscriberRef{ID:1};ctx:=context.Background()

  before,_:=c.Get(who,101);calls:=0
  c.SetSink(ReliableEnvelopeSinkFunc(func(context.Context,[]DeliveryEnvelope)error{calls++;if failure=="panic"{panic("review")};return errors.New("reject")}))
  if failure=="cancel"{x,cancel:=context.WithCancel(ctx);cancel();ctx=x}
  var err error
  switch op{case "subscribe":_,err=c.Subscribe(ctx,who,s,p);case "profile":_,err=c.Subscribe(ctx,who,s,entity.SyncProfile{Key:"b"});case "unsubscribe":err=c.Unsubscribe(ctx,who,101);case "batch":s.MarkDirty(1);q,e:=s.Prepare([]entity.SyncProfile{p});if e!=nil{t.Fatal(e)};err=c.Distribute(ctx,q)}
  if err==nil{t.Fatal("expected rejection")};if failure=="cancel"&&(!errors.Is(err,context.Canceled)||calls!=0){t.Fatalf("cancel err=%v calls=%d",err,calls)}
  after,ok:=c.Get(who,101);if op=="subscribe"{if ok||len(c.Profiles(101))!=0{t.Fatal("pending leaked")}}else if !ok||after.State!=SubscriptionActive||after.Profile!=before.Profile||after.ContentVersion!=before.ContentVersion{t.Fatalf("membership lost: %+v",after)}
  if op=="batch"&&(!s.PendingDirty()||s.Version()!=0){t.Fatal("failed admission consumed content")}
  c.SetSink(sink);ctx=context.Background()
  switch op{case "subscribe":_,err=c.Subscribe(ctx,who,s,p);case "profile":_,err=c.Subscribe(ctx,who,s,entity.SyncProfile{Key:"b"});case "unsubscribe":err=c.Unsubscribe(ctx,who,101);case "batch":err=c.FlushSubject(ctx,s)}
  if err!=nil{t.Fatal(err)};if op=="batch"&&(s.Version()!=1||s.PendingDirty()){t.Fatal("retry failed to commit")};if op=="unsubscribe"{if _,ok:=c.Get(who,101);ok{t.Fatal("leave retry retained member")}}
 })}}
