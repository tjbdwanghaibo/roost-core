## Remote 复现及 tracker 基准

首次实测 queue=29、slots=29。select 随机分支决定具体遗留数，64 次提高触发概率。测试末尾清理仅避免测试泄漏。

```go
package remoteentity

import (
	"context"
	"encoding/binary"
	"fmt"
	"github.com/tjbdwanghaibo/roost-core/entity"
	"testing"
	"time"
)

func TestReviewLateBatchCloseAfterFinalizerStop(t *testing.T) {
	var raw []int64
	for i := 0; i < 64; i++ {
		raw = append(raw, int64(7300+i))
	}
	mgr, ids := batchHarness(t, 239, raw...)
	ctx := context.Background()
	var batches []entity.RemoteWriteBatch
	for i, id := range ids {
		b, err := mgr.PrepareRemoteWriteBatch(ctx, []int64{id})
		if err != nil {
			t.Fatal(err)
		}
		if err = b.FinalizeLocked(entity.NewRemoteTransactionOutcome(remoteTestTxID(byte(i+1)), "review-late-close", "", true, 2)); err != nil {
			t.Fatal(err)
		}
		if err = b.Indeterminate(ctx, fmt.Errorf("uncertain")); err != nil {
			t.Fatal(err)
		}
		batches = append(batches, b)
	}
	stopCtx, cancel := context.WithTimeout(ctx, time.Second)
	defer cancel()
	if err := mgr.StopFinalizer(stopCtx); err != nil {
		t.Fatal(err)
	}
	for _, b := range batches {
		if err := b.Close(ctx); err != nil {
			t.Fatal(err)
		}
	}
	queued, slots := len(mgr.remote.finalizeQueue), len(mgr.remote.finalizeSlots)
	// Test cleanup only: finalizer is already gone; reclaim stranded items.
	for len(mgr.remote.finalizeQueue) > 0 {
		item := <-mgr.remote.finalizeQueue
		mgr.releaseRemoteEntriesObserved(ctx, item.entries)
		mgr.releaseRemoteFinalizeSlot()
	}
	if queued != 0 || slots != 0 {
		t.Fatalf("Close returned success after Stop but stranded queue=%d slots=%d with no finalizer", queued, slots)
	}
}
func BenchmarkReviewTrackerAtCapacity(b *testing.B) {
	for _, size := range []int{64, 1024, 8192, 65536} {
		b.Run(fmt.Sprint(size), func(b *testing.B) {
			cfg := DefaultConfig()
			cfg.TransactionTrackLimit = size
			mgr := NewManager(newMockVersionedLockFactory(), cfg, 1000)
			makeID := func(n int) entity.RemoteTransactionID {
				var id entity.RemoteTransactionID
				binary.BigEndian.PutUint64(id[8:], uint64(n))
				return id
			}
			for i := 1; i <= size; i++ {
				id := makeID(i)
				mgr.completeRemoteTransaction(id, entity.RemoteCommitStatus{TransactionID: id, State: entity.RemoteCommitCommitted})
			}
			b.ReportAllocs()
			b.ResetTimer()
			for i := 0; i < b.N; i++ {
				id := makeID(size + i + 1)
				if err := mgr.trackRemoteTransaction(id); err != nil {
					b.Fatal(err)
				}
				mgr.completeRemoteTransaction(id, entity.RemoteCommitStatus{TransactionID: id, State: entity.RemoteCommitCommitted})
			}
		})
	}
}
```

## Nest 延迟请求停机

已注册有效 handler 的最终版本 race 失败：Shutdown 成功后 50ms 无回复，取消才退出。Request 自身仍有同步超时，不是永久无界等待。

```go
package nest

import (
	"context"
	"github.com/tjbdwanghaibo/roost-core/entity"
	"testing"
	"time"
)

func TestReviewShutdownAnswersAcceptedDelayedRequest(t *testing.T) {
	MustRegisterMemoryHandler(NewHandlerName("review-delayed-stop"), func([]entity.IThreadSafeEntity, []any, ...HandlerOption) (any, error) { return "ok", nil })
	getter := newMockGetter()
	id := mustBuildCastID(t, 7381, entity.EntityCategory(1), nestLocalKind)
	getter.Add(newMockEntity(id, entity.EntityCategory(1)))
	mgr := NewEngine(NestOptionWithGetter(getter), NestOptionWithWorkerNumAndMsgCap(1, 0, 8), NestOptionWithTickDuration(time.Hour))
	if err := mgr.Start(); err != nil {
		t.Fatal(err)
	}
	ctx, cancel := context.WithCancel(context.Background())
	defer cancel()
	done := make(chan error, 1)
	go func() {
		_, err := mgr.Request(ctx, NewHandlerName("review-delayed-stop"), id, nil, SendOptionWithDelay(time.Hour))
		done <- err
	}()
	deadline := time.Now().Add(time.Second)
	for mgr.dispatcher.delayedCount() == 0 {
		if time.Now().After(deadline) {
			t.Fatal("request was not admitted to delay queue")
		}
		time.Sleep(time.Millisecond)
	}
	stopCtx, stopCancel := context.WithTimeout(context.Background(), time.Second)
	defer stopCancel()
	if err := mgr.Shutdown(stopCtx); err != nil {
		t.Fatal(err)
	}
	select {
	case <-done:
	case <-time.After(50 * time.Millisecond):
		cancel()
		<-done
		t.Fatal("Shutdown returned success but accepted delayed Request received no result; waits until its own cancellation/timeout")
	}
}

```
