## 最终结果与假设

- TestReviewRealFanout/jetstream_same_bus：12 条消息应向两个本地 handler 各交付一份，实际 6/6。普通 NATS 和不同 SID 两个对照通过。
- TestReviewRealPacketFanout/identity、gzip：51/18 个分片均发布确认且消费者已全部结算，两边均得到 0 个 Packet。
- 其余 32 个叶子通过。恢复测试允许旧消息重复，要求新消息最终到达；“确认后注入一次 error”和 Apply 错误是契约观察，通过不意味着 exactly-once 或业务自动恢复。
- 最终执行 7.877s，exit 1，无 race 报告。第一次重连断言错误要求首条必为新消息，已纠正，初稿两个失败不计产品缺陷。
- “确认丢失”由 IJetStream 包装器在真实 Publish 成功后人为返回一次 error，不是 TCP 层丢包。Apply 错误仅观察 400ms 无重投，结合源码说明包装契约，不由短窗口断言永久无重投。

	for _, mode := range []string{"active_consumer", "durable_offline", "client_recreate"} {
		t.Run(mode, func(t *testing.T) {
			b := reviewNewBroker(t)
			name := "R_" + mode
			b.stream(name)
			cfg := b.consumer(name)
			received := make(chan string, 8)

			sub, e := b.js.Subscribe(context.Background(), cfg, handler)

				t.Fatal(e)
			}
			defer sub.Stop()

				t.Fatal("first payload")
			}

				sub.Stop()

			}
			if mode == "durable_offline" {

			}
			if mode == "client_recreate" {
				b.client.Close()
				c := fn.DefaultConfig(b.url())

				if e != nil {
					t.Fatal(e)

					t.Fatal(e)
				}

				b.start()
				reviewEventually(t, b.client.Connected)
				reviewEventually(t, b.raw.IsConnected)
			}
			if mode != "active_consumer" {
				next, e := b.js.Subscribe(context.Background(), cfg, handler)
				if e != nil {
					t.Fatal(e)
				}
				defer next.Stop()
			}

				b.publish(name, "two")
			}
			duplicates := 0

				got := reviewReceive(t, received)

					break
				}

					t.Fatalf("unexpected resumed payload=%s repeats=%d", got, duplicates)
				}
				duplicates++
			}
			t.Logf("recovered new message; old-message redeliveries=%d", duplicates)
			b.settled(name, 2)
		})
	}

func TestReviewRealPlainRequest(t *testing.T) {
	b := reviewNewBroker(t)
	for _, mode := range []string{"response", "no_responders", "timeout", "closed", "queue"} {
		t.Run(mode, func(t *testing.T) {
			subject := "Q." + mode
			switch mode {
			case "response":
				sub, e := b.raw.Subscribe(subject, func(m *gonats.Msg) { b.raw.Publish(m.Reply, []byte("reply")) })
				if e != nil {
					t.Fatal(e)
				}
				defer sub.Unsubscribe()
				b.raw.Flush()
				got, e := b.client.Request(subject, []byte("request"), time.Second)
				if e != nil || string(got) != "reply" {
					t.Fatalf("reply=%s err=%v", got, e)
				}
			case "no_responders":
				_, e := b.client.Request(subject, nil, time.Second)
				if !errors.Is(e, fn.ErrNoResponders) {
					t.Fatal(e)
				}
			case "timeout":
				sub, e := b.raw.Subscribe(subject, func(*gonats.Msg) {})
				if e != nil {
					t.Fatal(e)
				}
				defer sub.Unsubscribe()
				b.raw.Flush()
				_, e = b.client.Request(subject, nil, 50*time.Millisecond)
				if !errors.Is(e, fn.ErrTimeout) {
					t.Fatal(e)
				}
			case "closed":
				c, e := nd.NewClient(fn.DefaultConfig(b.url()), nd.ClientOptions{})
				if e != nil {
					t.Fatal(e)
				}
				c.Close()
				_, e = c.Request(subject, nil, time.Second)
				if !errors.Is(e, fn.ErrClosed) {
					t.Fatal(e)
				}
			case "queue":
				received := make(chan struct{}, 32)
				a, e := b.client.QueueSubscribe(subject, "workers", func(*fn.Msg) { received <- struct{}{} })
				if e != nil {
					t.Fatal(e)
				}
				defer a.Unsubscribe()
				z, e := b.client.QueueSubscribe(subject, "workers", func(*fn.Msg) { received <- struct{}{} })
				if e != nil {
					t.Fatal(e)
				}
				defer z.Unsubscribe()
				time.Sleep(100 * time.Millisecond)
				for i := 0; i < 12; i++ {
					if e = b.client.Publish(subject, nil); e != nil {
						t.Fatal(e)
					}
				}
				for i := 0; i < 12; i++ {
					reviewReceive(t, received)
				}
				time.Sleep(100 * time.Millisecond)
				if len(received) != 0 {
					t.Fatal("queue duplicated delivery")
				}
			}
		})
	}
}

func TestReviewRealPacketFanout(t *testing.T) {
	b := reviewNewBroker(t)
	for _, mode := range []string{"identity", "gzip"} {
		t.Run(mode, func(t *testing.T) {
			name := "M_" + mode
			receiver, e := NewJetStreamSyncBus(context.Background(), b.js, JetStreamSyncConfig{LocalSid: 7, Prefix: name, Stream: name})
			if e != nil {
				t.Fatal(e)
			}
			defer receiver.Stop()
			a, z := make(chan ss.Packet, 4), make(chan ss.Packet, 4)
			u, e := ss.Subscribe(receiver, "state", func(p ss.Packet) error { a <- p; return nil })
			if e != nil {
				t.Fatal(e)
			}
			defer u()
			v, e := ss.Subscribe(receiver, "state", func(p ss.Packet) error { z <- p; return nil })
			if e != nil {
				t.Fatal(e)
			}
			defer v()
			opt := ss.PublisherOptions{FromSID: 9, RequireConfirmation: true, MaxFrameBytes: 10}
			if mode == "gzip" {
				opt.CompressionThreshold = 1
			}
			pub, e := ss.NewPublisherWithOptions(receiver, opt)
			if e != nil {
				t.Fatal(e)
			}
			time.Sleep(100 * time.Millisecond)
			if e = pub.Publish(ss.Packet{Stream: ss.Stream{Topic: "state", Key: 1}, Epoch: 1, Sequence: 1, Payload: bytes.Repeat([]byte("part"), 60)}); e != nil {
				t.Fatal(e)
			}
			reviewEventually(t, func() bool {
				consumer, err := b.admin.Consumer(context.Background(), name, durableSyncName("state", 7))
				if err != nil {
					return false
				}
				info, err := consumer.Info(context.Background())
				return err == nil && info.NumPending == 0 && info.NumAckPending == 0 && info.Delivered.Consumer >= pub.Metrics().Frames
			})
			if len(a) != 1 || len(z) != 1 {
				t.Fatalf("each local subscriber needs a complete packet; got %d/%d with %d broker-confirmed frames", len(a), len(z), pub.Metrics().Frames)
			}
		})
	}
}

func TestReviewRealDrain(t *testing.T) {
	for _, mode := range []string{"normal", "cancel_blocked_handler", "already_closed"} {
		t.Run(mode, func(t *testing.T) {
			b := reviewNewBroker(t)
			if mode == "already_closed" {
				b.client.Close()
				if e := b.client.DrainWithContext(context.Background()); e != nil {
					t.Logf("already closed drain returns %v", e)
					if !errors.Is(e, gonats.ErrConnectionClosed) {
						t.Fatal(e)
					}
				}
				return
			}
			entered, release, finished := make(chan struct{}), make(chan struct{}), make(chan struct{})
			var released atomic.Bool
			unblock := func() {
				if released.CompareAndSwap(false, true) {
					close(release)
				}
			}
			defer unblock()
			_, e := b.client.Subscribe("drain.event", func(*fn.Msg) { close(entered); <-release; close(finished) })
			if e != nil {
				t.Fatal(e)
			}
			time.Sleep(50 * time.Millisecond)
			if e = b.client.Publish("drain.event", nil); e != nil {
				t.Fatal(e)
			}
			reviewReceive(t, entered)
			ctx, cancel := context.WithTimeout(context.Background(), 2*time.Second)
			defer cancel()
			done := make(chan error, 1)
			go func() { done <- b.client.DrainWithContext(ctx) }()
			if mode == "normal" {
				select {
				case e := <-done:
					t.Fatalf("drain returned before handler: %v", e)
				case <-time.After(30 * time.Millisecond):
				}
				unblock()
				if e = reviewReceive(t, done); e != nil {
					t.Fatal(e)
				}
				reviewReceive(t, finished)
			} else {
				cancel()
				if e = reviewReceive(t, done); !errors.Is(e, context.Canceled) {
					t.Fatalf("drain error=%v", e)
				}
				if b.client.Connected() {
					t.Fatal("deadline did not close client")
				}
				unblock()
				reviewReceive(t, finished)
			}
		})
	}
}

type reviewBroker struct {
	t      *testing.T
	port   int
	dir    string
	cmd    *exec.Cmd
	log    *os.File
	client *nd.Client
	js     *nd.JetStreamClient
	raw    *gonats.Conn
	admin  gojs.JetStream
}

func reviewEventually(t *testing.T, f func() bool) {
	t.Helper()
	deadline := time.Now().Add(8 * time.Second)
	for time.Now().Before(deadline) {
		if f() {
			return
		}
		time.Sleep(10 * time.Millisecond)
	}
	t.Fatal("condition not reached within 8s")
}
func reviewReceive[T any](t *testing.T, ch <-chan T) T {
	t.Helper()
	select {
	case v := <-ch:
		return v
	case <-time.After(8 * time.Second):
		t.Fatal("delivery timeout")
		var zero T
		return zero
	}
}
func reviewNewBroker(t *testing.T) *reviewBroker {
	t.Helper()
	ln, e := net.Listen("tcp", "127.0.0.1:0")
	if e != nil {
		t.Fatal(e)
	}
	port := ln.Addr().(*net.TCPAddr).Port
	ln.Close()
	b := &reviewBroker{t: t, port: port, dir: t.TempDir()}
	t.Cleanup(func() {
		if b.client != nil {
			b.client.Close()
		}
		if b.raw != nil {
			b.raw.Close()
		}
		b.stop()
	})
	b.start()
	cfg := fn.DefaultConfig(b.url())
	cfg.ReconnectWait = 20 * time.Millisecond
	b.client, e = nd.NewClient(cfg, nd.ClientOptions{IgnoreDiscoveredServers: true})
	if e != nil {
		t.Fatal(e)
	}
	b.js, e = nd.NewJetStreamClient(b.client)
	if e != nil {
		t.Fatal(e)
	}
	b.raw, e = gonats.Connect(b.url(), gonats.ReconnectWait(20*time.Millisecond), gonats.MaxReconnects(-1))
	if e != nil {
		t.Fatal(e)
	}
	b.admin, e = gojs.New(b.raw)
	if e != nil {
		t.Fatal(e)
	}
	return b
}
func (b *reviewBroker) url() string { return fmt.Sprintf("nats://127.0.0.1:%d", b.port) }
func (b *reviewBroker) start() {
	b.t.Helper()
	var e error
	b.log, e = os.OpenFile(filepath.Join(b.dir, "server.log"), os.O_CREATE|os.O_APPEND|os.O_WRONLY, 0600)
	if e != nil {
		b.t.Fatal(e)
	}
	b.cmd = exec.Command(os.Getenv("ROOST_REVIEW_NATS"), "-js", "-a", "127.0.0.1", "-p", strconv.Itoa(b.port), "-sd", filepath.Join(b.dir, "data"))
	b.cmd.SysProcAttr = &syscall.SysProcAttr{HideWindow: true}
	b.cmd.Stdout = b.log
	b.cmd.Stderr = b.log
	if e = b.cmd.Start(); e != nil {
		b.t.Fatal(e)
	}
	reviewEventually(b.t, func() bool {
		c, e := net.DialTimeout("tcp", fmt.Sprintf("127.0.0.1:%d", b.port), 30*time.Millisecond)
		if e == nil {
			c.Close()
		}
		return e == nil
	})
}
func (b *reviewBroker) stop() {
	if b.cmd != nil {
		b.cmd.Process.Kill()
		b.cmd.Wait()
		b.cmd = nil
	}
	if b.log != nil {
		b.log.Close()
		b.log = nil
	}
}
func (b *reviewBroker) stream(name string) {
	b.t.Helper()
	if e := b.js.EnsureStream(context.Background(), fn.JetStreamConfig{Name: name, Subjects: []string{name + ".>"}, Storage: fn.JetStreamStorageFile, Duplicates: time.Second}); e != nil {
		b.t.Fatal(e)
	}
}
func (b *reviewBroker) publish(name, id string) fn.JetStreamPublishAck {
	b.t.Helper()
	a, e := b.js.Publish(context.Background(), name+".event", []byte(id), fn.JetStreamPublishOptions{MsgID: id})
	if e != nil {
		b.t.Fatal(e)
	}
	return a
}
func (b *reviewBroker) consumer(name string) fn.JetStreamConsumerConfig {
	return fn.JetStreamConsumerConfig{Stream: name, Name: "worker", Durable: "worker", FilterSubject: name + ".event", AckWait: 150 * time.Millisecond, MaxDeliver: 4}
}
func (b *reviewBroker) settled(name string, delivered uint64) {
	b.t.Helper()
	reviewEventually(b.t, func() bool {
		c, e := b.admin.Consumer(context.Background(), name, "worker")
		if e != nil {
			return false
		}
		i, e := c.Info(context.Background())
		return e == nil && i.NumAckPending == 0 && i.NumPending == 0 && i.Delivered.Consumer >= delivered
	})
}

func TestReviewRealPublish(t *testing.T) {
	b := reviewNewBroker(t)
	for _, mode := range []string{"ack", "same_id", "new_id", "cancel", "no_stream", "dedup_window", "restart_persistence"} {
		t.Run(mode, func(t *testing.T) {
			name := "P_" + mode
			b.stream(name)
			switch mode {
			case "ack":
				a := b.publish(name, "one")
				if a.Stream != name || a.Sequence != 1 || a.Duplicate {
					t.Fatalf("ack=%+v", a)
				}
			case "same_id":
				a := b.publish(name, "one")
				z := b.publish(name, "one")
				if !z.Duplicate || a.Sequence != z.Sequence {
					t.Fatalf("acks=%+v %+v", a, z)
				}
			case "new_id":
				a := b.publish(name, "one")
				z := b.publish(name, "two")
				if z.Duplicate || z.Sequence != a.Sequence+1 {
					t.Fatalf("acks=%+v %+v", a, z)
				}
			case "cancel":
				ctx, cancel := context.WithCancel(context.Background())
				cancel()
				_, e := b.js.Publish(ctx, name+".event", nil, fn.JetStreamPublishOptions{})
				if !errors.Is(e, context.Canceled) {
					t.Fatalf("error=%v", e)
				}
			case "no_stream":
				ctx, cancel := context.WithTimeout(context.Background(), 200*time.Millisecond)
				defer cancel()
				_, e := b.js.Publish(ctx, "absent.event", nil, fn.JetStreamPublishOptions{})
				if e == nil {
					t.Fatal("unpersisted publish succeeded")
				}
			case "dedup_window":
				b.publish(name, "one")
				time.Sleep(1100 * time.Millisecond)
				z := b.publish(name, "one")
				if z.Duplicate || z.Sequence != 2 {
					t.Fatalf("ack=%+v", z)
				}
			case "restart_persistence":
				b.publish(name, "one")
				b.stop()
				reviewEventually(t, func() bool { return !b.client.Connected() })
				b.start()
				reviewEventually(t, b.client.Connected)
				reviewEventually(t, b.raw.IsConnected)
				s, e := b.admin.Stream(context.Background(), name)
				if e != nil {
					t.Fatal(e)
				}
				info, e := s.Info(context.Background())
				if e != nil || info.State.Msgs != 1 {
					t.Fatalf("state=%+v err=%v", info, e)
				}
				z := b.publish(name, "two")
				if z.Sequence != 2 {
					t.Fatal(z)
				}
			}
		})
	}
}

func TestReviewRealDelivery(t *testing.T) {
	b := reviewNewBroker(t)
	for _, mode := range []string{"transient", "panic", "permanent", "max_deliver", "delayed", "deliver_new", "durable_resume", "stop_inflight"} {
		t.Run(mode, func(t *testing.T) {
			name := "D_" + mode
			b.stream(name)
			cfg := b.consumer(name)
			calls := make(chan uint64, 16)
			var seen atomic.Int32
			handler := func(ctx context.Context, m *fn.JetStreamMsg) error {
				n := seen.Add(1)
				calls <- m.NumDelivered
				switch mode {
				case "transient", "delayed":
					if n == 1 {
						return errors.New("retry")
					}
				case "panic":
					if n == 1 {
						panic("retry")
					}
				case "permanent":
					return fn.Permanent(errors.New("stop"))
				case "max_deliver":
					return errors.New("retry")
				case "stop_inflight":
					if n == 1 {
						<-ctx.Done()
						return ctx.Err()
					}
				}
				return nil
			}
			if mode == "max_deliver" {
				cfg.MaxDeliver = 2
			}
			if mode == "delayed" {
				cfg.NakBackoffMin = 100 * time.Millisecond
				cfg.NakBackoffMax = 100 * time.Millisecond
			}
			if mode == "deliver_new" {
				b.publish(name, "old")
				cfg.DeliverPolicy = fn.JetStreamDeliverNew
			}
			sub, e := b.js.Subscribe(context.Background(), cfg, handler)
			if e != nil {
				t.Fatal(e)
			}
			defer sub.Stop()
			start := time.Now()
			b.publish(name, "one")
			if n := reviewReceive(t, calls); n != 1 {
				t.Fatalf("first=%d", n)
			}
			switch mode {
			case "transient", "panic", "delayed", "max_deliver":
				if n := reviewReceive(t, calls); n != 2 {
					t.Fatalf("retry=%d", n)
				}
				if mode == "delayed" && time.Since(start) < 90*time.Millisecond {
					t.Fatal("backoff ignored")
				}
				b.settled(name, 2)
			case "permanent", "deliver_new":
				b.settled(name, 1)
				time.Sleep(250 * time.Millisecond)
				if seen.Load() != 1 {
					t.Fatalf("unexpected repeat=%d", seen.Load())
				}
			case "durable_resume":
				b.settled(name, 1)
				sub.Stop()
				reviewReceive(t, sub.Closed())
				b.publish(name, "two")
				next, e := b.js.Subscribe(context.Background(), cfg, handler)
				if e != nil {
					t.Fatal(e)
				}
				defer next.Stop()
				reviewReceive(t, calls)
				b.settled(name, 2)
				if seen.Load() != 2 {
					t.Fatalf("calls=%d", seen.Load())
				}
			case "stop_inflight":
				sub.Stop()
				reviewReceive(t, sub.Closed())
				next, e := b.js.Subscribe(context.Background(), cfg, handler)
				if e != nil {
					t.Fatal(e)
				}
				defer next.Stop()
				if n := reviewReceive(t, calls); n != 2 {
					t.Fatalf("resume delivery=%d", n)
				}
				b.settled(name, 2)
			}
		})
	}
}

				if mode == "jetstream_distinct_sid" {
					two, e := NewJetStreamSyncBus(context.Background(), b.js, JetStreamSyncConfig{LocalSid: 8, Prefix: name, Stream: name})
					if e != nil {
						t.Fatal(e)
					}
					defer two.Stop()
					right = two
				}
			}
			a, c := make(chan struct{}, 64), make(chan struct{}, 64)
			u, e := left.Subscribe("state", func(*sb.SyncMsg) error { a <- struct{}{}; return nil })
			if e != nil {
				t.Fatal(e)
			}
			defer u()
			v, e := right.Subscribe("state", func(*sb.SyncMsg) error { c <- struct{}{}; return nil })
			if e != nil {
				t.Fatal(e)
			}
			defer v()
			time.Sleep(100 * time.Millisecond)
			for i := 1; i <= 12; i++ {
				if e = left.Publish(&sb.SyncMsg{MessageID: fmt.Sprintf("m%d", i), Topic: "state", FromSid: 9, Version: int64(i)}); e != nil {
					t.Fatal(e)

			deadline := time.Now().Add(time.Second)

type reviewUncertainJS struct {
	fn.IJetStream
	fired atomic.Bool
}

func (j *reviewUncertainJS) Publish(ctx context.Context, s string, data []byte, opts fn.JetStreamPublishOptions) (fn.JetStreamPublishAck, error) {
	ack, e := j.IJetStream.Publish(ctx, s, data, opts)
	if e == nil && j.fired.CompareAndSwap(false, true) {
		return fn.JetStreamPublishAck{}, errors.New("injected lost successful confirmation")
	}
	return ack, e
}
func TestReviewRealStreamRecovery(t *testing.T) {
	b := reviewNewBroker(t)
	for _, mode := range []string{"fragment_identity", "fragment_gzip", "apply_error_observation", "retry_uncertain_observation"} {
		t.Run(mode, func(t *testing.T) {
			name := "S_" + mode
			var js fn.IJetStream = b.js
			if mode == "retry_uncertain_observation" {
				js = &reviewUncertainJS{IJetStream: b.js}
			}
			sender, e := NewJetStreamSyncBus(context.Background(), js, JetStreamSyncConfig{LocalSid: 1, Prefix: name, Stream: name})
			if e != nil {
				t.Fatal(e)
			}
			defer sender.Stop()
			receiver, e := NewJetStreamSyncBus(context.Background(), b.js, JetStreamSyncConfig{LocalSid: 2, Prefix: name, Stream: name})
			if e != nil {
				t.Fatal(e)
			}
			defer receiver.Stop()
			received := make(chan ss.Packet, 8)
			u, e := ss.SubscribeWithOptions(receiver, "state", ss.SubscribeOptions{RequireChecksum: true}, func(p ss.Packet) error {
				received <- p
				if mode == "apply_error_observation" {
					return errors.New("apply failed")
				}
				return nil
			})
			if e != nil {
				t.Fatal(e)
			}
			defer u()
			options := ss.PublisherOptions{FromSID: 1, RequireConfirmation: true}
			if mode == "fragment_identity" || mode == "fragment_gzip" {
				options.MaxFrameBytes = 37
			}
			if mode == "fragment_gzip" {
				options.CompressionThreshold = 1
			}
			publisher, e := ss.NewPublisherWithOptions(sender, options)
			if e != nil {
				t.Fatal(e)
			}
			packet := ss.Packet{Stream: ss.Stream{Topic: "state", Key: 1}, Epoch: 1, Sequence: 1, Full: true, Payload: bytes.Repeat([]byte("state-"), 100)}
			if mode == "retry_uncertain_observation" {
				buffer, e := ss.NewBufferedPublisher(publisher, ss.BufferedPublisherOptions{MaxAttempts: 2})
				if e != nil {
					t.Fatal(e)
				}
				defer buffer.Close(context.Background())
				e = buffer.Publish(packet)
			} else {
				e = publisher.Publish(packet)
			}
			if e != nil {
				t.Fatal(e)
			}
			p := reviewReceive(t, received)
			if !bytes.Equal(p.Payload, packet.Payload) {
				t.Fatal("payload mismatch")
			}
			if mode == "retry_uncertain_observation" {
				p = reviewReceive(t, received)
				if p.Sequence != 1 {
					t.Fatal("unexpected retry sequence")
				}
				t.Log("same business packet delivered twice after injected lost confirmation")
			}
			if mode == "apply_error_observation" {
				time.Sleep(400 * time.Millisecond)
				select {
				case <-received:
					t.Fatal("unexpected retry against syncbus contract")
				default:
				}
				t.Log("Apply error was acknowledged by syncbus wrapper; no automatic business recovery")
			}
			if options.MaxFrameBytes > 0 && publisher.Metrics().Frames < 2 {
				t.Fatal("fragmentation not exercised")
			}
		})
	}
}
