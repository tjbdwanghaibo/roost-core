package mirror

import (
 "context"
 "errors"
 "sync"
 "sync/atomic"
 "testing"
 "time"
 sb "github.com/tjbdwanghaibo/roost-core/syncbus"
)

type reviewLifecycleBus struct {
 mu sync.Mutex
 h sb.Handler
 attempts, unsubs int
 fail bool
}
func (b *reviewLifecycleBus) Publish(m *sb.SyncMsg) error {
 b.mu.Lock(); h := b.h; b.mu.Unlock()
 if h != nil { return h(m) }; return nil
}
func (b *reviewLifecycleBus) Subscribe(_ string,h sb.Handler)(func(),error){
 b.mu.Lock(); defer b.mu.Unlock(); b.attempts++
 if b.fail { b.fail=false; return nil,errors.New("subscribe failed") }
 b.h=h
 return func(){ b.mu.Lock(); b.h=nil; b.unsubs++; b.mu.Unlock() },nil
}
type reviewLifecycleStore struct {
 entered chan context.Context
 release chan struct{}
 count atomic.Int32
}
func (s *reviewLifecycleStore) ApplyReplica(ctx context.Context,_ Envelope)error{
 if s.entered != nil { s.entered<-ctx; <-s.release }
 s.count.Add(1); return nil
}
func TestReviewStopDoesNotDrainApply(t *testing.T){
 b:= &reviewLifecycleBus{}
 s:= &reviewLifecycleStore{entered:make(chan context.Context,1),release:make(chan struct{})}
 r:=New(b,"review",s); if err:=r.Start();err!=nil{t.Fatal(err)}
 done:=make(chan error,1)
 go func(){done<-b.Publish(&sb.SyncMsg{Topic:"review",Key:1,Version:1})}()
 var ctx context.Context
 select {case ctx= <-s.entered: case <-time.After(time.Second):t.Fatal("apply not entered")}
 stopped:=make(chan struct{}); go func(){r.Stop();close(stopped)}()
 select {case <-stopped: case <-time.After(time.Second):close(s.release);t.Fatal("Stop unexpectedly drains")}
 if ctx.Err()!=nil{t.Errorf("apply context unexpectedly canceled: %v",ctx.Err())}
 if s.count.Load()!=0{t.Error("apply completed before release")}
 close(s.release)
 if err:=<-done;err!=nil{t.Fatal(err)}
 if s.count.Load()!=1{t.Fatal("apply did not complete after Stop")}
}
func TestReviewOldCallbackSurvivesRestart(t *testing.T){
 b:= &reviewLifecycleBus{};s:= &reviewLifecycleStore{};r:=New(b,"review",s)
 if err:=r.Start();err!=nil{t.Fatal(err)}
 b.mu.Lock(); old:=b.h; b.mu.Unlock()
 r.Stop();if err:=r.Start();err!=nil{t.Fatal(err)};defer r.Stop()
 if err:=old(&sb.SyncMsg{Topic:"review",Key:1,Version:1});err!=nil{t.Fatal(err)}
 if s.count.Load()!=1{t.Fatal("old callback no longer applies; revisit boundary")}
}
func TestReviewStartFailureRetryAndIdempotence(t *testing.T){
 b:= &reviewLifecycleBus{fail:true};s:= &reviewLifecycleStore{};r:=New(b,"review",s)
 if r.Start()==nil{t.Fatal("expected failure")}
 if err:=r.Start();err!=nil{t.Fatal(err)}
 if err:=r.Start();err!=nil{t.Fatal(err)}
 r.Stop();r.Stop()
 if b.attempts!=2||b.unsubs!=1{t.Fatalf("attempts=%d unsubs=%d",b.attempts,b.unsubs)}
}
