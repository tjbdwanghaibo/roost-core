package remoteentity

import (
	"context"
	"encoding/json"
	"github.com/tjbdwanghaibo/roost-core/entity"

	"github.com/tjbdwanghaibo/roost-core/syncbus"
	"testing"
	"time"

type reviewBus struct {
	handlers map[string]syncbus.Handler
	messages []*syncbus.SyncMsg
}

func (b *reviewBus) Subscribe(s string, h syncbus.Handler) (func(), error) {
	b.handlers[s] = h
	return func() { delete(b.handlers, s) }, nil
}
func (b *reviewBus) Publish(m *syncbus.SyncMsg) error {
	copyMsg := *m
	copyMsg.Data = append([]byte(nil), m.Data...)
	b.messages = append(b.messages, &copyMsg)
	return nil
}
func (b *reviewBus) deliver(t *testing.T, i int) {
	t.Helper()
	m := b.messages[i]
	if err := b.handlers[m.Topic](m); err != nil {
		t.Fatal(err)
	}
}
func reviewSetup(t *testing.T) (*Manager, *reviewBus, entity.RemoteSnapshotKey) {
	t.Helper()
	const kind entity.EntityKind = 228
	entity.MustRegisterEntityKindDefs(entity.EntityKindDef{Kind: kind, Category: 1, RemotePolicy: entity.RemotePolicyManaged})
	id, e := entity.BuildEntityID(923001, kind)
	if e != nil {
		t.Fatal(e)
	}
	m := NewManager(nil, DefaultConfig(), 1001)
	b := &reviewBus{handlers: map[string]syncbus.Handler{}}
	s, i := m.BindSync(b)
	if e = s.Start(); e != nil {
		t.Fatal(e)
	}
	if e = i.Start(); e != nil {
		t.Fatal(e)
	}
	t.Cleanup(func() { s.Stop(); i.Stop(); m.remote.finalizeCancel() })
	return m, b, entity.RemoteSnapshotKey{EntityID: id, Kind: kind, Scope: 1}
}
func reviewUpdate(key entity.RemoteSnapshotKey, v uint64) entity.RemoteSnapshotRecord {
	return entity.RemoteSnapshotRecord{Key: key, StateVersion: v, MarkerEpoch: 1, RouteEpoch: 1, Schema: 1, Codec: 1, Full: true, Data: []byte("state")}
}
func reviewPublish(t *testing.T, m *Manager, key entity.RemoteSnapshotKey, v uint64) {
	t.Helper()
	if err := m.remote.interests.renew(entity.RemoteSnapshotInterest{ConsumerSID: 2, Key: key, ExpiresAt: time.Now().Add(time.Hour).UnixNano()}); err != nil {
		t.Fatal(err)
	}
	if err := m.syncer.(entity.IRemoteSnapshotPublisher).PublishRemoteSnapshot(context.Background(), reviewUpdate(key, v)); err != nil {
		t.Fatal(err)
	}
}
func TestReviewDelayedDeleteKeepsNewSnapshot(t *testing.T) {
	m, b, key := reviewSetup(t)
	s := m.syncer.(entity.IRemoteSnapshotPublisher)
	if e := s.DeleteRemoteSnapshot(context.Background(), key, 2); e != nil {
		t.Fatal(e)

	}
}

func TestReviewSnapshotWireRejectsForeignScope(t *testing.T) {
	m, b, key := reviewSetup(t)
	reviewPublish(t, m, key, 1)
	msg := b.messages[0]
	var env mirror.Envelope
	if e := json.Unmarshal(msg.Data, &env); e != nil {
		t.Fatal(e)
	}
	var wire remoteSnapshotWire
	if e := json.Unmarshal(env.Payload, &wire); e != nil {
		t.Fatal(e)
	}
	other := key
	other.Scope = 2
	wire.Update.Key = other
	raw, e := json.Marshal(wire)
	if e != nil {
		t.Fatal(e)
	}
	env.Payload = raw
	msg.Data, e = json.Marshal(env)
	if e != nil {
		t.Fatal(e)
	}
	err := b.handlers[msg.Topic](msg)
	_, ok, _ := m.remote.cache.Get(context.Background(), other, entity.RemoteReadCached, 0)
	if err == nil || ok {
		t.Fatalf("payload scope differs from wire key but accepted: err=%v foreignCached=%v", err, ok)
	}
}
func TestReviewSnapshotAndInterestOrderedControl(t *testing.T) {
	m, b, key := reviewSetup(t)
	reviewPublish(t, m, key, 1)
	reviewPublish(t, m, key, 2)
	b.deliver(t, 1)
	b.deliver(t, 0)
	got, ok, e := m.remote.cache.Get(context.Background(), key, entity.RemoteReadCached, 0)

		t.Fatalf("upsert order control: %+v %v %v", got, ok, e)

		t.Fatal(e)
	}
	b.deliver(t, 2)
	_, ok, e = m.remote.cache.Get(context.Background(), key, entity.RemoteReadCached, 0)
	if e != nil || ok {
		t.Fatal("ordered delete failed")
	}

	interest := entity.RemoteSnapshotInterest{ConsumerSID: 33, Key: key, ExpiresAt: time.Now().Add(time.Hour).UnixNano()}

		t.Fatal(e)
	}
	b.deliver(t, 4)
	if _, ok = m.remote.interests.entries[key][33]; ok {
		t.Fatal("ordered release failed")
	}
}

	}

package entity

import (
	"context"
	"errors"
	"runtime"
	"testing"
	"time"
)

func TestReviewSnapshotCanceledWaiterReturnsSlot(t *testing.T) {
	const kind EntityKind = 227
	MustRegisterEntityKindDefs(EntityKindDef{Kind: kind, Category: 1, RemotePolicy: RemotePolicyManaged})
	id, e := BuildEntityID(913002, kind)
	if e != nil {
		t.Fatal(e)
	}
	key := RemoteSnapshotKey{EntityID: id, Kind: kind}
	started := make(chan struct{})
	release := make(chan struct{})
	leaderDone := make(chan error, 1)
	c := NewRemoteSnapshotCache(RemoteSnapshotCacheConfig{MaxWaiters: 1, LoadTimeout: time.Minute}, nil, func(ctx context.Context, k RemoteSnapshotKey, _ RemoteReadConsistency, _ uint64) (RemoteSnapshotEnvelope, bool, error) {
		close(started)
		select {
		case <-release:
		case <-ctx.Done():
			return RemoteSnapshotEnvelope{}, false, ctx.Err()
		}
		return RemoteSnapshotEnvelope{Key: k, StateVersion: 1, MarkerEpoch: 1, RouteEpoch: 1, Schema: 1, Full: true, Payload: CopyFrozenRemoteSnapshotPayload([]byte("ok"))}, true, nil
	})
	go func() { _, _, err := c.Get(context.Background(), key, RemoteReadMonotonic, 1); leaderDone <- err }()
	<-started
	defer func() {
		close(release)
		if err := <-leaderDone; err != nil {
			t.Error(err)
		}
	}()
	ctx, cancel := context.WithCancel(context.Background())
	defer cancel()
	done := make(chan error, 1)
	go func() { _, _, err := c.Get(ctx, key, RemoteReadMonotonic, 1); done <- err }()
	deadline := time.Now().Add(time.Second)
	for {
		c.loadMu.Lock()
		n := c.loads[remoteSnapshotLoadKey{key: key, minVersion: 1}].waiters
		c.loadMu.Unlock()
		if n == 1 {
			break
		}
		if time.Now().After(deadline) {
			t.Fatal("waiter never joined")
		}
		runtime.Gosched()
	}
	cancel()
	if err := <-done; !errors.Is(err, context.Canceled) {
		t.Fatal(err)
	}
	ctx2, cancel2 := context.WithTimeout(context.Background(), 20*time.Millisecond)
	defer cancel2()
	_, _, err := c.Get(ctx2, key, RemoteReadMonotonic, 1)
	if errors.Is(err, ErrRemoteOverloaded) {
		t.Fatal("no active follower, but replacement rejected with ErrRemoteOverloaded")
	}
	if !errors.Is(err, context.DeadlineExceeded) {
		t.Fatalf("replacement did not wait: %v", err)
	}
}
