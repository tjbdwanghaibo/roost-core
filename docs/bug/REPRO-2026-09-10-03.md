```go
package remoteentity

import (
	"context"
	"fmt"
	"github.com/tjbdwanghaibo/roost-core/entity"
	"testing"
	"time"
)

type reviewSelectBarrier struct {
	context.Context
	entered chan struct{}
	resume  chan struct{}
}

func (c *reviewSelectBarrier) Done() <-chan struct{} { close(c.entered); <-c.resume; return nil }
func TestReviewTransactionWaitSurvivesEviction(t *testing.T) {
	cfg := DefaultConfig()
	cfg.TransactionTrackLimit = 1
	mgr := NewManager(newMockVersionedLockFactory(), cfg, 1000)
	first, second := remoteTestTxID(201), remoteTestTxID(202)
	barrier := &reviewSelectBarrier{Context: context.Background(), entered: make(chan struct{}), resume: make(chan struct{})}
	result := make(chan string, 1)
	go func() {
		defer func() {
			if p := recover(); p != nil {
				result <- fmt.Sprintf("panic: %v", p)
			}
		}()
		err := mgr.FlushRemoteTransaction(barrier, first)
		if err != nil {
			result <- err.Error()
		} else {
			result <- "ok"
		}
	}()
	select {
	case <-barrier.entered:
	case <-time.After(time.Second):
		t.Fatal("wait did not reach select")
	}
	mgr.completeRemoteTransaction(first, entity.RemoteCommitStatus{TransactionID: first, State: entity.RemoteCommitCommitted})
	if err := mgr.trackRemoteTransaction(second); err != nil {
		t.Fatal(err)
	}
	mgr.remote.txMu.Lock()
	_, retained := mgr.remote.txs[first]
	mgr.remote.txMu.Unlock()
	if retained {
		t.Fatal("control: completed tracker should have been evicted")
	}
	close(barrier.resume)
	select {
	case got := <-result:
		if got != "ok" {
			t.Fatalf("completed transaction waiter lost its result after cache eviction: %s", got)
		}
	case <-time.After(time.Second):
		t.Fatal("waiter stuck")
	}
}
```

```go
package skill

import (
	"fmt"
	"testing"
)

func reviewWaitProgram(t *testing.T, name string, ticks int) (*Program, CompileEnvironment) {
	return compileRuntimeJSON(t, asyncSkillJSON(name, `{"type":"none"}`, fmt.Sprintf(`{"flow":"wait","ticks":%d,"then":{"flow":"finish"}}`, ticks)))
}
func TestReviewCheckpointKeepsCompletionRetentionOrder(t *testing.T) {
	slow, env := reviewWaitProgram(t, "review-slow", 10)
	fast, _ := reviewWaitProgram(t, "review-fast", 1)
	host := runtimeTestHost(env)
	original := NewRuntime(host, RuntimeOptions{CompletedCastLimit: 2})
	slowID, err := original.Activate(slow, CastInput{Caster: 1})
	if err != nil {
		t.Fatal(err)
	}
	fastID, err := original.Activate(fast, CastInput{Caster: 1})
	if err != nil {
		t.Fatal(err)
	}
	if err = original.Advance(10); err != nil {
		t.Fatal(err)
	}
	cp, err := original.Checkpoint()
	if err != nil {
		t.Fatal(err)
	}
	resolver := ProgramResolverFunc(func(id, digest string) (*Program, error) {
		for _, p := range []*Program{slow, fast} {
			if p.id == id && p.identity.gameplayDigest == digest {
				return p, nil
			}
		}
		return nil, ErrCheckpointProgram
	})
	recoveryHost := runtimeTestHost(env)
	recoverySource := NewRuntime(recoveryHost, RuntimeOptions{CompletedCastLimit: 2})
	if _, err = recoverySource.Activate(slow, CastInput{Caster: 1}); err != nil {
		t.Fatal(err)
	}
	if _, err = recoverySource.Activate(fast, CastInput{Caster: 1}); err != nil {
		t.Fatal(err)
	}
	if err = recoverySource.Advance(10); err != nil {
		t.Fatal(err)
	}
	cp, err = recoverySource.Checkpoint()
	if err != nil {
		t.Fatal(err)
	}
	restored, err := RestoreRuntime(recoveryHost, RuntimeOptions{}, cp, resolver)
	if err != nil {
		t.Fatal(err)
	}
	// Each runtime runs only a wait/finish flow: no host mutation or asset effect.
	for _, r := range []*Runtime{original, restored} {
		if _, err = r.Activate(fast, CastInput{Caster: 1}); err != nil {
			t.Fatal(err)
		}
		if err = r.Advance(11); err != nil {
			t.Fatal(err)
		}
	}
	_, originalSlow := original.InspectCast(slowID)
	_, restoredSlow := restored.InspectCast(slowID)
	_, originalFast := original.InspectCast(fastID)
	_, restoredFast := restored.InspectCast(fastID)
	if originalSlow != restoredSlow || originalFast != restoredFast {
		t.Fatalf("retained casts diverged after restore: original slow=%v fast=%v; restored slow=%v fast=%v", originalSlow, originalFast, restoredSlow, restoredFast)
	}
}

		t.Fatal(err)

		t.Fatal(err)

	if err != nil {
		t.Fatal(err)
	}
	restored, err := RestoreRuntime(host, RuntimeOptions{}, cp, ProgramResolverFunc(func(id, digest string) (*Program, error) {
		if id == p.id && digest == p.identity.gameplayDigest {
			return p, nil

	}
}
```

```go
package mail

import (
	"context"
	"errors"
	"fmt"
	"github.com/tjbdwanghaibo/roost-core/versionstore"
	"testing"
)

type reviewStampLedger struct {
	SendLedger
	fail bool
}

func (s *reviewStampLedger) Update(ctx context.Context, key string, mutate versionstore.Mutate[SentRecord]) (versionstore.Versioned[SentRecord], bool, error) {
	if s.fail {
		s.fail = false
		return versionstore.Versioned[SentRecord]{}, false, fmt.Errorf("injected delivery stamp failure")
	}
	return s.SendLedger.Update(ctx, key, mutate)
}
func TestReviewStampFailureReplayPreservesClaim(t *testing.T) {
	ledger := &reviewStampLedger{SendLedger: versionstore.NewMemoryStore[string, SentRecord](), fail: true}
	h := newHarness(t, func(cfg *Config) { cfg.Sends = ledger })
	ctx := context.Background()
	req := withAttachment(directTo(1), "reward")
	env := mustSend(t, h, req)
	stamp, _, err := ledger.Get(ctx, req.RequestID)
	if err != nil || stamp.Value.DeliveredAtUnix != 0 {
		t.Fatal("control: missing stamp not reproduced", err)
	}
	claim, err := h.service.ReserveClaim(ctx, 1, env.ID, "")
	if err != nil {
		t.Fatal(err)
	}
	if _, err = h.service.CommitClaim(ctx, 1, env.ID, claim.Token); err != nil {
		t.Fatal(err)
	}
	again, err := h.service.Send(ctx, req)
	if err != nil || again.ID != env.ID {
		t.Fatal("send replay failed", err)
	}
	if _, err = h.service.ReserveClaim(ctx, 1, env.ID, ""); !errors.Is(err, ErrAlreadyClaimed) {
		t.Fatalf("replay resurrected claim: %v", err)
	}
	stamp, _, err = ledger.Get(ctx, req.RequestID)
	if err != nil || stamp.Value.DeliveredAtUnix == 0 {
		t.Fatal("stamp not repaired", err)
	}
	box, _, err := h.service.Mailbox(ctx, 1)
	if err != nil || box.Unread != 0 || box.Entries[env.ID].ClaimToken != claim.Token {
		t.Fatal("replay changed status/token", err)
	}
}
func TestReviewPartialBroadcastRetryKeepsFirstRecipientClaimed(t *testing.T) {
	var service *Service
	attempts := 0
	h := newHarness(t, func(cfg *Config) {
		cfg.Broadcast = DelivererFunc(func(ctx context.Context, e Envelope) error {
			attempts++
			if err := service.Deliver(ctx, 1, e.ID, 0); err != nil {
				return err
			}
			if attempts == 1 {
				return fmt.Errorf("injected partial fanout failure")
			}
			return service.Deliver(ctx, 2, e.ID, 0)
		})
	})
	service = h.service
	ctx := context.Background()
	req := SendRequest{Audience: AudienceBroadcast, Scope: "server-review", Subject: "reward", ExpiresInSeconds: 3600, RequestID: "review-partial", Attachment: []byte("reward")}
	env, err := service.Send(ctx, req)
	if err == nil || env.ID == "" {
		t.Fatal("control: partial failure missing")
	}
	claim, err := service.ReserveClaim(ctx, 1, env.ID, "server-review")
	if err != nil {
		t.Fatal(err)
	}
	if _, err = service.CommitClaim(ctx, 1, env.ID, claim.Token); err != nil {
		t.Fatal(err)
	}
	if _, err = service.Send(ctx, req); err != nil {
		t.Fatal(err)
	}
	first, _, err := service.Mailbox(ctx, 1)
	if err != nil || first.Unread != 0 || first.Entries[env.ID].Status != StatusClaimed || first.Entries[env.ID].ClaimToken != claim.Token {
		t.Fatal("first recipient claim changed", err)
	}
	second, _, err := service.Mailbox(ctx, 2)
	if err != nil || second.Unread != 1 {
		t.Fatal("second recipient missing", err)
	}
	if _, err = service.Send(ctx, req); err != nil || attempts != 2 {
		t.Fatal("successful fanout replayed again", err, attempts)
	}
}
```
