## 饱和回退：子进程与正常对照

```go
package nest

import (
	"context"
	"fmt"
	"os"
	"os/exec"
	"testing"
	"time"
)

func TestReviewSaturatedCompletionIsPanicSafe(t *testing.T) {
	for _, mode := range []string{"normal", "panic"} {
		t.Run(mode, func(t *testing.T) {
			cmd := exec.Command(os.Args[0], "-test.run=^TestReviewSaturatedCompletionChild$", "-test.timeout=10s", "-test.v")
			cmd.Env = append(os.Environ(), "ROOST_REVIEW_COMPLETION_CHILD="+mode)
			out, err := cmd.CombinedOutput()
			if err != nil {
				t.Fatalf("saturated completion child failed: %v\n%s", err, out)
			}
		})
	}
}
func TestReviewSaturatedCompletionChild(t *testing.T) {
	mode := os.Getenv("ROOST_REVIEW_COMPLETION_CHILD")
	if mode == "" {
		t.Skip("subprocess only")
	}
	p := newCompletionPump(1, 1)
	p.start()
	entered := make(chan struct{})
	release := make(chan struct{})
	if err := p.pool.Dispatch(1, &completionTask{run: func() { close(entered); <-release }}); err != nil {
		t.Fatal(err)
	}
	<-entered
	queued := 0
	for ; queued < 10000; queued++ {
		if err := p.pool.Dispatch(1, &completionTask{run: func() {}}); err != nil {
			break
		}
	}
	if queued == 10000 {
		t.Fatal("queue never saturated")
	}
	fmt.Printf("CONTROL: blocked worker, %d queued tasks, admission refused\n", queued)
	ticket := &fakeCommitTicket{lsn: 1, done: make(chan struct{})}
	close(ticket.done)
	tx := NewRollbackTx(RollbackUndo)
	tx.AfterCommit(func() {
		if mode == "panic" {
			panic("review saturated callback")
		}
	})
	msg, ch := GenSyncMsg(MsgTypeSingle)
	msg.Tid = 1
	deferred, _, _ := prepareCompletion(p, msg, nil, tx, ticket, "review-saturated", 17)
	if !deferred {
		t.Fatal("pump queue not accepted")
	}
	select {
	case ret := <-ch:
		if ret != 17 {
			t.Fatal(ret)
		}
	case <-time.After(time.Second):
		t.Fatal("reply missing")
	}
	close(release)
	if err := p.stop(context.Background()); err != nil {
		t.Fatal(err)
	}
	recycleMsg(msg)
}
```

## 真实 WAL：释放、截止、Sync 与重开

```go
package nestwal

import (
	"context"
	"errors"
	corenest "github.com/tjbdwanghaibo/roost-core/nest"
	"sync"
	"sync/atomic"
	"testing"
	"time"
)

func TestReviewHeldHeadBlocksSuccessorUntilRelease(t *testing.T) {
	w, err := Open(testOptions(t.TempDir()))
	if err != nil {
		t.Fatal(err)
	}
	var applies atomic.Int32
	c, err := NewCommitter(w, MutationApplyFunc(func(context.Context, corenest.TransactionID, corenest.EntityMutation) error {
		applies.Add(1)
		return nil
	}), EffectPublishFunc(func(context.Context, corenest.TransactionID, corenest.Effect) error { return nil }), DefaultCommitterOptions())
	if err != nil {
		t.Fatal(err)
	}
	defer c.Close(context.Background())
	a, b := testRecord(61, corenest.DurabilityPipelined), testRecord(62, corenest.DurabilityPipelined)
	for _, r := range []corenest.CommitRecord{a, b} {
		ticket, err := c.Enqueue(context.Background(), r)
		if err != nil {
			t.Fatal(err)
		}
		if err := waitTicket(t, ticket); err != nil {
			t.Fatal(err)
		}
	}
	c.TransactionReleased(b.ID)
	ctx, cancel := context.WithTimeout(context.Background(), time.Second)
	defer cancel()
	if err := c.Flush(ctx); !errors.Is(err, errTransactionHeld) {
		t.Fatalf("flush=%v", err)
	}
	if applies.Load() != 0 {
		t.Fatal("successor overtook held head")
	}
	c.TransactionReleased(a.ID)
	if err := c.Flush(ctx); err != nil {
		t.Fatal(err)
	}
	if applies.Load() != 2 {
		t.Fatalf("apply=%d", applies.Load())
	}
	if err := c.Close(ctx); err != nil {
		t.Fatal(err)
	}
	reopened, err := Open(testOptions(w.opts.Dir))
	if err != nil {
		t.Fatal(err)
	}
	defer reopened.Close(context.Background())
	remaining := 0
	if err := reopened.Replay(ctx, func(corenest.CommitFence, corenest.CommitRecord) error { remaining++; return nil }); err != nil {
		t.Fatal(err)
	}
	if remaining != 0 {
		t.Fatalf("ack not retained: remaining=%d", remaining)
	}
}
func TestReviewShutdownDeadlineDuringBackgroundApply(t *testing.T) {
	w, err := Open(testOptions(t.TempDir()))
	if err != nil {
		t.Fatal(err)
	}
	entered := make(chan struct{})
	release := make(chan struct{})
	var once sync.Once
	applier := MutationApplyFunc(func(ctx context.Context, _ corenest.TransactionID, _ corenest.EntityMutation) error {
		once.Do(func() { close(entered) })
		select {
		case <-release:
			return nil
		case <-ctx.Done():
			return ctx.Err()
		}
	})
	c, err := NewCommitter(w, applier, EffectPublishFunc(func(context.Context, corenest.TransactionID, corenest.Effect) error { return nil }), DefaultCommitterOptions())
	if err != nil {
		t.Fatal(err)
	}
	defer c.Close(context.Background())
	r := testRecord(63, corenest.DurabilityPipelined)
	ticket, err := c.Enqueue(context.Background(), r)
	if err != nil {
		t.Fatal(err)
	}
	if err := waitTicket(t, ticket); err != nil {
		t.Fatal(err)
	}
	c.TransactionReleased(r.ID)
	select {
	case <-entered:
	case <-time.After(time.Second):
		close(release)
		t.Fatal("background apply not entered")
	}
	ctx, cancel := context.WithTimeout(context.Background(), 20*time.Millisecond)
	defer cancel()
	done := make(chan error, 1)
	go func() { done <- c.Shutdown(ctx) }()
	<-ctx.Done()
	select {
	case err := <-done:
		close(release)
		if !errors.Is(err, context.DeadlineExceeded) {
			t.Fatalf("shutdown=%v", err)
		}
	case <-time.After(50 * time.Millisecond):
		close(release)
		<-done
		t.Fatal("Shutdown stayed blocked behind background replay after its context expired; returned only after releasing applier")
	}
}

func TestReviewSyncWaitsForAdmittedTicket(t *testing.T) {
	opts := testOptions(t.TempDir())
	opts.BatchDelay = time.Hour
	w, err := Open(opts)
	if err != nil {
		t.Fatal(err)
	}
	defer w.Close(context.Background())
	ticket, err := w.Enqueue(context.Background(), testRecord(64, corenest.DurabilityPipelined))
	if err != nil {
		t.Fatal(err)
	}
	ctx, cancel := context.WithTimeout(context.Background(), 30*time.Millisecond)
	defer cancel()
	err = w.Sync(ctx)
	if err == nil {
		select {
		case <-ticket.Done():
			if ticket.Err() != nil {
				t.Fatal(ticket.Err())
			}
		default:
			t.Fatalf("Sync returned nil before admitted ticket resolved: admitted=%d appended=%d durable_lsn=%d ticket_lsn=%d", w.Stats().Admitted, w.Stats().Appended, w.DurableLSN(), ticket.LSN())
		}
	} else if !errors.Is(err, context.DeadlineExceeded) {
		t.Fatal(err)
	}
}
func TestReviewCloseDrainsAdmittedTicketControl(t *testing.T) {
	opts := testOptions(t.TempDir())
	opts.BatchDelay = time.Hour
	w, err := Open(opts)
	if err != nil {
		t.Fatal(err)
	}
	defer w.Close(context.Background())
	ticket, err := w.Enqueue(context.Background(), testRecord(65, corenest.DurabilityPipelined))
	if err != nil {
		t.Fatal(err)
	}
	ctx, cancel := context.WithTimeout(context.Background(), time.Second)
	defer cancel()
	if err := w.Close(ctx); err != nil {
		t.Fatal(err)
	}
	select {
	case <-ticket.Done():
		if ticket.Err() != nil {
			t.Fatal(ticket.Err())
		}
	default:
		t.Fatal("Close left ticket unresolved")
	}
	if w.DurableLSN() < ticket.LSN() {
		t.Fatal("durability watermark missing")
	}
}

func TestReviewReopenRecoversDurableUnreleasedRecords(t *testing.T) {
	opts := testOptions(t.TempDir())
	w, err := Open(opts)
	if err != nil {
		t.Fatal(err)
	}
	var applies atomic.Int32
	applier := MutationApplyFunc(func(context.Context, corenest.TransactionID, corenest.EntityMutation) error {
		applies.Add(1)
		return nil
	})
	publisher := EffectPublishFunc(func(context.Context, corenest.TransactionID, corenest.Effect) error { return nil })
	c, err := NewCommitter(w, applier, publisher, DefaultCommitterOptions())
	if err != nil {
		t.Fatal(err)
	}
	defer c.Close(context.Background())
	for _, seq := range []byte{66, 67} {
		r := testRecord(seq, corenest.DurabilityPipelined)
		ticket, err := c.Enqueue(context.Background(), r)
		if err != nil {
			t.Fatal(err)
		}
		if err := waitTicket(t, ticket); err != nil {
			t.Fatal(err)
		}
	}
	if err := c.Close(context.Background()); err != nil {
		t.Fatal(err)
	}
	if applies.Load() != 0 {
		t.Fatal("unreleased records applied")
	}
	w2, err := Open(opts)
	if err != nil {
		t.Fatal(err)
	}
	c2, err := NewCommitter(w2, applier, publisher, DefaultCommitterOptions())
	if err != nil {
		t.Fatal(err)
	}
	defer c2.Close(context.Background())
	ctx, cancel := context.WithTimeout(context.Background(), time.Second)
	defer cancel()
	if err := c2.Flush(ctx); err != nil {
		t.Fatal(err)
	}
	if applies.Load() != 2 {
		t.Fatalf("reopen apply count=%d", applies.Load())
	}
}
```
