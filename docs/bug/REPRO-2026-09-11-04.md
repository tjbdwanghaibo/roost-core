## 新增测试

Panic 用例预期失败；慢 ticket 和慢 hook 两个子用例通过。panic 是合法 handler 注册的故障注入回调，worker 自身记录日志并恢复；未在测试中自行恢复该 panic。释放通知计数由 committer 替身记录，不能代表真实 WAL 投影已执行。

```go
package nest

import (
	"context"
	"errors"
	"github.com/tjbdwanghaibo/roost-core/entity"
	"sync/atomic"
	"testing"
	"time"
)

type reviewNotifier struct {
	*pipelinedTestCommitter
	releasedCount atomic.Int32
}

func (c *reviewNotifier) TransactionReleased(TransactionID) { c.releasedCount.Add(1) }
func reviewCompletionEngine(t *testing.T, name string, c *reviewNotifier, hook func()) (*NestMgr, int64, *rollbackTestEntity) {
	t.Helper()
	getter := newMockGetter()
	id, ent := newAsyncPilotEntity(t, 9901, 0)
	getter.Add(ent)
	MustRegisterHandlerWithMeta(NewHandlerName(name), func(es []entity.IThreadSafeEntity, _ []any, _ ...HandlerOption) (any, error) {
		e := es[0].(*rollbackTestEntity)
		old := e.dao.Value
		if !RecordUndo(e.dao, 1, func() error { e.dao.Value = old; return nil }) {
			return nil, errors.New("missing tx")
		}
		e.dao.Value++
		if err := MarkPersist(e.dao, 1); err != nil {
			return nil, err
		}
		if hook != nil {
			AfterCommit(hook)
		}
		return e.dao.Value, nil
	}, HandlerMeta{Rollback: RollbackUndo, Durability: DurabilityPipelined})
	mgr := NewEngine(NestOptionWithGetter(getter), NestOptionWithTransactionCommitter(c), NestOptionWithPipelinedAsyncCompletion(1, 8), NestOptionWithWorkerNumAndMsgCap(1, 0, 8), NestOptionWithTickDuration(time.Hour))
	if err := mgr.Start(); err != nil {
		t.Fatal(err)
	}
	return mgr, id, ent
}
func TestReviewAfterCommitPanicPreservesReplyAndRelease(t *testing.T) {
	c := &reviewNotifier{pipelinedTestCommitter: newPipelinedTestCommitter(true)}
	entered := make(chan struct{})
	mgr, id, ent := reviewCompletionEngine(t, "review-completion-panic", c, func() { close(entered); panic("review hook panic") })
	ctx, cancel := context.WithCancel(context.Background())
	defer cancel()
	reply := make(chan error, 1)
	go func() { _, err := mgr.Request(ctx, NewHandlerName("review-completion-panic"), id, nil); reply <- err }()
	select {
	case <-entered:
	case <-time.After(time.Second):
		t.Fatal("hook never ran")
	}
	stopCtx, stopCancel := context.WithTimeout(context.Background(), time.Second)
	defer stopCancel()
	if err := mgr.Shutdown(stopCtx); err != nil {
		t.Fatal(err)
	}
	var missing bool
	select {
	case <-reply:
	case <-time.After(30 * time.Millisecond):
		missing = true
		cancel()
		<-reply
	}
	if missing || c.releasedCount.Load() != 1 {
		t.Fatalf("durable completion lost obligations: missing_reply=%v release_count=%d durable_lsn=%d value=%d", missing, c.releasedCount.Load(), c.DurableLSN(), ent.dao.Value)
	}
}
func TestReviewShutdownRetryWaitsForSlowCompletion(t *testing.T) {
	for _, phase := range []string{"ticket", "hook"} {
		t.Run(phase, func(t *testing.T) {
			c := &reviewNotifier{pipelinedTestCommitter: newPipelinedTestCommitter(false)}
			entered := make(chan struct{})
			release := make(chan struct{})
			var hook func()
			if phase == "hook" {
				hook = func() { close(entered); <-release }
			}
			name := "review-slow-" + phase
			mgr, id, ent := reviewCompletionEngine(t, name, c, hook)
			defer func() {
				c.resolveAll(nil)
				select {
				case <-release:
				default:
					close(release)
				}
				mgr.Shutdown(context.Background())
			}()
			reply := make(chan error, 1)
			go func() {
				ret, err := mgr.Request(context.Background(), NewHandlerName(name), id, nil)
				if err == nil && ret != 1 {
					err = errors.New("wrong result")
				}
				reply <- err
			}()
			select {
			case <-c.enqueued:
			case <-time.After(time.Second):
				t.Fatal("not enqueued")
			}
			if phase == "hook" {
				c.resolveAll(nil)
				select {
				case <-entered:
				case <-time.After(time.Second):
					t.Fatal("hook not entered")
				}
			}
			for i := 0; i < 2; i++ {
				ctx, cancel := context.WithTimeout(context.Background(), 20*time.Millisecond)
				err := mgr.Shutdown(ctx)
				cancel()
				if !errors.Is(err, context.DeadlineExceeded) {
					t.Fatalf("shutdown %d=%v before completion", i, err)
				}
			}
			c.resolveAll(nil)
			close(release)
			ctx, cancel := context.WithTimeout(context.Background(), time.Second)
			defer cancel()
			if err := mgr.Shutdown(ctx); err != nil {
				t.Fatal(err)
			}
			select {
			case err := <-reply:
				if err != nil {
					t.Fatal(err)
				}
			case <-ctx.Done():
				t.Fatal("reply lost")
			}
			if c.releasedCount.Load() != 1 || ent.dao.Value != 1 {
				t.Fatal("release or value changed")
			}
		})
	}
}
```
