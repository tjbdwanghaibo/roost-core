## 1. RR-20260926-10：卸载后投影前重载（真实隔离 Mongo）

运行方式：把 `codegen/internal/entity/testdata/dataengine/*.go` 与本探针复制到临时目录，定义里的库名替换为 `roost_generated_it_probe_<pid>_<ts>`，
用 `go run ./codegen/cmd/dao` / `./codegen/cmd/entity` 生成后，在本地 `roost-it` 环境下执行
`ROOST_GENERATED_PHASE=probe go test -mod=mod -count=1 -run '^TestZZProbeUnloadReloadBeforeProjection$' -v .`（需先 source 本机集成环境变量文件，不入库）。
2026-09-26 复审方二次实跑三种策略均复现，输出见 RR-10；跑后探针库已不存在。

```go
package persistflow

import (
	"context"
	"errors"
	"os"
	"strings"
	"testing"
	"time"

	"github.com/tjbdwanghaibo/roost-core/dataengine/engine"
	"github.com/tjbdwanghaibo/roost-core/entity"
	fmongo "github.com/tjbdwanghaibo/roost-core/mongo"
	"github.com/tjbdwanghaibo/roost-core/mongo/driver"
	"github.com/tjbdwanghaibo/roost-core/nest"
)

func TestZZProbeUnloadReloadBeforeProjection(t *testing.T) {
	uri := os.Getenv("ROOST_DATAENGINE_IT_MONGO_URI")
	if uri == "" || os.Getenv("ROOST_GENERATED_PHASE") != "probe" {
		t.Skip("probe only")
	}
	if db := NewWalletDao().DbName(); !strings.HasPrefix(db, "roost_generated_it_") || db == "roost_generated_it_placeholder" {
		t.Fatal("refuse non-isolated database")
	}
	ctx, cancel := context.WithTimeout(context.Background(), 60*time.Second)
	defer cancel()
	client, err := driver.NewClient(fmongo.DefaultConfig(uri), driver.IndexMigrationPolicy{})
	if err != nil {
		t.Fatal(err)
	}
	defer client.Close(context.Background())
	defer func() { _ = client.Database(NewWalletDao().DbName()).Drop(context.Background()) }()
	RegisterEntity()
	for _, policy := range []nest.DurabilityPolicy{nest.DurabilityAsync, nest.DurabilityStrict, nest.DurabilityPipelined} {
		t.Run(policy.String(), func(t *testing.T) {
			h := newTradeFixture(t, ctx, client, t.TempDir(), "unload", policy, false)
			ids := h.seed(t, 31000+int(policy)*10, 2)
			pause := h.gate.pause()
			defer pause.resume()
			// Transaction A: committed to WAL (and completed to the caller); its projection is held at the gate.
			if err := h.request("trade_transfer", ids...); err != nil {
				t.Fatal(err)
			}
			waitTradeRecord(t, ctx, pause, ids)
			before := h.access.Manager().Get(ids[0]).(*Trader)
			beforeCoins, beforeVersion := before.wallet.GetCoins(), before.wallet.DirtyTracker().Version()
			t.Logf("memory after A: coins=%d version=%d", beforeCoins, beforeVersion)
			// Ordinary unload (logout / eviction): deleteFromDB=false.
			for _, id := range ids {
				if err := h.access.Destroy(ctx, h.access.Manager().Get(id), entity.DestroyReasonCommon, false); err != nil {
					t.Fatalf("unload %d: %v", id, err)
				}
			}
			// Reload through the DataEngine repository while A is still unprojected.
			for _, id := range ids {
				if _, err := h.runtime.Repository.LoadEntity(ctx, id, EntityKindTrader); err != nil {
					t.Fatalf("reload %d: %v", id, err)
				}
			}
			after := h.access.Manager().Get(ids[0]).(*Trader)
			t.Logf("memory after unload+reload: coins=%d version=%d (A's effect visible? %v)", after.wallet.GetCoins(), after.wallet.DirtyTracker().Version(), after.wallet.GetCoins() == beforeCoins)
			// Transaction B built on the reloaded (stale) state.
			errB := h.request("trade_transfer", ids...)
			t.Logf("transaction B on reloaded entity: err=%v", errB)
			pause.resume()
			select {
			case fatal := <-h.fatal:
				t.Logf("projector fatal after resume: conflict=%v err=%v", errors.Is(fatal, engine.ErrProjectionConflict), fatal)
			case <-time.After(5 * time.Second):
				t.Logf("no fatal within 5s")
			}
			h.close(t, nil)
		})
	}
}
```

## 2. RR-20260926-13：Remote 提交成功、释放失败后 Sync 冻结（nest 包）

`GOWORK=off go test -race ./nest -run 'TestProbeRemoteCloseFailureFreezesEntitySync' -count=1 -v`（periodic 变体与 on_change 版仅 Mode 不同，此处给 on_change 版）。

```go
package nest

import (
	"context"
	"errors"
	"testing"
	"time"

	"github.com/tjbdwanghaibo/roost-core/entity"
	"github.com/tjbdwanghaibo/roost-core/sync/entitysync"
)

type closeFailBatch struct{ stagedRemoteBatch }

func (b *closeFailBatch) Close(context.Context) error {
	b.closed.Add(1)
	return entity.ErrRemoteReleaseIncomplete
}

func TestProbeRemoteCloseFailureFreezesEntitySync(t *testing.T) {
	localID, e := newAsyncPilotEntity(t, 9950, 10)
	m, frames, _ := syncTestManager(t, e, entitysync.ModeOnChange)
	getter := newMockGetter()
	remoteID := mustBuildCastID(t, 9951, entity.EntityCategoryRemote, nestRemoteManagedKind)
	getter.Add(newMockEntity(remoteID, entity.EntityCategoryRemote))
	getter.Add(e)
	fail := true
	manager := stagedRemoteManager{prepare: func(context.Context) (entity.RemoteWriteBatch, error) {
		if fail {
			return &closeFailBatch{}, nil
		}
		return &stagedRemoteBatch{}, nil
	}}
	mgr := NewEngine(NestOptionWithGetter(getter), NestOptionWithRemoteEntityManager(manager), NestOptionWithEntitySync(m), NestOptionWithWorkerNumAndMsgCap(1, 1, 16))
	name := NewHandlerName("probe_remote_close")
	mgr.MustRegisterHandlerWithMeta(name, func(es []entity.IThreadSafeEntity, _ []any, _ ...HandlerOption) (any, error) {
		e.dao.Value++
		e.MarkSyncDirty(1)
		return "ok", nil
	}, HandlerMeta{Rollback: RollbackNone, Durability: DurabilityMemory})
	if err := m.Start(context.Background()); err != nil {
		t.Fatal(err)
	}
	if err := mgr.Start(); err != nil {
		t.Fatal(err)
	}
	defer mgr.Shutdown(context.Background())
	send := func() any {
		msg, ch := GenSyncMsg(MsgTypeMulti)
		msg.Name = name.String()
		msg.Tids = []int64{remoteID, localID}
		msg.Cost = true
		msg.HasRemote = true
		if err := mgr.dispatcher.TrySendMsg(msg); err != nil {
			t.Fatal(err)
		}
		return stagedWait(t, ch)
	}
	got := send()
	t.Logf("first reply (commit ok, close failed): %v", got)
	t.Logf("after close failure: SyncCommitReady=%v value=%d", e.Sync().SyncCommitReady(), e.dao.Value)
	fail = false
	got = send()
	t.Logf("second reply (all ok): %v", got)
	_ = m.Flush(context.Background())
	t.Logf("after successful second tx: SyncCommitReady=%v value=%d", e.Sync().SyncCommitReady(), e.dao.Value)
	select {
	case raw := <-frames:
		t.Logf("frame delivered value=%d", syncValue(t, raw))
	case <-time.After(300 * time.Millisecond):
		t.Errorf("no frame delivered after a later successful transaction: entity sync frozen")
	}
	_ = errors.New
}
```

## 3. RR-20260926-14：持久提交后 AfterCommit panic，Remote 批次仍 Abort（nest 包）

`GOWORK=off go test -race ./nest -run TestProbeRemoteAbortAfterDurableCommit -count=1 -v`

```go
package nest

import (
	"context"
	"testing"

	"github.com/tjbdwanghaibo/roost-core/entity"
)

func TestProbeRemoteAbortAfterDurableCommit(t *testing.T) {
	localID, e := newAsyncPilotEntity(t, 9960, 10)
	getter := newMockGetter()
	remoteID := mustBuildCastID(t, 9961, entity.EntityCategoryRemote, nestRemoteManagedKind)
	getter.Add(newMockEntity(remoteID, entity.EntityCategoryRemote))
	getter.Add(e)
	batch := &stagedRemoteBatch{}
	var committed bool
	batch.commit = func() { committed = true }
	manager := stagedRemoteManager{prepare: func(context.Context) (entity.RemoteWriteBatch, error) { return batch, nil }}
	committer := &recordingCommitter{}
	mgr := NewEngine(NestOptionWithGetter(getter), NestOptionWithRemoteEntityManager(manager), NestOptionWithTransactionCommitter(committer), NestOptionWithWorkerNumAndMsgCap(1, 1, 16))
	name := NewHandlerName("probe_remote_abort")
	mgr.MustRegisterHandlerWithMeta(name, func(es []entity.IThreadSafeEntity, _ []any, _ ...HandlerOption) (any, error) {
		old := e.dao.Value
		RecordUndo(e.dao, 1, func() error { e.dao.Value = old; return nil })
		e.dao.Value++
		if err := MarkPersist(e.dao, 1); err != nil {
			return nil, err
		}
		CurrentRollbackTx().AfterAdmission(func() { panic("lifecycle finalizer panic") })
		return "ok", nil
	}, HandlerMeta{Rollback: RollbackUndo, Durability: DurabilityStrict})
	if err := mgr.Start(); err != nil {
		t.Fatal(err)
	}
	defer mgr.Shutdown(context.Background())
	msg, ch := GenSyncMsg(MsgTypeMulti)
	msg.Name = name.String()
	msg.Tids = []int64{remoteID, localID}
	msg.Cost = true
	msg.HasRemote = true
	if err := mgr.dispatcher.TrySendMsg(msg); err != nil {
		t.Fatal(err)
	}
	got := stagedWait(t, ch)
	t.Logf("reply=%v", got)
	t.Logf("WAL record committed: mutations=%d; local value=%d; remote batch Commit called=%v Abort called=%v", len(committer.record.Mutations), e.dao.Value, committed, batch.aborted.Load())
	if len(committer.record.Mutations) > 0 && batch.aborted.Load() {
		t.Errorf("remote batch aborted after the transaction was durably committed")
	}
}
```

```go
package entitysync

import (
	"context"
	"sync/atomic"
	"testing"
	"time"

	"github.com/tjbdwanghaibo/roost-core/entity"
	"github.com/tjbdwanghaibo/roost-core/sync/nettransport"
)

type probeReliable struct{ sent atomic.Int32 }

func (p *probeReliable) SendReliable(ctx context.Context, _ nettransport.SessionID, _ []byte) error {
	p.sent.Add(1)
	return nil
}

func TestProbeAsyncTransportSameIDReopen(t *testing.T) {
	down := &probeReliable{}
	async, err := nettransport.NewAsyncTransport(down, nettransport.DefaultAsyncTransportConfig())
	if err != nil {
		t.Fatal(err)
	}
	defer async.Close(context.Background())
	tr, _ := NewAsyncTransport(async)
	var lost atomic.Int32
	m, err := NewManager(ManagerConfig{Transport: tr, Interval: time.Hour, SessionLost: func(SessionID, error) { lost.Add(1) }})
	if err != nil {
		t.Fatal(err)
	}
	packs := 0
	s := testSubject(t, 7, &packs)
	if err := m.Register(s); err != nil {
		t.Fatal(err)
	}
	if err := m.OpenSession(1); err != nil {
		t.Fatal(err)
	}
	if err := m.Subscribe(1, 7, entity.SyncProfile{}); err != nil {
		t.Fatal(err)
	}
	if err := m.Flush(context.Background()); err != nil {
		t.Fatal(err)
	}
	// reconnect: close then immediately reopen with the same session id
	m.CloseSession(1)
	openErr := m.OpenSession(1)
	subErr := m.Subscribe(1, 7, entity.SyncProfile{})
	flushErr := m.Flush(context.Background())
	time.Sleep(50 * time.Millisecond)
	st := m.Stats()
	t.Logf("reopen err=%v subscribe err=%v flush err=%v; SessionLost calls=%d; downstream sends=%d; SessionsLost=%d", openErr, subErr, flushErr, lost.Load(), down.sent.Load(), st.SessionsLost)
	if lost.Load() > 0 {
		t.Errorf("reopened session with the same id was lost by the transport")
	}
}
```
