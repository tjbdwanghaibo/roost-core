```go
package entity

import (
	"context"
	"errors"
	"testing"
	"time"
)

func reviewSnapshot(t *testing.T) RemoteSnapshotEnvelope {
	t.Helper()
	const kind EntityKind = 225
	MustRegisterEntityKindDefs(EntityKindDef{Kind: kind, Category: 1, RemotePolicy: RemotePolicyManaged})
	id, e := BuildEntityID(913004, kind)
	if e != nil {
		t.Fatal(e)
	}
	a := RemoteSnapshotEnvelope{Key: RemoteSnapshotKey{EntityID: id, Kind: kind}, StateVersion: 1, MarkerEpoch: 1, RouteEpoch: 1, Schema: 1, Codec: 1, Full: true, Payload: CopyFrozenRemoteSnapshotPayload([]byte("A"))}
	a.Checksum = RemoteSnapshotChecksum(a.Payload.BytesCopy())
	return a
}

type reviewLayer struct {
	value            RemoteSnapshotEnvelope
	started, release chan struct{}
	err              error
}

func (s *reviewLayer) Get(ctx context.Context, k RemoteSnapshotKey) (RemoteSnapshotEnvelope, bool, error) {
	if s.started != nil {
		close(s.started)
		select {
		case <-s.release:
		case <-ctx.Done():
			return RemoteSnapshotEnvelope{}, false, ctx.Err()
		}
	}
	return s.value, s.err == nil, s.err
}
func (s *reviewLayer) Set(context.Context, RemoteSnapshotEnvelope) error { return s.err }
func (s *reviewLayer) Delete(context.Context, RemoteSnapshotKey) error   { return s.err }
func TestReviewL2RefillMustNotOverwriteSameVersion(t *testing.T) {
	ctx := context.Background()
	a := reviewSnapshot(t)
	l2 := &reviewLayer{value: a, started: make(chan struct{}), release: make(chan struct{})}
	c := NewRemoteSnapshotCache(RemoteSnapshotCacheConfig{LoadTimeout: time.Minute}, l2, nil)
	done := make(chan error, 1)
	go func() { _, _, e := c.Get(ctx, a.Key, RemoteReadCached, 0); done <- e }()
	<-l2.started
	b := a
	b.Payload = CopyFrozenRemoteSnapshotPayload([]byte("B"))
	if e := c.Publish(ctx, b); e != nil {
		close(l2.release)
		<-done
		t.Fatal(e)
	}
	close(l2.release)
	loadErr := <-done
	got, ok, e := c.Get(ctx, a.Key, RemoteReadCached, 0)
	if e != nil || !ok || string(got.Payload.BytesCopy()) != "B" {
		t.Fatalf("inflight L2 A replaced published B at same version: payload=%s loadErr=%v err=%v", got.Payload.BytesCopy(), loadErr, e)
	}
}
func TestReviewSnapshotExpiresAtHonored(t *testing.T) {
	ctx := context.Background()
	a := reviewSnapshot(t)
	a.ExpiresAt = time.Now().Add(-time.Second).UnixNano()
	c := NewRemoteSnapshotCache(RemoteSnapshotCacheConfig{TTL: time.Minute}, nil, nil)
	if e := c.Publish(ctx, a); e != nil {
		return
	}
	got, ok, e := c.Get(ctx, a.Key, RemoteReadCached, 0)
	if e == nil && ok && got.Expired(time.Now()) {
		t.Fatal("expired envelope returned as cache hit despite ExpiresAt")
	}
}
func TestReviewNetworkErrorDegradesControl(t *testing.T) {
	ctx := context.Background()
	a := reviewSnapshot(t)
	c := NewRemoteSnapshotCache(RemoteSnapshotCacheConfig{}, &reviewLayer{err: errors.New("network unavailable")}, nil)
	if e := c.Publish(ctx, a); e != nil {
		t.Fatal(e)
	}
	got, ok, e := c.Get(ctx, a.Key, RemoteReadCached, 0)
	if e != nil || !ok || string(got.Payload.BytesCopy()) != "A" {
		t.Fatal("network fallback failed")
	}
}
func TestReviewLocalSchemaConflictControl(t *testing.T) {
	ctx := context.Background()
	a := reviewSnapshot(t)
	c := NewRemoteSnapshotCache(RemoteSnapshotCacheConfig{}, nil, nil)
	if e := c.Publish(ctx, a); e != nil {
		t.Fatal(e)
	}
	a.Schema = 2
	if e := c.Publish(ctx, a); !errors.Is(e, ErrRemoteVersionConflict) {
		t.Fatalf("local conflict=%v", e)
	}
}
func TestReviewAuthoritativeFailureThenRetry(t *testing.T) {
	ctx := context.Background()
	a := reviewSnapshot(t)
	attempt := 0
	failure := errors.New("temporary loader failure")
	c := NewRemoteSnapshotCache(RemoteSnapshotCacheConfig{}, nil, func(context.Context, RemoteSnapshotKey, RemoteReadConsistency, uint64) (RemoteSnapshotEnvelope, bool, error) {
		attempt++
		if attempt == 1 {
			return RemoteSnapshotEnvelope{}, false, failure
		}
		return a, true, nil
	})
	if _, _, e := c.Get(ctx, a.Key, RemoteReadMonotonic, 1); !errors.Is(e, failure) {
		t.Fatal(e)
	}
	got, ok, e := c.Get(ctx, a.Key, RemoteReadMonotonic, 1)
	if e != nil || !ok || got.StateVersion != 1 || attempt != 2 {
		t.Fatalf("retry: %v %v attempts=%d", ok, e, attempt)
	}
}
```
