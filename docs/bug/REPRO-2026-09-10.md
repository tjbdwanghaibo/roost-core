## 过期旧请求重放与新请求入队（通过）

```go
package match
import("context";"testing";"time")
func TestReviewSweepReleasesSubjectAndKeepsReplay(t *testing.T){
 s,c:=newStore(t,func(cfg *Config){cfg.TicketTTL=time.Second});ctx:=context.Background()
 first:=enqueue(t,s,ranked(),player(1,100),"old")
 c.advance(2*time.Second)
 n,err:=s.Sweep(ctx,ranked(),1);if err!=nil||n!=1{t.Fatalf("sweep=%d %v",n,err)}
 replay,err:=s.Enqueue(ctx,ranked(),player(1,100),"old");if err!=nil||replay.ID!=first.ID||replay.State!=TicketExpired{t.Fatalf("replay=%+v %v",replay,err)}
 fresh:=enqueue(t,s,ranked(),player(1,100),"new")
 if fresh.ID==first.ID{t.Fatal("new intent reused expired ticket")}
 candidates,err:=s.Candidates(ctx,ranked(),10);if err!=nil||len(candidates)!=1||candidates[0].ID!=fresh.ID{t.Fatalf("candidates=%+v %v",candidates,err)}
}
```
