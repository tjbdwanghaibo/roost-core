// 临时探针（RR 复现附录用，跑完删除）：Slow 消息的快阶段继承了慢阶段注入的
// local executor，entity.RunLocal 在快 worker 上再次投递快续行并同步等待。

import (

// zzRunLocalStacks 只打印与本问题相关的 goroutine 堆栈片段。
func zzRunLocalStacks(t *testing.T) {
	t.Helper()
	buf := make([]byte, 1<<20)
	n := runtime.Stack(buf, true)
	for _, g := range strings.Split(string(buf[:n]), "\n\n") {
		if !strings.Contains(g, "dispatchFastContinuation") && !strings.Contains(g, "dispatchQueue).work") {
			continue
		}
		if strings.Contains(g, "sync.Cond.Wait") && !strings.Contains(g, "dispatchFastContinuation") {
			t.Logf("---- idle worker: %s", strings.SplitN(g, "\n", 2)[0])
			continue
		}
		var keep []string
		for _, line := range strings.Split(g, "\n") {
			if strings.HasPrefix(line, "goroutine ") || strings.Contains(line, "roost-core/") {
				keep = append(keep, line)
			}
		}
		t.Logf("---- stack\n%s", strings.Join(keep, "\n"))
	}
}

func zzShutdown(t *testing.T, mgr *NestMgr) {
	t.Helper()
	ctx, cancel := context.WithTimeout(context.Background(), 500*time.Millisecond)
	defer cancel()
	t.Logf("Shutdown(500ms) = %v", mgr.Shutdown(ctx))
	t.Logf("after shutdown Stats.Queue = %+v", mgr.Stats().Queue)
}

func zzRegisterRunLocal(mgr *NestMgr, name string, ran *atomic.Int32) {
	mgr.MustRegisterHandlerWithMeta(NewHandlerName(name), func(_ []entity.IThreadSafeEntity, _ []any, _ ...HandlerOption) (any, error) {
		err := entity.RunLocal(nestBaseContext(), func() { ran.Add(1) })
		return "ok", err
	}, HandlerMeta{Rollback: RollbackNone, Durability: DurabilityMemory})
}

// 用例 3：真实 ManagerAccess + loader。loader 按 dataengine/engine/entity_repository.go:189
// 的做法把 Create/发布交给 entity.RunLocal(ctx, ...)。Slow handler 用 fctx Base
// （与生成 lifecycle.Get/GetOrCreate 的 ctx 来源相同）直接调 ManagerAccess.Get 访问冷实体。
type zzRunLocalLoader struct {
	manager *entity.EntityManager
	loads   atomic.Int32
}

func (l *zzRunLocalLoader) LoadEntity(ctx context.Context, id int64, kind entity.EntityKind) (entity.IThreadSafeEntity, error) {
	l.loads.Add(1)
	var created entity.IThreadSafeEntity
	var createErr error
	dispatchErr := entity.RunLocal(ctx, func() {
		value := newMockEntityWithKind(id, entity.ResolveEntityID(id).Category, kind)
		createErr = l.manager.TryAdd(value)
		created = value
	})
	if dispatchErr != nil || createErr != nil {
		return nil, fmt.Errorf("create: %v / %v", dispatchErr, createErr)
	}
	return created, nil
}

func TestZZRunLocalManagerAccessColdGetInSlowHandler(t *testing.T) {
	manager := entity.NewEntityManager()
	access := entity.NewManagerAccess(manager)
	loader := &zzRunLocalLoader{manager: manager}
	if _, err := access.ConfigureLoader(loader); err != nil {
		t.Fatal(err)
	}
	hot := mustBuildCastID(t, 7821, entity.EntityCategory(1), nestLocalKind)
	var cold atomic.Int64
	cold.Store(mustBuildCastID(t, 7822, entity.EntityCategory(1), nestLocalKind))
	if err := manager.TryAdd(newMockEntityWithKind(hot, entity.EntityCategory(1), nestLocalKind)); err != nil {
		t.Fatal(err)
	}

		value, err := access.Get(ctx, cold.Load(), entity.EntityCategory(1))
		getErr.Store(fmt.Sprintf("value=%v err=%v", value != nil, err))

		t.Fatal(err)
	}
	// 对照：普通快消息里同样调用，local executor 未注入，RunLocal 就地执行、可完成（但冷加载 I/O 在快 worker 上）。

import (

	"github.com/tjbdwanghaibo/roost-core/entity"

// RR 复现：scheduleSnapshots 在任何 Push 前对全部会话预扣冷快照额度；
// 第一个会话 Push 即 RetryLater 后，同一 on_change 窗口的第二次 Flush 无额度可用。
func TestProbeRRRetryLaterPrechargesWindowBudget(t *testing.T) {
	r := newRecordingTransport()
	m := newTestManager(t, r, ManagerConfig{Mode: ModeOnChange, Interval: time.Hour, SnapshotBudget: SnapshotBudget{MaxObjects: 3}})
	open(t, m, 1, 2, 3)
	packs := 0
	for sid := SessionID(1); sid <= 3; sid++ {
		id := int64(sid) * 10
		if err := m.Register(testSubject(t, id, &packs)); err != nil {
			t.Fatal(err)
		}
		mustSubscribe(t, m, sid, id, entity.SyncProfile{})
	}
	r.setRetry(ErrRetryLater)
	err := m.Flush(t.Context())
	r.setRetry(nil)
	if !errors.Is(err, ErrRetryLater) {
		t.Fatalf("err=%v", err)
	}
	delivered := len(r.take(1)) + len(r.take(2)) + len(r.take(3))
	t.Logf("after RetryLater: windowAllowance=%+v windowSessions=%v snapshotCursor=%v delivered=%d", m.windowAllowance, m.windowSessions, m.snapshotCursor, delivered)
	mustFlush(t, m) // 同一窗口（Interval=1h，未轮转）
	got := len(r.take(1)) + len(r.take(2)) + len(r.take(3))
	s := m.Stats()
	t.Logf("same-window second Flush: frames=%d CreatesAdmitted=%d SnapshotsDeferred=%d PendingSnapshots=%d WaitingSnapshotSubjects=%d", got, s.CreatesAdmitted, s.SnapshotsDeferred, s.PendingSnapshots, s.WaitingSnapshotSubjects)
	if got == 0 {
		t.Fatalf("RetryLater on the first Push consumed the whole window budget for never-attempted sessions")
	}
}
