## 可靠追帧

TCP 使用本机真实 socket，但写后错误是适配器注入。长度前缀是测试封装，不是框架新增协议。

```go

import("context";"encoding/binary";"errors";"io";"net";"testing";"time";corestate "github.com/tjbdwanghaibo/roost-core/statesync")
type reviewTCPSender struct{conn net.Conn;calls int}
func(s *reviewTCPSender)SendReliable(_ context.Context,_ corestate.SessionID,p []byte)error{
 var header [4]byte;binary.BigEndian.PutUint32(header[:],uint32(len(p)))
 if _,err:=s.conn.Write(header[:]);err!=nil{return err};if _,err:=s.conn.Write(p);err!=nil{return err}
 s.calls++;if s.calls==1{return errors.New("adapter reply lost after real TCP write")};return nil
}
func TestReviewReliableTCPRetry(t *testing.T){
 listener,err:=net.Listen("tcp4","127.0.0.1:0");if err!=nil{t.Fatal(err)};defer listener.Close()
 send,err:=net.Dial("tcp4",listener.Addr().String());if err!=nil{t.Fatal(err)};defer send.Close()
```
