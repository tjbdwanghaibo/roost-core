package room

import (
	"context"
	"encoding/json"
	"errors"
	fn "github.com/tjbdwanghaibo/roost-core/nats"
	sb "github.com/tjbdwanghaibo/roost-core/syncbus"
	"reflect"
	"sync/atomic"
	"testing"
	"time"
)

type reviewSub struct{ stops atomic.Int32 }

func (s *reviewSub) Stop()                   { s.stops.Add(1) }
func (s *reviewSub) Drain()                  { s.Stop() }
func (s *reviewSub) Closed() <-chan struct{} { c := make(chan struct{}); close(c); return c }

type reviewJS struct {
	*fakeJetStream
	sub              *reviewSub
	entered, release chan struct{}
	pubErr           error
	deadline         bool
	value            any
	subErr           error
}

func (j *reviewJS) Subscribe(ctx context.Context, c fn.JetStreamConsumerConfig, h fn.JetStreamHandler) (fn.IJetStreamSubscription, error) {
	if j.entered != nil {
		close(j.entered)
		<-j.release
	}
	if j.subErr != nil {
		return nil, j.subErr
	}
	j.fakeJetStream.Subscribe(ctx, c, h)
	if j.sub == nil {
		j.sub = &reviewSub{}
	}
	return j.sub, nil
}
func (j *reviewJS) Publish(ctx context.Context, s string, b []byte, o fn.JetStreamPublishOptions) (fn.JetStreamPublishAck, error) {
	_, j.deadline = ctx.Deadline()
	j.value = ctx.Value("review")
	if j.pubErr != nil {
		return fn.JetStreamPublishAck{}, j.pubErr
	}
	return j.fakeJetStream.Publish(ctx, s, b, o)
}
func TestReviewDurableNamespace(t *testing.T) {
	for _, mode := range []string{"different_prefix", "different_sid", "different_topic", "different_stream"} {
		t.Run(mode, func(t *testing.T) {
			j := newFakeJetStream()
			a := JetStreamSyncConfig{LocalSid: 7, Prefix: "one", Stream: "SHARED"}
			b := a
			topic := "state"
			switch mode {
			case "different_prefix":
				b.Prefix = "two"
			case "different_sid":
				b.LocalSid = 8
			case "different_topic":
				topic = "other"
			case "different_stream":
				b.Stream = "SECOND"
			}
			left, e := NewJetStreamSyncBus(context.Background(), j, a)
			if e != nil {
				t.Fatal(e)
			}
			right, e := NewJetStreamSyncBus(context.Background(), j, b)
			if e != nil {
				t.Fatal(e)
			}
			if _, e = left.Subscribe("state", func(*sb.SyncMsg) error { return nil }); e != nil {
				t.Fatal(e)
			}
			if _, e = right.Subscribe(topic, func(*sb.SyncMsg) error { return nil }); e != nil {
				t.Fatal(e)
			}
			x, y := j.consumers[0], j.consumers[1]
			if x.Stream == y.Stream && x.Durable == y.Durable && x.FilterSubject != y.FilterSubject {
				t.Fatalf("distinct routes share consumer identity: (%s,%s): %s versus %s", x.Stream, x.Durable, x.FilterSubject, y.FilterSubject)
			}
		})
	}
}
func TestReviewSyncBusLifecycle(t *testing.T) {
	for _, mode := range []string{"unsubscribe_idempotent", "stop_registered", "subscribe_failure_retry", "inflight_stop_observation"} {
		t.Run(mode, func(t *testing.T) {
			j := &reviewJS{fakeJetStream: newFakeJetStream()}
			b, e := NewJetStreamSyncBus(context.Background(), j, JetStreamSyncConfig{LocalSid: 1})
			if e != nil {
				t.Fatal(e)
			}
			if mode == "inflight_stop_observation" {
				j.entered = make(chan struct{})
				j.release = make(chan struct{})
				done := make(chan func(), 1)
				errs := make(chan error, 1)
				go func() { u, e := b.Subscribe("x", func(*sb.SyncMsg) error { return nil }); done <- u; errs <- e }()
				<-j.entered
				b.Stop()
				close(j.release)
				u := <-done
				if e = <-errs; e != nil {
					t.Fatal(e)
				}
				defer u()
				b.mu.Lock()
				n := len(b.subs)
				b.mu.Unlock()
				t.Logf("Stop returned before pending Subscribe registered: tracked=%d stops=%d", n, j.sub.stops.Load())
				if n != 1 || j.sub.stops.Load() != 0 {
					t.Fatal("observed lifecycle changed")
				}
				return
			}
			if mode == "subscribe_failure_retry" {
				j.subErr = errors.New("setup")
				if _, e = b.Subscribe("x", func(*sb.SyncMsg) error { return nil }); e == nil {
					t.Fatal("expected error")
				}
				j.subErr = nil
			}
			u, e := b.Subscribe("x", func(*sb.SyncMsg) error { return nil })
			if e != nil {
				t.Fatal(e)
			}
			if mode == "stop_registered" {
				b.Stop()
				if len(b.subs) != 0 || j.sub.stops.Load() != 1 {
					t.Fatal("stop did not clear")
				}
			} else {
				u()
				u()
				if len(b.subs) != 0 || j.sub.stops.Load() != 1 {
					t.Fatal("unsubscribe not idempotent")
				}
			}
		})
	}
}
func TestReviewJetStreamPublishMatrix(t *testing.T) {
	for _, mode := range []string{"confirmed", "cancel", "deadline_context", "explicit_id", "fallback_parts", "source_sid", "source_ownership", "publish_error"} {
		t.Run(mode, func(t *testing.T) {
			j := &reviewJS{fakeJetStream: newFakeJetStream()}
			b, e := NewJetStreamSyncBus(context.Background(), j, JetStreamSyncConfig{LocalSid: 7, PublishTime: time.Second})
			if e != nil {
				t.Fatal(e)
			}
			m := &sb.SyncMsg{Topic: "state", Key: 1, Version: 1, Data: []byte("x")}
			before := *m
			before.Data = append([]byte(nil), m.Data...)
			ctx := context.WithValue(context.Background(), "review", "value")
			switch mode {
			case "cancel":
				x, cancel := context.WithCancel(ctx)
				cancel()
				ctx = x
			case "explicit_id":
				m.MessageID = "delivery"
			case "fallback_parts":
				m.FromSid = 7
				m.Parts = 2
				if syncMsgID(m) == "" {
					t.Fatal("empty fallback")
				}
				first := syncMsgID(m)
				m.Part = 1
				if first == syncMsgID(m) {
					t.Fatal("parts collided")
				}
				return
			case "source_sid":
				m.FromSid = 9
			case "publish_error":
				j.pubErr = errors.New("broker reject")
			}
			if mode == "confirmed" {
				e = b.PublishConfirmed(m)
			} else {
				e = b.PublishContext(ctx, m)
			}
			if mode == "cancel" {
				if !errors.Is(e, context.Canceled) || len(j.publishes) != 0 {
					t.Fatal(e)
				}
				return
			}
			if mode == "publish_error" {
				if !errors.Is(e, j.pubErr) {
					t.Fatal(e)
				}
				return
			}
			if e != nil {
				t.Fatal(e)
			}
			if len(j.publishes) != 1 || !j.deadline {
				t.Fatal("publish confirmation/deadline path")
			}
			var owned sb.SyncMsg
			if e = json.Unmarshal(j.publishes[0].data, &owned); e != nil {
				t.Fatal(e)
			}
			if mode == "explicit_id" && j.publishes[0].opts.MsgID != "delivery" {
				t.Fatal("id changed")
			}
			if mode == "source_sid" && owned.FromSid != 9 {
				t.Fatal("sid overwritten")
			}
			if mode == "deadline_context" && j.value != "value" {
				t.Fatal("context dropped")
			}
			if mode == "source_ownership" && !reflect.DeepEqual(before, *m) {
				t.Fatal("caller mutated")
			}
		})
	}
}
func TestReviewJetStreamReceiveMatrix(t *testing.T) {
	for _, mode := range []string{"remote", "self", "malformed", "nil", "handler_error", "duplicate"} {
		t.Run(mode, func(t *testing.T) {
			j := newFakeJetStream()
			b, e := NewJetStreamSyncBus(context.Background(), j, JetStreamSyncConfig{LocalSid: 7})
			if e != nil {
				t.Fatal(e)
			}
			calls := 0
			u, e := b.Subscribe("state", func(*sb.SyncMsg) error {
				calls++
				if mode == "handler_error" {
					return errors.New("application")
				}
				return nil
			})
			if e != nil {
				t.Fatal(e)
			}
			defer u()
			h := j.handlers["roost.sync.state"]
			msg := &sb.SyncMsg{Topic: "state", FromSid: 8}
			if mode == "self" {
				msg.FromSid = 7
			}
			data, _ := json.Marshal(msg)
			if mode == "malformed" {
				data = []byte("{")
			}
			raw := &fn.JetStreamMsg{Data: data}
			if mode == "nil" {
				raw = nil
			}
			if e = h(context.Background(), raw); e != nil {
				t.Fatal("wrapper contract", e)
			}
			want := 1
			if mode == "self" || mode == "malformed" || mode == "nil" {
				want = 0
			}
			if mode == "duplicate" {
				h(context.Background(), raw)
				want = 2
			}
			if calls != want {
				t.Fatalf("calls=%d want=%d", calls, want)
			}
		})
	}
}

package driver

import (
	"context"
	"errors"
	fn "github.com/tjbdwanghaibo/roost-core/nats"
	"testing"
	"time"
)

func TestReviewJetStreamSettlement(t *testing.T) {
	for _, mode := range []string{"ack", "transient", "permanent", "max_deliver", "delayed", "ack_error", "nak_error", "term_error", "panic", "cancel"} {
		t.Run(mode, func(t *testing.T) {
			m := &settleMsg{}
			wrapped := jetStreamMsg(m)
			wrapped.NumDelivered = 1
			cfg := fn.JetStreamConsumerConfig{MaxDeliver: 3}
			var err error
			want := "ack"
			switch mode {
			case "transient":
				err = errors.New("transient")
				want = "nak"
			case "permanent":
				err = fn.Permanent(errors.New("bad"))
				want = "term"
			case "max_deliver":
				err = errors.New("transient")
				wrapped.NumDelivered = 3
				want = "term"
			case "delayed":
				err = errors.New("transient")
				cfg.NakBackoffMin = time.Millisecond
				cfg.NakBackoffMax = time.Second
				want = "nak_delay"
			case "ack_error":
				m.ackErr = errors.New("wire")
			case "nak_error":
				err = errors.New("app")
				m.nakErr = errors.New("wire")
				want = "nak"
			case "term_error":
				err = fn.Permanent(errors.New("app"))
				m.termErr = errors.New("wire")
				want = "term"
			case "panic":
				err = invokeJetStreamHandler(context.Background(), func(context.Context, *fn.JetStreamMsg) error { panic("app") }, wrapped)
				want = "nak"
			case "cancel":
				err = context.Canceled
				want = "nak"
			}
			settleJetStreamDelivery(m, wrapped, cfg, err)
			if len(m.calls) != 1 || m.calls[0] != want {
				t.Fatalf("calls=%v want=%s", m.calls, want)
			}
		})
	}
}
