package platform

import (
 "context"
 "testing"
 "time"
)

// Reuses only the package's in-memory fixture and signed callback builder.
// The owner run loop and its 30-second ticker are unchanged.
func TestReviewPlatformRecovery(t *testing.T) {
 for _, mode := range []string{"manual_control", "provider_retry_control", "owner_timer"} {
  t.Run(mode, func(t *testing.T) {
   h:=newHarness(t)
   h.deliverer.failFor["review-paid"]=1
   raw,sig:=callback(t,"review-paid",42,100)
   _,err:=h.service.HandleCallback(context.Background(),raw,sig)
   if err==nil {t.Fatal("fixture must fail first delivery")}
   h.clock.advance(time.Minute)
   switch mode {
   case "manual_control":
    receipt,err:=h.service.AttemptDelivery(context.Background(),"review-paid")
    if err!=nil || !receipt.Delivered {t.Fatalf("manual: %+v %v",receipt,err)}
   case "provider_retry_control":
    receipt,err:=h.service.HandleCallback(context.Background(),raw,sig)
    if err!=nil || !receipt.Delivered {t.Fatalf("provider: %+v %v",receipt,err)}
   case "owner_timer":
    s:=&Server{service:h.service}
    ctx,cancel:=context.WithTimeout(context.Background(),RetryInterval+2*time.Second)
    defer cancel()
    if err:=s.run(ctx);err!=nil {t.Fatal(err)}
   }
   if n:=h.deliverer.grantsFor("review-paid"); n!=1 {
    order,_,_:=h.service.Order(context.Background(),"review-paid")
    t.Fatalf("paid order not recovered: grants=%d order=%+v",n,order)
   }
  })
 }
}
