func TestAud5IsolatedInReleaseHookOfRemoteMessage(t *testing.T) {

		name       string
		meta       HandlerMeta
		outerFails bool
	}{

					return nil, errors.New("probe premise: no guard scope")

					hookRan.Store(true)
					_, hookErr = RunIsolatedTransaction(context.Background(), iso, "aud5_iso_in_release_hook", func() (any, error) {

						RecordUndo(e.dao, 1, func() error { e.dao.Value = old; return nil })
						e.dao.Value = 55
						return nil, MarkPersist(e.dao, 1)
					})
				})
				if tc.outerFails {
					return nil, errors.New("outer business failed")
				}
				return "ok", nil
			}, tc.meta)
			if err := mgr.Start(); err != nil {
				t.Fatal(err)
			}
			defer func() { _ = mgr.Shutdown(context.Background()) }()
			msg, ch := GenSyncMsg(MsgTypeMulti)

			if err := mgr.dispatcher.TrySendMsg(msg); err != nil {
				t.Fatal(err)

			reply := stagedWait(t, ch)
			t.Logf("PROBE A %s: hookRan=%v hookErr=%v isoCommits=%d outerCommits=%d FinalizeLocked=%v batch Commit=%d Abort=%d reply=%v afterCommit=%v",

				}
			}
		})
	}
