实际：`successful replacement run-3 lost claim; third Enter=(run-4,<nil>); stored states=(open,open)`。

```go
package session

import (
	"context"
	"errors"
	"github.com/tjbdwanghaibo/roost-core/versionstore"
	"testing"
)

type reviewRunsDeleteHook struct {
	versionstore.Store[string, Run]
	after func(versionstore.Versioned[Run])
}

func (s *reviewRunsDeleteHook) Delete(ctx context.Context, key string, expect versionstore.Versioned[Run]) error {
	if err := s.Store.Delete(ctx, key, expect); err != nil {
		return err
	}
	if s.after != nil {
		s.after(expect)
	}
	return nil
}

func TestReviewCollisionCleanupPreservesReacquiredClaim(t *testing.T) {
	ctx := context.Background()
	ledger := &barrierLedger{Store: versionstore.NewMemoryStore[string, LedgerEntry](), ready: make(chan struct{})}
	runs := &reviewRunsDeleteHook{Store: versionstore.NewMemoryStore[string, Run]()}
	h := newHarness(t, func(cfg *Config) { cfg.Requests = ledger; cfg.Runs = runs })
	var replacement Run
	var replacementErr error
	runs.after = func(old versionstore.Versioned[Run]) {
		// The rejected run is gone, but Enter has not released its claim yet.
		// A new Enter legitimately resolves that orphan and takes a fresh claim.
		replacement, replacementErr = h.service.Enter(ctx, old.Value.OwnerID, enterReq("replacement-request"))
	}
	type result struct {
		run Run
		err error
	}
	done := make(chan result, 2)
	for _, owner := range []int64{1, 2} {
		go func(id int64) { r, e := h.service.Enter(ctx, id, enterReq("collision-request")); done <- result{r, e} }(owner)
	}
	a, b := <-done, <-done
	if (a.err == nil) == (b.err == nil) {
		t.Fatalf("expected exactly one collision winner: %v / %v", a.err, b.err)
	}
	if !errors.Is(errors.Join(a.err, b.err), ErrRequestInvalid) {
		t.Fatalf("missing collision error")
	}
	if replacementErr != nil || replacement.ID == "" {
		t.Fatalf("replacement Enter failed: %v", replacementErr)
	}
	claim, found, err := h.service.cfg.Claims.Get(ctx, replacement.OwnerID)
	if err != nil {
		t.Fatal(err)
	}
	if !found || claim.Value.RunID != replacement.ID {
		// A third call shows why losing this claim breaks the one-live-run contract.
		third, thirdErr := h.service.Enter(ctx, replacement.OwnerID, enterReq("third-request"))
		storedReplacement, _, _ := runs.Get(ctx, replacement.ID)
		storedThird, _, _ := runs.Get(ctx, third.ID)
		t.Fatalf("successful replacement %s lost claim; third Enter=(%s,%v); stored states=(%s,%s)", replacement.ID, third.ID, thirdErr, storedReplacement.Value.State, storedThird.Value.State)
	}
}
```
