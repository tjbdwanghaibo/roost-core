## Remote：Close 与 Stop 并发

```go
package remoteentity

import (
	"context"
	"fmt"
	"github.com/tjbdwanghaibo/roost-core/entity"
	"sync"
	"testing"
	"time"
)

func TestReviewBatchCloseRacesFinalizerStop(t *testing.T) {
	var raw []int64
	for i := 0; i < 64; i++ {
		raw = append(raw, int64(7300+i))
	}
	mgr, ids := batchHarness(t, 239, raw...)
	ctx := context.Background()
	var batches []entity.RemoteWriteBatch
	for i, id := range ids {
		b, err := mgr.PrepareRemoteWriteBatch(ctx, []int64{id})
		if err != nil {
			t.Fatal(err)
		}
		if err = b.FinalizeLocked(entity.NewRemoteTransactionOutcome(remoteTestTxID(byte(i+1)), "review-late-close", "", true, 2)); err != nil {
			t.Fatal(err)
		}
		if err = b.Indeterminate(ctx, fmt.Errorf("uncertain")); err != nil {
			t.Fatal(err)
		}
		batches = append(batches, b)
	}
	stopCtx, cancel := context.WithTimeout(ctx, time.Second)
	defer cancel()
	start := make(chan struct{})
	var wg sync.WaitGroup
	errs := make(chan error, len(batches)+1)
	for _, b := range batches {
		wg.Add(1)
		go func(b entity.RemoteWriteBatch) { defer wg.Done(); <-start; errs <- b.Close(ctx) }(b)
	}
	wg.Add(1)
	go func() { defer wg.Done(); <-start; errs <- mgr.StopFinalizer(stopCtx) }()
	close(start)
	wg.Wait()
	close(errs)
	for err := range errs {
		if err != nil {
			t.Fatal(err)
		}
	}
	queued, slots := len(mgr.remote.finalizeQueue), len(mgr.remote.finalizeSlots)
	// Test cleanup only: finalizer is already gone; reclaim stranded items.
	for len(mgr.remote.finalizeQueue) > 0 {
		item := <-mgr.remote.finalizeQueue
		mgr.releaseRemoteEntriesObserved(ctx, item.entries)
		mgr.releaseRemoteFinalizeSlot()
	}
	if queued != 0 || slots != 0 {
		t.Fatalf("Close returned success after Stop but stranded queue=%d slots=%d with no finalizer", queued, slots)
	}
}
```

## Nest：内部重排与停止回复

```go
package nest

import (
	"context"
	"errors"
	"github.com/tjbdwanghaibo/roost-core/entity"
	"sync/atomic"
	"testing"
	"time"
)

func TestReviewInternalRequeueGetsStoppedReply(t *testing.T) {
	getter := newMockGetter()
	id := mustBuildCastID(t, 7781, entity.EntityCategory(1), nestLocalKind)
	ent := newMockEntity(id, entity.EntityCategory(1))
	if !ent.Base().BeginGroupTransition(entity.EntityGroupTransitionJoin, 9871) {
		t.Fatal("pending transition")
	}
	getter.Add(ent)
	var called atomic.Int32
	name := NewHandlerName("review-internal-requeue-shutdown")
	MustRegisterMemoryHandler(name, func([]entity.IThreadSafeEntity, []any, ...HandlerOption) (any, error) { called.Add(1); return nil, nil })
	mgr := NewEngine(NestOptionWithGetter(getter), NestOptionWithWorkerNumAndMsgCap(1, 0, 8), NestOptionWithTickDuration(time.Hour))
	if err := mgr.Start(); err != nil {
		t.Fatal(err)
	}
	ctx, cancel := context.WithTimeout(context.Background(), time.Second)
	defer cancel()
	defer mgr.Shutdown(context.Background())
	done := make(chan error, 1)
	go func() { _, err := mgr.Request(ctx, name, id, nil); done <- err }()
	deadline := time.Now().Add(time.Second)
	for mgr.dispatcher.delayedCount() == 0 {
		if time.Now().After(deadline) {
			t.Fatal("request never requeued")
		}
		time.Sleep(time.Millisecond)
	}
	if err := mgr.Shutdown(ctx); err != nil {
		t.Fatal(err)
	}
	select {
	case err := <-done:
		if !errors.Is(err, ErrNestStopped) {
			t.Fatalf("reply=%v", err)
		}
	case <-ctx.Done():
		t.Fatal("request not answered")
	}
	if called.Load() != 0 {
		t.Fatal("handler ran while transition pending")
	}
}
func TestReviewRequeueAfterStopRepliesExactlyOnce(t *testing.T) {
	d := NewDispatcher("review-stopped-requeue", 1, 0, 8, nil)
	d.OnInit()
	if err := d.OnDestroyWithContext(context.Background()); err != nil {
		t.Fatal(err)
	}
	mgr := &NestMgr{dispatcher: d}
	msg, ch := GenSyncMsg(MsgTypeSingle)
	defer recycleMsg(msg)
	if !requeueTransientDispatch(mgr, msg, ErrEntityLockGroupChanged) {
		t.Fatal("not requeued")
	}
	if msg.RetChan != nil {
		t.Fatal("original still owns reply")
	}
	select {
	case got := <-ch:
		err, ok := got.(error)
		if !ok || !errors.Is(err, ErrNestStopped) {
			t.Fatalf("reply=%v", got)
		}
	case <-time.After(time.Second):
		t.Fatal("no reply")
	}
	select {
	case got := <-ch:
		t.Fatalf("duplicate reply=%v", got)
	default:
	}
}
```

## Mail：原修复、拒绝原子性与对照

```go
package mail

import (
	"context"
	"errors"
	"fmt"
	"github.com/tjbdwanghaibo/roost-core/versionstore"
	"reflect"
	"testing"
	"time"
)

func reviewSettleAndEvict(t *testing.T, h *harness) (Envelope, Claim) {
	t.Helper()
	ctx := context.Background()
	e := mustSend(t, h, withAttachment(directTo(1), "reward"))
	c, err := h.service.ReserveClaim(ctx, 1, e.ID, "")
	if err != nil {
		t.Fatal(err)
	}
	if _, err = h.service.CommitClaim(ctx, 1, e.ID, c.Token); err != nil {
		t.Fatal(err)
	}
	for i := 0; i < MaxMailboxEntries; i++ {
		if err = h.service.Deliver(ctx, 1, fmt.Sprintf("filler-%d", i), 0); err != nil {
			t.Fatal(err)
		}
	}
	box, _, err := h.service.Mailbox(ctx, 1)
	if err != nil {
		t.Fatal(err)
	}
	if _, ok := box.Entries[e.ID]; ok {
		t.Fatal("not evicted")
	}
	if _, ok := box.SettledClaims[e.ID]; !ok {
		t.Fatal("no tombstone")
	}
	return e, c
}
func TestReviewMailboxSnapshotOwnsSettledClaims(t *testing.T) {
	h := newHarness(t)
	e, _ := reviewSettleAndEvict(t, h)
	ctx := context.Background()
	a, _, err := h.service.Mailbox(ctx, 1)
	if err != nil {
		t.Fatal(err)
	}
	delete(a.SettledClaims, e.ID)
	b, _, err := h.service.Mailbox(ctx, 1)
	if err != nil {
		t.Fatal(err)
	}
	if _, ok := b.SettledClaims[e.ID]; !ok {
		t.Fatal("mutating returned mailbox deleted stored settled claim")
	}
}
func TestReviewTombstoneCapacityOutlivesValidEnvelope(t *testing.T) {
	h := newHarness(t)
	ctx := context.Background()
	e, first := reviewSettleAndEvict(t, h)
	if _, err := h.service.ReserveClaim(ctx, 1, e.ID, ""); !errors.Is(err, ErrAlreadyClaimed) {
		t.Fatalf("control: %v", err)
	}
	for i := 0; i < MaxSettledClaims; i++ {
		h.clock.advance(time.Second)
		if _, err := h.service.Delete(ctx, 1, fmt.Sprintf("filler-%d", i)); err != nil {
			t.Fatal(err)
		}
		req := withAttachment(directTo(1), "new")
		req.RequestID = fmt.Sprintf("new-%d", i)
		n := mustSend(t, h, req)
		c, err := h.service.ReserveClaim(ctx, 1, n.ID, "")
		if err != nil {
			t.Fatal(err)
		}
		if _, err = h.service.CommitClaim(ctx, 1, n.ID, c.Token); err != nil {
			t.Fatal(err)
		}
		h.clock.advance(time.Second)
		if err = h.service.Deliver(ctx, 1, fmt.Sprintf("next-filler-%d", i), 0); errors.Is(err, ErrClaimHistoryFull) {
			break
		} else if err != nil {
			t.Fatal(err)
		}
	}
	box, _, err := h.service.Mailbox(ctx, 1)
	if err != nil {
		t.Fatal(err)
	}
	if _, ok := box.SettledClaims[e.ID]; !ok {
		t.Fatal("oldest protected tombstone disappeared")
	}
	if h.clock.Now().Unix() >= e.ExpiresAtUnix {
		t.Fatal("control: envelope expired")
	}
	if _, err = h.service.Delete(ctx, 1, "next-filler-0"); err != nil {
		t.Fatal(err)
	}
	if err = h.service.Deliver(ctx, 1, e.ID, 0); err != nil && !errors.Is(err, ErrClaimHistoryFull) {
		t.Fatal(err)
	}
	again, err := h.service.ReserveClaim(ctx, 1, e.ID, "")
	if err == nil {
		t.Fatalf("valid claimed mail got new token after tombstone eviction: old=%s new=%s elapsed=400s", first.Token, again.Token)
	}
	if !errors.Is(err, ErrAlreadyClaimed) {
		t.Fatal(err)
	}
}

func reviewAssertRefusalAtomic(t *testing.T, h *harness) {
	ctx := context.Background()
	_, _ = reviewSettleAndEvict(t, h)
	for i := 0; i < MaxSettledClaims; i++ {
		h.clock.advance(time.Second)
		if _, err := h.service.Delete(ctx, 1, fmt.Sprintf("filler-%d", i)); err != nil {
			t.Fatal(err)
		}
		req := withAttachment(directTo(1), "new")
		req.RequestID = fmt.Sprintf("new-%d", i)
		n := mustSend(t, h, req)
		c, err := h.service.ReserveClaim(ctx, 1, n.ID, "")
		if err != nil {
			t.Fatal(err)
		}
		if _, err = h.service.CommitClaim(ctx, 1, n.ID, c.Token); err != nil {
			t.Fatal(err)
		}
		h.clock.advance(time.Second)
		before, _, err := h.service.Mailbox(ctx, 1)
		if err != nil {
			t.Fatal(err)
		}
		versionBefore, _, _ := h.mailboxes.Get(ctx, 1)
		incoming := fmt.Sprintf("next-filler-%d", i)
		err = h.service.Deliver(ctx, 1, incoming, 0)
		if errors.Is(err, ErrClaimHistoryFull) {
			after, _, readErr := h.service.Mailbox(ctx, 1)
			if readErr != nil {
				t.Fatal(readErr)
			}
			versionAfter, _, _ := h.mailboxes.Get(ctx, 1)
			_, exists := after.Entries[incoming]
			unread := int32(0)
			for _, entry := range after.Entries {
				if entry.Status == StatusUnread {
					unread++
				}
			}
			if !reflect.DeepEqual(before, after) || versionBefore.Version != versionAfter.Version {
				t.Fatalf("refused delivery changed mailbox: inserted=%v settled=%d->%d version=%d->%d unread_field=%d actual_unread=%d", exists, len(before.SettledClaims), len(after.SettledClaims), versionBefore.Version, versionAfter.Version, after.Unread, unread)
			}
			return
		}
		if err != nil {
			t.Fatal(err)
		}
	}
	t.Fatal("control: no history refusal")
}

func TestReviewClaimHistoryRefusalIsAtomic(t *testing.T) {
	reviewAssertRefusalAtomic(t, newHarness(t))
}

type reviewIsolatedMailboxStore struct{ MailboxStore }

func (s *reviewIsolatedMailboxStore) Update(ctx context.Context, key int64, mutate versionstore.Mutate[Mailbox]) (versionstore.Versioned[Mailbox], bool, error) {
	return s.MailboxStore.Update(ctx, key, func(current Mailbox, found bool) (Mailbox, bool, error) { return mutate(current.clone(), found) })
}
func TestReviewRefusalAtomicWithIsolatedCallbackControl(t *testing.T) {
	h := newHarness(t, func(cfg *Config) { cfg.Mailboxes = &reviewIsolatedMailboxStore{cfg.Mailboxes} })
	reviewAssertRefusalAtomic(t, h)
}
```
