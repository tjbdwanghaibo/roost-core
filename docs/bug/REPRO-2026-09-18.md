```go
package fixture

import "github.com/tjbdwanghaibo/roost-core/entity"

const ReviewKind entity.EntityKind = 201

//roost:entity entityKind=ReviewKind category=entity.EntityCategoryPlayer sync=true subjectPacker=NewPacker
type Player struct {
	*entity.EntityBase
	entity.ComponentManager
	entity.DaoManager
	Value byte
}

func NewPacker(e entity.IThreadSafeEntity) entity.SubjectSyncPacker {
	p := e.(*Player)
	return entity.SubjectSyncPackFunc{
		Snapshot: func(entity.SyncProfile) (entity.FrozenSyncPayload, error) {
			return entity.CopyFrozenSyncPayload(1, []byte{p.Value}), nil
		},
		Delta: func(entity.SyncProfile, uint64) (entity.FrozenSyncPayload, error) {
			return entity.CopyFrozenSyncPayload(1, []byte{p.Value}), nil
		},
	}
}
```

## composition_test.go

```go
package reviewsync_test

import (
	"context"
	"errors"
	fixture "example.com/reviewsync/syncoff"
	"github.com/tjbdwanghaibo/roost-core/entity"
	es "github.com/tjbdwanghaibo/roost-core/entitysync"
	nt "github.com/tjbdwanghaibo/roost-core/nettransport"
	"github.com/tjbdwanghaibo/roost-core/room"
	ss "github.com/tjbdwanghaibo/roost-core/statesync"
	"sync"
	"testing"
	"time"
)

var ctx = context.Background()
var sub = es.SubscriberRef{ID: 9}

type transport struct {
	mu     sync.Mutex
	frames []nt.OutboundFrame
	fail   bool
}

func (r *transport) AdmitBatch(_ context.Context, f []nt.OutboundFrame) error {
	r.mu.Lock()
	defer r.mu.Unlock()
	if r.fail {
		return errors.New("review reject")
	}
	r.frames = append(r.frames, f...)
	return nil
}
func (r *transport) reset()        { r.mu.Lock(); r.frames = nil; r.mu.Unlock() }
func (r *transport) count() int    { r.mu.Lock(); defer r.mu.Unlock(); return len(r.frames) }
func (r *transport) reject(v bool) { r.mu.Lock(); r.fail = v; r.mu.Unlock() }
func (r *transport) last() nt.OutboundFrame {
	r.mu.Lock()
	defer r.mu.Unlock()
	return r.frames[len(r.frames)-1]
}

type harness struct {
	p    *fixture.Player
	m    *room.RoomManager
	r    *room.RoomBroadcaster
	s    *room.RoomTransportSink
	wire *transport
}

func player(t *testing.T) *fixture.Player {
	t.Helper()
	fixture.RegisterEntity()
	id, err := entity.BuildEntityID(42, fixture.ReviewKind)
	if err != nil {
		t.Fatal(err)
	}
	e, err := entity.BuildEntity(&entity.EntityCreateParam{Id: id, Kind: fixture.ReviewKind, IsCreate: true})
	if err != nil {
		t.Fatal(err)
	}
	p := e.(*fixture.Player)
	p.Value = 1
	p.EnableSync(entity.EntitySyncCreateParam{Enabled: true, EntityID: p.ID(), Packer: fixture.NewPacker(p)})
	return p
}
func setup(t *testing.T) *harness {
	t.Helper()
	h := &harness{p: player(t), wire: &transport{}}
	var err error
	h.s, err = room.NewRoomTransportSink(room.RoomTransportSinkConfig{Transport: h.wire, Sessions: room.RoomSessionResolverFunc(func(_ context.Context, s es.SubscriberRef) (ss.SessionID, error) { return ss.SessionID(s.ID), nil }), SlowConsumerPolicy: room.SlowConsumerFailBatch})
	if err != nil {
		t.Fatal(err)
	}
	h.m, err = room.NewRoomManager(room.RoomManagerConfig{Downstream: h.s, ReplicationInterval: time.Hour, SweepInterval: time.Hour})
	if err != nil {
		t.Fatal(err)
	}
	if err = h.m.Start(ctx); err != nil {
		t.Fatal(err)
	}
	h.r, err = h.m.Create(77)
	if err != nil {
		t.Fatal(err)
	}
	if err = h.r.RegisterSubject(h.p.Sync()); err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() {
		c, cancel := context.WithTimeout(ctx, time.Second)
		defer cancel()
		if err := h.m.Close(c); err != nil {
			t.Error(err)
		}
		if err := h.s.Close(c); err != nil {
			t.Error(err)
		}
		h.p.Sync().Close()
	})
	return h
}
func (h *harness) subscribe(t *testing.T) {
	t.Helper()
	if _, err := h.r.Subscribe(ctx, sub, h.p.ID(), entity.SyncProfile{}); err != nil {
		t.Fatal(err)
	}
}
func mutate(p *fixture.Player, lsn uint64) {
	p.GetMutex().Lock()
	defer p.GetMutex().Unlock()
	p.Value++
	p.SetLastCommitLSN(lsn)
	p.MarkSyncDirty(1)
}
func decode(t *testing.T, f nt.OutboundFrame) ss.DeltaFrame {
	t.Helper()
	raw := f.Reliable
	if len(raw) == 0 {
		r := ss.NewReassembler(ss.DefaultLimits(), time.Second)
		for _, d := range f.Datagrams {
			b, done, _, err := r.PushFor(f.Session, d, time.Now())
			if err != nil {
				t.Fatal(err)
			}
			if done {
				raw = b
			}
		}
	}
	_, _, d, err := room.DecodeRoomWireFrame(raw, ss.DefaultLimits())
	if err != nil {
		t.Fatal(err)
	}
	return d
}
func TestGeneratedManualComposition(t *testing.T) {
	for _, mode := range []string{"snapshot", "delta", "batch", "profile", "unsubscribe", "retire", "close", "admission_retry"} {
		t.Run(mode, func(t *testing.T) {
			h := setup(t)
			h.subscribe(t)
			d := decode(t, h.wire.last())
			if len(d.Objects) != 1 || d.Objects[0].Operation != ss.ObjectCreate {
				t.Fatalf("snapshot: %+v", d)
			}
			u, err := room.DecodeRoomSubjectUpdate(d.Objects[0].Components[0].Data, 0)
			if err != nil || u.SubjectID != h.p.ID() || string(u.Payload.BytesCopy()) != "\x01" {
				t.Fatalf("snapshot payload: %+v %v", u, err)
			}
			h.wire.reset()
			switch mode {
			case "snapshot":
				return
			case "delta", "batch", "admission_retry":
				mutate(h.p, 0)
				if mode == "admission_retry" {
					h.wire.reject(true)
					if h.r.FlushSubject(ctx, h.p.ID()) == nil {
						t.Fatal("rejection hidden")
					}
					if !h.p.Sync().PendingDirty() || h.wire.count() != 0 {
						t.Fatal("rejection consumed state")
					}
					h.wire.reject(false)
				}
				if mode == "batch" {
					err = h.r.FlushDirty(ctx)
				} else {
					err = h.r.FlushSubject(ctx, h.p.ID())
				}
				if err != nil {
					t.Fatal(err)
				}
				d = decode(t, h.wire.last())
				if d.Objects[0].Operation != ss.ObjectUpdate {
					t.Fatal("not update")
				}
				u, err = room.DecodeRoomSubjectUpdate(d.Objects[0].Components[0].Data, 0)
				if err != nil || string(u.Payload.BytesCopy()) != "\x02" {
					t.Fatalf("delta: %+v %v", u, err)
				}
			case "profile":
				_, err = h.r.Subscribe(ctx, sub, h.p.ID(), entity.SyncProfile{Key: "near", LOD: 1})
				if err != nil {
					t.Fatal(err)
				}
				d = decode(t, h.wire.last())
				u, err = room.DecodeRoomSubjectUpdate(d.Objects[0].Components[0].Data, 0)
				if err != nil || u.Profile.Key != "near" {
					t.Fatal("profile not updated")
				}
			case "unsubscribe", "retire":
				if mode == "retire" {
					err = h.r.RetireSubject(ctx, h.p.ID())
				} else {
					err = h.r.Unsubscribe(ctx, sub, h.p.ID())
				}
				if err != nil {
					t.Fatal(err)
				}
				d = decode(t, h.wire.last())
				if d.Objects[0].Operation != ss.ObjectRemove || len(h.r.Subscribers(h.p.ID())) != 0 {
					t.Fatal("leave failed")
				}
				if mode == "unsubscribe" {
					if err = h.r.UnregisterSubject(h.p.ID()); err != nil {
						t.Fatal(err)
					}
				}
				if h.m.Stats().ActiveSubjects != 0 || h.m.Stats().ActiveSubscribers != 0 {
					t.Fatal("budget not released")
				}
			case "close":
				if err = h.m.Close(ctx); err != nil {
					t.Fatal(err)
				}
				if h.m.Stats().ActiveSubjects != 0 || h.m.Stats().ActiveSubscribers != 0 {
					t.Fatal("close budget")
				}
				if _, err = h.r.Subscribe(ctx, sub, h.p.ID(), entity.SyncProfile{}); !errors.Is(err, room.ErrRoomBroadcasterStopped) {
					t.Fatalf("late subscribe %v", err)
				}
			}
		})
	}
}
func TestRoomPipelinedGate(t *testing.T) {
	for _, mode := range []string{"snapshot", "delta", "batch"} {
		t.Run(mode, func(t *testing.T) {
			h := setup(t)
			if mode != "snapshot" {
				h.subscribe(t)
				h.wire.reset()
			}
			mutate(h.p, 10)
			// Deployment durable LSN is 9. Room config exposes no hook to install it.
			var err error
			if mode == "snapshot" {
				_, err = h.r.Subscribe(ctx, sub, h.p.ID(), entity.SyncProfile{})
			} else if mode == "batch" {
				err = h.r.FlushDirty(ctx)
			} else {
				err = h.r.FlushSubject(ctx, h.p.ID())
			}
			if err != nil && !errors.Is(err, es.ErrDurabilityDeferred) {
				t.Fatal(err)
			}
			if h.wire.count() != 0 {
				t.Fatalf("LSN 10 escaped before durable LSN 9: frames=%d", h.wire.count())
			}
		})
	}
}
func TestCoordinatorGateControl(t *testing.T) {
	for _, mode := range []string{"snapshot", "delta", "batch"} {
		t.Run(mode, func(t *testing.T) {
			p := player(t)
			defer p.Sync().Close()
			sent := 0
			watermark := uint64(9)
			c := es.NewSubscriptionCoordinator(es.ReliableEnvelopeSinkFunc(func(_ context.Context, e []es.DeliveryEnvelope) error { sent += len(e); return nil }))
			defer c.Close()
			c.SetDurableWatermark(func() uint64 { return watermark })
			if mode != "snapshot" {
				if _, err := c.Subscribe(ctx, sub, p.Sync(), entity.SyncProfile{}); err != nil {
					t.Fatal(err)
				}
				sent = 0
			}
			mutate(p, 10)
			run := func() error {
				if mode == "snapshot" {
					_, err := c.Subscribe(ctx, sub, p.Sync(), entity.SyncProfile{})
					return err
				}
				if mode == "batch" {
					prepared, err := p.Sync().Prepare(c.Profiles(p.ID()))
					if err != nil {
						return err
					}
					return c.DistributeBatch(ctx, []*entity.PreparedSubjectSync{prepared})
				}
				return c.FlushSubject(ctx, p.Sync())
			}
			if err := run(); err != nil && !errors.Is(err, es.ErrDurabilityDeferred) {
				t.Fatal(err)
			}
			if sent != 0 || !p.Sync().PendingDirty() {
				t.Fatalf("gate sent=%d dirty=%v", sent, p.Sync().PendingDirty())
			}
			watermark = 10
			if err := run(); err != nil {
				t.Fatal(err)
			}
			if sent != 1 {
				t.Fatalf("durable retry sent=%d", sent)
			}
		})
	}
}
```
