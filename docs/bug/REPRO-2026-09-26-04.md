// errseq semantics: an error is reported once per open file). Ack does not
// check terminal and advances the checkpoint over bytes whose fsync failed.
func TestProbeAckAfterFsyncFailureAdvancesCheckpoint(t *testing.T) {
	dir := t.TempDir()
	opts := testOptions(dir)
	opts.GroupCommitInterval = time.Hour
	var fatal atomic.Value
	opts.OnFatal = func(err error) { fatal.Store(err) }
	w, err := Open(opts)
	if err != nil {
		t.Fatal(err)
	}
	fence1, err := w.Append(context.Background(), testRecord(1, corenest.DurabilityAsync))
	if err != nil {
		t.Fatal(err)
	}
	var calls atomic.Int32
	orig := probeFileSync
	probeFileSync = func(f *os.File) error {
		if calls.Add(1) == 1 {
			return &os.PathError{Op: "sync", Path: f.Name(), Err: syscall.EIO}
		}
		return nil // Linux: error already reported to this fd -> 0
	}
	defer func() { probeFileSync = orig }()
	_, err = w.Append(context.Background(), testRecord(2, corenest.DurabilityStrict))
	t.Logf("strict append after injected EIO: err=%v", err)
	t.Logf("terminal=%v unsynced=%v", w.terminal(), w.unsynced)
	fence2 := corenest.CommitFence{Segment: w.segment, Offset: w.offset}
	ackErr := w.Ack(context.Background(), fence2)
	cp, _ := loadCheckpoint(dir)
	t.Logf("Ack(fence=%d/%d) on terminal WAL: err=%v; stored checkpoint=%d/%d; fsync calls=%d", fence2.Segment, fence2.Offset, ackErr, cp.fence.Segment, cp.fence.Offset, calls.Load())
	_ = w.Close(context.Background())
	// Simulate the lost writeback: on disk nothing after the first record survived
	// (the failed pages never reached the platter).
	path := filepath.Join(dir, segmentName(fence1.Segment))
	if err := os.Truncate(path, 0); err != nil {
		t.Fatal(err)
	}
	probeFileSync = orig
	_, openErr := Open(opts)
	t.Logf("reopen after lost writeback: err=%v", openErr)

		t.Errorf("DEFECT: Ack succeeded on terminal WAL and the reopen refuses (RR-16 outcome)")
	}
}

// touchExpiryStub: the first generation's lease is lost (touch -> -1); later
// generations renew normally. Counts successful renewals per token.
type touchExpiryStub struct {
	*unlockEvalStub
	tmu      sync.Mutex
	expired  map[string]bool
	renewals map[string]int
}

func (s *touchExpiryStub) Eval(ctx context.Context, script string, keys []string, args ...any) (any, error) {
	if script != versionedTouchLua {
		return s.unlockEvalStub.Eval(ctx, script, keys, args...)
	}
	token := args[0].(string)
	s.tmu.Lock()
	defer s.tmu.Unlock()
	if s.expired[token] {
		s.unlockEvalStub.mu.Lock()
		if s.hash["owner"] == token {
			delete(s.hash, "owner") // Redis key gone
		}
		s.unlockEvalStub.mu.Unlock()
		return int64(-1), nil
	}
	s.renewals[token]++
	return int64(1000), nil
}

func touchActive(l *versionedLock) bool {
	l.touchMu.Lock()
	defer l.touchMu.Unlock()
	return l.touchActive
}

// State injection: the old goroutine has decided to exit (lease expired, acquired=false)
// but its deferred touchActive=false has not run yet. A new acquisition in that window
// never starts a renewal goroutine.
func TestProbeTouchRestartWindowStateInjection(t *testing.T) {
	stub := &touchExpiryStub{unlockEvalStub: newUnlockEvalStub(), expired: map[string]bool{}, renewals: map[string]int{}}
	lock := newVersionedLock(stub, 42, fredis.VersionedLockOptions{TTL: 30 * time.Millisecond, AutoAsyncTouch: true, AsyncTouchInterval: 2 * time.Millisecond, AsyncTouchExtend: 15 * time.Millisecond})
	// Simulate: previous generation expired, goroutine between Touch return and its defer.
	lock.touchMu.Lock()
	lock.touchActive = true
	lock.touchCancel = func() {}
	lock.touchMu.Unlock()
	if err := lock.TryLock(context.Background()); err != nil {
		t.Fatal(err)
	}
	token := lock.token
	// old goroutine's defer now runs
	lock.touchMu.Lock()
	lock.touchActive = false
	lock.touchCancel = nil
	lock.touchMu.Unlock()
	time.Sleep(40 * time.Millisecond)
	stub.tmu.Lock()
	n := stub.renewals[token]
	stub.tmu.Unlock()
	t.Logf("new generation acquired=%v renewals in 40ms (interval 2ms)=%d touchActive=%v", lock.IsAcquired(), n, touchActive(lock))
}

		}
