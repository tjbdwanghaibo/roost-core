```go
package match_test

import (
 "context"
 "math"
 "testing"
 match "github.com/tjbdwanghaibo/roost-core/service/match"
)

func fresh(t *testing.T) match.Store {
 t.Helper()
 s, err := match.NewMemoryStore(match.Config{})
 if err != nil { t.Fatal(err) }
 return s
}
func enqueue(t *testing.T, s match.Store, q match.Queue, id int64) match.Ticket {
 t.Helper()
 v, err := s.Enqueue(context.Background(), q, match.Subject{Kind:"player", ID:id, Payload:[]byte("old")}, "")
 if err != nil { t.Fatal(err) }
 return v
}
func TestReviewQueueCollision(t *testing.T) {
 a := match.Queue{Mode:"a:2", GroupSize:3, Partition:"x"}
 b := match.Queue{Mode:"a", GroupSize:2, Partition:"3:x"}
 for _, q := range []match.Queue{a,b} { if err:=q.Validate(); err!=nil {t.Fatal(err)} }
 s:=fresh(t)
 x,y:=enqueue(t,s,a,1),enqueue(t,s,b,2)
 got,err:=s.Commit(context.Background(),b,[]string{x.ID,y.ID})
 if err==nil {t.Fatalf("cross-queue commit accepted: keys=%q/%q match=%+v",a.Key(),b.Key(),got)}
}
func TestReviewScoreOverflow(t *testing.T) {
 q:=match.Queue{Mode:"ranked",GroupSize:2}
 cs:=[]match.Ticket{
  {ID:"a",Queue:q,State:match.TicketWaiting,ExpiresAtUnix:2000,Subject:match.Subject{Kind:"player",ID:1,Score:0},CreatedAtUnix:1000},
  {ID:"b",Queue:q,State:match.TicketWaiting,ExpiresAtUnix:2000,Subject:match.Subject{Kind:"player",ID:2,Score:math.MinInt64},CreatedAtUnix:1000},
 }
 for _,c:=range cs {if err:=c.Subject.Validate();err!=nil{t.Fatal(err)}}
 g:=match.ScoreWindowGrouping{InitialWindow:10,NowUnix:func()int64{return 1000}}
 group,formed,err:=g.Group(q,cs)
 if err!=nil {t.Fatal(err)}
 if formed {t.Fatalf("distance outside window accepted: %+v",group)}
}
func TestReviewMemoryResultOwnership(t *testing.T) {
 s:=fresh(t);q:=match.Queue{Mode:"ranked",GroupSize:2}
 a,b:=enqueue(t,s,q,1),enqueue(t,s,q,2)
 m,err:=s.Commit(context.Background(),q,[]string{a.ID,b.ID})
 if err!=nil {t.Fatal(err)}
 m.Members[0].ID=999
 again,found,err:=s.Match(context.Background(),q,m.ID)
 if err!=nil||!found {t.Fatal(found,err)}
 if again.Members[0].ID!=1 {t.Fatalf("stored match changed through returned slice: %+v",again)}
}
```
