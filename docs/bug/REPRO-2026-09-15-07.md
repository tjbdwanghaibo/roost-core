package syncstream
import("errors";"os";"testing";"time";"reflect")
func TestReviewHistoryIdentity(t *testing.T){for _,mode:=range []string{"delete_stream","delete_observer","sweep_idle","rotate_epoch"}{t.Run(mode,func(t *testing.T){h:=NewHistory(HistoryOptions{Epoch:7,IdleTTL:time.Second,PruneAcknowledged:true});o:=Observer{ID:1};s:=Stream{Topic:"state",Key:1};old,e:=h.Append(Packet{Observer:o,Stream:s,Full:true,Payload:[]byte("old")});if e!=nil{t.Fatal(e)};switch mode{case "delete_stream":e=h.DeleteStream(o,s);case "delete_observer":_,e=h.DeleteObserver(o);case "sweep_idle":_,e=h.SweepIdle(time.Now().Add(2*time.Second));case "rotate_epoch":e=h.RotateEpoch(8)};if e!=nil{t.Fatal(e)};fresh,e:=h.Append(Packet{Observer:o,Stream:s,Full:true,Payload:[]byte("new")});if e!=nil{t.Fatal(e)};replay:=h.Resync(ResyncRequest{Observer:o,Stream:s,Epoch:old.Epoch,AfterSequence:old.Sequence});ack:=h.AcknowledgeEpoch(o,s,old.Epoch,old.Sequence);if !replay.FullRequired&&len(replay.Packets)==0&&ack==nil&&h.Status(o,s).Retained==0{t.Fatalf("old identity silently swallowed new packet: old=%+v fresh=%+v replay=%+v status=%+v",old,fresh,replay,h.Status(o,s))};if mode=="rotate_epoch"&&!errors.Is(ack,ErrAckEpochMismatch){t.Fatal(ack)}})}}
func TestReviewWALTailContinuation(t *testing.T){for _,mode:=range []string{"clean","partial_tail","partial_then_checkpoint"}{t.Run(mode,func(t *testing.T){dir:=t.TempDir();j,e:=NewFileHistoryJournal(dir,7);if e!=nil{t.Fatal(e)};h,e:=NewHistoryWithJournal(HistoryOptions{},j);if e!=nil{t.Fatal(e)};s:=Stream{Topic:"state",Key:1};if _,e=h.Append(Packet{Stream:s,Full:true});e!=nil{t.Fatal(e)};path:=j.walPath(j.generation);if e=j.Close();e!=nil{t.Fatal(e)};if mode!="clean"{f,e:=os.OpenFile(path,os.O_APPEND|os.O_WRONLY,0600);if e!=nil{t.Fatal(e)};_,e=f.WriteString(`{"Version":1,"Kind":"append"`);if e!=nil{t.Fatal(e)};f.Close()};j2,e:=NewFileHistoryJournal(dir,99);if e!=nil{t.Fatal(e)};h2,e:=NewHistoryWithJournal(HistoryOptions{},j2);if e!=nil{t.Fatal("first recovery",e)};if mode=="partial_then_checkpoint"{if e=h2.Checkpoint();e!=nil{t.Fatal(e)}};p,e:=h2.Append(Packet{Stream:s,Payload:[]byte("committed-after-recovery")});if e!=nil{t.Fatal("append",e)};j2.Close();j3,e:=NewFileHistoryJournal(dir,99);if e!=nil{t.Fatal(e)};defer j3.Close();h3,e:=NewHistoryWithJournal(HistoryOptions{},j3);if e!=nil{t.Fatalf("acknowledged sequence %d cannot recover: %v",p.Sequence,e)};if h3.Status(Observer{},s).LatestSequence!=2{t.Fatal("lost append")}})}}
type reviewFailJournal struct{snapshot HistorySnapshot;fail bool}
func(j *reviewFailJournal)Load()(HistorySnapshot,error){return j.snapshot,nil}

func TestReviewJournalRefusal(t *testing.T){for _,mode:=range []string{"append","ack","delete_stream","delete_observer","sweep","rotate"}{t.Run(mode,func(t *testing.T){j:=&reviewFailJournal{snapshot:HistorySnapshot{Version:HistorySnapshotVersion,Epoch:7}};h,e:=NewHistoryWithJournal(HistoryOptions{IdleTTL:time.Second,PruneAcknowledged:true},j);if e!=nil{t.Fatal(e)};o:=Observer{ID:1};s:=Stream{Topic:"state"};if _,e=h.Append(Packet{Observer:o,Stream:s,Full:true});e!=nil{t.Fatal(e)};before:=h.Export();j.fail=true;switch mode{case "append":_,e=h.Append(Packet{Observer:o,Stream:s});case "ack":e=h.AcknowledgeEpoch(o,s,7,1);case "delete_stream":e=h.DeleteStream(o,s);case "delete_observer":_,e=h.DeleteObserver(o);case "sweep":_,e=h.SweepIdle(time.Now().Add(2*time.Second));case "rotate":e=h.RotateEpoch(8)};if e==nil||!reflect.DeepEqual(before,h.Export()){t.Fatalf("refusal mutated history err=%v",e)}})}}
