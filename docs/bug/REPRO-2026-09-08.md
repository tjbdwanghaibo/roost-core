## session_review_test.go

```go
package session

import (
	"context"
	"github.com/tjbdwanghaibo/roost-core/versionstore"
	"sync/atomic"
	"testing"
)

// Both initial ledger reads complete before either caller can insert. All
// writes still use the real MemoryStore's atomic insert-only implementation.
type reviewBarrierLedger struct {
	versionstore.Store[string, LedgerEntry]
	reads atomic.Int32
	ready chan struct{}
}

func (s *reviewBarrierLedger) Get(ctx context.Context, key string) (versionstore.Versioned[LedgerEntry], bool, error) {
	value, ok, err := s.Store.Get(ctx, key)
	n := s.reads.Add(1)
	if n == 2 {
		close(s.ready)
	}
	if n <= 2 {
		<-s.ready
	}
	return value, ok, err
}
func TestReviewEnterLedgerCollisionMustNotSucceed(t *testing.T) {
	ledger := &reviewBarrierLedger{Store: versionstore.NewMemoryStore[string, LedgerEntry](), ready: make(chan struct{})}
	h := newHarness(t, func(cfg *Config) { cfg.Requests = ledger })
	type result struct {
		owner int64
		run   Run
		err   error
	}
	done := make(chan result, 2)
	for _, owner := range []int64{1, 2} {
		go func(id int64) {
			r, e := h.service.Enter(context.Background(), id, enterReq("same-request"))
			done <- result{id, r, e}
		}(owner)
	}
	a, b := <-done, <-done
	if a.err == nil && b.err == nil {
		entry, _, _ := ledger.Store.Get(context.Background(), "same-request")
		loser := a
		if loser.owner == entry.Value.OwnerID {
			loser = b
		}
		_, retryErr := h.service.Enter(context.Background(), loser.owner, enterReq("same-request"))
		t.Fatalf("both owners succeeded (runs %s, %s), ledger owner=%d; losing owner's retry=%v", a.run.ID, b.run.ID, entry.Value.OwnerID, retryErr)
	}
}
```

## cache_review_test.go

```go
package cache

import (
	"context"
	"errors"
	"runtime"
	"testing"
)

func TestReviewCanceledWaiterReleasesCapacity(t *testing.T) {
	started, release, finished := make(chan struct{}), make(chan struct{}), make(chan struct{})
	s := NewReadThroughStore[int, int](nil, nil, func(context.Context, int) (int, bool, error) {
		close(started)
		<-release
		return 7, true, nil
	}, StoreConfig[int, int]{KeyOf: func(v int) int { return v }}, ReadThroughOptions{MaxWaitersPerKey: 1})
	go func() { defer close(finished); s.Get(context.Background(), 7) }()
	<-started
	defer func() { close(release); <-finished }()
	ctx, cancel := context.WithCancel(context.Background())
	defer cancel()
	waiterDone := make(chan error, 1)
	go func() { _, _, err := s.Get(ctx, 7); waiterDone <- err }()
	for s.Stats().Coalesced < 1 {
		runtime.Gosched()
	}
	cancel()
	if err := <-waiterDone; !errors.Is(err, context.Canceled) {
		t.Fatalf("first canceled waiter: %v", err)
	}
	// A fresh live context must be admitted after the previous follower left.
	nextCtx, nextCancel := context.WithCancel(context.Background())
	defer nextCancel()
	nextDone := make(chan error, 1)
	go func() { _, _, err := s.Get(nextCtx, 7); nextDone <- err }()
	for s.Stats().Coalesced < 2 {
		select {
		case err := <-nextDone:
			t.Fatalf("no live waiters, but replacement rejected: %v", err)
		default:
			runtime.Gosched()
		}
	}
	nextCancel()
	if err := <-nextDone; !errors.Is(err, context.Canceled) {
		t.Fatalf("replacement cancellation: %v", err)
	}
}
```
