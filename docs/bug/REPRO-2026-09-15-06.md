 opts:=SubscribeOptions{RequireChecksum:true};switch mode{case "max_chunks":opts.MaxChunks=1;case "max_envelope":opts.MaxEnvelopeBytes=1;case "max_payload":opts.MaxPayloadBytes=1;case "observer":opts.ExpectedObserver=&Observer{ID:8}}
 bus:=&memoryBus{};calls:=0;unsub,e:=SubscribeWithOptions(bus,"state",opts,func(got Packet)error{calls++;if !bytes.Equal(got.Payload,packet.Payload){t.Fatal("payload changed")};return nil});if e!=nil{t.Fatal(e)};defer unsub()
 if mode=="gzip_reverse"{for i,j:=0,len(frames)-1;i<j;i,j=i+1,j-1{frames[i],frames[j]=frames[j],frames[i]}}
