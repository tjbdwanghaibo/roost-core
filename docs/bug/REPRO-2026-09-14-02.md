## WAL：失败传播与 Close 取消重试

关闭真实文件句柄是受控 I/O 错误注入，不是磁盘掉电。Enqueue 已强制同步，故最终同步失败测试使用 Append(pipelined)。

```go
package nestwal
import (
 "context"
 "errors"
 "testing"
 "time"
 corenest "github.com/tjbdwanghaibo/roost-core/nest"
)
func TestReviewFinalSyncFailure(t *testing.T){
 opts:=testOptions(t.TempDir());opts.GroupCommitInterval=time.Hour
 w,err:=Open(opts);if err!=nil{t.Fatal(err)}
 defer w.Close(context.Background())
 _,err=w.Append(context.Background(),testRecord(110,corenest.DurabilityPipelined));if err!=nil{t.Fatal(err)}
 // A following barrier confirms physical writes without syncing.
 if err=w.awaitWriteBarrier(context.Background());err!=nil{t.Fatal(err)}
 w.stateMu.Lock();err=w.active.Close();w.stateMu.Unlock();if err!=nil{t.Fatal(err)}
 if err=w.Close(context.Background());err==nil{t.Fatal("Close ignored final sync failure")}
 if err=w.Sync(context.Background());!errors.Is(err,corenest.ErrCommitIndeterminate){t.Fatalf("Sync=%v",err)}

}
func TestReviewCloseCancelRetry(t *testing.T){
 entered,resume:=make(chan struct{}),make(chan struct{})
 opts:=testOptions(t.TempDir());opts.BatchMaxRecords=1;opts.GroupCommitInterval=time.Hour
```

## Activity：窗口交错、跨轮派发和正常对照

Create 包装器仅插入一个合法调度顺序；时钟推进不使用墙钟等待。

```go
package activity
import (
 "context"
 "testing"
 "time"
 "github.com/tjbdwanghaibo/roost-core/versionstore"
)
type reviewOpeningStore struct {
 versionstore.Store[Key, Activity]
 beforeCreate func()
}
func (s *reviewOpeningStore) Create(ctx context.Context, k Key, v Activity) (versionstore.Versioned[Activity], bool, error) {
 s.beforeCreate()
 return s.Store.Create(ctx,k,v)
}
func TestReviewOpenSweepInterleave(t *testing.T) {
 s,c:=newActivityService(t)
 ctx:=context.Background()
 s.cfg.Activities=&reviewOpeningStore{Store:s.cfg.Activities,beforeCreate:func(){
  if _,err:=s.AdvanceExpired(ctx,"group-a",10);err!=nil{t.Fatal(err)}
 }}
 k:=activityKey("opening")
 openActivity(t,s,k,1,2)
 notify(t,s,k,1)
 c.advance(time.Minute)
 if _,err:=s.AdvanceExpired(ctx,"group-a",10);err!=nil{t.Fatal(err)}
 a,found,err:=s.LookupActivity(ctx,k)
 if err!=nil||!found{t.Fatalf("lookup %v %v",found,err)}
 keys,err:=s.PendingActivities(ctx,"group-a",10);if err!=nil{t.Fatal(err)}
 if a.Status!=StatusComplete{t.Fatalf("open succeeded but expired activity stranded: status=%s window=%v",a.Status,keys)}
}
func TestReviewDispatchAcrossSweeps(t *testing.T) {
 for _,mode:=range []string{"grace","notify","heal"}{t.Run(mode,func(t *testing.T){
  s,c:=newActivityService(t)
  ctx:=context.Background(); k:=activityKey(mode)
  if mode=="heal"{s.cfg.Dispatches=&dispatchesFailingOnce{Store:s.cfg.Dispatches}}
  openActivity(t,s,k,1,2);notify(t,s,k,1)
  switch mode {case "grace": c.advance(time.Minute)
  case "notify": notify(t,s,k,2)
  case "heal": if _,err:=s.NotifyPhase(ctx,k,2);err==nil{t.Fatal("expected injected create failure")}}
  server:=&Server{service:s}
  server.sweepGroup(ctx,s,"group-a")
  c.advance(time.Minute)
  server.sweepGroup(ctx,s,"group-a")
  d,found,err:=s.LookupDispatch(ctx,k,1)
  if err!=nil||!found{t.Fatalf("lookup %v %v",found,err)}
  if d.Attempts!=2{t.Fatalf("two due sweeps: attempts=%d want=2 state=%s due=%v",d.Attempts,d.State,d.Due(c.Now().Unix()))}
 })}
}
func TestReviewActivityControls(t *testing.T){
 s,c:=newActivityService(t);ctx:=context.Background();k:=activityKey("control")
 openActivity(t,s,k,1,2);notify(t,s,k,1);c.advance(time.Minute)
```
