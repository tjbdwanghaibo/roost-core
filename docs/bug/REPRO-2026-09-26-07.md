type aud4Result struct {
	ret any
	err error

func aud4Send(mgr *NestMgr, name HandlerName, id int64) chan aud4Result {
	out := make(chan aud4Result, 1)
	go func() {
		ctx, cancel := context.WithTimeout(context.Background(), 20*time.Second)

	}()

// 遗留风险 2：外层可回滚事务里 RunIsolatedTransaction 已持久提交对声明实体的修改（值 + tracker 版本），外层随后失败回滚。
type aud4RecordCommitter struct {
	mu      sync.Mutex
	records []CommitRecord
	err     error
}

func (c *aud4RecordCommitter) Commit(_ context.Context, r CommitRecord) error {
	c.mu.Lock()
	defer c.mu.Unlock()
	c.records = append(c.records, r)
	return c.err
}

// 遗留风险 3b：带 Remote 批次的消息里调用 RunIsolatedTransaction，嵌套事务走外层批次 finalize / remoteCommitted 分支。
type aud4RecordingBatch struct {
	entity.RemoteWriteBatch
	mu        sync.Mutex
	finalized []string
	commits   atomic.Int32
	aborts    atomic.Int32
	closed    atomic.Int32
}

func (b *aud4RecordingBatch) FinalizeLocked(o entity.RemoteTransactionOutcome) error {
	b.mu.Lock()
	defer b.mu.Unlock()
	b.finalized = append(b.finalized, fmt.Sprintf("%+v", o.Handler))
	return nil
}
func (b *aud4RecordingBatch) Commits() []entity.RemoteCommit { return nil }
func (b *aud4RecordingBatch) Commit(context.Context) ([]entity.RemoteCommitReceipt, error) {
	b.commits.Add(1)
	return nil, nil
}
func (b *aud4RecordingBatch) Abort(context.Context, error) error { b.aborts.Add(1); return nil }
func (b *aud4RecordingBatch) Close(context.Context) error       { b.closed.Add(1); return nil }

// 再次撤销发生在“已重新登记、政策阶段尚未交付重新提交”之间。
func secondRetractBeforeDelivery(t *testing.T, manager *entitysync.Manager, id int64) {
	t.Helper()
	again := subjectState(t, id)
	if err := manager.Register(again); err != nil {
		t.Fatal(err)
	}
	// 非政策订阅者（例如业务直接 Subscribe）让 SubjectAwaitsReload 为真，重载器才会调度。
	if err := manager.OpenSession(99); err != nil {
		t.Fatal(err)
	}
	if err := manager.Subscribe(99, id, entity.SyncProfile{}); err != nil {
		t.Fatal(err)
	}
	again.Close()
	if !manager.SubjectAwaitsReload(id) {
		t.Fatal("premise: reloader would not schedule")
	}
	if !manager.RetractUnloadedSubject(id) {
		t.Fatal("premise: second retract refused")
	}
	flushTimes(manager, 3)
	if got := manager.Subscribers(id); len(got) != 0 {
		t.Fatalf("premise: second retraction not finished: %v", got)
	}
	if err := manager.Register(subjectState(t, id)); err != nil {
		t.Fatal(err)
	}
	flushTimes(manager, 3)
}
