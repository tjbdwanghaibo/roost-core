# RR-20260927-26：并发 Touch 推迟清理时，`ReleaseCast(旧实例)` 按 ID 释放了同 ID 新实例的锁

日期：2026-09-27。基线：`d4234c9`（批次 4 修复方复现），现 main `00f67c7`。来源：OPEN-ITEMS B07。等级 **P3**：handler 内锁被中途释放、业务以为仍持有。引入：修前即有（按 ID 释放）。状态：已复现，未修复。

无并发时不触发：Destroy 返回前旧实例 ID 已清零，`ReleaseCast` 为空操作。触发条件：另一个 goroutine 正按 Cast 顺序 Touch 旧 X 并等它的锁，
清理被推迟到最后一次 UnTouch（`entity/entity_base.go` UnTouch / ClearBase），旧实例 GUId 仍非零；handler 内 Destroy X → 新建 X → `ReleaseCast(旧X)` 调
`ReleaseEntity(id)`（`nest/cast.go:205`、`entity/entity_guard.go` ReleaseEntity）把新 X 的锁在 handler 中途放掉：
`ReleaseCast(old X) released the re-created X's lock mid-handler`。

修复约束：`ReleaseCast` / `ReleaseEntity` 按实例释放（只在 Guard 登记的持有者就是传入的那个实例时释放），与 RR-67 “按实例判断持有”一致；按 ID 的旧入口保留语义或标注；不在快 worker 上等待；RR-67/81 与 B17/B18 回归不回退。

红探针（批次 4）：

```go
package nest

import (
	"context"
	"testing"
	"time"

	"github.com/tjbdwanghaibo/roost-core/entity"
)

// 探针 OPEN-ITEMS B07（形状二）：另一个 goroutine 正按 Cast 的方式 Touch 旧 X 并等待它的锁时，旧 X 的清理
// （ID 清零）推迟到最后一个 UnTouch；此刻 ReleaseCast(旧 X) 按 ID 释放的是新 X。
func TestProbeB07ReleaseCastWhileOldInstanceTouched(t *testing.T) {
	manager := entity.NewEntityManager()
	pilots := addPilots(t, manager, 37200, 1)
	access := entity.NewManagerAccess(manager)
	x := mustBuildCastID(t, 37205, castOtherCategory, higherGroupCreatedKind)
	param := func() *entity.EntityCreateParam {
		return &entity.EntityCreateParam{IsCreate: true, Id: x, Category: castOtherCategory, Kind: higherGroupCreatedKind}
	}
	if _, err := access.Create(param()); err != nil {
		t.Fatal(err)
	}
	mgr := NewEngine(NestOptionWithGetter(access), NestOptionWithTransactionCommitter(&recordingCommitter{}), NestOptionWithWorkerNumAndMsgCap(1, 1, 16))
	name := NewHandlerName("probe_b07_touched")
	waiterTouched := make(chan struct{})
	waiterDone := make(chan bool, 1)
	var oldGUID int64
	var freshLockedAfterCreate, freshLockedAfterRelease bool
	var fresh entity.IThreadSafeEntity
	mgr.MustRegisterHandlerWithMeta(name, func([]entity.IThreadSafeEntity, []any, ...HandlerOption) (any, error) {
		old, err := CastOne[*rollbackTestEntity](x)
		if err != nil {
			return nil, err
		}
		go func() { // Nest 之外的另一个访问者，按 Cast 的顺序：Touch → 等锁
			if !old.Touch() {
				close(waiterTouched)
				waiterDone <- false
				return
			}
			close(waiterTouched)
			var locked bool
			_ = entity.WithGuardScope("probe_waiter", func(scope *entity.GuardScope) error {
				locked = scope.Guard().RequireEntity(old)
				return nil
			})
			old.UnTouch()
			waiterDone <- locked
		}()
		<-waiterTouched
		if err := access.Destroy(context.Background(), old, entity.EntityDestroyReason(0), false); err != nil {
			return nil, err
		}
		if fresh, err = access.Create(param()); err != nil {
			return nil, err
		}
		oldGUID = old.GUId()
		freshLockedAfterCreate = !tryLockElsewhere(fresh)
		ReleaseCast(old)
		freshLockedAfterRelease = !tryLockElsewhere(fresh)
		return "ok", nil
	}, HandlerMeta{})
	if err := mgr.Start(); err != nil {
		t.Fatal(err)
	}
	defer func() { _ = mgr.Shutdown(context.Background()) }()
	if _, err := mgr.Request(context.Background(), name, pilots[0], nil); err != nil {
		t.Fatal(err)
	}
	select {
	case locked := <-waiterDone:
		t.Logf("waiter locked the destroyed instance: %v", locked)
	case <-time.After(3 * time.Second):
		t.Log("waiter still blocked")
	}
	t.Logf("oldGUID after Destroy=%d freshLockedAfterCreate=%v freshLockedAfterReleaseCast=%v", oldGUID, freshLockedAfterCreate, freshLockedAfterRelease)
	if freshLockedAfterCreate && !freshLockedAfterRelease {
		t.Fatal("ReleaseCast(old X) released the re-created X's lock mid-handler")
	}
}
```
