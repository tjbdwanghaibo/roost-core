package terrain

import (
	"context"
	"errors"
	"github.com/tjbdwanghaibo/roost-core/framework/entity"
	"github.com/tjbdwanghaibo/roost-core/infra/base/spatial"
	"testing"
)

// Models the final Recovered call of a second Refresh that captured the same
// generation. Actual commits, publication failures, and the first Refresh use
// the generated module. It does not run two complete Refresh calls concurrently.
func TestReviewTerrainRecoveryGenerationABA(t *testing.T) {
	for _, lateCompletion := range []bool{false, true} {
		name := "control"
		if lateCompletion {
			name = "late_completion"
		}
		t.Run(name, func(t *testing.T) {
			f := newTerrainFixture(t)
			world := f.world(2)
			publish := world.SiteCommit
			fail := func(entity.IThreadSafeEntity, spatial.Point, bool) error {
				return errors.New("review: publication unavailable")
			}
			world.SiteCommit = fail
			built := f.must(build(1))
			if !built.ProjectionPending {
				t.Fatal("initial failure not recorded")
			}
			oldGeneration := world.Failures()
			world.SiteCommit = publish
			if err := f.controller.runtime.Refresh(context.Background(), 2); err != nil {
				t.Fatal(err)
			}
			if world.Failures() != 0 {
				t.Fatal("first Refresh did not recover")
			}
			world.SiteCommit = fail
			command := build(2)
			command.Op = "remove"
			removed := f.must(command)
			if !removed.ProjectionPending {
				t.Fatal("new failure not recorded")
			}
			newGeneration := world.Failures()
			if lateCompletion {
				world.Recovered(oldGeneration)
			}
			retry := f.must(command)
			f.ledger(2, 0)
			t.Logf("old=%d new=%d after=%d duplicate=%t pending=%t", oldGeneration, newGeneration, world.Failures(), retry.Duplicate, retry.ProjectionPending)
			world.SiteCommit = publish
			if !retry.ProjectionPending {
				t.Fatal("stale recovery completion erased an unrepaired committed removal")
			}
		})
	}
}
