for name, f, old, new, run in muts:
    p = root + f
    src = open(p).read()
    assert src.count(old) == 1, (name, src.count(old))
    open(p, 'w').write(src.replace(old, new))
    try:
        r = subprocess.run(['go', 'test', '-count=1', '-run', run, '.'], cwd=root, capture_output=True, text=True, env={**__import__('os').environ, 'GOWORK': 'off'})

        fails = [l.strip() for l in r.stdout.splitlines() if l.startswith('--- FAIL') or 'build failed' in l or l.startswith('#')]
        print(f"{name}: {verdict}  [{run}]  {'; '.join(fails[:3])}", flush=True)
    finally:
        open(p, 'w').write(src)
