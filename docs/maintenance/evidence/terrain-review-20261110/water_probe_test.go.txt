package terrain

import (
	"fmt"
	"math"
	"testing"
)

func TestReviewWaterMetadataCopy(t *testing.T) {
	meta := Meta{BaseRevision: 1, ChunksX: 1, ChunksY: 1, WaterLevel: 0, WaterRegions: []WaterRegion{{MaxX: 1, MaxY: 1, Level: 100}}}
	v := newWalkView(t, meta, 0)
	copy := v.Meta()
	copy.WaterRegions[0].Level = 200
	if got := v.Meta().WaterLevelOf(ChunkKey{}); got != v.WaterLevelOf(ChunkKey{}) {
		t.Fatalf("returned metadata aliases View: metadata level=%d, actual water=%d", got, v.WaterLevelOf(ChunkKey{}))
	}
}
func TestReviewWaterDepthNeverNegative(t *testing.T) {
	for _, level := range []int32{0, math.MaxInt32} {
		t.Run(fmt.Sprint(level), func(t *testing.T) {
			meta := Meta{BaseRevision: 1, ChunksX: 1, ChunksY: 1, HeightBase: -10000, WaterLevel: level}
			if err := meta.Validate(); err != nil {
				t.Fatal(err)
			}
			v := newWalkView(t, meta, 0)
			d, err := v.WaterDepth(0, 0)
			if err != nil {
				t.Fatal(err)
			}
			if d < 0 {
				t.Fatalf("valid water metadata produces negative depth: %d", d)
			}
		})
	}
}
