#!/bin/sh
# 删掉演练写进隔离环境 Redis 的键（前缀 roost:lrmp:），删前删后各数一次。地址取环境变量，不打印。
. "$HOME/.roost-it/roost-dataengine-it/env.sh"
h=${ROOST_DATAENGINE_IT_REDIS_ADDR%:*}; p=${ROOST_DATAENGINE_IT_REDIS_ADDR##*:}
echo "before: $(redis-cli -h "$h" -p "$p" --scan --pattern 'roost:lrmp:*' | wc -l | tr -d ' ') keys"
if [ "${1:-}" = "--delete" ]; then
  redis-cli -h "$h" -p "$p" --scan --pattern 'roost:lrmp:*' | while read -r k; do redis-cli -h "$h" -p "$p" DEL "$k" > /dev/null; done
  echo "after: $(redis-cli -h "$h" -p "$p" --scan --pattern 'roost:lrmp:*' | wc -l | tr -d ' ') keys"
fi
