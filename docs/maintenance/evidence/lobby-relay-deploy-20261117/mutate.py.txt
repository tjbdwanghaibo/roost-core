def run(pattern):
    r = subprocess.run(["go", "test", "-count=1", "-run", "^(" + pattern + ")$", "./codegen/internal/roost/"], cwd=CORE, env=dict(os.environ, GOWORK="off"), capture_output=True, text=True)
    return r.returncode, r.stdout + r.stderr
ok = True
for name, path, old, new, pattern in MUTATIONS:
    src = open(path).read()
    if src.count(old) != 1:
        print(f"{name}: anchor found {src.count(old)} times; SKIPPED-BROKEN"); ok = False; continue
    open(path, "w").write(src.replace(old, new))
    try:
        code, out = run(pattern)
    finally:
        open(path, "w").write(src)
    fails = [l.strip() for l in out.splitlines() if l.startswith("--- FAIL")]
    print(f"{name}: {'RED' if code != 0 else 'GREEN(!)'} {fails}")
    if code == 0: ok = False
code, out = run("|".join(sorted({m[4] for m in MUTATIONS})))
print("restored:", "GREEN" if code == 0 else "RED\n" + out[-2000:])
sys.exit(0 if ok and code == 0 else 1)
