env = {**os.environ, 'GOWORK': 'off'}
def run(test):
    return subprocess.run(['go', 'test', '-mod=mod', '-count=1', '-run', '^' + test + '$', './game/handler'], cwd=root, capture_output=True, text=True, env=env)
for name, path, old, new, test in muts:
    f = root / path
    text = f.read_text()
    assert text.count(old) == 1, name
    f.write_text(text.replace(old, new))
    try:
        r = run(test)
        out = r.stdout + r.stderr
        reason = [l.strip() for l in out.splitlines() if ('mmo_' in l or 'robot_' in l) and '.go:' in l][:1]
        print(f'{name}\n  {test}: mutated -> {"FAIL (red)" if r.returncode else "PASS (NOT RED!)"} {reason}', flush=True)
    finally:
        f.write_text(text)
    r = run(test)
    print(f'  restored -> {"PASS (green)" if r.returncode == 0 else "FAIL"}', flush=True)
