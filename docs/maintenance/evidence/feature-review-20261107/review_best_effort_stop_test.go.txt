package dataengine

import (
	"context"
	"fmt"
	"github.com/tjbdwanghaibo/roost-core/framework/nestwal"
	"testing"
	"time"
)

func TestReviewBestEffortStopWaitsForRunningHandler(t *testing.T) {
	for _, workers := range []int{1, 2} {
		t.Run(fmt.Sprint(workers), func(t *testing.T) {
			client := &bestEffortNATS{}
			entered, release, done := make(chan struct{}), make(chan struct{}), make(chan struct{})
			sub, err := subscribeBestEffort(client, "review.effect", "review", EffectConsumer{MaxAckPending: 4, AckWait: time.Minute}, workers, func(context.Context, nestwal.EffectEnvelope) error { close(entered); <-release; return nil })
			if err != nil {
				t.Fatal(err)
			}
			go func() { defer close(done); client.deliver(t, "review-1", "1") }()
			<-entered
			sub.Stop()
			early := false
			select {
			case <-sub.Closed():
				early = true
			case <-time.After(50 * time.Millisecond):
			}
			close(release)
			<-done
			select {
			case <-sub.Closed():
			case <-time.After(time.Second):
				t.Fatal("never closed after releasing handler")
			}
			if early {
				t.Fatal("Closed signaled while accepted handler was still running")
			}
		})
	}
}
