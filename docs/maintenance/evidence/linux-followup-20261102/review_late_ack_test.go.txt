package testdata
import("testing";"bytes";"errors")
func TestReviewFullCaptureLateAckAndRollback(t *testing.T){
 cases:=[]struct{name string;replace,child func(*HeroDao)}{
 {"pointer",func(d *HeroDao){d.SetMount(&EquipInfo{level:1})},func(d *HeroDao){d.GetMount().SetStar(99)}},
 {"list",func(d *HeroDao){d.SetSquadAll([]*EquipInfo{{level:1},{level:2}})},func(d *HeroDao){v,_:=d.GetSquad(1);v.SetStar(99)}},
 {"map",func(d *HeroDao){d.SetEquips(2,&EquipInfo{level:1})},func(d *HeroDao){v,_:=d.GetEquips(2);v.SetStar(99)}},
 }
 for _,tc:=range cases{t.Run(tc.name,func(t *testing.T){
 d:=collectionHero(8);baseline,_:=UnmarshalSyncHeroDao(d.MarshalSync(^uint64(0)))
 if err:=syncMutation(t,func()error{tc.replace(d);return nil});err!=nil{t.Fatal(err)}
 captured:=d.tracker.SyncChangeVersion();raw:=d.MarshalSyncChanges(^uint64(0));patch,_:=UnmarshalSyncHeroDao(raw)
 first,err:=ApplySyncHeroDao(patch,baseline);if err!=nil{t.Fatal(err)}
 if err:=syncMutation(t,func()error{tc.child(d);return nil});err!=nil{t.Fatal(err)}
 refused:=errors.New("rollback")
 if err:=syncMutation(t,func()error{tc.replace(d);d.tracker.AcknowledgeSyncChanges(captured);return refused});!errors.Is(err,refused){t.Fatal(err)}
 nextPatch,err:=UnmarshalSyncHeroDao(d.MarshalSyncChanges(^uint64(0)));if err!=nil{t.Fatal(err)}
 result,err:=ApplySyncHeroDao(nextPatch,first);if err!=nil{t.Fatal(err)}
 got,_:=MarshalSyncHeroDao(result);want:=d.MarshalSync(^uint64(0));if !bytes.Equal(got,want){t.Fatal("old confirmation or rollback lost a subsequent child write")}
 })}
}
