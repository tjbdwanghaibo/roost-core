only = sys.argv[1:]
for name, path, old, new, pkg, test in M:
    if only and name.split()[0] not in only: continue
    src = open(path).read()
    assert src.count(old) == 1, (name, src.count(old))
    open(path, 'w').write(src.replace(old, new))
    try:
        r = subprocess.run(['go', 'test', '-count=1', pkg, '-run', '^' + test + '$'], capture_output=True, text=True, timeout=300)
        out = (r.stdout + r.stderr).strip().splitlines()
        if r.returncode != 0 and any('[build failed]' in l or l.startswith('# ') for l in out):
            status = 'BUILD-FAILED(!)'
        else:
            status = 'RED' if r.returncode != 0 else 'GREEN(!)'
        fail = [l for l in out if '_test.go:' in l][:1]
        print(f"{name}\t{path}\t{test}\t{status}\t{fail[0].strip() if fail else out[-1] if out else ''}", flush=True)
    finally:
        open(path, 'w').write(src)
    r = subprocess.run(['go', 'test', '-count=1', pkg, '-run', '^' + test + '$'], capture_output=True, text=True, timeout=300)
    print(f"  restored -> {'GREEN' if r.returncode == 0 else 'RED(!)'}", flush=True)
