cmake_minimum_required(VERSION 3.20)
project(roost_client LANGUAGES CXX)
option(BUILD_SHARED_LIBS "Build a shared engine plugin" OFF)
add_library(roost_client src/client.cpp)
target_compile_features(roost_client PUBLIC cxx_std_17)
target_include_directories(roost_client PUBLIC include)
set_target_properties(roost_client PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
if(BUILD_SHARED_LIBS)
 target_compile_definitions(roost_client PUBLIC ROOST_SHARED PRIVATE ROOST_BUILD)
endif()
if(WIN32)
 target_link_libraries(roost_client PRIVATE ws2_32)
else()
 find_package(Threads REQUIRED)
 target_link_libraries(roost_client PRIVATE Threads::Threads)
endif()
option(ROOST_BUILD_TESTS "Build native loopback and Sync tests" OFF)
option(ROOST_PROTOBUF_TESTS "Build generated DAO protobuf tests" OFF)
set(ROOST_DEMO_ROOT "" CACHE PATH "Generated game-demo used by the Gate integration client")
if(ROOST_BUILD_TESTS)
 enable_testing()
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 add_executable(roost_native_smoke tests/smoke.cpp)
 target_link_libraries(roost_native_smoke PRIVATE roost_client)
 add_test(NAME native_client COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/run.py $<TARGET_FILE:roost_native_smoke>)
endif()
if(ROOST_PROTOBUF_TESTS OR ROOST_DEMO_ROOT)
 find_package(Protobuf CONFIG REQUIRED)
 function(roost_proto target source)
  get_filename_component(directory "${source}" DIRECTORY)
  get_filename_component(name "${source}" NAME_WE)
  add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/${name}.pb.cc" "${CMAKE_CURRENT_BINARY_DIR}/${name}.pb.h"
   COMMAND protobuf::protoc "--proto_path=${directory}" "--cpp_out=${CMAKE_CURRENT_BINARY_DIR}" "${source}"
   DEPENDS "${source}" protobuf::protoc VERBATIM)
  target_sources(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/${name}.pb.cc")
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_link_libraries(${target} PRIVATE protobuf::libprotobuf)
 endfunction()
endif()
if(ROOST_PROTOBUF_TESTS)
 enable_testing()
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 add_executable(roost_native_protobuf tests/protobuf.cpp)
 target_link_libraries(roost_native_protobuf PRIVATE roost_client)
 roost_proto(roost_native_protobuf "${CMAKE_CURRENT_SOURCE_DIR}/../../codegen/internal/dao/testdata/golden/dao_sync.proto")
 add_test(NAME native_protobuf COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/protobuf.py $<TARGET_FILE:roost_native_protobuf>)
endif()
if(ROOST_DEMO_ROOT)
 add_executable(roost_gate_client tests/gate.cpp)
 target_link_libraries(roost_gate_client PRIVATE roost_client)
 roost_proto(roost_gate_client "${ROOST_DEMO_ROOT}/protocol/proto/protocol.proto")
 roost_proto(roost_gate_client "${ROOST_DEMO_ROOT}/db/dao_sync.proto")
endif()
