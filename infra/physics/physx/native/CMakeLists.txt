cmake_minimum_required(VERSION 3.16)
project(RoostPhysX LANGUAGES C CXX)
set(PHYSX_ROOT_DIR "${PHYSX_SOURCE}/physx" CACHE PATH "" FORCE)
set(PXSHARED_PATH "${PHYSX_SOURCE}/pxshared" CACHE PATH "" FORCE)
set(CMAKEMODULES_PATH "${PHYSX_SOURCE}/externals/cmakemodules" CACHE PATH "" FORCE)
set(CMAKEMODULES_VERSION "1.27" CACHE STRING "" FORCE)
set(PX_GENERATE_STATIC_LIBRARIES ON CACHE BOOL "" FORCE)
set(PX_BUILDSNIPPETS OFF CACHE BOOL "" FORCE)
set(PX_BUILDPUBLICSAMPLES OFF CACHE BOOL "" FORCE)
set(PX_OUTPUT_LIB_DIR "${CMAKE_BINARY_DIR}/lib" CACHE PATH "" FORCE)
set(PX_OUTPUT_BIN_DIR "${CMAKE_BINARY_DIR}/bin" CACHE PATH "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(APPLE)
 set(TARGET_BUILD_PLATFORM mac CACHE STRING "" FORCE)
else()
 set(TARGET_BUILD_PLATFORM linux CACHE STRING "" FORCE)
endif()
add_subdirectory("${PHYSX_ROOT_DIR}/compiler/public" sdk)
add_library(roost_physx SHARED bridge.cpp)
target_compile_features(roost_physx PRIVATE cxx_std_14)
target_compile_definitions(roost_physx PRIVATE NDEBUG PX_PHYSX_STATIC_LIB)
target_include_directories(roost_physx PRIVATE "${PHYSX_ROOT_DIR}/include" "${PXSHARED_PATH}/include")
target_link_libraries(roost_physx PRIVATE PhysXExtensions PhysX PhysXCommon PhysXFoundation PhysXPvdSDK)
set_target_properties(roost_physx PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${ROOST_LIB_DIR}")
